工作流

工作流

复杂的租户操作（开通、升级）可能需要审批流程。本节实现工作流的发起和审批操作，以及操作日志搜索页。

🤔 导师提问：工作流引入了"审批"环节，让开通和升级操作不再是即时生效的。在后端，你可能用状态机（Pending → Approved/Rejected）来建模。前端的工作流页面与直接调用 tenantsApi.provision 的租户管理页面有什么关系？两者都能开通租户——什么场景下用工作流，什么场景下直接开通？

工作流审批页

// src/pages/ops/workflow/index.tsx
import { useState, useEffect } from "react"
import { Play, ArrowUpCircle, CheckCircle, XCircle } from "lucide-react"
import { Button } from "@/components/ui/button"
import { Input } from "@/components/ui/input"
import { Label } from "@/components/ui/label"
import { Textarea } from "@/components/ui/textarea"
import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card"
import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from "@/components/ui/select"
import { toast } from "sonner"
import {
  startProvisionWorkflow, startUpgradeWorkflow, approveWorkflow, rejectWorkflow,
  type StartProvisionWorkflowRequest, type StartUpgradeWorkflowRequest, type ApprovalRequest,
} from "@/api/ops/workflow"
import { getEditionList, type EditionDto } from "@/api/ops/editions"

export default function OpsWorkflowPage() {
  const [editions, setEditions] = useState<EditionDto[]>([])
  const [provisionForm, setProvisionForm] = useState<StartProvisionWorkflowRequest>({
    tenantName: "", tenantIdentifier: "", editionId: "",
  })
  const [upgradeForm, setUpgradeForm] = useState<StartUpgradeWorkflowRequest>({
    tenantId: "", targetEditionId: "",
  })
  const [approvalForm, setApprovalForm] = useState<ApprovalRequest>({
    workflowId: "", approved: true, reason: "",
  })
  const [submitting, setSubmitting] = useState<string | null>(null)

  useEffect(() => {
    getEditionList().then((list) => setEditions(list.filter((e) => e.isActive)))
  }, [])

  const handleStartProvision = async () => {
    if (!provisionForm.tenantName || !provisionForm.tenantIdentifier || !provisionForm.editionId) {
      toast.error("请填写完整的开通信息"); return
    }
    setSubmitting("provision")
    try { await startProvisionWorkflow(provisionForm); toast.success("开通工作流已发起") }
    catch { toast.error("发起失败") } finally { setSubmitting(null) }
  }

  const handleStartUpgrade = async () => {
    if (!upgradeForm.tenantId || !upgradeForm.targetEditionId) {
      toast.error("请填写完整的升级信息"); return
    }
    setSubmitting("upgrade")
    try { await startUpgradeWorkflow(upgradeForm); toast.success("升级工作流已发起") }
    catch { toast.error("发起失败") } finally { setSubmitting(null) }
  }

  const handleApproval = async (approved: boolean) => {
    if (!approvalForm.workflowId) { toast.error("请输入工作流 ID"); return }
    setSubmitting("approval")
    try {
      const fn = approved ? approveWorkflow : rejectWorkflow
      await fn({ ...approvalForm, approved })
      toast.success(approved ? "已审批通过" : "已审批拒绝")
    } catch { toast.error("审批操作失败") } finally { setSubmitting(null) }
  }

  return (
    <div className="p-6 space-y-6">
      <h1 className="text-2xl font-bold">工作流审批</h1>

      <div className="grid grid-cols-1 lg:grid-cols-3 gap-6">
        {/* 发起开通工作流 */}
        <Card>
          <CardHeader><CardTitle className="flex items-center gap-2 text-base">
            <Play className="h-5 w-5" /> 发起开通</CardTitle></CardHeader>
          <CardContent>
            <div className="space-y-3">
              <div className="space-y-2"><Label>租户名称</Label>
                <Input value={provisionForm.tenantName}
                  onChange={(e) => setProvisionForm((prev) => ({ ...prev, tenantName: e.target.value }))} /></div>
              <div className="space-y-2"><Label>租户标识</Label>
                <Input value={provisionForm.tenantIdentifier}
                  onChange={(e) => setProvisionForm((prev) => ({ ...prev, tenantIdentifier: e.target.value }))} /></div>
              <div className="space-y-2"><Label>目标版本</Label>
                <Select value={provisionForm.editionId}
                  onValueChange={(v) => setProvisionForm((prev) => ({ ...prev, editionId: v }))}>
                  <SelectTrigger><SelectValue placeholder="选择版本" /></SelectTrigger>
                  <SelectContent>
                    {editions.map((e) => (<SelectItem key={e.id} value={e.id}>{e.displayName}</SelectItem>))}
                  </SelectContent></Select></div>
              <Button className="w-full" onClick={handleStartProvision} disabled={submitting !== null}>
                {submitting === "provision" ? "提交中..." : "发起开通"}</Button>
            </div>
          </CardContent>
        </Card>

        {/* 发起升级工作流 */}
        <Card>
          <CardHeader><CardTitle className="flex items-center gap-2 text-base">
            <ArrowUpCircle className="h-5 w-5" /> 发起升级</CardTitle></CardHeader>
          <CardContent>
            <div className="space-y-3">
              <div className="space-y-2"><Label>租户 ID</Label>
                <Input value={upgradeForm.tenantId}
                  onChange={(e) => setUpgradeForm((prev) => ({ ...prev, tenantId: e.target.value }))} /></div>
              <div className="space-y-2"><Label>目标版本</Label>
                <Select value={upgradeForm.targetEditionId}
                  onValueChange={(v) => setUpgradeForm((prev) => ({ ...prev, targetEditionId: v }))}>
                  <SelectTrigger><SelectValue placeholder="选择版本" /></SelectTrigger>
                  <SelectContent>
                    {editions.map((e) => (<SelectItem key={e.id} value={e.id}>{e.displayName}</SelectItem>))}
                  </SelectContent></Select></div>
              <Button className="w-full" onClick={handleStartUpgrade} disabled={submitting !== null}>
                {submitting === "upgrade" ? "提交中..." : "发起升级"}</Button>
            </div>
          </CardContent>
        </Card>

        {/* 审批操作 */}
        <Card>
          <CardHeader><CardTitle className="flex items-center gap-2 text-base">
            <CheckCircle className="h-5 w-5" /> 审批操作</CardTitle></CardHeader>
          <CardContent>
            <div className="space-y-3">
              <div className="space-y-2"><Label>工作流 ID</Label>
                <Input value={approvalForm.workflowId}
                  onChange={(e) => setApprovalForm((prev) => ({ ...prev, workflowId: e.target.value }))} /></div>
              <div className="space-y-2"><Label>审批原因</Label>
                <Textarea value={approvalForm.reason}
                  onChange={(e) => setApprovalForm((prev) => ({ ...prev, reason: e.target.value }))} rows={3} /></div>
              <div className="flex gap-2">
                <Button className="flex-1" onClick={() => handleApproval(true)} disabled={submitting !== null}>
                  <CheckCircle className="mr-2 h-4 w-4" /> 通过</Button>
                <Button variant="destructive" className="flex-1" onClick={() => handleApproval(false)} disabled={submitting !== null}>
                  <XCircle className="mr-2 h-4 w-4" /> 拒绝</Button>
              </div>
            </div>
          </CardContent>
        </Card>
      </div>
    </div>
  )
}

工作流流程

发起开通/升级 → 创建工作流记录 → 等待审批
                                       ↓
                              审批通过 → 执行操作
                              审批拒绝 → 工作流终止

后端类比：工作流就像 IWorkflowExecutor + 状态机——先创建待审批记录（Pending），审批通过后执行实际操作（Provision/Upgrade）。审批拒绝则标记为 Rejected。

🤔 导师提问：工作流的三个 Card（开通/升级/审批）都共用一个 submitting 状态来防止重复提交。这和功能开关页面的独立 loading 策略不同——三个按钮共用一个 string | null。为什么工作流页面选择全局互斥，而不是三个独立 loading？如果两个运营员同时审批不同的工作流，后端如何处理并发审批？

🤔 导师提问：工作流的状态流转（Pending → Approved → Executed / Rejected）是经典的有限状态机。在后端，你通常用数据库事务保证状态转换的原子性。但前端没有事务概念——如果 approveWorkflow 请求发出后网络超时，前端无法知道后端是否已处理。这种情况在前端应该如何处理？提示：考虑轮询工作流状态或乐观更新 + 回滚。

操作日志搜索页

// src/pages/ops/search/index.tsx
import { useState } from "react"
import { Search } from "lucide-react"
import { Button } from "@/components/ui/button"
import { Input } from "@/components/ui/input"
import { Table, TableBody, TableCell, TableHead, TableHeader, TableRow } from "@/components/ui/table"
import { Badge } from "@/components/ui/badge"
import { toast } from "sonner"
import opsHttp from "@/api/ops/http"
import type { TenantOperationLogDto } from "@/api/ops/tenants"

export default function OpsSearchPage() {
  const [keyword, setKeyword] = useState("")
  const [results, setResults] = useState<TenantOperationLogDto[]>([])
  const [loading, setLoading] = useState(false)

  const handleSearch = async () => {
    if (!keyword.trim()) { toast.error("请输入搜索关键词"); return }
    setLoading(true)
    try { setResults(await opsHttp.get<TenantOperationLogDto[]>("/search", { params: { keyword } })) }
    catch { toast.error("搜索失败") } finally { setLoading(false) }
  }

  return (
    <div className="p-6 space-y-6">
      <h1 className="text-2xl font-bold">操作日志搜索</h1>
      <div className="flex gap-2">
        <Input value={keyword} onChange={(e) => setKeyword(e.target.value)}
          placeholder="输入关键词搜索操作日志..." className="max-w-md"
          onKeyDown={(e) => e.key === "Enter" && handleSearch()} />
        <Button onClick={handleSearch} disabled={loading}>
          <Search className="mr-2 h-4 w-4" /> 搜索</Button>
      </div>
      <Table><TableHeader><TableRow>
        <TableHead>操作类型</TableHead><TableHead>详情</TableHead>
        <TableHead>操作人</TableHead><TableHead>IP 地址</TableHead><TableHead>时间</TableHead>
      </TableRow></TableHeader>
      <TableBody>
        {results.map((log, i) => (
          <TableRow key={i}>
            <TableCell><Badge variant="outline">{log.operationType}</Badge></TableCell>
            <TableCell className="max-w-md truncate">{log.detail}</TableCell>
            <TableCell>{log.operatorName}</TableCell>
            <TableCell className="font-mono text-xs">{log.clientIpAddress}</TableCell>
            <TableCell className="text-sm">{new Date(log.creationTime).toLocaleString()}</TableCell>
          </TableRow>
        ))}
      </TableBody></Table>
    </div>
  )
}

🤔 导师提问：操作日志搜索页直接使用了 opsHttp.get（原始 axios 实例），而不是通过 API 模块封装。这和 02 节中 data-exchange.ts 的做法类似。什么情况下应该把 API 调用封装到独立模块，什么情况下可以直接在组件中使用 opsHttp？两种方式的优缺点是什么？

项目审视

回顾本章，我们构建了一个功能完整的 SaaS 租户运营中心。思考以下问题：

  1. 状态枚举的一致性：TENANT_STATUS、HEALTH_STATUS 等枚举值由后端定义，前端如何保证与后端同步？如果后端新增了一个状态，前端会怎样？考虑使用 OpenAPI 代码生成工具自动同步。

  2. 权限粒度：当前所有运营操作都要求运营员角色，但实际中"暂停租户"和"查看日志"的权限级别不同。如何在前端实现更细粒度的权限控制？回顾篇章 4 的权限中心，设计一个 usePermission Hook。

  3. 轮询 vs 推送：租户健康状态页目前需要手动刷新。如果需要实时监控，应该使用 WebSocket 还是 SSE？结合篇章 8 通知中心的 SignalR 基础设施，考虑复用方案。

  4. 表单复杂度边界：EditionFormDialog 已经有动态模块行和配额行，如果未来版本需要支持"嵌套动态行"（如每个配额下还有子项），当前的 useState + 数组方案是否还能维护？何时应该引入 Form 状态管理库（如 react-hook-form 的 useFieldArray）？

  5. 国际化准备：所有状态徽章的文字（"活跃"、"已暂停"）都是硬编码中文。如果系统需要支持多语言，状态枚举到显示文字的映射应该如何设计？

🤔 导师提问：审批操作中，handleApproval 函数同时处理通过和拒绝——const fn = approved ? approveWorkflow : rejectWorkflow。这和后端用一个 Action 处理多种操作（如 [HttpPatch] approve 和 [HttpPatch] reject 合并为 [HttpPatch] process）的取舍一样。合并处理的简洁性 vs 分离处理的清晰性，你会如何选择？

🔍 验证步骤

  1. 在工作流审批页面填写租户名称、标识符，选择版本，点击"发起开通"——观察 Network 面板发送 POST /ops-api/api/tenant-ops/workflow/provision/start，返回 instanceId 和 status: Pending
  2. 复制返回的 instanceId，粘贴到审批操作卡的工作流 ID 输入框，填写审批原因，点击"通过"——确认 toast 提示"已审批通过"
  3. 再发起一个开通工作流，这次点击"拒绝"——确认 toast 提示"已审批拒绝"
  4. 在操作日志搜索页输入关键词搜索——确认返回日志列表包含操作类型、操作人、IP 和时间
  5. 验证全局互斥：在开通工作流提交中（按钮显示"提交中..."），确认升级和审批按钮也被禁用

小结

功能实现
开通工作流租户信息 + 版本选择 + 发起
升级工作流租户 ID + 目标版本 + 发起
审批操作工作流 ID + 原因 + 通过/拒绝
操作日志搜索关键词搜索 + Table 展示
