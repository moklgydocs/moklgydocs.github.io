架构级开发模式：KeepAlive、URL 驱动状态、文件上传与 i18n

TL;DR
中后台系统有几个"所有项目迟早要面对"的架构级难题——多标签页 KeepAlive（React 官方不支持）、筛选条件与 URL 同步（用户能分享/刷新不丢失）、大文件分片上传、多语言国际化。本章给出每个难题的生产级解决方案。

1. KeepAlive 多标签页缓存

痛点

中后台用户习惯同时打开多个标签页（订单列表 → 订单详情 → 回到列表），切换回列表时期望看到之前的筛选状态和滚动位置。但 React 的默认行为是——组件卸载时销毁所有状态。

方案：基于 display:none 的页面缓存

// src/components/layout/KeepAlive.tsx — 简洁的 KeepAlive 实现
import { useRef, type ReactNode, type ReactElement } from "react";
import { useLocation } from "react-router-dom";

interface CacheEntry {
  key: string;
  element: ReactElement;
}

interface KeepAliveProps {
  include?: string[];  // 需要缓存的路由路径
  maxCache?: number;   // 最大缓存数
  children: ReactNode;
}

function KeepAlive({ include, maxCache = 10, children }: KeepAliveProps) {
  const location = useLocation();
  const cacheRef = useRef<CacheEntry[]>([]);

  const currentPath = location.pathname;
  const shouldCache = !include || include.some((p) => currentPath.startsWith(p));

  // 更新缓存
  if (shouldCache && children) {
    const existingIndex = cacheRef.current.findIndex(
      (entry) => entry.key === currentPath
    );

    if (existingIndex >= 0) {
      // 已缓存，更新内容
      cacheRef.current[existingIndex].element = children as ReactElement;
    } else {
      // 新增缓存
      cacheRef.current.push({
        key: currentPath,
        element: children as ReactElement,
      });

      // 超出最大缓存数时移除最旧的
      if (cacheRef.current.length > maxCache) {
        cacheRef.current.shift();
      }
    }
  }

  return (
    <>
      {cacheRef.current.map((entry) => (
        <div
          key={entry.key}
          style={{
            display: entry.key === currentPath ? "block" : "none",
            height: entry.key === currentPath ? "auto" : 0,
            overflow: entry.key === currentPath ? "visible" : "hidden",
          }}
        >
          {entry.element}
        </div>
      ))}
      {/* 不在缓存中的页面正常渲染 */}
      {!shouldCache && children}
    </>
  );
}

export { KeepAlive };
// 在 AdminLayout 中使用
function AdminLayout() {
  return (
    <div className="flex h-screen">
      <Sidebar />
      <main className="flex-1 overflow-auto p-6">
        <KeepAlive include={["/orders", "/dashboard", "/inventory"]}>
          <Outlet />
        </KeepAlive>
      </main>
    </div>
  );
}

import { useEffect, useRef } from "react";

function useActivated(callback: () => void) {
  const isFirstMount = useRef(true);

  useEffect(() => {
    // 利用 IntersectionObserver 检测 display:none → display:block 的切换
    // 简化方案：利用 document.visibilityState 或自定义事件

    if (isFirstMount.current) {
      isFirstMount.current = false;
      return;
    }
    callback();
  });
}

2. URL 驱动状态模式

痛点

运营同学在订单列表页设置了筛选条件（状态=待处理、日期=最近7天、搜索=手机），然后把 URL 分享给同事——同事打开是空白筛选条件。或者用户按 F5 刷新页面，筛选条件丢失。

方案：所有列表页的筛选/排序/分页都同步到 URL

// src/hooks/use-table-url-state.ts — 表格状态与 URL 双向同步
import { useSearchParams } from "react-router-dom";
import { useMemo, useCallback } from "react";

interface TableUrlState {
  page: number;
  pageSize: number;
  sortBy: string;
  sortOrder: "asc" | "desc";
  filters: Record<string, string>;
}

const DEFAULT_STATE: TableUrlState = {
  page: 1,
  pageSize: 20,
  sortBy: "",
  sortOrder: "desc",
  filters: {},
};

function useTableUrlState(
  filterKeys: string[] = []
): [TableUrlState, (updates: Partial<TableUrlState>) => void, () => void] {
  const [searchParams, setSearchParams] = useSearchParams();

  const state = useMemo<TableUrlState>(() => {
    const filters: Record<string, string> = {};
    for (const key of filterKeys) {
      const value = searchParams.get(key);
      if (value) filters[key] = value;
    }

    return {
      page: Number(searchParams.get("page")) || DEFAULT_STATE.page,
      pageSize: Number(searchParams.get("pageSize")) || DEFAULT_STATE.pageSize,
      sortBy: searchParams.get("sortBy") || DEFAULT_STATE.sortBy,
      sortOrder: (searchParams.get("sortOrder") as "asc" | "desc") || DEFAULT_STATE.sortOrder,
      filters,
    };
  }, [searchParams, filterKeys]);

  const setState = useCallback(
    (updates: Partial<TableUrlState>) => {
      setSearchParams((prev) => {
        const next = new URLSearchParams(prev);

        if (updates.page !== undefined) next.set("page", String(updates.page));
        if (updates.pageSize !== undefined) {
          next.set("pageSize", String(updates.pageSize));
          next.set("page", "1"); // 改页大小回到第一页
        }
        if (updates.sortBy !== undefined) next.set("sortBy", updates.sortBy);
        if (updates.sortOrder !== undefined) next.set("sortOrder", updates.sortOrder);
        if (updates.filters !== undefined) {
          // 先清除旧的 filter 参数
          for (const key of filterKeys) {
            next.delete(key);
          }
          // 设置新的
          for (const [key, value] of Object.entries(updates.filters)) {
            if (value) {
              next.set(key, value);
            }
          }
          next.set("page", "1"); // 改筛选回到第一页
        }

        return next;
      });
    },
    [setSearchParams, filterKeys]
  );

  const resetState = useCallback(() => {
    setSearchParams({});
  }, [setSearchParams]);

  return [state, setState, resetState];
}

export { useTableUrlState };
// 使用：订单列表页
function OrderListPage() {
  const [tableState, setTableState, resetState] = useTableUrlState([
    "status", "keyword", "dateFrom", "dateTo"
  ]);

  // URL: /orders?page=2&pageSize=20&sortBy=createdAt&sortOrder=desc&status=pending&keyword=手机
  // 刷新后筛选条件自动恢复！
  // 分享给同事，同事看到的是一样的筛选结果！

  const { data, isLoading } = useOrderList({
    page: tableState.page,
    pageSize: tableState.pageSize,
    sortBy: tableState.sortBy,
    sortOrder: tableState.sortOrder,
    filters: tableState.filters,
  });

  return (
    <div className="space-y-4">
      {/* 筛选栏 */}
      <div className="flex gap-4">
        <input
          value={tableState.filters.keyword || ""}
          onChange={(e) =>
            setTableState({ filters: { ...tableState.filters, keyword: e.target.value } })
          }
          placeholder="搜索订单号或客户..."
        />
        <select
          value={tableState.filters.status || ""}
          onChange={(e) =>
            setTableState({ filters: { ...tableState.filters, status: e.target.value } })
          }
        >
          <option value="">全部状态</option>
          <option value="pending">待处理</option>
          <option value="shipped">已发货</option>
        </select>
        <button onClick={resetState}>重置</button>
      </div>

      {/* 表格 */}
      <DataTable
        data={data?.list ?? []}
        total={data?.total ?? 0}
        loading={isLoading}
        pagination={{ pageIndex: tableState.page - 1, pageSize: tableState.pageSize }}
        onPaginationChange={(p) =>
          setTableState({ page: p.pageIndex + 1, pageSize: p.pageSize })
        }
      />
    </div>
  );
}

3. 文件上传模式

3.1 基础上传组件

// src/components/business/file-upload.tsx
import { useState, useRef, useCallback, type ChangeEvent } from "react";
import { Button } from "@/components/ui/button";
import { request } from "@/api/request";
import { formatFileSize } from "@/utils/format";

interface UploadFile {
  id: string;
  name: string;
  size: number;
  progress: number;
  status: "pending" | "uploading" | "success" | "error";
  url?: string;
  error?: string;
}

interface FileUploadProps {
  accept?: string;           // ".jpg,.png,.pdf"
  maxSize?: number;          // 最大文件大小（字节）
  maxCount?: number;         // 最大文件数量
  multiple?: boolean;
  uploadUrl?: string;
  onSuccess?: (files: UploadFile[]) => void;
}

function FileUpload({
  accept,
  maxSize = 10 * 1024 * 1024, // 默认 10MB
  maxCount = 5,
  multiple = false,
  uploadUrl = "/upload",
  onSuccess,
}: FileUploadProps) {
  const [files, setFiles] = useState<UploadFile[]>([]);
  const inputRef = useRef<HTMLInputElement>(null);

  const handleFileSelect = useCallback(
    (e: ChangeEvent<HTMLInputElement>) => {
      const selectedFiles = Array.from(e.target.files || []);

      // 校验
      const validFiles = selectedFiles.filter((file) => {
        if (file.size > maxSize) {
          console.warn(`${file.name} 超过最大文件大小 ${formatFileSize(maxSize)}`);
          return false;
        }
        return true;
      });

      if (files.length + validFiles.length > maxCount) {
        console.warn(`最多上传 ${maxCount} 个文件`);
        return;
      }

      // 创建上传任务
      const newFiles: UploadFile[] = validFiles.map((file) => ({
        id: Math.random().toString(36).slice(2),
        name: file.name,
        size: file.size,
        progress: 0,
        status: "pending",
      }));

      setFiles((prev) => [...prev, ...newFiles]);

      // 逐个上传
      validFiles.forEach((file, index) => {
        uploadFile(file, newFiles[index].id);
      });

      // 清空 input
      if (inputRef.current) inputRef.current.value = "";
    },
    [files, maxSize, maxCount]
  );

  async function uploadFile(file: File, fileId: string) {
    const formData = new FormData();
    formData.append("file", file);

    try {
      setFiles((prev) =>
        prev.map((f) =>
          f.id === fileId ? { ...f, status: "uploading" as const } : f
        )
      );

      const { data } = await request.post(uploadUrl, formData, {
        headers: { "Content-Type": "multipart/form-data" },
        onUploadProgress: (progressEvent) => {
          const progress = Math.round(
            ((progressEvent.loaded || 0) / (progressEvent.total || 1)) * 100
          );
          setFiles((prev) =>
            prev.map((f) => (f.id === fileId ? { ...f, progress } : f))
          );
        },
      });

      setFiles((prev) =>
        prev.map((f) =>
          f.id === fileId
            ? { ...f, status: "success" as const, progress: 100, url: data.data.url }
            : f
        )
      );
    } catch (err) {
      setFiles((prev) =>
        prev.map((f) =>
          f.id === fileId
            ? { ...f, status: "error" as const, error: "上传失败" }
            : f
        )
      );
    }
  }

  const removeFile = useCallback((fileId: string) => {
    setFiles((prev) => prev.filter((f) => f.id !== fileId));
  }, []);

  return (
    <div className="space-y-3">
      <div
        className="flex cursor-pointer items-center justify-center rounded-lg border-2 border-dashed p-8 transition-colors hover:border-primary hover:bg-primary/5"
        onClick={() => inputRef.current?.click()}
      >
        <div className="text-center">
          <p className="text-sm text-muted-foreground">
            点击或拖拽文件到此处上传
          </p>
          <p className="mt-1 text-xs text-muted-foreground">
            单文件最大 {formatFileSize(maxSize)}，最多 {maxCount} 个
          </p>
        </div>
      </div>

      <input
        ref={inputRef}
        type="file"
        accept={accept}
        multiple={multiple}
        className="hidden"
        onChange={handleFileSelect}
      />

      {/* 文件列表 */}
      {files.map((file) => (
        <div key={file.id} className="flex items-center gap-3 rounded border p-3">
          <div className="flex-1">
            <p className="text-sm">{file.name}</p>
            <p className="text-xs text-muted-foreground">{formatFileSize(file.size)}</p>
            {file.status === "uploading" && (
              <div className="mt-1 h-1.5 w-full overflow-hidden rounded-full bg-muted">
                <div
                  className="h-full rounded-full bg-primary transition-all"
                  style={{ width: `${file.progress}%` }}
                />
              </div>
            )}
          </div>
          <span className="text-xs">
            {file.status === "uploading" && `${file.progress}%`}
            {file.status === "success" && "✓"}
            {file.status === "error" && <span className="text-destructive">✗</span>}
          </span>
          <button
            onClick={() => removeFile(file.id)}
            className="text-sm text-muted-foreground hover:text-destructive"
          >
            删除
          </button>
        </div>
      ))}
    </div>
  );
}

export { FileUpload, type UploadFile };

// src/i18n/index.tsx — i18n 核心
import { createContext, useContext, useCallback, type ReactNode } from "react";
import { zhCN } from "./locales/zh-CN";
import { enUS } from "./locales/en-US";
import { useAppSettings } from "@/store/app-settings";

type Locale = typeof zhCN;
type LocaleKey = "zh-CN" | "en-US";

const locales: Record<LocaleKey, Locale> = {
  "zh-CN": zhCN,
  "en-US": enUS,
};

// 用点号路径访问嵌套对象
type NestedKeyOf<T, Prefix extends string = ""> = T extends Record<string, unknown>
  ? {
      [K in keyof T & string]: T[K] extends Record<string, unknown>
        ? NestedKeyOf<T[K], `${Prefix}${K}.`>
        : `${Prefix}${K}`;
    }[keyof T & string]
  : never;

type TranslationKey = NestedKeyOf<Locale>;

function getNestedValue(obj: Record<string, unknown>, path: string): string {
  return path.split(".").reduce<unknown>((acc, key) => {
    if (acc && typeof acc === "object") {
      return (acc as Record<string, unknown>)[key];
    }
    return path;
  }, obj) as string;
}

function useTranslation() {
  const { language } = useAppSettings();
  const locale = locales[language];

  const t = useCallback(
    (key: TranslationKey): string => {
      return getNestedValue(locale as unknown as Record<string, unknown>, key) || key;
    },
    [locale]
  );

  return { t, language };
}

export { useTranslation, type TranslationKey };
// 使用：
function OrderPage() {

  return (
    <div>
      <h1>{t("order.title")}</h1>
      <Button>{t("order.create")}</Button>
    </div>
  );
}

5. Loading/Skeleton 策略

// src/components/ui/skeleton.tsx — 骨架屏组件
import { cn } from "@/lib/utils";

function Skeleton({ className, ...props }: React.HTMLAttributes<HTMLDivElement>) {
  return (
    <div
      className={cn("animate-pulse rounded-md bg-muted", className)}
      {...props}
    />
  );
}

// 表格骨架屏
function TableSkeleton({ rows = 5, cols = 4 }: { rows?: number; cols?: number }) {
  return (
    <div className="space-y-3">
      {/* 表头骨架 */}
      <div className="flex gap-4">
        {Array.from({ length: cols }).map((_, i) => (
          <Skeleton key={i} className="h-4 flex-1" />
        ))}
      </div>
      {/* 行骨架 */}
      {Array.from({ length: rows }).map((_, i) => (
        <div key={i} className="flex gap-4">
          {Array.from({ length: cols }).map((_, j) => (
            <Skeleton key={j} className="h-8 flex-1" />
          ))}
        </div>
      ))}
    </div>
  );
}

// 统计卡片骨架
function StatCardSkeleton() {
  return (
    <div className="rounded-lg border p-4">
      <Skeleton className="h-4 w-24" />
      <Skeleton className="mt-2 h-8 w-32" />
      <Skeleton className="mt-2 h-3 w-20" />
    </div>
  );
}

export { Skeleton, TableSkeleton, StatCardSkeleton };

6. 模式选择指南

模式解决的问题复杂度优先级
URL 驱动状态筛选条件刷新丢失、无法分享低高
Loading/Skeleton白屏闪烁、加载体验差低高
文件上传上传进度、大文件处理中中
KeepAlive标签页切换状态丢失高中
国际化 i18n多语言支持中看业务需求
