业务数据处理实战

业务数据处理实战

LINQ 最大的价值不在于语法优雅，而在于它能把日常业务中"遍历 + 判断 + 累加"的样板代码压缩成一条声明式管道。本章用四个真实业务域——电商订单、ERP 数据、后台管理、ETL 清洗——展示 LINQ 如何解决实际问题。

一、电商订单分析

数据模型

public enum OrderStatus { Pending, Paid, Shipped, Delivered, Cancelled }

public class Order
{
    public int OrderId { get; set; }
    public int CustomerId { get; set; }
    public string CustomerName { get; set; } = "";
    public DateTime OrderDate { get; set; }
    public OrderStatus Status { get; set; }
    public List<OrderItem> Items { get; set; } = new();
    public decimal TotalAmount => Items.Sum(i => i.LineTotal);
}

public class OrderItem
{
    public int ProductId { get; set; }
    public string ProductName { get; set; } = "";
    public int Quantity { get; set; }
    public decimal UnitPrice { get; set; }
    public decimal LineTotal => Quantity * UnitPrice;
}

案例1：按时间范围筛选订单

需求：查询 2025 年 Q1 已付款及之后状态的订单。

var start = new DateTime(2025, 1, 1);
var end = new DateTime(2025, 4, 1);

var q1Orders = orders
    .Where(o => o.OrderDate >= start && o.OrderDate < end)
    .Where(o => o.Status >= OrderStatus.Paid)
    .OrderByDescending(o => o.OrderDate)
    .ToList();

Console.WriteLine($"Q1 有效订单：{q1Orders.Count} 笔，合计 {q1Orders.Sum(o => o.TotalAmount):C}");

要点：两个 Where 分开写比合并为一个更清晰——日期条件和状态条件是两个独立维度。

案例2：客户消费排名

需求：按消费总额排名前 10 客户。

var topCustomers = orders
    .Where(o => o.Status != OrderStatus.Cancelled)
    .GroupBy(o => new { o.CustomerId, o.CustomerName })
    .Select(g => new
    {
        g.Key.CustomerId,
        g.Key.CustomerName,
        TotalSpent = g.Sum(o => o.TotalAmount),
        OrderCount = g.Count(),
        AvgOrderValue = g.Average(o => o.TotalAmount)
    })
    .OrderByDescending(c => c.TotalSpent)
    .Take(10)
    .ToList();

foreach (var c in topCustomers)
{
    Console.WriteLine($"{c.CustomerName,-12} 总消费 {c.TotalSpent,10:C}  " +
                      $"{c.OrderCount} 单  客单价 {c.AvgOrderValue:C}");
}

要点：GroupBy 匿名类型做复合键 → Select 投影聚合 → OrderByDescending 排名 → Take 截取。

案例3：商品销售排行榜

需求：统计各商品销量和销售额，按销售额降序。

var productRanking = orders
    .Where(o => o.Status != OrderStatus.Cancelled)
    .SelectMany(o => o.Items, (order, item) => new { order.OrderDate, item })
    .GroupBy(x => new { x.item.ProductId, x.item.ProductName })
    .Select(g => new
    {
        g.Key.ProductId,
        g.Key.ProductName,
        TotalQuantity = g.Sum(x => x.item.Quantity),
        TotalRevenue = g.Sum(x => x.item.LineTotal),
        OrderCount = g.Select(x => x.OrderDate).Distinct().Count()
    })
    .OrderByDescending(p => p.TotalRevenue)
    .Take(20)
    .ToList();

要点：SelectMany 是关键——先把订单明细"展平"为行级数据，再按商品聚合。

案例4：月度销售趋势

需求：按年月聚合销售额，观察趋势。

var monthlyTrend = orders
    .Where(o => o.Status != OrderStatus.Cancelled)
    .GroupBy(o => new { o.OrderDate.Year, o.OrderDate.Month })
    .Select(g => new
    {
        g.Key.Year,
        g.Key.Month,
        Revenue = g.Sum(o => o.TotalAmount),
        OrderCount = g.Count(),
        AvgValue = Math.Round(g.Average(o => o.TotalAmount), 2)
    })
    .OrderBy(t => t.Year).ThenBy(t => t.Month)
    .ToList();

// 环比增长率
foreach (var t in monthlyTrend)
{
    var label = $"{t.Year}-{t.Month:D2}";
    Console.WriteLine($"{label}  营收 {t.Revenue,12:C}  {t.OrderCount} 单  均价 {t.AvgValue:C}");
}

案例5：客单价分析

需求：按客户分组，计算客单价，并与全局均值对比。

var globalAvg = orders
    .Where(o => o.Status != OrderStatus.Cancelled)
    .Average(o => o.TotalAmount);

var customerAnalysis = orders
    .Where(o => o.Status != OrderStatus.Cancelled)
    .GroupBy(o => o.CustomerName)
    .Select(g => new
    {
        Customer = g.Key,
        AvgValue = g.Average(o => o.TotalAmount),
        DiffFromGlobal = g.Average(o => o.TotalAmount) - globalAvg,
        Level = g.Average(o => o.TotalAmount) switch
        {
            > 500 => "高价值",
            > 200 => "中价值",
            _ => "低价值"
        }
    })
    .OrderByDescending(c => c.AvgValue)
    .ToList();

案例1：BOM 展开（递归展平）

需求：将多级 BOM 树展开为所有末级物料清单。

public static IEnumerable<BomItem> FlattenBom(BomItem root)
{
    yield return root;
    foreach (var child in root.Children.SelectMany(FlattenBom))
        yield return child;
}

// 使用
var allMaterials = FlattenBom(rootBom)
    .Where(b => b.Children.Count == 0)  // 只取末级
    .GroupBy(b => new { b.ChildId, b.ChildName })
    .Select(g => new
    {
        g.Key.ChildId,
        g.Key.ChildName,
        TotalQty = g.Sum(x => x.Quantity)
    })
    .ToList();

案例3：采购单比对

需求：比对新旧采购单差异——新增行、删除行、变更行。

var oldDict = oldLines.ToDictionary(l => l.Sku);
var newDict = newLines.ToDictionary(l => l.Sku);

var added = newLines
    .Where(l => !oldDict.ContainsKey(l.Sku))
    .ToList();

var removed = oldLines
    .Where(l => !newDict.ContainsKey(l.Sku))
    .ToList();

var changed = newLines
    .Where(l => oldDict.TryGetValue(l.Sku, out var old)
             && (old.Quantity != l.Quantity || old.UnitPrice != l.UnitPrice))
    .Select(l => new
    {
        l.Sku,
        OldQty = oldDict[l.Sku].Quantity,
        NewQty = l.Quantity,
        OldPrice = oldDict[l.Sku].UnitPrice,
        NewPrice = l.UnitPrice
    })
    .ToList();

案例4：MRP 简易计算

需求：根据需求量和现有库存计算采购建议。

var purchaseSuggestions = demandItems
    .GroupBy(d => d.Sku)
    .Select(g => new
    {
        Sku = g.Key,
        DemandQty = g.Sum(d => d.Quantity),
        StockQty = inventoryBySku.TryGetValue(g.Key, out var inv) ? inv : 0
    })

    .ToList();

案例5：应收账龄分析

需求：按账龄区间统计未收金额。

var today = DateTime.Today;

var agingAnalysis = receivables
    .Where(r => !r.IsPaid)
    .Select(r => new
    {
        r.CustomerId,
        r.CustomerName,
        r.Amount,
        OverdueDays = (today - r.DueDate).Days,
        AgingBucket = (today - r.DueDate).Days switch
        {
            <= 0 => "未到期",
            <= 30 => "0-30天",
            <= 60 => "31-60天",
            <= 90 => "61-90天",
            _ => "90天以上"
        }
    })
    .GroupBy(r => r.AgingBucket)
    .Select(g => new
    {
        AgingBucket = g.Key,
        TotalAmount = g.Sum(r => r.Amount),
        Count = g.Count(),
        Customers = g.Select(r => r.CustomerName).Distinct().Count()
    })
    .OrderBy(g => g.AgingBucket)
    .ToList();

案例1：多条件动态查询

需求：根据前端传入的多个可选条件，动态构建查询。

public static IQueryable<Order> BuildQuery(
    IQueryable<Order> source,
    DateTime? startDate,
    DateTime? endDate,
    OrderStatus? status,
    string? keyword)
{
    var query = source;

    if (startDate.HasValue)
        query = query.Where(o => o.OrderDate >= startDate.Value);

    if (endDate.HasValue)
        query = query.Where(o => o.OrderDate < endDate.Value);

    if (status.HasValue)
        query = query.Where(o => o.Status == status.Value);

    if (!string.IsNullOrWhiteSpace(keyword))
        query = query.Where(o => o.CustomerName.Contains(keyword));

    return query;
}

更灵活的方式——使用 Expression 动态组合：

public static Expression<Func<T, bool>> And<T>(
    this Expression<Func<T, bool>> left,
    Expression<Func<T, bool>> right)
{
    var param = Expression.Parameter(typeof(T));
    var body = Expression.AndAlso(
        Expression.Invoke(left, param),
        Expression.Invoke(right, param));
    return Expression.Lambda<Func<T, bool>>(body, param);
}

// 使用
Expression<Func<Order, bool>> predicate = o => true;

if (startDate.HasValue)
    predicate = predicate.And(o => o.OrderDate >= startDate.Value);
if (status.HasValue)
    predicate = predicate.And(o => o.Status == status.Value);

var result = orders.AsQueryable().Where(predicate).ToList();

案例2：权限过滤

需求：不同角色只能看到对应范围的数据。

public static IQueryable<Order> ApplyPermission(
    IQueryable<Order> query,
    User currentUser,
    List<Department> departments)
{
    return currentUser.Role switch
    {
        "Admin" => query,  // 管理员看全部
        "Manager" => query.Where(o =>
            departments.Any(d => d.DepartmentId == currentUser.DepartmentId
                             && d.CustomerIds.Contains(o.CustomerId))),
        _ => query.Where(o => o.CustomerId == currentUser.UserId)
    };
}

案例1：数据去重

需求：按 Id 去重，保留最新记录。

// .NET 6+ 可用 DistinctBy
var deduped = rawRecords
    .OrderByDescending(r => r.CreatedAt)
    .DistinctBy(r => r.Id)
    .ToList();

// .NET 6 以下
var dedupedLegacy = rawRecords
    .GroupBy(r => r.Id)
    .Select(g => g.OrderByDescending(r => r.CreatedAt).First())
    .ToList();

案例2：数据校验

需求：找出不符合规则的记录。

var invalidRecords = rawRecords
    .Select((r, idx) => new { Record = r, Line = idx + 1 })
    .Where(x => string.IsNullOrWhiteSpace(x.Record.Id)
             || string.IsNullOrWhiteSpace(x.Record.Name)
             || !x.Record.Email.Contains('@'))
    .Select(x => new
    {
        x.Line,
        x.Record.Id,
        Errors = new[]
        {
            string.IsNullOrWhiteSpace(x.Record.Id) ? "Id为空" : "",
            string.IsNullOrWhiteSpace(x.Record.Name) ? "名称为空" : "",
            !x.Record.Email.Contains('@') ? "邮箱格式错误" : ""
        }.Where(e => e != "").ToList()
    })
    .ToList();

小结

业务域核心操作典型模式
电商订单SelectMany + GroupBy + 聚合展平→分组→汇总
ERP 数据递归展平 + 字典查找 + 集合比对展开→聚合→差异
后台管理动态Where + 投影Select + 分页过滤→投影→分页
ETL 清洗Distinct + 校验Where + 格式Select校验→转换→合并

关键心得：

  1. SelectMany 是处理嵌套数据的利器——订单明细、BOM 子件，都需要先展平再聚合
  2. GroupBy + Select 是分析型查询的标准范式——分组后投影聚合值，几乎覆盖所有统计场景
  3. 动态查询用条件式 Where 叠加——比拼装 SQL 字符串安全，比 Expression 树简单
  4. 数据清洗永远是 Where → Select 的管道——先过滤异常，再投影为目标格式
