复杂业务SQL建模

第五章：复杂业务SQL建模

真正区分 SQL 高手和新手的，不是语法，而是"如何用 SQL 表达复杂业务逻辑"。

一、行转列 / 列转行

1.1 PIVOT —— 行转列

-- 场景：按月统计每个产品的销售额
-- 原始数据：ProductName, Month, Amount（多行）
-- 目标：ProductName, 1月, 2月, 3月...（一行）

SELECT ProductName, [1] AS Jan, [2] AS Feb, [3] AS Mar, 
       [4] AS Apr, [5] AS May, [6] AS Jun
FROM (
    SELECT ProductName, MONTH(OrderDate) AS M, Amount
    FROM OrderDetails
    WHERE YEAR(OrderDate) = 2024
) src
PIVOT (
    SUM(Amount) FOR M IN ([1],[2],[3],[4],[5],[6])
) pvt;

1.2 动态PIVOT

-- 月份不确定时，用动态SQL
DECLARE @cols NVARCHAR(MAX), @sql NVARCHAR(MAX);

SELECT @cols = STRING_AGG(QUOTENAME(M), ',') WITHIN GROUP (ORDER BY M)
FROM (SELECT DISTINCT MONTH(OrderDate) AS M FROM OrderDetails WHERE YEAR(OrderDate) = 2024) t;

SET @sql = N'
SELECT ProductName, ' + @cols + '
FROM (
    SELECT ProductName, MONTH(OrderDate) AS M, Amount
    FROM OrderDetails WHERE YEAR(OrderDate) = 2024
) src
PIVOT (SUM(Amount) FOR M IN (' + @cols + ')) pvt';

EXEC sp_executesql @sql;

1.3 UNPIVOT —— 列转行

-- 把宽表变成长表
SELECT ProductName, MonthName, Amount
FROM ProductMonthlySales
UNPIVOT (
    Amount FOR MonthName IN (Jan, Feb, Mar, Apr, May, Jun)
) unpvt;

二、连续问题模式

2.1 连续登录天数（经典）

;WITH LoginDays AS (
    SELECT DISTINCT UserId, CAST(LoginTime AS DATE) AS LoginDate
    FROM UserLogins
),
Grouped AS (
    SELECT UserId, LoginDate,
        DATEADD(DAY, -ROW_NUMBER() OVER(PARTITION BY UserId ORDER BY LoginDate), LoginDate) AS Grp
    FROM LoginDays
)
SELECT UserId, MIN(LoginDate) StartDate, MAX(LoginDate) EndDate, COUNT(*) ConsecutiveDays
FROM Grouped
GROUP BY UserId, Grp
HAVING COUNT(*) >= 7  -- 连续登录7天以上
ORDER BY ConsecutiveDays DESC;

2.2 连续增长的销售额

;WITH DailySales AS (
    SELECT SaleDate, SUM(Amount) AS DayAmount FROM Sales GROUP BY SaleDate
),
Flagged AS (
    SELECT *, 
        CASE WHEN DayAmount > LAG(DayAmount) OVER(ORDER BY SaleDate) THEN 0 ELSE 1 END AS BreakFlag
    FROM DailySales
),
Grouped AS (
    SELECT *, SUM(BreakFlag) OVER(ORDER BY SaleDate) AS Grp
    FROM Flagged
)
SELECT Grp, MIN(SaleDate) StartDate, MAX(SaleDate) EndDate, 
       COUNT(*) GrowthDays, MIN(DayAmount) MinAmount, MAX(DayAmount) MaxAmount
FROM Grouped
GROUP BY Grp
HAVING COUNT(*) >= 3
ORDER BY GrowthDays DESC;

三、漏斗分析

-- 电商漏斗：浏览 → 加购 → 下单 → 支付
;WITH Funnel AS (
    SELECT 
        CAST(EventTime AS DATE) AS EventDate,
        COUNT(DISTINCT CASE WHEN EventType = 'view' THEN UserId END) AS ViewUsers,
        COUNT(DISTINCT CASE WHEN EventType = 'add_cart' THEN UserId END) AS CartUsers,
        COUNT(DISTINCT CASE WHEN EventType = 'order' THEN UserId END) AS OrderUsers,
        COUNT(DISTINCT CASE WHEN EventType = 'payment' THEN UserId END) AS PayUsers
    FROM UserEvents
    WHERE EventTime >= '2024-01-01' AND EventTime < '2024-02-01'
    GROUP BY CAST(EventTime AS DATE)
)
SELECT 
    EventDate,
    ViewUsers,
    CartUsers,
    OrderUsers,
    PayUsers,
    -- 转化率
    CAST(CartUsers * 100.0 / NULLIF(ViewUsers, 0) AS DECIMAL(5,2)) AS ViewToCart,
    CAST(OrderUsers * 100.0 / NULLIF(CartUsers, 0) AS DECIMAL(5,2)) AS CartToOrder,
    CAST(PayUsers * 100.0 / NULLIF(OrderUsers, 0) AS DECIMAL(5,2)) AS OrderToPay,
    CAST(PayUsers * 100.0 / NULLIF(ViewUsers, 0) AS DECIMAL(5,2)) AS OverallConversion
FROM Funnel
ORDER BY EventDate;

四、排名与Top N

4.1 每组Top N

-- 每个类目销量前5的商品
SELECT * FROM (
    SELECT CategoryName, ProductName, SalesCount,
        ROW_NUMBER() OVER(PARTITION BY CategoryId ORDER BY SalesCount DESC) AS Rn
    FROM Products p
    INNER JOIN Categories c ON p.CategoryId = c.CategoryId
) t WHERE Rn <= 5;

4.2 动态排名（随查询时间变化）

-- 实时销售排行榜
SELECT 
    p.ProductName,
    SUM(oi.Quantity) AS TotalSold,
    SUM(oi.Quantity * oi.UnitPrice) AS TotalRevenue,
    RANK() OVER(ORDER BY SUM(oi.Quantity * oi.UnitPrice) DESC) AS RevenueRank,
    RANK() OVER(ORDER BY SUM(oi.Quantity) DESC) AS QuantityRank
FROM OrderItems oi
INNER JOIN Products p ON oi.ProductId = p.ProductId
INNER JOIN Orders o ON oi.OrderId = o.OrderId
WHERE o.OrderDate >= DATEADD(DAY, -30, GETDATE())
GROUP BY p.ProductId, p.ProductName
ORDER BY TotalRevenue DESC;

五、库存与账务

5.1 库存流水与余额

-- 计算每次出入库后的实时余额
SELECT 
    ProductId, TransDate, TransType, Quantity,
    SUM(CASE WHEN TransType = 'IN' THEN Quantity ELSE -Quantity END) 
        OVER(PARTITION BY ProductId ORDER BY TransDate, TransId) AS Balance
FROM StockTransactions
ORDER BY ProductId, TransDate;

5.2 先进先出（FIFO）成本计算

-- FIFO: 按入库顺序消耗库存
;WITH InStock AS (
    SELECT *, ROW_NUMBER() OVER(PARTITION BY ProductId ORDER BY PurchaseDate) AS BatchNo,
           SUM(Quantity) OVER(PARTITION BY ProductId ORDER BY PurchaseDate) AS CumulativeQty
    FROM PurchaseRecords
),
Consumed AS (
    SELECT ProductId, SUM(Quantity) AS TotalConsumed
    FROM SalesRecords
    GROUP BY ProductId
)
SELECT i.ProductId, i.BatchNo, i.UnitCost, i.Quantity,
    CASE 
        WHEN i.CumulativeQty <= c.TotalConsumed THEN i.Quantity  -- 全部消耗
        WHEN i.CumulativeQty - i.Quantity >= c.TotalConsumed THEN 0  -- 未消耗
        ELSE c.TotalConsumed - (i.CumulativeQty - i.Quantity)  -- 部分消耗
    END AS ConsumedQty
FROM InStock i
INNER JOIN Consumed c ON i.ProductId = c.ProductId;

六、日期补零与间隔分析

-- 没有数据的日期也要显示（补零）
;WITH DateRange AS (
    SELECT CAST('2024-01-01' AS DATE) AS Dt
    UNION ALL
    SELECT DATEADD(DAY, 1, Dt) FROM DateRange WHERE Dt < '2024-01-31'
)
SELECT 
    d.Dt AS OrderDate,
    ISNULL(COUNT(o.OrderId), 0) AS OrderCount,
    ISNULL(SUM(o.TotalAmount), 0) AS DayAmount
FROM DateRange d
LEFT JOIN Orders o ON CAST(o.OrderDate AS DATE) = d.Dt
GROUP BY d.Dt
ORDER BY d.Dt
OPTION (MAXRECURSION 31);

七、JSON 处理（SQL Server 2016+）

-- 解析JSON
DECLARE @json NVARCHAR(MAX) = N'[
    {"id": 1, "name": "张三", "tags": ["admin", "vip"]},
    {"id": 2, "name": "李四", "tags": ["user"]}
]';

SELECT * FROM OPENJSON(@json)
WITH (
    Id INT '$.id',
    Name NVARCHAR(50) '$.name',
    Tags NVARCHAR(MAX) '$.tags' AS JSON
);

-- 生成JSON
SELECT OrderId, OrderDate, TotalAmount
FROM Orders
WHERE CustomerId = 1
FOR JSON PATH, ROOT('orders');
