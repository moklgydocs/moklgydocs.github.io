ASP.NET Core 配置系统（Configuration）：源码级深入

一、核心认知

配置系统的本质就是一个"扁平化的键值对字典"，不管你的配置来自 JSON 文件、环境变量、命令行还是数据库，最终都被拍平成 string key → string value 的形式。

appsettings.json:                    内存中的表示：
{                                    Key                         Value
  "Logging": {                       ─────────────────────────── ──────
    "LogLevel": {                    "Logging:LogLevel:Default"  "Information"
      "Default": "Information"       "Logging:LogLevel:Microsoft" "Warning"
    }                                "ConnectionStrings:Default"  "Server=..."
  },                                 "Jwt:Secret"                "my-secret"
  "ConnectionStrings": {             "Jwt:Issuer"                "https://sso.example.com"
    "Default": "Server=..."          "Jwt:ExpiresInMinutes"      "60"
  },
  "Jwt": {
    "Secret": "my-secret",
    "Issuer": "https://sso.example.com",
    "ExpiresInMinutes": 60
  }
}

分隔符是冒号 :。不管配置源是什么格式，最终都用冒号分隔层级。

四、ConfigurationRoot —— 配置的核心实现

// 📁 源码位置: src/libraries/Microsoft.Extensions.Configuration/
//              src/ConfigurationRoot.cs

public class ConfigurationRoot : IConfigurationRoot, IDisposable
{
    private readonly IList<IConfigurationProvider> _providers;
    private readonly IList<IDisposable> _changeTokenRegistrations;
    private ConfigurationReloadToken _changeToken = new();

    public ConfigurationRoot(IList<IConfigurationProvider> providers)
    {
        _providers = providers;

        // 🔥 加载所有 Provider 的数据
        foreach (var provider in providers)
        {
            provider.Load();
            
            // 🔥 监听每个 Provider 的变更
            _changeTokenRegistrations.Add(
                ChangeToken.OnChange(
                    () => provider.GetReloadToken(),
                    () => RaiseChanged()   // 任何一个Provider变了，就通知
                )
            );
        }
    }

    // 🔥 读取配置值 —— 从后往前遍历 Providers
    public string? this[string key]
    {
        get
        {
            // 从最后一个Provider开始找（后注册的优先级高）
            for (var i = _providers.Count - 1; i >= 0; i--)
            {
                if (_providers[i].TryGet(key, out var value))
                {
                    return value;  // 找到了就返回
                }
            }
            return null;  // 所有Provider都没有
        }
        set
        {
            // 设置值时，写入所有Provider
            foreach (var provider in _providers)
            {
                provider.Set(key, value);
            }
        }
    }

    // 获取子节点
    public IConfigurationSection GetSection(string key)
        => new ConfigurationSection(this, key);

    // 变更通知Token
    public IChangeToken GetReloadToken() => _changeToken;

    // 手动重新加载
    public void Reload()
    {
        foreach (var provider in _providers)
        {
            provider.Load();
        }
        RaiseChanged();
    }

    private void RaiseChanged()
    {
        // 🔥 触发变更通知
        var previousToken = Interlocked.Exchange(
            ref _changeToken, new ConfigurationReloadToken());
        previousToken.OnReload();
    }
}

关键发现：优先级的实现极其简单 —— 从后往前遍历！

Providers 列表（按添加顺序）：
[0] appsettings.json           → {"Jwt:Secret": "from-json"}
[1] appsettings.Development.json → {"Jwt:Secret": "from-dev-json"}
[2] User Secrets               → (无此Key)
[3] 环境变量                    → (无此Key)
[4] 命令行                     → {"Jwt:Secret": "from-cmdline"}

config["Jwt:Secret"] 的查找过程：
  [4] 命令行: TryGet → "from-cmdline" ✅ 找到！直接返回

如果命令行没设置：
  [4] 命令行: TryGet → false
  [3] 环境变量: TryGet → false
  [2] User Secrets: TryGet → false
  [1] appsettings.Development.json: TryGet → "from-dev-json" ✅

优先级 = 注册顺序的倒序 = 后注册的覆盖先注册的

JsonConfigurationFileParser —— JSON 拍平的核心

// 📁 源码位置: src/libraries/Microsoft.Extensions.Configuration.Json/
//              src/JsonConfigurationFileParser.cs

internal sealed class JsonConfigurationFileParser
{
    private readonly Dictionary<string, string?> _data = new(StringComparer.OrdinalIgnoreCase);
    private readonly Stack<string> _paths = new();

    public static IDictionary<string, string?> Parse(Stream input)
    {
        var parser = new JsonConfigurationFileParser();
        // 使用 System.Text.Json 解析
        var jsonDocumentOptions = new JsonDocumentOptions
        {
            CommentHandling = JsonCommentHandling.Skip,
            AllowTrailingCommas = true,
        };
        using var doc = JsonDocument.Parse(input, jsonDocumentOptions);
        
        if (doc.RootElement.ValueKind != JsonValueKind.Object)
            throw new FormatException("Top-level JSON element must be an object.");
            
        parser.VisitObjectElement(doc.RootElement);
        return parser._data;
    }

    private void VisitObjectElement(JsonElement element)
    {
        foreach (var property in element.EnumerateObject())
        {
            EnterContext(property.Name);  // 压入路径栈
            VisitValue(property.Value);
            ExitContext();                // 弹出路径栈
        }
    }

    private void VisitValue(JsonElement value)
    {
        switch (value.ValueKind)
        {
            case JsonValueKind.Object:
                VisitObjectElement(value);  // 递归
                break;

            case JsonValueKind.Array:
                var index = 0;
                foreach (var arrayElement in value.EnumerateArray())
                {
                    EnterContext(index.ToString());  // 数组用索引作为Key
                    VisitValue(arrayElement);
                    ExitContext();
                    index++;
                }
                break;

            // 🔥 叶子节点 → 拍平成键值对
            case JsonValueKind.Number:
            case JsonValueKind.String:
            case JsonValueKind.True:
            case JsonValueKind.False:
            case JsonValueKind.Null:
                var key = _currentPath;  // 用冒号连接的路径
                if (_data.ContainsKey(key))
                    throw new FormatException($"Duplicated key '{key}'");
                _data[key] = value.ToString();
                break;
        }
    }

    // 路径管理
    private void EnterContext(string context) => _paths.Push(context);
    private void ExitContext() => _paths.Pop();
    
    // 🔥 将路径栈拼成 "Section:SubSection:Key" 的形式
    private string _currentPath => ConfigurationPath.Combine(_paths.Reverse());
}

六、环境变量 Provider —— 特殊的键映射

// 📁 源码位置: src/libraries/Microsoft.Extensions.Configuration.EnvironmentVariables/
//              src/EnvironmentVariablesConfigurationProvider.cs

public class EnvironmentVariablesConfigurationProvider : ConfigurationProvider
{
    private readonly string _prefix;

    public override void Load()
    {
        var data = new Dictionary<string, string?>(StringComparer.OrdinalIgnoreCase);
        
        var envVariables = Environment.GetEnvironmentVariables();
        
        foreach (DictionaryEntry envVariable in envVariables)
        {
            var key = (string)envVariable.Key;
            
            // 🔥 过滤前缀（如果设置了）
            if (!string.IsNullOrEmpty(_prefix))
            {
                if (!key.StartsWith(_prefix, StringComparison.OrdinalIgnoreCase))
                    continue;
                key = key.Substring(_prefix.Length);
            }
            
            // 🔥 关键：将 __ (双下划线) 替换为 : (冒号)
            // 因为环境变量名不允许包含冒号
            key = key.Replace("__", ConfigurationPath.KeyDelimiter);
            //                 ^^                     ^
            //            双下划线                    冒号
            
            data[key] = (string?)envVariable.Value;
        }

        Data = data;
    }
}

这就是为什么环境变量用双下划线：

# 在Docker/Kubernetes中设置环境变量：
export ConnectionStrings__Default="Server=prod-db;..."
export Jwt__Secret="super-secret-key"
export OpenIddict__Endpoints__Token="/connect/token"

# 等价于 appsettings.json 中的：
# "ConnectionStrings:Default": "Server=prod-db;..."
# "Jwt:Secret": "super-secret-key"  
# "OpenIddict:Endpoints:Token": "/connect/token"

# 而且环境变量优先级比JSON高，所以会覆盖JSON中的值
# 这就是12-Factor App推荐的配置方式 ✅

七、ConfigurationManager —— .NET 6+ 的新实现

在 .NET 6 之前，ConfigurationBuilder 和 ConfigurationRoot 是分开的。.NET 6 引入了 ConfigurationManager，它同时是 Builder 和 Root：

// 📁 源码位置: src/libraries/Microsoft.Extensions.Configuration/
//              src/ConfigurationManager.cs

public sealed class ConfigurationManager : 
    IConfigurationBuilder,   // 可以添加配置源
    IConfigurationRoot,      // 可以读取配置值
    IDisposable
{
    private readonly ConfigurationSources _sources;
    private readonly ConfigurationProviders _providers;

    // 🔥 关键设计：添加配置源时，立即加载！
    // 不需要等到 Build()
    
    private class ConfigurationSources : IList<IConfigurationSource>
    {
        private readonly List<IConfigurationSource> _sources = new();
        private readonly ConfigurationManager _config;

        public void Add(IConfigurationSource source)
        {
            _sources.Add(source);
            
            // 🔥 立即构建Provider并加载数据
            var provider = source.Build(_config);
            provider.Load();
            _config._providers.Add(provider);
            
            // 🔥 触发变更通知
            _config.RaiseChanged();
        }
    }

    // 读取配置 —— 和 ConfigurationRoot 一样，从后往前遍历
    public string? this[string key]
    {
        get
        {
            for (var i = _providers.Count - 1; i >= 0; i--)
            {
                if (_providers[i].TryGet(key, out var value))
                    return value;
            }
            return null;
        }
    }
}

这个设计的好处：

var builder = WebApplication.CreateBuilder(args);

// 在 CreateBuilder 之后，配置已经可以读取了！
// 因为 ConfigurationManager 添加源时就立即加载

// ✅ 可以立即使用，不用等 Build()

// 你还可以继续添加配置源

// 立即加载 custom.json 的内容
// 之后读取就能拿到 custom.json 中的值

// 这在旧版的 ConfigurationBuilder 中做不到
// 旧版必须先 Build() 才能读取

八、配置热更新（File Watch）—— reloadOnChange 的实现

// 你写的代码：
builder.Configuration.AddJsonFile("appsettings.json", 
    optional: true, 
    reloadOnChange: true);  // 👈 文件变更时自动重新加载

FileConfigurationProvider 的监听机制

// 📁 源码位置: src/libraries/Microsoft.Extensions.Configuration.FileExtensions/
//              src/FileConfigurationProvider.cs

public abstract class FileConfigurationProvider : ConfigurationProvider, IDisposable
{
    private readonly IDisposable? _changeTokenRegistration;

    public FileConfigurationProvider(FileConfigurationSource source)
    {
        Source = source;

        if (Source.ReloadOnChange && Source.FileProvider != null)
        {
            // 🔥 监听文件变更
            _changeTokenRegistration = ChangeToken.OnChange(
                // 监听什么？文件系统的变更通知
                () => Source.FileProvider.Watch(Source.Path!),
                // 变更时做什么？重新加载
                () =>
                {
                    // 延迟一小段时间（避免文件写入一半就触发）
                    Thread.Sleep(Source.ReloadDelay);  // 默认250ms
                    
                    // 🔥 重新加载文件内容
                    Load(reload: true);
                }
            );
        }
    }

    private void Load(bool reload)
    {
        // 读取文件
        var file = Source.FileProvider?.GetFileInfo(Source.Path!);
        if (file == null || !file.Exists)
        {
            if (Source.Optional || reload)
            {
                Data = new Dictionary<string, string?>(StringComparer.OrdinalIgnoreCase);
            }
            else
            {
                throw new FileNotFoundException($"The configuration file '{Source.Path}' was not found.");
            }
        }
        else
        {
            using var stream = file.CreateReadStream();
            Load(stream);  // 子类实现（如JsonConfigurationProvider）
        }

        // 🔥 通知所有监听者：配置已变更
        OnReload();  // 触发 ConfigurationRoot 的 RaiseChanged()
    }
}

ChangeToken 机制 —— 变更通知的核心

// 📁 源码位置: src/libraries/Microsoft.Extensions.Primitives/src/ChangeToken.cs

public static class ChangeToken
{
    // 🔥 核心方法：监听变更，变更时执行回调
    public static IDisposable OnChange(
        Func<IChangeToken> changeTokenProducer,
        Action changeTokenConsumer)
    {
        return new ChangeTokenRegistration<Action>(
            changeTokenProducer, 
            callback => callback(), 
            changeTokenConsumer);
    }
}

// ChangeToken 的内部实现
private sealed class ChangeTokenRegistration<TState> : IDisposable
{
    private readonly Func<IChangeToken> _changeTokenProducer;
    private readonly Action<TState> _changeTokenConsumer;
    private readonly TState _state;

    public ChangeTokenRegistration(
        Func<IChangeToken> changeTokenProducer,
        Action<TState> changeTokenConsumer,
        TState state)
    {
        _changeTokenProducer = changeTokenProducer;
        _changeTokenConsumer = changeTokenConsumer;
        _state = state;

        // 获取第一个 ChangeToken 并注册回调
        var token = changeTokenProducer();
        RegisterChangeTokenCallback(token);
    }

    private void RegisterChangeTokenCallback(IChangeToken token)
    {
        // 🔥 注册回调：变更发生时执行
        token.RegisterChangeCallback(_ =>
        {
            // 1. 执行消费者回调
            _changeTokenConsumer(_state);
            
            // 2. 🔥 获取新的 ChangeToken 并重新注册
            //    因为每个 ChangeToken 只能触发一次！
            var newToken = _changeTokenProducer();
            RegisterChangeTokenCallback(newToken);
            
        }, state: null);
    }
}

变更传播链路：

文件系统 (FileSystemWatcher)
    │  检测到 appsettings.json 被修改
    ▼
PhysicalFileProvider.Watch()
    │  返回 IChangeToken（一次性的）
    ▼
FileConfigurationProvider
    │  收到回调 → 重新 Load() → 解析JSON → 更新 Data 字典
    │  调用 OnReload()
    ▼
ConfigurationRoot.RaiseChanged()
    │  创建新的 ConfigurationReloadToken
    │  旧 Token 触发回调
    ▼
IOptionsMonitor<T>
    │  收到变更通知 → 重新绑定配置到强类型对象
    │  触发 OnChange 回调
    ▼
你的代码
    │  optionsMonitor.OnChange(newOptions => { /* 使用新配置 */ })
    ▼
配置热更新完成 ✅

九、Options 模式 —— 将配置绑定到强类型对象

为什么需要 Options 模式？

// ❌ 到处读字符串，容易出错

    public TokenService(IOptions<JwtOptions> options)

三种 Options 接口

// 📁 源码位置: src/libraries/Microsoft.Extensions.Options/src/

// ========== IOptions<T> ==========
// Singleton 生命周期，应用启动后值不变
public interface IOptions<out TOptions> where TOptions : class
{
    TOptions Value { get; }  // 只有一个属性
}

// ========== IOptionsSnapshot<T> ==========
// Scoped 生命周期，每个请求重新从配置中读取
// 同一请求内多次访问 Value 得到同一个实例
public interface IOptionsSnapshot<out TOptions> : IOptions<TOptions> 
    where TOptions : class
{
    TOptions Get(string? name);  // 支持命名选项
}

// ========== IOptionsMonitor<T> ==========
// Singleton 生命周期，但能感知配置变更
// 通过 OnChange 回调通知
public interface IOptionsMonitor<out TOptions> where TOptions : class
{
    TOptions CurrentValue { get; }  // 当前最新值
    TOptions Get(string? name);
    IDisposable? OnChange(Action<TOptions, string?> listener);  // 变更回调
}

三者对比：

┌──────────────────────┬───────────────┬──────────────────┬──────────────────────┐
│                      │ IOptions<T>   │ IOptionsSnapshot │ IOptionsMonitor<T>   │
├──────────────────────┼───────────────┼──────────────────┼──────────────────────┤
│ DI 生命周期           │ Singleton     │ Scoped           │ Singleton            │
│ 何时读取配置          │ 首次访问      │ 每个请求Scope     │ 实时（配置变更时）    │

IOptionsMonitor 的源码实现

// 📁 源码位置: src/libraries/Microsoft.Extensions.Options/src/OptionsMonitor.cs

public sealed class OptionsMonitor<TOptions> : IOptionsMonitor<TOptions>, IDisposable
    where TOptions : class
{
    private readonly IOptionsFactory<TOptions> _factory;
    private readonly IEnumerable<IOptionsChangeTokenSource<TOptions>> _sources;
    
    // 🔥 缓存当前值
    internal event Action<TOptions, string?>? _onChange;
    private readonly ConcurrentDictionary<string, Lazy<TOptions>> _cache = new();

    public OptionsMonitor(
        IOptionsFactory<TOptions> factory,
        IEnumerable<IOptionsChangeTokenSource<TOptions>> sources,
        IOptionsMonitorCache<TOptions> cache)
    {
        _factory = factory;
        _sources = sources;

        // 🔥 为每个变更源注册回调
        foreach (var source in sources)
        {
            var registration = ChangeToken.OnChange(
                () => source.GetChangeToken(),
                (name) =>
                {
                    // 配置变更时：
                    // 1. 清除缓存
                    _cache.TryRemove(name ?? Options.DefaultName, out _);
                    
                    // 2. 创建新的Options实例
                    var options = Get(name);
                    
                    // 3. 🔥 触发 OnChange 回调
                    _onChange?.Invoke(options, name);
                    
                }, source.Name);
        }
    }

    // 获取当前值
    public TOptions CurrentValue => Get(Options.DefaultName);

    // 获取命名选项
    public TOptions Get(string? name)
    {
        name ??= Options.DefaultName;
        
        // 从缓存中获取或创建
        return _cache.GetOrAdd(name, 
            static (name, factory) => new Lazy<TOptions>(() => factory.Create(name)),
            _factory).Value;
    }

    // 注册变更回调
    public IDisposable? OnChange(Action<TOptions, string?> listener)
    {
        var disposable = new ChangeTrackerDisposable(this, listener);
        _onChange += listener;
        return disposable;
    }
}

public static IServiceCollection AddOptions(this IServiceCollection services)
{
    services.TryAdd(ServiceDescriptor.Singleton(
        typeof(IOptions<>), typeof(UnnamedOptionsManager<>)));
        
    services.TryAdd(ServiceDescriptor.Scoped(
        typeof(IOptionsSnapshot<>), typeof(OptionsManager<>)));
        
    services.TryAdd(ServiceDescriptor.Singleton(
        typeof(IOptionsMonitor<>), typeof(OptionsMonitor<>)));
        
    services.TryAdd(ServiceDescriptor.Transient(
        typeof(IOptionsFactory<>), typeof(OptionsFactory<>)));
        
    services.TryAdd(ServiceDescriptor.Singleton(
        typeof(IOptionsMonitorCache<>), typeof(OptionsCache<>)));
        
    return services;
}

OptionsFactory —— 创建 Options 实例的工厂

// 📁 源码位置: src/libraries/Microsoft.Extensions.Options/src/OptionsFactory.cs

public class OptionsFactory<TOptions> : IOptionsFactory<TOptions>
    where TOptions : class
{
    private readonly IConfigureOptions<TOptions>[] _setups;      // Configure回调
    private readonly IPostConfigureOptions<TOptions>[] _postConfigures;
    private readonly IValidateOptions<TOptions>[] _validations;

    public TOptions Create(string name)
    {
        // 🔥 第1步：创建空的 Options 实例
        var options = CreateInstance(name);  // new TOptions()
        
        // 🔥 第2步：依次执行所有 Configure 回调
        foreach (var setup in _setups)
        {
            if (setup is IConfigureNamedOptions<TOptions> namedSetup)
            {
                namedSetup.Configure(name, options);
            }
            else if (name == Options.DefaultName)
            {
                setup.Configure(options);
            }
        }

        // 🔥 第3步：执行 PostConfigure 回调
        foreach (var post in _postConfigures)
        {
            post.PostConfigure(name, options);
        }

        // 🔥 第4步：执行验证
        if (_validations.Length > 0)
        {
            var failures = new List<string>();
            foreach (var validate in _validations)
            {
                var result = validate.Validate(name, options);
                if (result is { Failed: true })
                {
                    failures.AddRange(result.Failures);
                }
            }
            if (failures.Count > 0)
            {
                throw new OptionsValidationException(name, typeof(TOptions), failures);
            }
        }

        return options;
    }
}

配置绑定（Bind）的源码

// 📁 源码位置: src/libraries/Microsoft.Extensions.Configuration.Binder/
//              src/ConfigurationBinder.cs

public static class ConfigurationBinder
{
    public static void Bind(
        this IConfiguration configuration, 
        object? instance)
    {
        BindInstance(configuration, instance, new BinderOptions());
    }

    private static void BindInstance(
        IConfiguration configuration, 
        object? instance,
        BinderOptions options)
    {
        if (instance == null) return;
        
        var modelType = instance.GetType();
        
        // 遍历所有可写属性
        foreach (var property in modelType.GetProperties(BindingFlags.Public | BindingFlags.Instance))
        {
            if (!property.CanWrite) continue;
            
            // 🔥 根据属性名找配置的子节点
            var childSection = configuration.GetSection(property.Name);
            // 属性名 "Secret" → 配置键 "Jwt:Secret"
            
            if (childSection.Value != null)
            {
                // 简单类型 → 直接转换
                var converted = ConvertValue(property.PropertyType, childSection.Value);
                property.SetValue(instance, converted);
            }
            else if (childSection.GetChildren().Any())
            {
                // 复杂类型 → 递归绑定
                var subInstance = property.GetValue(instance) 
                    ?? Activator.CreateInstance(property.PropertyType);
                BindInstance(childSection, subInstance, options);
                property.SetValue(instance, subInstance);
            }
        }
    }
}

ValidateOnStart 的实现

// 📁 源码位置: src/libraries/Microsoft.Extensions.Options/
//              src/OptionsBuilderExtensions.cs

public static OptionsBuilder<TOptions> ValidateOnStart<TOptions>(
    this OptionsBuilder<TOptions> optionsBuilder) 
    where TOptions : class
{
    // 注册一个 IHostedService，在应用启动时触发验证
    optionsBuilder.Services.AddTransient<IStartupValidator>(sp =>
    {
        return new StartupValidator(
            () => sp.GetRequiredService<IOptionsMonitor<TOptions>>().Get(optionsBuilder.Name));
    });
    
    // 注册 ValidateOptionsStartupHostedService
    optionsBuilder.Services.AddHostedService<ValidateOptionsStartupHostedService>();
    
    return optionsBuilder;
}

// 这个 HostedService 在启动时执行所有验证
internal sealed class ValidateOptionsStartupHostedService : IHostedService
{
    private readonly IEnumerable<IStartupValidator> _validators;
    
    public Task StartAsync(CancellationToken cancellationToken)
    {
        var exceptions = new List<Exception>();
        
        foreach (var validator in _validators)
        {
            try
            {
                validator.Validate();  // 🔥 触发 Options 创建和验证
            }
            catch (OptionsValidationException ex)
            {
                exceptions.Add(ex);
            }
        }
        
        if (exceptions.Count > 0)
        {
            // 🔥 启动失败！配置有问题
            throw new AggregateException(exceptions);
        }
        
        return Task.CompletedTask;
    }
}

// ========== 从数据库加载配置的自定义配置源 ==========

// 第1步：定义 Source
public class DatabaseConfigurationSource : IConfigurationSource
{
    public string ConnectionString { get; set; } = "";
    public TimeSpan RefreshInterval { get; set; } = TimeSpan.FromMinutes(5);

    public IConfigurationProvider Build(IConfigurationBuilder builder)
    {
        return new DatabaseConfigurationProvider(this);
    }
}

// 第2步：实现 Provider
public class DatabaseConfigurationProvider : ConfigurationProvider, IDisposable
{
    private readonly DatabaseConfigurationSource _source;
    private Timer? _refreshTimer;

    public DatabaseConfigurationProvider(DatabaseConfigurationSource source)
    {
        _source = source;
    }

    public override void Load()
    {
        // 从数据库加载配置
        using var connection = new SqlConnection(_source.ConnectionString);
        connection.Open();
        
        using var command = new SqlCommand(
            "SELECT [Key], [Value] FROM AppConfiguration", connection);
        using var reader = command.ExecuteReader();

        var data = new Dictionary<string, string?>(StringComparer.OrdinalIgnoreCase);
        while (reader.Read())
        {
            data[reader.GetString(0)] = reader.GetString(1);
        }

        Data = data;

        // 设置定时刷新
        _refreshTimer ??= new Timer(_ =>
        {
            Load();
            OnReload();  // 🔥 触发变更通知 → 热更新
        }, null, _source.RefreshInterval, _source.RefreshInterval);
    }

    public void Dispose() => _refreshTimer?.Dispose();
}

// 第3步：扩展方法
public static class DatabaseConfigurationExtensions
{
    public static IConfigurationBuilder AddDatabase(
        this IConfigurationBuilder builder,
        string connectionString,
        TimeSpan? refreshInterval = null)
    {
        builder.Add(new DatabaseConfigurationSource
        {
            ConnectionString = connectionString,
            RefreshInterval = refreshInterval ?? TimeSpan.FromMinutes(5)
        });
        return builder;
    }
}

// 使用：
builder.Configuration.AddDatabase(
    connectionString: builder.Configuration.GetConnectionString("ConfigDb")!,
    refreshInterval: TimeSpan.FromMinutes(1));

// 后台服务中使用 IOptionsMonitor
public class ClientSyncService : BackgroundService
{
    private readonly IOptionsMonitor<List<SsoClientOptions>> _clientOptions;
    private readonly IServiceScopeFactory _scopeFactory;

    public ClientSyncService(
        IOptionsMonitor<List<SsoClientOptions>> clientOptions,
        IServiceScopeFactory scopeFactory)
    {
        _clientOptions = clientOptions;
        _scopeFactory = scopeFactory;

        // 🔥 监听配置变更，自动同步到数据库
        _clientOptions.OnChange(async (clients, _) =>
        {
            using var scope = _scopeFactory.CreateScope();
            var manager = scope.ServiceProvider
                .GetRequiredService<IOpenIddictApplicationManager>();
            
            await SyncClientsToDatabase(manager, clients);
        });
    }

    protected override async Task ExecuteAsync(CancellationToken stoppingToken)
    {
        // 启动时同步一次
        using var scope = _scopeFactory.CreateScope();
        var manager = scope.ServiceProvider
            .GetRequiredService<IOpenIddictApplicationManager>();
        
        await SyncClientsToDatabase(manager, _clientOptions.CurrentValue);
    }
    
    private async Task SyncClientsToDatabase(
        IOpenIddictApplicationManager manager,

                {
                    ClientId = client.ClientId,
                    DisplayName = client.DisplayName,

                });
            }
            // else update...
        }
    }

十四、检验你的理解

  1. 配置系统底层的数据结构是什么？
    （答案：Dictionary<string, string?> —— 扁平化的键值对字典，Key 用冒号分隔层级）

  2. 为什么环境变量用双下划线 __ 而不是冒号 : ？
    （答案：因为大多数操作系统的环境变量名不支持冒号。EnvironmentVariablesProvider 在加载时会自动把 __ 替换成 :）

  3. IOptions<T> 和 IOptionsMonitor<T> 都是 Singleton，但行为不同，为什么？
    （答案：IOptions<T> 内部只在第一次访问时创建实例并缓存，之后不变。IOptionsMonitor<T> 通过 ChangeToken.OnChange 监听配置源变更，变更时清除缓存并重新创建实例。两者都是 Singleton 但缓存策略不同）

  4. reloadOnChange: true 的文件监听是谁实现的？变更通知是怎么传播到 IOptionsMonitor 的？
    （答案：PhysicalFileProvider.Watch() 用 FileSystemWatcher 监听文件。变更时 FileConfigurationProvider.Load() 重新加载数据，调用 OnReload() → ConfigurationRoot.RaiseChanged() → IChangeToken 触发回调 → IOptionsChangeTokenSource 传递给 OptionsMonitor → 清除缓存 + 触发 OnChange）

  5. 如果你在 appsettings.json、appsettings.Development.json 和环境变量中都设置了同一个 Key，最终生效的是哪个？为什么？
    （答案：环境变量。因为 ConfigurationRoot 从后往前遍历 Providers，环境变量的 Provider 在 JSON Provider 之后注册，优先级更高。后注册的覆盖先注册的）

  6. Options 验证（ValidateOnStart）是在什么时机执行的？为什么选择这个时机？
    （答案：通过 IHostedService 在应用启动时执行。Host.StartAsync() 会按顺序启动所有 IHostedService，ValidateOptionsStartupHostedService 在启动时触发所有验证。选择这个时机是因为能在应用开始接受请求之前就发现配置错误，快速失败）
