ASP.NET Core 授权系统（Authorization）：源码级深入

一、核心认知

授权系统回答的是："这个人能不能做这件事？"

认证告诉你"他是谁"（ClaimsPrincipal），授权告诉你"他能干什么"。

认证系统的输出：
  context.User = ClaimsPrincipal
  {
      Name: "zhangsan",
      Claims: [
          { Type: "sub",   Value: "user-guid-123" },
          { Type: "role",  Value: "Admin" },
          { Type: "role",  Value: "Editor" },
          { Type: "email", Value: "zs@example.com" },
          { Type: "scope", Value: "openid profile api" },
          { Type: "department", Value: "Engineering" }
      ]
  }

                    ↓ 传给授权系统 ↓

授权系统的判断：
  [Authorize(Roles = "Admin")]           → ✅ 有Admin角色，通过
  [Authorize(Policy = "RequireEditor")]  → ✅ 有Editor角色，通过
  [Authorize(Policy = "Over18")]         → ❓ 要看策略怎么定义
  [AllowAnonymous]                       → ✅ 任何人都可以
  没有[Authorize]                        → ✅ 默认允许

4.2 IAuthorizationHandler —— 处理器

public interface IAuthorizationHandler
{
    Task HandleAsync(AuthorizationHandlerContext context);
}

// AuthorizationHandlerContext 包含：
// - User: ClaimsPrincipal
// - Requirements: 需要满足的所有需求
// - Resource: 被访问的资源（可选）
// - PendingRequirements: 还未满足的需求
// - HasSucceeded: 是否所有需求都已满足

4.3 AuthorizationHandler<TRequirement> —— 泛型基类

// 📁 源码位置: src/Security/Authorization/Core/src/AuthorizationHandler.cs

// 这是你写自定义 Handler 时最常继承的基类
public abstract class AuthorizationHandler<TRequirement> : IAuthorizationHandler
    where TRequirement : IAuthorizationRequirement
{
    public virtual async Task HandleAsync(AuthorizationHandlerContext context)
    {
        // 🔥 遍历所有 Pending Requirements
        foreach (var req in context.Requirements.OfType<TRequirement>())
        {
            // 只处理匹配自己类型的 Requirement
            await HandleRequirementAsync(context, req);
        }
    }

    // 🔥 子类实现这个方法
    protected abstract Task HandleRequirementAsync(
        AuthorizationHandlerContext context,
        TRequirement requirement);
    
    // 在子类中：
    // context.Succeed(requirement)   → 标记此需求已满足
    // context.Fail()                 → 显式失败（阻止其他Handler成功）
    // 什么都不做                      → 此Handler不满足，但其他Handler可能满足
}

// 还有一个带资源的版本
public abstract class AuthorizationHandler<TRequirement, TResource> : IAuthorizationHandler
    where TRequirement : IAuthorizationRequirement
{
    protected abstract Task HandleRequirementAsync(
        AuthorizationHandlerContext context,
        TRequirement requirement,
        TResource resource);  // 👈 多了一个资源参数
}

4.4 AuthorizationPolicy —— 策略

// 📁 源码位置: src/Security/Authorization/Core/src/AuthorizationPolicy.cs

public class AuthorizationPolicy
{
    // 🔥 需要满足的所有需求（必须全部满足）
    public IReadOnlyList<IAuthorizationRequirement> Requirements { get; }
    
    // 🔥 需要使用的认证方案
    public IReadOnlyList<string> AuthenticationSchemes { get; }

    public AuthorizationPolicy(
        IEnumerable<IAuthorizationRequirement> requirements,
        IEnumerable<string> authenticationSchemes)
    {
        Requirements = new List<IAuthorizationRequirement>(requirements).AsReadOnly();
        AuthenticationSchemes = new List<string>(authenticationSchemes).AsReadOnly();
    }

    // 🔥 合并多个策略
    public static AuthorizationPolicy Combine(params AuthorizationPolicy[] policies)
    {
        var requirements = new List<IAuthorizationRequirement>();
        var schemes = new List<string>();
        
        foreach (var policy in policies)
        {
            requirements.AddRange(policy.Requirements);
            schemes.AddRange(policy.AuthenticationSchemes);
        }
        
        return new AuthorizationPolicy(requirements, schemes.Distinct());
    }
}

4.5 AuthorizationPolicyBuilder —— 策略构建器

// 📁 源码位置: src/Security/Authorization/Core/src/AuthorizationPolicyBuilder.cs

public class AuthorizationPolicyBuilder
{
    private readonly List<IAuthorizationRequirement> _requirements = new();
    private readonly List<string> _authenticationSchemes = new();

    // ============================
    // 内置的便捷方法
    // ============================
    
    // 🔥 要求已认证用户（最常用）
    public AuthorizationPolicyBuilder RequireAuthenticatedUser()
    {
        _requirements.Add(new DenyAnonymousAuthorizationRequirement());
        return this;
    }

    // 🔥 要求特定角色
    public AuthorizationPolicyBuilder RequireRole(params string[] roles)
    {
        _requirements.Add(new RolesAuthorizationRequirement(roles));
        return this;
    }

    // 🔥 要求特定声明
    public AuthorizationPolicyBuilder RequireClaim(string claimType, params string[] allowedValues)
    {
        if (allowedValues.Length == 0)
        {
            // 只要有这个声明就行，不管值
            _requirements.Add(new ClaimsAuthorizationRequirement(claimType, null));
        }
        else
        {
            // 声明的值必须在允许列表中
            _requirements.Add(new ClaimsAuthorizationRequirement(claimType, allowedValues));
        }
        return this;
    }

    // 🔥 要求用户名
    public AuthorizationPolicyBuilder RequireUserName(string userName)
    {
        _requirements.Add(new NameAuthorizationRequirement(userName));
        return this;
    }

    // 🔥 自定义断言
    public AuthorizationPolicyBuilder RequireAssertion(Func<AuthorizationHandlerContext, bool> handler)
    {
        _requirements.Add(new AssertionRequirement(handler));
        return this;
    }

    // 🔥 添加自定义需求
    public AuthorizationPolicyBuilder AddRequirements(params IAuthorizationRequirement[] requirements)
    {
        _requirements.AddRange(requirements);
        return this;
    }

    // 🔥 指定认证方案
    public AuthorizationPolicyBuilder AddAuthenticationSchemes(params string[] schemes)
    {
        _authenticationSchemes.AddRange(schemes);
        return this;
    }

    // 构建
    public AuthorizationPolicy Build()
    {
        return new AuthorizationPolicy(_requirements, _authenticationSchemes.Distinct());
    }
}

五、内置 Requirement 和 Handler 的源码

5.1 DenyAnonymousAuthorizationRequirement

// 📁 源码位置: src/Security/Authorization/Core/src/
//              DenyAnonymousAuthorizationRequirement.cs

// RequireAuthenticatedUser() 添加的就是这个
public class DenyAnonymousAuthorizationRequirement :
    AuthorizationHandler<DenyAnonymousAuthorizationRequirement>,
    IAuthorizationRequirement
    // ☝️ 既是 Requirement 又是 Handler！自己处理自己
{
    protected override Task HandleRequirementAsync(
        AuthorizationHandlerContext context,
        DenyAnonymousAuthorizationRequirement requirement)
    {
        var user = context.User;
        
        // 🔥 检查是否有已认证的身份
        var userIsAnonymous =
            user?.Identity == null ||
            !user.Identities.Any(i => i.IsAuthenticated);
        
        if (!userIsAnonymous)
        {
            context.Succeed(requirement);  // ✅ 已认证
        }
        // 如果匿名 → 不调用 Succeed，也不调用 Fail
        // 这意味着此需求未满足，但不会阻止其他Handler
        
        return Task.CompletedTask;
    }
}

5.2 RolesAuthorizationRequirement

// 📁 源码位置: src/Security/Authorization/Core/src/
//              RolesAuthorizationRequirement.cs

public class RolesAuthorizationRequirement :
    AuthorizationHandler<RolesAuthorizationRequirement>,
    IAuthorizationRequirement
{
    public IEnumerable<string> AllowedRoles { get; }

    public RolesAuthorizationRequirement(IEnumerable<string> allowedRoles)
    {
        AllowedRoles = allowedRoles;
    }

    protected override Task HandleRequirementAsync(
        AuthorizationHandlerContext context,
        RolesAuthorizationRequirement requirement)
    {
        if (context.User != null)
        {
            // 🔥 检查用户是否拥有任意一个要求的角色
            foreach (var role in requirement.AllowedRoles)
            {
                if (context.User.IsInRole(role))
                {
                    context.Succeed(requirement);  // ✅ 有角色
                    return Task.CompletedTask;
                }
            }
        }
        // 没有匹配的角色 → 不调用 Succeed
        
        return Task.CompletedTask;
    }
}

// System.Security.Claims.ClaimsPrincipal

public virtual bool IsInRole(string role)
{
    // 遍历所有身份
    for (int i = 0; i < _identities.Count; i++)
    {
        if (_identities[i] != null && _identities[i].HasClaim(
            _identities[i].RoleClaimType, role))
        //                 ^^^^^^^^^^^^^^^
        // 默认是 ClaimTypes.Role = "http://schemas.microsoft.com/ws/2008/06/identity/claims/role"
        // 但 JWT 通常用 "role" 而不是完整URI
        // OpenIddict 默认也用 "role"
        {
            return true;
        }
    }
    return false;
}

5.3 ClaimsAuthorizationRequirement

// 📁 源码位置: src/Security/Authorization/Core/src/
//              ClaimsAuthorizationRequirement.cs

public class ClaimsAuthorizationRequirement :
    AuthorizationHandler<ClaimsAuthorizationRequirement>,
    IAuthorizationRequirement
{
    public string ClaimType { get; }
    public IEnumerable<string>? AllowedValues { get; }

    protected override Task HandleRequirementAsync(
        AuthorizationHandlerContext context,
        ClaimsAuthorizationRequirement requirement)
    {
        if (context.User != null)
        {
            if (requirement.AllowedValues == null || !requirement.AllowedValues.Any())
            {
                // 🔥 只检查是否存在此类型的声明（不管值）
                if (context.User.Claims.Any(c =>
                    string.Equals(c.Type, requirement.ClaimType, StringComparison.OrdinalIgnoreCase)))
                {
                    context.Succeed(requirement);
                }
            }
            else
            {
                // 🔥 检查声明值是否在允许列表中
                if (context.User.Claims.Any(c =>
                    string.Equals(c.Type, requirement.ClaimType, StringComparison.OrdinalIgnoreCase) &&
                    requirement.AllowedValues.Contains(c.Value, StringComparer.Ordinal)))
                {
                    context.Succeed(requirement);
                }
            }
        }
        
        return Task.CompletedTask;
    }
}

6.2 从 [Authorize] 到 Policy 的转换过程

// 📁 源码位置: src/Security/Authorization/Policy/src/AuthorizationPolicy.cs

public class AuthorizationPolicy
{
    // 🔥 将 [Authorize] 特性数据转换为 Policy
    public static async Task<AuthorizationPolicy?> CombineAsync(
        IAuthorizationPolicyProvider policyProvider,
        IEnumerable<IAuthorizeData> authorizeData)
    {
        AuthorizationPolicyBuilder? policyBuilder = null;

        foreach (var authorizeDatum in authorizeData)
        {
            policyBuilder ??= new AuthorizationPolicyBuilder();

            // 🔥 处理 Policy 属性
            if (!string.IsNullOrWhiteSpace(authorizeDatum.Policy))
            {
                var policy = await policyProvider.GetPolicyAsync(authorizeDatum.Policy);
                if (policy == null)
                {
                    throw new InvalidOperationException(
                        $"The AuthorizationPolicy named: '{authorizeDatum.Policy}' was not found.");
                }
                policyBuilder.Combine(policy);
            }

            // 🔥 处理 Roles 属性
            if (!string.IsNullOrWhiteSpace(authorizeDatum.Roles))
            {
                var roles = authorizeDatum.Roles.Split(',')
                    .Select(r => r.Trim())
                    .Where(r => r.Length > 0);
                    
                policyBuilder.RequireRole(roles.ToArray());
                // [Authorize(Roles = "Admin,Editor")]
                // → RequireRole("Admin", "Editor")
                // → 用户有 Admin 或 Editor 任意一个就通过
            }

            // 🔥 处理 AuthenticationSchemes 属性
            if (!string.IsNullOrWhiteSpace(authorizeDatum.AuthenticationSchemes))
            {
                var schemes = authorizeDatum.AuthenticationSchemes.Split(',')
                    .Select(s => s.Trim())
                    .Where(s => s.Length > 0);
                    
                policyBuilder.AddAuthenticationSchemes(schemes.ToArray());
            }
        }

        if (policyBuilder == null)
        {
            return null;
        }

        // 🔥 如果没有显式设置 Policy，合并默认策略
        var builtPolicy = policyBuilder.Build();
        
        if (builtPolicy.Requirements.Count == 0 && 
            builtPolicy.AuthenticationSchemes.Count == 0)
        {
            // [Authorize] 没有参数 → 使用默认策略
            var defaultPolicy = await policyProvider.GetDefaultPolicyAsync();
            return defaultPolicy;
            // 默认策略 = RequireAuthenticatedUser()
        }

        return builtPolicy;
    }
}

各种 [Authorize] 写法对应的 Policy：

[Authorize]
→ 默认策略：RequireAuthenticatedUser()
→ Requirements: [ DenyAnonymousAuthorizationRequirement ]

[Authorize(Roles = "Admin")]
→ Requirements: [ RolesAuthorizationRequirement { AllowedRoles: ["Admin"] } ]
  注意：没有 DenyAnonymous！但角色检查本身隐含了要认证

[Authorize(Roles = "Admin,Editor")]
→ Requirements: [ RolesAuthorizationRequirement { AllowedRoles: ["Admin", "Editor"] } ]
  任意一个角色即可（OR关系）

[Authorize(Roles = "Admin")]
[Authorize(Roles = "Editor")]  （两个 [Authorize]）
→ Requirements: [
      RolesAuthorizationRequirement { AllowedRoles: ["Admin"] },
      RolesAuthorizationRequirement { AllowedRoles: ["Editor"] }
  ]
  两个都要满足（AND关系）！必须同时是Admin和Editor

[Authorize(Policy = "CanEditArticle")]
→ 从 IAuthorizationPolicyProvider 获取名为 "CanEditArticle" 的策略
→ 策略中定义了什么需求就是什么需求

[Authorize(AuthenticationSchemes = "Bearer")]
→ AuthenticationSchemes: ["Bearer"]
→ 只用 JWT Bearer 认证，不用 Cookie
→ Requirements: (取决于默认策略)

[Authorize(Policy = "AdminOnly", AuthenticationSchemes = "Bearer")]
→ AuthenticationSchemes: ["Bearer"]
→ Requirements: 来自 "AdminOnly" 策略的需求

[AllowAnonymous]
→ 跳过所有授权检查

// 📁 源码位置: src/Security/Authorization/Policy/src/AuthorizationMiddleware.cs

public class AuthorizationMiddleware
{
    private readonly RequestDelegate _next;
    private readonly IAuthorizationPolicyProvider _policyProvider;

    public async Task Invoke(HttpContext context)
    {
        // 🔥 第1步：获取当前 Endpoint
        var endpoint = context.GetEndpoint();
        
        if (endpoint == null)
        {
            // 没有匹配到 Endpoint → 直接继续
            await _next(context);
            return;
        }

        // 🔥 第2步：获取 Endpoint 上的授权元数据
        var authorizeData = endpoint.Metadata.GetOrderedMetadata<IAuthorizeData>();
        // 也就是 [Authorize] 特性的数据

        // 🔥 第3步：检查是否有 [AllowAnonymous]
        if (endpoint.Metadata.GetMetadata<IAllowAnonymous>() != null)
        {
            // 有 [AllowAnonymous] → 跳过授权
            await _next(context);
            return;
        }

        // 🔥 第4步：构建授权策略
        var policy = await AuthorizationPolicy.CombineAsync(
            _policyProvider, authorizeData);
        
        if (policy == null)
        {
            // 没有 [Authorize] → 检查回退策略
            var fallbackPolicy = await _policyProvider.GetFallbackPolicyAsync();
            
            if (fallbackPolicy == null)
            {
                // 没有回退策略 → 允许访问
                await _next(context);
                return;
            }
            
            policy = fallbackPolicy;
        }

        // 🔥 第5步：如果策略指定了认证方案，先用指定方案认证
        if (policy.AuthenticationSchemes.Any())
        {
            ClaimsPrincipal? newPrincipal = null;
            
            foreach (var scheme in policy.AuthenticationSchemes)
            {
                var result = await context.AuthenticateAsync(scheme);
                if (result?.Succeeded ?? false)
                {
                    newPrincipal = SecurityHelper.MergeUserPrincipal(
                        newPrincipal, result.Principal);
                }
            }
            
            if (newPrincipal != null)
            {
                context.User = newPrincipal;
                // 用指定方案的认证结果替换 context.User
            }
        }

        // 🔥 第6步：执行授权检查
        var authorizeResult = await _policyEvaluator.AuthorizeAsync(
            policy, 
            context.AuthenticateAsync,  // 认证结果
            context, 
            resource: endpoint);

        // 🔥 第7步：根据结果处理
        if (authorizeResult.Challenged)
        {
            // 未认证 → Challenge
            if (policy.AuthenticationSchemes.Any())
            {
                foreach (var scheme in policy.AuthenticationSchemes)
                {
                    await context.ChallengeAsync(scheme);
                }
            }
            else
            {
                await context.ChallengeAsync();
                // 使用默认 Challenge 方案
                // Cookie → 302 到登录页
                // JWT → 401
            }
            return;  // ⛔ 短路！
        }
        else if (authorizeResult.Forbidden)
        {
            // 已认证但无权限 → Forbid
            if (policy.AuthenticationSchemes.Any())
            {
                foreach (var scheme in policy.AuthenticationSchemes)
                {
                    await context.ForbidAsync(scheme);
                }
            }
            else
            {
                await context.ForbidAsync();
                // 通常返回 403
            }
            return;  // ⛔ 短路！
        }

        // 🔥 授权通过 → 继续管道
        await _next(context);
    }
}

// 📁 源码位置: src/Security/Authorization/Policy/src/PolicyEvaluator.cs

public class PolicyEvaluator : IPolicyEvaluator
{
    private readonly IAuthorizationService _authorization;

    // 🔥 认证阶段
    public async Task<AuthenticateResult> AuthenticateAsync(
        AuthorizationPolicy policy, HttpContext context)
    {
        if (policy.AuthenticationSchemes.Any())
        {
            // 使用策略指定的方案认证
            ClaimsPrincipal? newPrincipal = null;
            DateTimeOffset? minExpiresUtc = null;

            foreach (var scheme in policy.AuthenticationSchemes)
            {
                var result = await context.AuthenticateAsync(scheme);
                if (result != null && result.Succeeded)
                {
                    newPrincipal = SecurityHelper.MergeUserPrincipal(
                        newPrincipal, result.Principal);
                }
            }
            
            if (newPrincipal != null)
            {
                context.User = newPrincipal;
                return AuthenticateResult.Success(
                    new AuthenticationTicket(newPrincipal, 
                        string.Join(";", policy.AuthenticationSchemes)));
            }
            
            return AuthenticateResult.NoResult();
        }

        // 使用默认认证（已在AuthenticationMiddleware中完成）
        return context.User?.Identity?.IsAuthenticated ?? false
            ? AuthenticateResult.Success(
                new AuthenticationTicket(context.User, "context.User"))
            : AuthenticateResult.NoResult();
    }

    // 🔥 授权阶段
    public async Task<PolicyAuthorizationResult> AuthorizeAsync(
        AuthorizationPolicy policy,
        AuthenticateResult authenticationResult,
        HttpContext context,
        object? resource)
    {
        // 🔥🔥🔥 调用 IAuthorizationService
        var result = await _authorization.AuthorizeAsync(
            context.User!, 
            resource, 
            policy);

        if (result.Succeeded)
        {
            return PolicyAuthorizationResult.Success();
        }

        // 区分 Challenge 和 Forbid
        if (authenticationResult.Succeeded)
        {
            // 已认证但授权失败 → Forbid (403)
            return PolicyAuthorizationResult.Forbid(result.Failure);
        }
        else
        {
            // 未认证 → Challenge (要求登录)
            return PolicyAuthorizationResult.Challenge();
        }
    }
}

8.2 DefaultAuthorizationService —— 授权判断的核心逻辑

// 📁 源码位置: src/Security/Authorization/Core/src/DefaultAuthorizationService.cs

public class DefaultAuthorizationService : IAuthorizationService
{
    private readonly IAuthorizationPolicyProvider _policyProvider;
    private readonly IAuthorizationHandlerProvider _handlers;
    private readonly IAuthorizationHandlerContextFactory _contextFactory;
    private readonly AuthorizationOptions _options;
    private readonly IAuthorizationEvaluator _evaluator;

    public async Task<AuthorizationResult> AuthorizeAsync(
        ClaimsPrincipal user,
        object? resource,
        IEnumerable<IAuthorizationRequirement> requirements)
    {
        // 🔥 第1步：创建授权上下文
        var authContext = _contextFactory.CreateContext(requirements, user, resource);
        // AuthorizationHandlerContext 包含：
        //   - User: ClaimsPrincipal
        //   - Requirements: 所有需求
        //   - Resource: 被访问的资源
        //   - PendingRequirements: 还未满足的需求
        
        // 🔥 第2步：获取所有注册的 Handler
        var handlers = await _handlers.GetHandlersAsync(authContext);

        // 🔥 第3步：逐个执行 Handler
        foreach (var handler in handlers)
        {
            await handler.HandleAsync(authContext);
            
            // 如果配置了 InvokeHandlersAfterFailure = false
            // 且已经有明确的失败，就不继续执行其他Handler
            if (!_options.InvokeHandlersAfterFailure && authContext.HasFailed)
            {
                break;
            }
        }

        // 🔥 第4步：评估结果
        var result = _evaluator.Evaluate(authContext);
        
        // 评估逻辑：
        // - 如果 HasFailed = true → 失败（有Handler显式调用了Fail()）
        // - 如果所有 Requirements 都被 Succeed 了 → 成功
        // - 否则 → 失败（有未满足的Requirement）
        
        if (result.Succeeded)
        {
            _logger.UserAuthorizationSucceeded();
        }
        else
        {
            _logger.UserAuthorizationFailed(result.Failure);
        }

        return result;
    }
}

8.3 AuthorizationHandlerContext —— 授权上下文

// 📁 源码位置: src/Security/Authorization/Core/src/AuthorizationHandlerContext.cs

public class AuthorizationHandlerContext
{
    private readonly HashSet<IAuthorizationRequirement> _pendingRequirements;
    private bool _failCalled;
    private bool _succeedCalled;

    public AuthorizationHandlerContext(
        IEnumerable<IAuthorizationRequirement> requirements,
        ClaimsPrincipal user,
        object? resource)
    {
        Requirements = requirements;
        User = user;
        Resource = resource;
        _pendingRequirements = new HashSet<IAuthorizationRequirement>(requirements);
    }

    public ClaimsPrincipal User { get; }
    public object? Resource { get; }
    public IEnumerable<IAuthorizationRequirement> Requirements { get; }
    
    // 还未满足的需求
    public IEnumerable<IAuthorizationRequirement> PendingRequirements 
        => _pendingRequirements;

    // 🔥 是否所有需求都已满足
    public bool HasSucceeded 
        => !_failCalled && _succeedCalled && !_pendingRequirements.Any();
    
    public bool HasFailed => _failCalled;

    // 🔥 标记某个需求已满足
    public void Succeed(IAuthorizationRequirement requirement)
    {
        _succeedCalled = true;
        _pendingRequirements.Remove(requirement);
    }

    // 🔥 显式标记失败（即使其他需求都满足了也会失败）
    public void Fail()
    {
        _failCalled = true;
    }
    
    // ⚠️ 注意没有 "Skip" 方法
    // 如果Handler什么都不做 → 需求仍然 Pending → 最终 HasSucceeded = false
}

Succeed、Fail 和 "什么都不做" 的区别：

│                                                                      │
│  场景：Policy有3个Requirement [R1, R2, R3]                           │
│        有4个Handler [H1, H2, H3, H4]                                │
│                                                                      │
│  ─────────── 正常成功 ───────────                                    │
│  H1: context.Succeed(R1)    → PendingRequirements: [R2, R3]         │

│  H3: 什么都不做              → PendingRequirements: [R3]             │
│  H4: context.Succeed(R3)    → PendingRequirements: []               │
│  结果：HasSucceeded = true ✅                                        │
│                                                                      │
│  ─────────── 需求未满足 ───────────                                  │
│  H1: context.Succeed(R1)    → PendingRequirements: [R2, R3]         │
│  H2: context.Succeed(R2)    → PendingRequirements: [R3]             │
│  H3: 什么都不做              → PendingRequirements: [R3]             │
│  H4: 什么都不做              → PendingRequirements: [R3]  ← 仍有未满足│
│  结果：HasSucceeded = false ❌ (R3没有被任何Handler满足)              │
│                                                                      │
│  ─────────── 显式失败 ───────────                                    │
│  H1: context.Succeed(R1)    → PendingRequirements: [R2, R3]         │
│  H2: context.Fail()          → _failCalled = true                    │
│  H3: context.Succeed(R2)    → PendingRequirements: [R3]             │
│  H4: context.Succeed(R3)    → PendingRequirements: []               │
│  结果：HasSucceeded = false ❌ (即使所有R都被Succeed了)              │
│         因为 HasSucceeded = !_failCalled && !pending.Any()           │
│         Fail() 是一票否决 ⛔                                         │
│                                                                      │
│  ⚠️ Fail() 的用途：例如账户被锁定、IP黑名单等                       │
│     不管用户有什么角色/声明，直接拒绝                                  │
│                                                                      │

{
    public string RequiredScope { get; }
    
    public ScopeRequirement(string requiredScope)
    {
        RequiredScope = requiredScope;
    }
}

// ========== Handler ==========
public class ScopeAuthorizationHandler : AuthorizationHandler<ScopeRequirement>
{
    protected override Task HandleRequirementAsync(
        AuthorizationHandlerContext context,
        ScopeRequirement requirement)
    {
        // 🔥 检查 Token 中的 scope 声明
        // OpenIddict 签发的Token中，scope可能是空格分隔的字符串
        // 也可能是多个独立的 scope claim
        
        var scopeClaims = context.User.FindAll("scope");
        
        foreach (var claim in scopeClaims)
        {
            // 处理空格分隔的情况："openid profile api.read"
            var scopes = claim.Value.Split(' ', StringSplitOptions.RemoveEmptyEntries);
            
            if (scopes.Contains(requirement.RequiredScope, StringComparer.Ordinal))
            {
                context.Succeed(requirement);
                return Task.CompletedTask;
            }
        }
        
        // 也检查 "scp" 声明（某些 OIDC Provider 用这个）
        var scpClaims = context.User.FindAll("scp");
        foreach (var claim in scpClaims)
        {
            if (string.Equals(claim.Value, requirement.RequiredScope, StringComparison.Ordinal))
            {
                context.Succeed(requirement);
                return Task.CompletedTask;
            }
        }
        
        return Task.CompletedTask;
    }
}

    public ArticleOperationRequirement(string operation)
    {
        Operation = operation;
    }
    
    public static readonly ArticleOperationRequirement Read = new("Read");
    public static readonly ArticleOperationRequirement Edit = new("Edit");
    public static readonly ArticleOperationRequirement Delete = new("Delete");
}

// ========== Resource ==========
public class Article
{
    public string Id { get; set; } = "";
    public string Title { get; set; } = "";
    public string AuthorId { get; set; } = "";   // 👈 文章作者
    public bool IsPublished { get; set; }
}

// ========== Handler ==========
public class ArticleAuthorizationHandler :
    AuthorizationHandler<ArticleOperationRequirement, Article>
    //                                                ^^^^^^^ 资源类型
{
    protected override Task HandleRequirementAsync(
        AuthorizationHandlerContext context,
        ArticleOperationRequirement requirement,
        Article resource)
    {
        var userId = context.User.FindFirst("sub")?.Value;

        switch (requirement.Operation)
        {
            case "Read":
                // 已发布的文章任何人可读，未发布的只有作者可读
                if (resource.IsPublished || resource.AuthorId == userId)
                {
                    context.Succeed(requirement);
                }
                break;

            case "Edit":
                // 作者可以编辑自己的文章
                if (resource.AuthorId == userId)
                {
                    context.Succeed(requirement);
                }
                // Admin也可以编辑任何文章
                if (context.User.IsInRole("Admin"))
                {
                    context.Succeed(requirement);
                }
                break;

            case "Delete":
                // 只有Admin可以删除
                if (context.User.IsInRole("Admin"))
                {
                    context.Succeed(requirement);
                }
                break;
        }

        return Task.CompletedTask;
    }
}

// ========== Handler ==========
public class PermissionAuthorizationHandler :
    AuthorizationHandler<PermissionRequirement>
{
    private readonly IServiceScopeFactory _scopeFactory;
    
    public PermissionAuthorizationHandler(IServiceScopeFactory scopeFactory)
    {
        _scopeFactory = scopeFactory;
    }
    
    protected override async Task HandleRequirementAsync(
        AuthorizationHandlerContext context,
        PermissionRequirement requirement)
    {
        var userId = context.User.FindFirst("sub")?.Value;
        if (userId == null) return;

        // 从数据库查询用户权限
        using var scope = _scopeFactory.CreateScope();
        var permissionService = scope.ServiceProvider
            .GetRequiredService<IPermissionService>();

        var hasPermission = await permissionService
            .UserHasPermissionAsync(userId, requirement.Permission);

        if (hasPermission)
        {
            context.Succeed(requirement);
        }
    }
}

public class PermissionPolicyProvider : IAuthorizationPolicyProvider
{
    private readonly DefaultAuthorizationPolicyProvider _fallbackProvider;

    public PermissionPolicyProvider(IOptions<AuthorizationOptions> options)
    {
        // 回退到默认Provider（处理非Permission策略）
        _fallbackProvider = new DefaultAuthorizationPolicyProvider(options);
    }

    public Task<AuthorizationPolicy> GetDefaultPolicyAsync()
        => _fallbackProvider.GetDefaultPolicyAsync();

    public Task<AuthorizationPolicy?> GetFallbackPolicyAsync()
        => _fallbackProvider.GetFallbackPolicyAsync();

    public Task<AuthorizationPolicy?> GetPolicyAsync(string policyName)
    {
        // 🔥 检查是否是 Permission 策略
        if (policyName.StartsWith("Permission:", StringComparison.OrdinalIgnoreCase))
        {
            var permission = policyName.Substring("Permission:".Length);
            
            // 动态创建策略
            var policy = new AuthorizationPolicyBuilder()
                .RequireAuthenticatedUser()
                .AddRequirements(new PermissionRequirement(permission))
                .Build();
            
            return Task.FromResult<AuthorizationPolicy?>(policy);
        }
        
        // 不是Permission策略 → 回退到默认Provider
        return _fallbackProvider.GetPolicyAsync(policyName);
    }
}

十六、自己实现一个极简授权系统

// ========== 极简实现 ==========

// Requirement
interface IRequirement { }

class RoleRequirement : IRequirement
{
    public string Role { get; }
    public RoleRequirement(string role) => Role = role;
}

class AgeRequirement : IRequirement
{
    public int MinAge { get; }
    public AgeRequirement(int minAge) => MinAge = minAge;
}

// Policy
class Policy
{
    public List<IRequirement> Requirements { get; } = new();
}

// Handler Context
class AuthContext
{
    public Dictionary<string, string> UserClaims { get; }
    public List<IRequirement> Requirements { get; }
    public HashSet<IRequirement> Pending { get; }
    public bool Failed { get; private set; }
    
    public AuthContext(Dictionary<string, string> claims, List<IRequirement> reqs)
    {
        UserClaims = claims;
        Requirements = reqs;
        Pending = new HashSet<IRequirement>(reqs);
    }

    public void Succeed(IRequirement req) => Pending.Remove(req);
    public void Fail() => Failed = true;
    public bool HasSucceeded => !Failed && !Pending.Any();
}

// Handlers
interface IHandler { void Handle(AuthContext context); }

class RoleHandler : IHandler
{
    public void Handle(AuthContext context)
    {
        foreach (var req in context.Requirements.OfType<RoleRequirement>())
        {
            if (context.UserClaims.TryGetValue("role", out var role) 
                && role == req.Role)
            {
                context.Succeed(req);
            }
        }
    }
}

class AgeHandler : IHandler
{
    public void Handle(AuthContext context)
    {
        foreach (var req in context.Requirements.OfType<AgeRequirement>())
        {
            if (context.UserClaims.TryGetValue("age", out var ageStr) 
                && int.Parse(ageStr) >= req.MinAge)
            {
                context.Succeed(req);
            }
        }
    }
}

// Authorization Service
class MiniAuthService
{
    private readonly List<IHandler> _handlers;
    
    public MiniAuthService(List<IHandler> handlers) => _handlers = handlers;
    
    public bool Authorize(Dictionary<string, string> userClaims, Policy policy)
    {
        var context = new AuthContext(userClaims, policy.Requirements);
        
        foreach (var handler in _handlers)
        {
            handler.Handle(context);
        }
        
        return context.HasSucceeded;
    }
}

十七、检验你的理解

  1. [Authorize(Roles = "Admin,Editor")] 和连写两个 [Authorize(Roles = "Admin")][Authorize(Roles = "Editor")] 有什么区别？
    （答案：前者是 OR 关系——用户有 Admin 或 Editor 任一角色即可通过，因为它们在同一个 RolesAuthorizationRequirement 中，AllowedRoles = ["Admin", "Editor"]。后者是 AND 关系——用户必须同时有 Admin 和 Editor 两个角色，因为生成了两个独立的 RolesAuthorizationRequirement，每个都必须被 Succeed。）

  2. 在 AuthorizationHandler 中，调用 context.Succeed(requirement) 和什么都不做有什么区别？可以有多个Handler处理同一个Requirement吗？
    （答案：Succeed 将 Requirement 从 PendingRequirements 中移除。什么都不做则 Requirement 仍然 Pending。可以有多个Handler处理同一个Requirement——只要其中一个Handler Succeed了就行。例如一个Handler检查是否是作者，另一个Handler检查是否是Admin，任一个成功都可以编辑文章。）

  3. DefaultPolicy 和 FallbackPolicy 分别影响哪些端点？SSO项目中应该怎么设置？
    （答案：DefaultPolicy 影响标注了 [Authorize] 但没指定 Policy 的端点。FallbackPolicy 影响既没有 [Authorize] 也没有 [AllowAnonymous] 的端点。SSO授权服务器的 FallbackPolicy 应为 null（允许匿名，因为登录页等需要匿名访问）。API资源服务器的 FallbackPolicy 应为 RequireAuthenticatedUser（所有API默认需要Token）。）

  4. 基于资源的授权（Resource-based Authorization）为什么不能通过 [Authorize] 特性实现？
    （答案：因为 [Authorize] 特性在编译时就确定了，而资源（如文章对象）需要在运行时从数据库获取后才能判断。比如"用户是否是文章的作者"这个判断，需要先拿到文章对象才能比较。所以必须在Controller方法内部调用 IAuthorizationService.AuthorizeAsync(User, article, requirement) 来进行基于资源的授权。）

  5. IAuthorizationPolicyProvider 为什么是 Singleton？如果权限数据在数据库中且可能变化，怎么处理？
    （答案：PolicyProvider 是 Singleton 因为它被 AuthorizationMiddleware 使用，不依赖请求级别的数据。如果权限在数据库中，有两种方式：① PolicyProvider 动态生成策略（策略结构固定，如 PermissionRequirement），实际的数据库查询在 Handler 中完成（Handler 可以通过 IServiceScopeFactory 创建 Scope 来访问数据库）。② 使用 IOptionsMonitor 监听配置变化。）

  6. 在 AuthenticationMiddleware 的第一阶段，OpenIddict Handler 处理 /connect/token 请求并短路。这时候 UseAuthorization 中间件有没有执行？Token 端点的安全性由谁保证？
    （答案：UseAuthorization 没有执行，因为请求在 UseAuthentication 层就被短路了。Token 端点的安全性完全由 OpenIddict Server Handler 自己保证——它验证 client_id、client_secret、authorization_code、PKCE 等。这是 OAuth 2.0 协议层面的安全，不依赖 ASP.NET Core 的授权系统。）
