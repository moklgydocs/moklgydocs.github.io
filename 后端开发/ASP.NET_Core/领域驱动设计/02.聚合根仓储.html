namespace MokFramework.EntityFrameworkCore;

/// <summary>
/// 当前用户接口
/// 
/// 审计拦截器通过此接口获取当前操作者
/// 
/// 实现方式:
///   Web 项目: 从 HttpContext.User 的 Claims 中读取
///   后台任务: 返回 null 或系统用户ID
///   单元测试: Mock
///   
/// 为什么不直接注入 IHttpContextAccessor:
///   1. MokFramework.EFCore 不应该依赖 ASP.NET Core
///   2. 后台任务没有 HttpContext
///   3. 解耦 ✅
/// </summary>
public interface ICurrentUser
{
    /// <summary>
    /// 当前用户ID（未登录为 null）
    /// </summary>
    Guid? Id { get; }

    /// <summary>
    /// 当前租户ID（非多租户场景为 null）
    /// </summary>
    Guid? TenantId { get; }

    /// <summary>
    /// 是否已认证
    /// </summary>
    bool IsAuthenticated { get; }
}

PlatformDbContext — 所有项目的基础 DbContext

// ============================================
// DbContext/PlatformDbContext.cs
// ============================================

using Microsoft.EntityFrameworkCore;
using MokFramework.Auditing;
using MokFramework.Concurrency;
using MokFramework.Data;
using MokFramework.Entities;
using MokFramework.MultiTenancy;
using MokFramework.SoftDelete;
using MokFramework.EntityFrameworkCore.ValueConverters;

namespace MokFramework.EntityFrameworkCore;

/// <summary>
/// 平台基础 DbContext
/// 
/// 所有项目的 DbContext 继承此类
/// 自动获得:
///   ✅ 软删除全局过滤
///   ✅ 多租户全局过滤
///   ✅ ExtraProperties JSON 转换
///   ✅ ConcurrencyStamp 并发检测
///   ✅ DateTime UTC 标准化
///   ✅ 审计字段自动配置
///   
/// 用法:
///   public class SsoDbContext : PlatformDbContext&lt;SsoDbContext&gt;
///   {
///       public DbSet&lt;SsoUser&gt; Users => Set&lt;SsoUser&gt;();
///       public SsoDbContext(DbContextOptions&lt;SsoDbContext&gt; options) : base(options) { }
///   }
/// </summary>
public abstract class PlatformDbContext<TContext> : DbContext
    where TContext : DbContext
{
    private readonly ICurrentUser _currentUser;

    protected PlatformDbContext(DbContextOptions<TContext> options)
        : base(options)
    {
        // 允许无 ICurrentUser 的场景（Migration 工具等）
        _currentUser = new NullCurrentUser();
    }

    protected PlatformDbContext(
        DbContextOptions<TContext> options,
        ICurrentUser currentUser)
        : base(options)
    {
        _currentUser = currentUser;
    }

    /// <summary>
    /// 是否启用软删除过滤（默认启用）
    /// 某些场景需要查已删除数据时可覆盖
    /// </summary>
    protected virtual bool IsSoftDeleteFilterEnabled => true;

    /// <summary>
    /// 是否启用多租户过滤（默认启用）
    /// </summary>
    protected virtual bool IsMultiTenantFilterEnabled => true;

    protected override void OnModelCreating(ModelBuilder modelBuilder)
    {
        base.OnModelCreating(modelBuilder);

        foreach (var entityType in modelBuilder.Model.GetEntityTypes())
        {
            var clrType = entityType.ClrType;

            // ============ 软删除全局过滤 ============
            ConfigureSoftDeleteFilter(modelBuilder, clrType);

            // ============ 多租户全局过滤 ============
            ConfigureMultiTenantFilter(modelBuilder, clrType);

            // ============ 审计字段配置 ============
            ConfigureAuditProperties(modelBuilder, clrType);

            // ============ 并发戳配置 ============
            ConfigureConcurrencyStamp(modelBuilder, clrType);

            // ============ 扩展属性配置 ============
            ConfigureExtraProperties(modelBuilder, clrType);
        }
    }

    #region 全局过滤器配置

    private void ConfigureSoftDeleteFilter(ModelBuilder modelBuilder, Type clrType)
    {
        if (!typeof(ISoftDelete).IsAssignableFrom(clrType))
            return;

        // 构建: entity => !entity.IsDeleted || !IsSoftDeleteFilterEnabled
        // EF Core 的全局过滤在查询时动态求值 IsSoftDeleteFilterEnabled

        modelBuilder.Entity(clrType).AddQueryFilter<ISoftDelete>(
            e => !e.IsDeleted);
    }

    private void ConfigureMultiTenantFilter(ModelBuilder modelBuilder, Type clrType)
    {
        if (!typeof(IMultiTenant).IsAssignableFrom(clrType))
            return;

        var currentTenantId = _currentUser.TenantId;

        modelBuilder.Entity(clrType).AddQueryFilter<IMultiTenant>(
            e => e.TenantId == currentTenantId);
    }

    #endregion

    #region 审计字段自动配置

    private static void ConfigureAuditProperties(ModelBuilder modelBuilder, Type clrType)
    {
        var builder = modelBuilder.Entity(clrType);

        if (typeof(IHasCreationTime).IsAssignableFrom(clrType))
        {
            builder.Property(nameof(IHasCreationTime.CreationTime))
                   .HasColumnName("CreationTime")
                   .IsRequired();
        }

        if (typeof(IMayHaveCreator).IsAssignableFrom(clrType))
        {
            builder.Property(nameof(IMayHaveCreator.CreatorId))
                   .HasColumnName("CreatorId")
                   .IsRequired(false);
        }

        if (typeof(IHasModificationTime).IsAssignableFrom(clrType))
        {
            builder.Property(nameof(IHasModificationTime.LastModificationTime))
                   .HasColumnName("LastModificationTime")
                   .IsRequired(false);
        }

        if (typeof(IModificationAudited).IsAssignableFrom(clrType))
        {
            builder.Property(nameof(IModificationAudited.LastModifierId))
                   .HasColumnName("LastModifierId")
                   .IsRequired(false);
        }

        if (typeof(ISoftDelete).IsAssignableFrom(clrType))
        {
            builder.Property(nameof(ISoftDelete.IsDeleted))
                   .HasColumnName("IsDeleted")
                   .HasDefaultValue(false)
                   .IsRequired();

            // 索引：大部分查询都过滤 IsDeleted
            builder.HasIndex(nameof(ISoftDelete.IsDeleted));
        }

        if (typeof(IHasDeletionTime).IsAssignableFrom(clrType))
        {
            builder.Property(nameof(IHasDeletionTime.DeletionTime))
                   .HasColumnName("DeletionTime")
                   .IsRequired(false);
        }

        if (typeof(IDeletionAudited).IsAssignableFrom(clrType))
        {
            builder.Property(nameof(IDeletionAudited.DeleterId))
                   .HasColumnName("DeleterId")
                   .IsRequired(false);
        }

        if (typeof(IMultiTenant).IsAssignableFrom(clrType))
        {
            builder.Property(nameof(IMultiTenant.TenantId))
                   .HasColumnName("TenantId")
                   .IsRequired(false);

            builder.HasIndex(nameof(IMultiTenant.TenantId));
        }
    }

    #endregion

    #region 并发戳

    private static void ConfigureConcurrencyStamp(ModelBuilder modelBuilder, Type clrType)
    {
        if (!typeof(IHasConcurrencyStamp).IsAssignableFrom(clrType))
            return;

        modelBuilder.Entity(clrType)
            .Property(nameof(IHasConcurrencyStamp.ConcurrencyStamp))
            .HasMaxLength(40)
            .IsConcurrencyToken(); // EF Core 乐观并发检测
    }

    #endregion

    #region 扩展属性

    private static void ConfigureExtraProperties(ModelBuilder modelBuilder, Type clrType)
    {
        if (!typeof(IHasExtraProperties).IsAssignableFrom(clrType))
            return;

        modelBuilder.Entity(clrType)
            .Property(nameof(IHasExtraProperties.ExtraProperties))
            .HasColumnName("ExtraProperties")
            .HasConversion(new ExtraPropertiesValueConverter())
            .HasMaxLength(int.MaxValue)   // NVARCHAR(MAX) / TEXT
            .IsRequired(false)
            .HasColumnType("text");       // PostgreSQL 用 text
    }

    #endregion
}

全局过滤器扩展方法

// ============================================
// Extensions/QueryFilterExtensions.cs
// 解决 EF Core 多个过滤器合并问题
// ============================================

using System.Linq.Expressions;
using Microsoft.EntityFrameworkCore;
using Microsoft.EntityFrameworkCore.Metadata.Builders;

namespace MokFramework.EntityFrameworkCore.Extensions;

/// <summary>
/// EF Core 全局过滤器扩展
/// 
/// EF Core 的限制: 一个实体只能有一个 HasQueryFilter
/// 多次调用会覆盖
/// 
/// ABP 的解决方案: 合并多个过滤器为一个表达式
/// 我们复刻此方案
/// </summary>
public static class QueryFilterExtensions
{
    /// <summary>
    /// 添加查询过滤器（支持多个过滤器合并）
    /// </summary>
    public static void AddQueryFilter<TInterface>(
        this EntityTypeBuilder entityTypeBuilder,
        Expression<Func<TInterface, bool>> filter)
    {
        var clrType = entityTypeBuilder.Metadata.ClrType;

        // 将 TInterface 的表达式转换为具体实体类型的表达式
        var parameter = Expression.Parameter(clrType, "e");
        var body = ReplacingExpressionVisitor.Replace(
            filter.Parameters[0],
            parameter,
            filter.Body);

        var newFilter = Expression.Lambda(body, parameter);

        // 获取已有的过滤器
        var existingFilter = entityTypeBuilder.Metadata.GetQueryFilter();

        if (existingFilter != null)
        {
            // 合并: existing AND new
            var existingBody = ReplacingExpressionVisitor.Replace(
                existingFilter.Parameters[0],
                parameter,
                existingFilter.Body);

            var combinedBody = Expression.AndAlso(existingBody, body);
            var combinedFilter = Expression.Lambda(combinedBody, parameter);

            entityTypeBuilder.HasQueryFilter(combinedFilter);
        }
        else
        {
            entityTypeBuilder.HasQueryFilter(newFilter);
        }
    }
}

/// <summary>
/// 表达式参数替换访问器
/// </summary>
internal class ReplacingExpressionVisitor : ExpressionVisitor
{
    private readonly Expression _oldValue;
    private readonly Expression _newValue;

    private ReplacingExpressionVisitor(Expression oldValue, Expression newValue)
    {
        _oldValue = oldValue;
        _newValue = newValue;
    }

    public static Expression Replace(
        Expression oldValue, Expression newValue, Expression expression)
    {
        return new ReplacingExpressionVisitor(oldValue, newValue).Visit(expression);
    }

    protected override Expression VisitParameter(ParameterExpression node)
    {
        return node == _oldValue ? _newValue : base.VisitParameter(node);
    }

    protected override Expression VisitMember(MemberExpression node)
    {
        if (node.Expression == _oldValue)
        {
            return Expression.MakeMemberAccess(_newValue, node.Member);
        }
        return base.VisitMember(node);
    }
}

using System.Text.Json;
using Microsoft.EntityFrameworkCore.Storage.ValueConversion;
using MokFramework.Data;

namespace MokFramework.EntityFrameworkCore.ValueConverters;

/// <summary>
/// ExtraPropertyDictionary ↔ JSON string 转换器
/// 
/// 数据库存储: {"NickName":"老张","Score":100}
/// C# 对象: ExtraPropertyDictionary { "NickName" → "老张", "Score" → 100 }
/// </summary>
public class ExtraPropertiesValueConverter
    : ValueConverter<ExtraPropertyDictionary, string>
{
    private static readonly JsonSerializerOptions JsonOptions = new()
    {
        PropertyNamingPolicy = JsonNamingPolicy.CamelCase,
        WriteIndented = false,
    };

    public ExtraPropertiesValueConverter()
        : base(
            d => Serialize(d),
            s => Deserialize(s))
    {
    }

    private static string Serialize(ExtraPropertyDictionary dict)
    {
        if (dict == null || dict.Count == 0)
            return "{}";

        return JsonSerializer.Serialize(dict, JsonOptions);
    }

    private static ExtraPropertyDictionary Deserialize(string json)
    {
        if (string.IsNullOrWhiteSpace(json))
            return new ExtraPropertyDictionary();

        try
        {
            var dict = JsonSerializer.Deserialize<Dictionary<string, object?>>(
                json, JsonOptions);
            return dict == null
                ? new ExtraPropertyDictionary()
                : new ExtraPropertyDictionary(dict);
        }
        catch
        {
            return new ExtraPropertyDictionary();
        }
    }
}

/// <summary>
/// DateTime UTC 转换器
/// 
/// 所有 DateTime 存入数据库时统一为 UTC
/// 从数据库读出时标记为 UTC Kind
/// 
/// 避免时区问题:
///   开发者写 DateTime.Now → 自动转 UTC 存储
///   查询时 → 自动标记为 UTC Kind
///   前端显示 → 前端根据用户时区转换
/// </summary>
public class DateTimeUtcConverter : ValueConverter<DateTime, DateTime>
{
    public DateTimeUtcConverter()
        : base(
            v => v.Kind == DateTimeKind.Utc ? v : v.ToUniversalTime(),
            v => DateTime.SpecifyKind(v, DateTimeKind.Utc))
    {
    }
}

public class NullableDateTimeUtcConverter : ValueConverter<DateTime?, DateTime?>
{
    public NullableDateTimeUtcConverter()
        : base(
            v => v.HasValue
                ? (v.Value.Kind == DateTimeKind.Utc ? v : v.Value.ToUniversalTime())
                : v,
            v => v.HasValue
                ? DateTime.SpecifyKind(v.Value, DateTimeKind.Utc)
                : v)
    {
    }
}

审计拦截器

// ============================================
// Interceptors/AuditingInterceptor.cs
// ============================================

using Microsoft.EntityFrameworkCore;
using Microsoft.EntityFrameworkCore.ChangeTracking;
using Microsoft.EntityFrameworkCore.Diagnostics;
using MokFramework.Auditing;
using MokFramework.SoftDelete;

namespace MokFramework.EntityFrameworkCore.Interceptors;

/// <summary>
/// 审计拦截器
/// 
/// 在 SaveChanges 时自动填充:
///   Added:    CreationTime, CreatorId
///   Modified: LastModificationTime, LastModifierId
///   
/// 开发者零侵入 ✅
/// 不需要手动设置任何审计字段
/// </summary>
public class AuditingInterceptor : SaveChangesInterceptor
{
    private readonly ICurrentUser _currentUser;

    public AuditingInterceptor(ICurrentUser currentUser)
    {
        _currentUser = currentUser;
    }

    public override InterceptionResult<int> SavingChanges(
        DbContextEventData eventData,
        InterceptionResult<int> result)
    {
        if (eventData.Context != null)
            ApplyAuditing(eventData.Context);

        return base.SavingChanges(eventData, result);
    }

    public override ValueTask<InterceptionResult<int>> SavingChangesAsync(
        DbContextEventData eventData,
        InterceptionResult<int> result,
        CancellationToken cancellationToken = default)
    {
        if (eventData.Context != null)
            ApplyAuditing(eventData.Context);

        return base.SavingChangesAsync(eventData, result, cancellationToken);
    }

    private void ApplyAuditing(DbContext context)
    {
        var now = DateTime.UtcNow;
        var userId = _currentUser.Id;

        foreach (var entry in context.ChangeTracker.Entries())
        {
            switch (entry.State)
            {
                case EntityState.Added:
                    SetCreationAudit(entry, now, userId);
                    break;

                case EntityState.Modified:
                    SetModificationAudit(entry, now, userId);
                    break;
            }
        }
    }

    private static void SetCreationAudit(EntityEntry entry, DateTime now, Guid? userId)
    {
        if (entry.Entity is IHasCreationTime hasCreationTime)
        {
            // 只在未手动设置时填充（允许显式指定创建时间）
            if (hasCreationTime.CreationTime == default)
            {
                entry.Property(nameof(IHasCreationTime.CreationTime))
                     .CurrentValue = now;
            }
        }

        if (entry.Entity is IMayHaveCreator mayHaveCreator)
        {
            if (mayHaveCreator.CreatorId == null)
            {
                entry.Property(nameof(IMayHaveCreator.CreatorId))
                     .CurrentValue = userId;
            }
        }
    }

    private static void SetModificationAudit(EntityEntry entry, DateTime now, Guid? userId)
    {
        if (entry.Entity is IHasModificationTime)
        {
            entry.Property(nameof(IHasModificationTime.LastModificationTime))
                 .CurrentValue = now;
        }

        if (entry.Entity is IModificationAudited)
        {
            entry.Property(nameof(IModificationAudited.LastModifierId))
                 .CurrentValue = userId;
        }
    }
}

软删除拦截器

// ============================================
// Interceptors/SoftDeleteInterceptor.cs
// ============================================

using Microsoft.EntityFrameworkCore;
using Microsoft.EntityFrameworkCore.Diagnostics;
using MokFramework.Auditing;
using MokFramework.SoftDelete;

namespace MokFramework.EntityFrameworkCore.Interceptors;

/// <summary>
/// 软删除拦截器
/// 
/// 拦截 Delete 操作 → 转为 Update（设置 IsDeleted=true）
/// 
/// 核心逻辑:
///   1. 遍历所有 Deleted 状态的实体
///   2. 如果实现了 ISoftDelete
///   3. 将状态改为 Modified
///   4. 设置 IsDeleted=true
///   5. 如果实现了 IDeletionAudited, 填充 DeletionTime + DeleterId
///   
/// 这样 _context.Remove(entity) 或 entry.State = Deleted
/// 都会自动走软删除 ✅
/// 
/// 开发者无需关心软删除逻辑
/// </summary>
public class SoftDeleteInterceptor : SaveChangesInterceptor
{
    private readonly ICurrentUser _currentUser;

    public SoftDeleteInterceptor(ICurrentUser currentUser)
    {
        _currentUser = currentUser;
    }

    public override InterceptionResult<int> SavingChanges(
        DbContextEventData eventData,
        InterceptionResult<int> result)
    {
        if (eventData.Context != null)
            ApplySoftDelete(eventData.Context);

        return base.SavingChanges(eventData, result);
    }

    public override ValueTask<InterceptionResult<int>> SavingChangesAsync(
        DbContextEventData eventData,
        InterceptionResult<int> result,
        CancellationToken cancellationToken = default)
    {
        if (eventData.Context != null)
            ApplySoftDelete(eventData.Context);

        return base.SavingChangesAsync(eventData, result, cancellationToken);
    }

    private void ApplySoftDelete(DbContext context)
    {
        var now = DateTime.UtcNow;
        var userId = _currentUser.Id;

        var deletedEntries = context.ChangeTracker.Entries()
            .Where(e => e.State == EntityState.Deleted &&
                        e.Entity is ISoftDelete)
            .ToList();

        foreach (var entry in deletedEntries)
        {
            // 关键: 将 Delete 转为 Update
            entry.State = EntityState.Modified;

            // 设置 IsDeleted
            entry.Property(nameof(ISoftDelete.IsDeleted)).CurrentValue = true;

            // 如果有删除审计
            if (entry.Entity is IHasDeletionTime)
            {
                entry.Property(nameof(IHasDeletionTime.DeletionTime))
                     .CurrentValue = now;
            }

            if (entry.Entity is IDeletionAudited)
            {
                entry.Property(nameof(IDeletionAudited.DeleterId))
                     .CurrentValue = userId;
            }

            // 同时设置修改审计（软删除也是一种修改）
            if (entry.Entity is IHasModificationTime)
            {
                entry.Property(nameof(IHasModificationTime.LastModificationTime))
                     .CurrentValue = now;
            }

            if (entry.Entity is IModificationAudited)
            {
                entry.Property(nameof(IModificationAudited.LastModifierId))
                     .CurrentValue = userId;
            }
        }
    }
}

并发戳拦截器

// ============================================
// Interceptors/ConcurrencyStampInterceptor.cs
// ============================================

using Microsoft.EntityFrameworkCore;
using Microsoft.EntityFrameworkCore.Diagnostics;
using MokFramework.Concurrency;

namespace MokFramework.EntityFrameworkCore.Interceptors;

/// <summary>
/// 并发戳拦截器
/// 
/// 每次 Update 时自动刷新 ConcurrencyStamp
/// 配合 EF Core 的 IsConcurrencyToken 实现乐观并发
/// 
/// 工作原理:
///   1. 张三读取 Order (ConcurrencyStamp = "aaa")
///   2. 李四读取 Order (ConcurrencyStamp = "aaa")
///   3. 张三保存 → ConcurrencyStamp 从 "aaa" 改为 "bbb"
///      UPDATE Orders SET ... WHERE Id=@id AND ConcurrencyStamp='aaa'
///      → 成功（影响 1 行）
///   4. 李四保存 → 仍带着旧的 "aaa"
///      UPDATE Orders SET ... WHERE Id=@id AND ConcurrencyStamp='aaa'
///      → 失败（影响 0 行，因为已经变成 "bbb"）
///      → EF Core 抛出 DbUpdateConcurrencyException
/// </summary>
public class ConcurrencyStampInterceptor : SaveChangesInterceptor
{
    public override InterceptionResult<int> SavingChanges(
        DbContextEventData eventData,
        InterceptionResult<int> result)
    {
        if (eventData.Context != null)
            RefreshConcurrencyStamps(eventData.Context);

        return base.SavingChanges(eventData, result);
    }

    public override ValueTask<InterceptionResult<int>> SavingChangesAsync(
        DbContextEventData eventData,
        InterceptionResult<int> result,
        CancellationToken cancellationToken = default)
    {
        if (eventData.Context != null)
            RefreshConcurrencyStamps(eventData.Context);

        return base.SavingChangesAsync(eventData, result, cancellationToken);
    }

    private static void RefreshConcurrencyStamps(DbContext context)
    {
        var entries = context.ChangeTracker.Entries()
            .Where(e => e.State == EntityState.Modified &&
                        e.Entity is IHasConcurrencyStamp);

        foreach (var entry in entries)
        {
            entry.Property(nameof(IHasConcurrencyStamp.ConcurrencyStamp))
                 .CurrentValue = Guid.NewGuid().ToString("N");
        }
    }

using Microsoft.EntityFrameworkCore;
using Microsoft.EntityFrameworkCore.Diagnostics;
using MokFramework.Entities;
using MokFramework.Events;

namespace MokFramework.EntityFrameworkCore.Interceptors;

/// <summary>
/// 领域事件拦截器
/// 
/// 在 SaveChanges 成功后自动分发领域事件
/// 
/// 时机选择:
///   SavedChanges（而不是 SavingChanges）
///   因为领域事件的处理可能触发新的数据库操作
///   必须在当前事务提交后再触发
///   
/// ABP 的做法:
///   支持 pre-save 和 post-save 两种事件
///   我们简化为只支持 post-save（更安全）
/// </summary>
public class DomainEventInterceptor : SaveChangesInterceptor
{
    private readonly IDomainEventDispatcher _dispatcher;

    public DomainEventInterceptor(IDomainEventDispatcher dispatcher)
    {
        _dispatcher = dispatcher;
    }

    public override int SavedChanges(
        SaveChangesCompletedEventData eventData,
        int result)
    {
        if (eventData.Context != null)
        {
            DispatchDomainEventsAsync(eventData.Context)
                .GetAwaiter().GetResult();
        }

        return base.SavedChanges(eventData, result);
    }

    public override async ValueTask<int> SavedChangesAsync(
        SaveChangesCompletedEventData eventData,
        int result,
        CancellationToken cancellationToken = default)
    {
        if (eventData.Context != null)
        {
            await DispatchDomainEventsAsync(eventData.Context, cancellationToken);
        }

        return await base.SavedChangesAsync(eventData, result, cancellationToken);
    }

    private async Task DispatchDomainEventsAsync(
        DbContext context,
        CancellationToken cancellationToken = default)
    {
        // 收集所有聚合根的领域事件
        var aggregateRoots = context.ChangeTracker.Entries()
            .Where(e => e.Entity is IAggregateRoot)
            .Select(e => (IAggregateRoot)e.Entity)
            .Where(ar => ar.DomainEvents.Count > 0)
            .ToList();

        if (aggregateRoots.Count == 0)
            return;

        var allEvents = aggregateRoots
            .SelectMany(ar => ar.DomainEvents)
            .ToList();

        // 先清除事件（防止重复分发）
        foreach (var aggregateRoot in aggregateRoots)
        {
            aggregateRoot.ClearDomainEvents();
        }

        // 分发事件
        await _dispatcher.DispatchAsync(allEvents, cancellationToken);
    }
}

using Microsoft.Extensions.DependencyInjection;
using Microsoft.Extensions.Logging;
using MokFramework.Events;

namespace MokFramework.EntityFrameworkCore.Events;

/// <summary>
/// 领域事件分发器
/// 
/// 通过 DI 容器找到所有 IDomainEventHandler&lt;TEvent&gt; 并执行
/// 
/// 设计决策:
///   同步执行（在同一请求中）
///   如果处理器失败，记录日志但不影响主流程
///   如果需要严格一致性，使用 Outbox Pattern（未来扩展）
/// </summary>
public class DomainEventDispatcher : IDomainEventDispatcher
{
    private readonly IServiceProvider _serviceProvider;
    private readonly ILogger<DomainEventDispatcher> _logger;

    public DomainEventDispatcher(
        IServiceProvider serviceProvider,
        ILogger<DomainEventDispatcher> logger)
    {
        _serviceProvider = serviceProvider;
        _logger = logger;
    }

    public async Task DispatchAsync(
        IDomainEvent domainEvent,
        CancellationToken cancellationToken = default)
    {
        var eventType = domainEvent.GetType();
        var handlerType = typeof(IDomainEventHandler<>).MakeGenericType(eventType);

        // 获取所有处理器
        var handlers = _serviceProvider.GetServices(handlerType);

        foreach (var handler in handlers)
        {
            try
            {
                _logger.LogDebug(
                    "Dispatching domain event {EventType} to handler {HandlerType}",
                    eventType.Name, handler!.GetType().Name);

                // 反射调用 HandleAsync
                var method = handlerType.GetMethod(nameof(IDomainEventHandler<IDomainEvent>.HandleAsync));
                var task = (Task)method!.Invoke(handler, [domainEvent, cancellationToken])!;
                await task;
            }
            catch (Exception ex)
            {
                _logger.LogError(ex,
                    "Error handling domain event {EventType} in handler {HandlerType}",
                    eventType.Name, handler!.GetType().Name);

                // 不抛出，不影响主流程
                // 如果需要严格一致性，改为抛出或使用补偿机制
            }
        }
    }

    public async Task DispatchAsync(
        IEnumerable<IDomainEvent> domainEvents,
        CancellationToken cancellationToken = default)
    {
        foreach (var domainEvent in domainEvents)
        {
            await DispatchAsync(domainEvent, cancellationToken);
        }
    }
}

using System.Text.Json;
using Microsoft.EntityFrameworkCore;
using Microsoft.EntityFrameworkCore.ChangeTracking;
using Microsoft.EntityFrameworkCore.Diagnostics;
using MokFramework.EntityFrameworkCore.AuditLog;
using MokFramework.SoftDelete;

namespace MokFramework.EntityFrameworkCore.Interceptors;

/// <summary>
/// 审计日志拦截器
/// 
/// 记录所有实体的变更历史
/// 在 SavingChanges 时收集变更信息
/// 在 SavedChanges 时异步持久化
/// 
/// 和 AuditingInterceptor 的区别:
///   AuditingInterceptor: 填充审计字段（CreationTime 等）
///   AuditLogInterceptor: 记录变更日志（谁在什么时候改了什么）
/// </summary>
public class AuditLogInterceptor : SaveChangesInterceptor
{
    private readonly ICurrentUser _currentUser;
    private readonly IAuditLogStore _store;

    // 暂存变更信息（SavingChanges 收集，SavedChanges 存储）
    private List<EntityChangeInfo>? _pendingChanges;

    public AuditLogInterceptor(ICurrentUser currentUser, IAuditLogStore store)
    {
        _currentUser = currentUser;
        _store = store;
    }

    public override InterceptionResult<int> SavingChanges(
        DbContextEventData eventData,
        InterceptionResult<int> result)
    {
        if (eventData.Context != null)
            _pendingChanges = CollectChanges(eventData.Context);

        return base.SavingChanges(eventData, result);
    }

    public override ValueTask<InterceptionResult<int>> SavingChangesAsync(
        DbContextEventData eventData,
        InterceptionResult<int> result,
        CancellationToken cancellationToken = default)
    {
        if (eventData.Context != null)
            _pendingChanges = CollectChanges(eventData.Context);

        return base.SavingChangesAsync(eventData, result, cancellationToken);
    }

    public override int SavedChanges(
        SaveChangesCompletedEventData eventData, int result)
    {
        if (_pendingChanges is { Count: > 0 })
        {
            _store.SaveAsync(_pendingChanges).GetAwaiter().GetResult();
            _pendingChanges = null;
        }

        return base.SavedChanges(eventData, result);
    }

    public override async ValueTask<int> SavedChangesAsync(
        SaveChangesCompletedEventData eventData,
        int result,
        CancellationToken cancellationToken = default)
    {
        if (_pendingChanges is { Count: > 0 })
        {
            await _store.SaveAsync(_pendingChanges, cancellationToken);
            _pendingChanges = null;
        }

        return await base.SavedChangesAsync(eventData, result, cancellationToken);
    }

    private List<EntityChangeInfo> CollectChanges(DbContext context)
    {
        var changes = new List<EntityChangeInfo>();
        var now = DateTime.UtcNow;

        foreach (var entry in context.ChangeTracker.Entries())
        {
            if (entry.State is not (EntityState.Added
                or EntityState.Modified or EntityState.Deleted))
                continue;

            // 跳过没有实际变更的 Modified 状态
            if (entry.State == EntityState.Modified &&
                !entry.Properties.Any(p => p.IsModified))
                continue;

            var changeInfo = new EntityChangeInfo
            {
                ChangeTime = now,
                ChangeType = entry.State switch
                {
                    EntityState.Added => EntityChangeType.Created,
                    EntityState.Modified => GetActualChangeType(entry),
                    EntityState.Deleted => EntityChangeType.Deleted,
                    _ => EntityChangeType.Updated
                },
                EntityTypeFullName = entry.Entity.GetType().FullName!,
                EntityId = GetEntityId(entry),
                UserId = _currentUser.Id,
            };

            // 收集属性变更
            foreach (var property in entry.Properties)
            {
                if (ShouldSkipProperty(property))
                    continue;

                if (entry.State == EntityState.Added)
                {
                    changeInfo.PropertyChanges.Add(new EntityPropertyChangeInfo
                    {
                        PropertyName = property.Metadata.Name,
                        PropertyTypeFullName = property.Metadata.ClrType.FullName!,
                        OriginalValue = null,
                        NewValue = Serialize(property.CurrentValue),
                    });
                }
                else if (entry.State == EntityState.Deleted)
                {
                    changeInfo.PropertyChanges.Add(new EntityPropertyChangeInfo
                    {
                        PropertyName = property.Metadata.Name,
                        PropertyTypeFullName = property.Metadata.ClrType.FullName!,
                        OriginalValue = Serialize(property.OriginalValue),
                        NewValue = null,
                    });
                }
                else if (property.IsModified &&
                         !Equals(property.OriginalValue, property.CurrentValue))
                {
                    changeInfo.PropertyChanges.Add(new EntityPropertyChangeInfo
                    {
                        PropertyName = property.Metadata.Name,
                        PropertyTypeFullName = property.Metadata.ClrType.FullName!,
                        OriginalValue = Serialize(property.OriginalValue),
                        NewValue = Serialize(property.CurrentValue),
                    });
                }
            }

            if (changeInfo.PropertyChanges.Count > 0 ||
                entry.State is EntityState.Added or EntityState.Deleted)
            {
                changes.Add(changeInfo);
            }
        }

        return changes;
    }

    /// <summary>
    /// 判断实际变更类型（软删除的 Modified 其实是 Deleted）
    /// </summary>
    private static EntityChangeType GetActualChangeType(EntityEntry entry)
    {
        if (entry.Entity is ISoftDelete softDelete && softDelete.IsDeleted)
        {
            var isDeletedProp = entry.Property(nameof(ISoftDelete.IsDeleted));
            if (isDeletedProp.IsModified &&
                isDeletedProp.OriginalValue is false)
            {
                return EntityChangeType.Deleted; // 软删除记录为 Deleted
            }
        }

        return EntityChangeType.Updated;
    }

    /// <summary>
    /// 跳过不需要记录的属性
    /// </summary>
    private static bool ShouldSkipProperty(PropertyEntry property)
    {
        var name = property.Metadata.Name;

        // 审计字段本身不记录（它们是由拦截器填充的）
        return name is nameof(MokFramework.Auditing.IHasCreationTime.CreationTime)
            or nameof(MokFramework.Auditing.IMayHaveCreator.CreatorId)
            or nameof(MokFramework.Auditing.IHasModificationTime.LastModificationTime)
            or nameof(MokFramework.Auditing.IModificationAudited.LastModifierId)
            or nameof(MokFramework.Auditing.IHasDeletionTime.DeletionTime)
            or nameof(MokFramework.Auditing.IDeletionAudited.DeleterId)
            or nameof(MokFramework.Concurrency.IHasConcurrencyStamp.ConcurrencyStamp)
            or nameof(MokFramework.Data.IHasExtraProperties.ExtraProperties);
    }

    private static string? GetEntityId(EntityEntry entry)
    {
        var keyProperties = entry.Properties
            .Where(p => p.Metadata.IsPrimaryKey())
            .Select(p => p.CurrentValue?.ToString())
            .ToArray();

        return keyProperties.Length switch
        {
            0 => null,
            1 => keyProperties[0],
            _ => string.Join(",", keyProperties)
        };
    }

    private static string? Serialize(object? value)
    {
        if (value == null) return null;

        return value switch
        {
            string s => s,
            DateTime dt => dt.ToString("O"),
            DateTimeOffset dto => dto.ToString("O"),
            _ when value.GetType().IsPrimitive => value.ToString(),
            Guid g => g.ToString(),
            _ => JsonSerializer.Serialize(value)
        };
    }
}

/// <summary>
/// EF Core 通用仓储实现
/// 
/// 所有项目共享此实现
/// 不需要为每个实体写仓储类（除非有特殊查询）
/// 
/// 用法:
///   services.AddScoped(typeof(IRepository<,>), typeof(EfCoreRepository<,,>));
///

///   private readonly IRepository&lt;Order, Guid&gt; _orderRepo;
///   var order = await _orderRepo.GetAsync(id);
/// </summary>
public class EfCoreRepository<TDbContext, TEntity, TKey>
    : IRepository<TEntity, TKey>
    where TDbContext : DbContext
    where TEntity : class, IEntity<TKey>
    where TKey : notnull
{
    protected readonly TDbContext DbContext;
    protected readonly DbSet<TEntity> DbSet;

    public EfCoreRepository(TDbContext dbContext)
    {
        DbContext = dbContext;
        DbSet = dbContext.Set<TEntity>();
    }

    #region 查询

    public virtual async Task<TEntity?> FindAsync(
        TKey id, CancellationToken cancellationToken = default)
    {

    }

    public virtual async Task<TEntity> GetAsync(

    {

        return entity ?? throw new EntityNotFoundException(typeof(TEntity), id);
    }

    public virtual async Task<List<TEntity>> GetListAsync(
        CancellationToken cancellationToken = default)
    {
        return await DbSet.ToListAsync(cancellationToken);
    }

    public virtual async Task<List<TEntity>> GetListAsync(
        Expression<Func<TEntity, bool>> predicate,
        CancellationToken cancellationToken = default)
    {
        return await DbSet.Where(predicate).ToListAsync(cancellationToken);
    }

    public virtual async Task<long> GetCountAsync(
        CancellationToken cancellationToken = default)
    {
        return await DbSet.LongCountAsync(cancellationToken);
    }

    public virtual async Task<long> GetCountAsync(
        Expression<Func<TEntity, bool>> predicate,
        CancellationToken cancellationToken = default)
    {
        return await DbSet.LongCountAsync(predicate, cancellationToken);
    }

    public virtual async Task<bool> AnyAsync(
        Expression<Func<TEntity, bool>> predicate,
        CancellationToken cancellationToken = default)
    {
        return await DbSet.AnyAsync(predicate, cancellationToken);
    }

    public virtual IQueryable<TEntity> AsQueryable()
    {
        return DbSet.AsQueryable();
    }

    public virtual async Task<List<TEntity>> GetPagedListAsync(
        int skipCount,
        int maxResultCount,
        Expression<Func<TEntity, bool>>? predicate = null,
        string? sorting = null,
        CancellationToken cancellationToken = default)
    {
        var query = DbSet.AsQueryable();

        if (predicate != null)
            query = query.Where(predicate);

        query = ApplySorting(query, sorting);

        return await query
            .Skip(skipCount)
            .Take(maxResultCount)
            .ToListAsync(cancellationToken);
    }

    #endregion

    #region 增删改

    public virtual async Task<TEntity> InsertAsync(
        TEntity entity, CancellationToken cancellationToken = default)
    {
        var entry = await DbSet.AddAsync(entity, cancellationToken);
        return entry.Entity;
    }

    public virtual async Task InsertManyAsync(
        IEnumerable<TEntity> entities,
        CancellationToken cancellationToken = default)
    {
        await DbSet.AddRangeAsync(entities, cancellationToken);
    }

    public virtual Task<TEntity> UpdateAsync(
        TEntity entity, CancellationToken cancellationToken = default)
    {
        DbContext.Attach(entity);
        DbSet.Update(entity);
        return Task.FromResult(entity);
    }

    public virtual Task UpdateManyAsync(
        IEnumerable<TEntity> entities,
        CancellationToken cancellationToken = default)
    {
        DbSet.UpdateRange(entities);
        return Task.CompletedTask;
    }

    public virtual Task DeleteAsync(
        TEntity entity, CancellationToken cancellationToken = default)
    {
        DbSet.Remove(entity);
        return Task.CompletedTask;
    }

    public virtual async Task DeleteAsync(
        TKey id, CancellationToken cancellationToken = default)
    {
        var entity = await GetAsync(id, cancellationToken);
        await DeleteAsync(entity, cancellationToken);
    }

    public virtual async Task DeleteManyAsync(
        IEnumerable<TKey> ids,
        CancellationToken cancellationToken = default)
    {
        var entities = await DbSet
            .Where(e => ids.Contains(e.Id))
            .ToListAsync(cancellationToken);

        DbSet.RemoveRange(entities);
    }

    public virtual async Task DeleteAsync(
        Expression<Func<TEntity, bool>> predicate,
        CancellationToken cancellationToken = default)
    {
        var entities = await DbSet
            .Where(predicate)
            .ToListAsync(cancellationToken);

        DbSet.RemoveRange(entities);
    }

    #endregion

    #region 排序

    /// <summary>
    /// 动态排序
    /// 支持格式: "Name ASC" / "CreationTime DESC" / "Name ASC, CreationTime DESC"
    /// </summary>
    protected virtual IQueryable<TEntity> ApplySorting(
        IQueryable<TEntity> query, string? sorting)
    {
        if (string.IsNullOrWhiteSpace(sorting))
        {
            return ApplyDefaultSorting(query);
        }

        var parts = sorting.Split(',', StringSplitOptions.RemoveEmptyEntries);
        IOrderedQueryable<TEntity>? orderedQuery = null;

        foreach (var part in parts)
        {
            var trimmed = part.Trim();
            var segments = trimmed.Split(' ', StringSplitOptions.RemoveEmptyEntries);
            var propertyName = segments[0];
            var descending = segments.Length > 1 &&
                             segments[1].Equals("DESC", StringComparison.OrdinalIgnoreCase);

            var parameter = Expression.Parameter(typeof(TEntity), "e");
            var property = Expression.Property(parameter, propertyName);
            var lambda = Expression.Lambda(property, parameter);

            var methodName = orderedQuery == null
                ? (descending ? "OrderByDescending" : "OrderBy")
                : (descending ? "ThenByDescending" : "ThenBy");

            var method = typeof(Queryable).GetMethods()
                .First(m => m.Name == methodName && m.GetParameters().Length == 2)
                .MakeGenericMethod(typeof(TEntity), property.Type);

            orderedQuery = (IOrderedQueryable<TEntity>)method.Invoke(
                null, [orderedQuery ?? query, lambda])!;
        }

        return orderedQuery ?? query;
    }

    /// <summary>
    /// 默认排序（子类可覆盖）
    /// </summary>
    protected virtual IQueryable<TEntity> ApplyDefaultSorting(IQueryable<TEntity> query)
    {
        // 默认按 Id 排序（如果有的话）
        return query;
    }

    #endregion
}

    }

    public async Task CommitTransactionAsync(
        CancellationToken cancellationToken = default)
    {
        if (_transaction == null)
            throw new InvalidOperationException("事务未开始");

        try
        {
            await _dbContext.SaveChangesAsync(cancellationToken);
            await _transaction.CommitAsync(cancellationToken);
        }
        catch
        {
            await RollbackTransactionAsync(cancellationToken);
            throw;
        }
        finally
        {
            await _transaction.DisposeAsync();
            _transaction = null;
        }
    }

    public async Task RollbackTransactionAsync(
        CancellationToken cancellationToken = default)
    {
        if (_transaction == null) return;

        try
        {
            await _transaction.RollbackAsync(cancellationToken);
        }
        finally
        {
            await _transaction.DisposeAsync();
            _transaction = null;
        }
    }
}

using Microsoft.EntityFrameworkCore;

namespace MokFramework.EntityFrameworkCore.Extensions;

/// <summary>
/// ModelBuilder 扩展
/// </summary>
public static class ModelBuilderExtensions
{
    /// <summary>
    /// 统一设置所有 DateTime 类型的值转换器
    /// 确保存入 UTC，读出标记 UTC Kind
    /// </summary>
    public static ModelBuilder UseUtcDateTime(this ModelBuilder modelBuilder)
    {
        var utcConverter = new ValueConverters.DateTimeUtcConverter();
        var nullableUtcConverter = new ValueConverters.NullableDateTimeUtcConverter();

        foreach (var entityType in modelBuilder.Model.GetEntityTypes())
        {
            foreach (var property in entityType.GetProperties())
            {
                if (property.ClrType == typeof(DateTime))
                {
                    property.SetValueConverter(utcConverter);
                }
                else if (property.ClrType == typeof(DateTime?))
                {
                    property.SetValueConverter(nullableUtcConverter);
                }
            }
        }

        return modelBuilder;
    }

    /// <summary>
    /// 统一设置表名前缀
    /// 例: builder.UseTablePrefix("Sso") → SsoUsers, SsoRoles...
    /// </summary>
    public static ModelBuilder UseTablePrefix(
        this ModelBuilder modelBuilder, string prefix)
    {
        foreach (var entityType in modelBuilder.Model.GetEntityTypes())
        {
            var tableName = entityType.GetTableName();
            if (tableName != null && !tableName.StartsWith(prefix))
            {
                entityType.SetTableName($"{prefix}{tableName}");
            }
        }

        return modelBuilder;
    }

    /// <summary>
    /// 统一设置 string 类型的默认最大长度
    /// 防止 nvarchar(max) 性能问题
    /// </summary>
    public static ModelBuilder UseDefaultStringMaxLength(
        this ModelBuilder modelBuilder, int maxLength = 256)
    {
        foreach (var entityType in modelBuilder.Model.GetEntityTypes())
        {
            foreach (var property in entityType.GetProperties())
            {
                if (property.ClrType == typeof(string) &&
                    property.GetMaxLength() == null)
                {
                    property.SetMaxLength(maxLength);
                }
            }
        }

        return modelBuilder;
    }
}

using Microsoft.EntityFrameworkCore;
using Microsoft.Extensions.DependencyInjection;
using MokFramework.EntityFrameworkCore.AuditLog;

using MokFramework.EntityFrameworkCore.Repositories;

namespace MokFramework.EntityFrameworkCore.Extensions;

/// <summary>
/// 平台 EF Core 层 DI 注册
/// 
/// 用法（SSO 项目 Program.cs）:
/// 
///   builder.Services.AddPlatformDbContext&lt;SsoDbContext&gt;(options =>
///       options.UseNpgsql(connectionString));
/// </summary>
public static class ServiceCollectionExtensions
{
    /// <summary>
    /// 注册平台 DbContext 及所有基础设施
    /// </summary>
    public static IServiceCollection AddPlatformDbContext<TDbContext>(
        this IServiceCollection services,
        Action<DbContextOptionsBuilder> optionsAction,
        Action<PlatformDbOptions>? configureOptions = null)
        where TDbContext : DbContext
    {
        var options = new PlatformDbOptions();
        configureOptions?.Invoke(options);

        // 注册 CurrentUser（如果未注册）
        services.TryAddScoped<ICurrentUser, NullCurrentUser>();

        // 注册拦截器
        services.AddScoped<AuditingInterceptor>();
        services.AddScoped<SoftDeleteInterceptor>();
        services.AddScoped<ConcurrencyStampInterceptor>();
        services.AddScoped<DomainEventInterceptor>();

        if (options.EnableAuditLog)
        {
            services.AddScoped<AuditLogInterceptor>();
            services.TryAddScoped<IAuditLogStore, DefaultAuditLogStore>();
        }

        // 注册 DbContext
        services.AddDbContext<TDbContext>((sp, dbOptions) =>
        {
            optionsAction(dbOptions);

            // 添加拦截器
            dbOptions.AddInterceptors(
                sp.GetRequiredService<AuditingInterceptor>(),
                sp.GetRequiredService<SoftDeleteInterceptor>(),
                sp.GetRequiredService<ConcurrencyStampInterceptor>(),
                sp.GetRequiredService<DomainEventInterceptor>());

            if (options.EnableAuditLog)
            {
                dbOptions.AddInterceptors(
                    sp.GetRequiredService<AuditLogInterceptor>());
            }
        });

        // 注册通用仓储
        services.AddScoped(
            typeof(IRepository<,>),
            typeof(EfCoreRepository<TDbContext,,>).MakeGenericType(
                typeof(TDbContext)));
        // 注意: 上面的写法在编译时不行，实际用下面的方式
        RegisterRepositories<TDbContext>(services);

        // 注册 UnitOfWork
        services.AddScoped<IUnitOfWork, EfCoreUnitOfWork<TDbContext>>();

        // 注册领域事件分发器
        services.TryAddScoped<IDomainEventDispatcher, DomainEventDispatcher>();

        // 注册 GUID 生成器
        services.TryAddSingleton<IGuidGenerator>(
            new SequentialGuidGenerator(options.SequentialGuidType));

        return services;
    }

    private static void RegisterRepositories<TDbContext>(
        IServiceCollection services)
        where TDbContext : DbContext
    {
        // 注册开放泛型仓储
        // IRepository<TEntity, TKey> → EfCoreRepository<TDbContext, TEntity, TKey>
        services.AddScoped(typeof(IRepository<,>), sp =>
        {
            // 这需要特殊处理，用工厂方式
            throw new InvalidOperationException(
                "请使用 AddRepository<TEntity, TKey>() 注册具体仓储，" +
                "或使用 AddPlatformRepositories<TDbContext>() 批量注册");
        });
    }

    /// <summary>
    /// 注册具体实体的仓储
    /// </summary>
    public static IServiceCollection AddRepository<TDbContext, TEntity, TKey>(
        this IServiceCollection services)
        where TDbContext : DbContext
        where TEntity : class, IEntity<TKey>
        where TKey : notnull
    {
        services.AddScoped<IRepository<TEntity, TKey>,
            EfCoreRepository<TDbContext, TEntity, TKey>>();

        services.AddScoped<IReadOnlyRepository<TEntity, TKey>,
            EfCoreRepository<TDbContext, TEntity, TKey>>();

        services.AddScoped<IBasicRepository<TEntity, TKey>,
            EfCoreRepository<TDbContext, TEntity, TKey>>();

        return services;
    }
}

// ============================================
// 这个文件放在各 Host 项目中（不在 MokFramework.EFCore 中）
// 因为 MokFramework.EFCore 不依赖 ASP.NET Core
// 
// 例: SSO.Host/Infrastructure/HttpContextCurrentUser.cs
// 或者单独做一个 MokFramework.AspNetCore 包
// ============================================

using System.Security.Claims;
using Microsoft.AspNetCore.Http;
using MokFramework.EntityFrameworkCore;

namespace MokFramework.AspNetCore;

/// <summary>
/// 从 HttpContext 获取当前用户
/// ASP.NET Core 项目使用此实现
/// </summary>
public class HttpContextCurrentUser : ICurrentUser
{
    private readonly IHttpContextAccessor _accessor;

    public HttpContextCurrentUser(IHttpContextAccessor accessor)
    {
        _accessor = accessor;
    }

    public Guid? Id
    {
        get
        {
            var sub = _accessor.HttpContext?.User
                .FindFirst(ClaimTypes.NameIdentifier)?.Value
                ?? _accessor.HttpContext?.User
                    .FindFirst("sub")?.Value;

            return sub != null && Guid.TryParse(sub, out var id)
                ? id
                : null;
        }
    }

    public Guid? TenantId
    {
        get
        {
            var tenantId = _accessor.HttpContext?.User
                .FindFirst("tenant_id")?.Value;

            return tenantId != null && Guid.TryParse(tenantId, out var id)
                ? id
                : null;
        }
    }

    public bool IsAuthenticated =>
        _accessor.HttpContext?.User.Identity?.IsAuthenticated ?? false;
}

拦截器执行顺序

SaveChangesAsync 被调用:

  ┌──────────────────────────────────────────────────┐
  │  SavingChanges 阶段（事务提交前）                  │
  │                                                  │
  │  1. AuditLogInterceptor.SavingChanges            │
  │     → 收集变更前的快照                             │
  │                                                  │
  │  2. SoftDeleteInterceptor.SavingChanges           │
  │     → Deleted → Modified (IsDeleted=true)        │
  │     → 填充 DeletionTime / DeleterId              │
  │                                                  │
  │  3. AuditingInterceptor.SavingChanges             │
  │     → 填充 CreationTime / CreatorId              │
  │     → 填充 LastModificationTime / LastModifierId │
  │                                                  │
  │  4. ConcurrencyStampInterceptor.SavingChanges     │
  │     → 刷新 ConcurrencyStamp                      │
  │                                                  │
  │  ── EF Core 执行 SQL → 数据库提交 ──             │
  │                                                  │
  └──────────────────────────────────────────────────┘
  
  ┌──────────────────────────────────────────────────┐
  │  SavedChanges 阶段（事务提交后）                   │
  │                                                  │
  │  5. AuditLogInterceptor.SavedChanges              │
  │     → 持久化变更日志                              │
  │                                                  │
  │  6. DomainEventInterceptor.SavedChanges           │
  │     → 收集领域事件                                │
  │     → 清除事件                                    │
  │     → 分发给处理器                                │
  │                                                  │
  └──────────────────────────────────────────────────┘
