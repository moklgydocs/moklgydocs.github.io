中间件测试

中间件测试

中间件是请求管道的核心组件，任何 bug 都可能影响所有请求。系统化的测试是保障中间件可靠性的关键。本章将介绍从单元测试到集成测试的完整方案。

一、为什么要测试中间件

1.1 中间件的特殊性

中间件不同于普通的服务类，它具有以下特征：

  • 与 HttpContext 紧密耦合：请求和响应的处理高度依赖 HttpContext 的状态
  • 执行顺序敏感：中间件的注册顺序直接决定行为
  • 副作用范围大：一个中间件的 bug 可能影响整个管道
  • 异步执行：请求在中间件之间异步传递，容易引入竞态条件

1.2 常见中间件 Bug

Bug 类型典型表现严重程度
顺序错误认证中间件在授权之后执行高 — 授权永远失败
响应体读取后未恢复后续中间件读到空 Body高 — 请求处理异常
异常未处理未 catch 的异常导致 500高 — 用户体验差
Response 已开始仍写入InvalidOperationException中 — 日志刷屏
Header 写入时机错误状态码已发送后写 Header中 — 运行时异常
内存泄漏缓存无限增长低(初期) — 逐渐恶化

1.3 测试金字塔

        ╱  集成测试  ╲          ← 少量，验证中间件在完整管道中的行为
       ╱──────────────╲
      ╱   组件测试      ╲        ← 适量，验证中间件与相邻组件的交互
     ╱──────────────────╲
    ╱     单元测试         ╲     ← 大量，验证中间件的核心逻辑
   ╱────────────────────────╲

public class RequestTimingMiddlewareTests : IAsyncLifetime
{
    private HttpClient _client = null!;
    private TestServer _server = null!;

    public Task InitializeAsync()
    {
        var builder = new WebHostBuilder()
            .ConfigureServices(services =>
            {
                services.AddSingleton<ILogger<RequestTimingMiddleware>>(
                    LoggerFactory.Create(b => b.AddDebug())
                        .CreateLogger<RequestTimingMiddleware>());
            })
            .Configure(app =>
            {
                app.UseMiddleware<RequestTimingMiddleware>();
                app.Run(async context =>
                {
                    await context.Response.WriteAsync("OK");
                });
            });

        _server = new TestServer(builder);
        _client = _server.CreateClient();
        return Task.CompletedTask;
    }

    public Task DisposeAsync()
    {
        _client.Dispose();
        _server.Dispose();
        return Task.CompletedTask;
    }
}

4.2 设置 Request/Response 属性

public static class HttpContextHelper
{
    /// <summary>
    /// 创建一个可用于测试的 HttpContext
    /// </summary>
    public static HttpContext CreateTestContext(
        string method = "GET",
        string path = "/",
        string? body = null,
        Dictionary<string, string>? headers = null)
    {
        var context = new DefaultHttpContext();

        // 请求属性
        context.Request.Method = method;
        context.Request.Path = path;
        context.Request.Scheme = "https";
        context.Request.Host = new HostString("localhost", 5001);

        // 请求体
        if (body != null)
        {
            context.Request.Body = new MemoryStream(Encoding.UTF8.GetBytes(body));
            context.Request.ContentType = "application/json";
            context.Request.ContentLength = body.Length;
        }

        // 请求头
        if (headers != null)
        {
            foreach (var (key, value) in headers)
            {
                context.Request.Headers[key] = value;
            }
        }

        // 响应体（用于后续读取响应内容）
        context.Response.Body = new MemoryStream();

        return context;
    }

    /// <summary>
    /// 从响应 Body 中读取内容
    /// </summary>
    public static string GetResponseBody(HttpContext context)
    {
        context.Response.Body.Seek(0, SeekOrigin.Begin);
        using var reader = new StreamReader(context.Response.Body);
        return reader.ReadToEnd();
    }
}

4.3 Mock RequestDelegate（_next）

RequestDelegate 是一个委托，可以用 Mock 也可以直接用匿名方法：

// 方式1：使用 Moq
var mockNext = new Mock<RequestDelegate>();
mockNext
    .Setup(n => n(It.IsAny<HttpContext>()))
    .Returns(Task.CompletedTask);

var middleware = new MyMiddleware(mockNext.Object);

// 验证 next 是否被调用
mockNext.Verify(n => n(context), Times.Once);

// 方式2：使用匿名委托
var nextCalled = false;
RequestDelegate next = ctx =>
{
    nextCalled = true;
    return Task.CompletedTask;
};

var middleware = new MyMiddleware(next);
await middleware.InvokeAsync(context);

Assert.True(nextCalled);

// 方式3：模拟 next 修改响应
RequestDelegate next = async ctx =>
{
    ctx.Response.StatusCode = 200;
    await ctx.Response.WriteAsync("Hello from next");
};

    [Fact]

public class RateLimitMiddlewareBoundaryTests
{
    [Fact]
    public async Task ExactlyAtLimit_StillAllowed()
    {
        // 刚好在限制上，应该允许
        var options = new RateLimitOptions { MaxRequests = 5 };
        var middleware = new RateLimitMiddleware(
            Mock.Of<RequestDelegate>(), Options.Create(options));

        for (int i = 0; i < 5; i++)
        {
            var context = HttpContextHelper.CreateTestContext(path: "/api/data");
            await middleware.InvokeAsync(context);
            Assert.NotEqual(429, context.Response.StatusCode);
        }
    }

    [Fact]
    public async Task OneOverLimit_Returns429()
    {
        // 超过限制一个，应该返回 429
        var options = new RateLimitOptions { MaxRequests = 5 };
        var middleware = new RateLimitMiddleware(
            Mock.Of<RequestDelegate>(), Options.Create(options));

        // 先发送 5 个合法请求
        for (int i = 0; i < 5; i++)
        {
            var context = HttpContextHelper.CreateTestContext(path: "/api/data");
            await middleware.InvokeAsync(context);
        }

        // 第 6 个请求应该被限流
        var exceededContext = HttpContextHelper.CreateTestContext(path: "/api/data");
        exceededContext.Response.Body = new MemoryStream();
        await middleware.InvokeAsync(exceededContext);
        Assert.Equal(429, exceededContext.Response.StatusCode);
    }

    [Fact]
    public async Task WindowResets_AllowsNewRequests()
    {
        // 时间窗口重置后，应该允许新请求
        var options = new RateLimitOptions
        {
            MaxRequests = 2,
            WindowSeconds = 1
        };
        var mockTimeProvider = new Mock<ITimeProvider>();
        var now = DateTime.UtcNow;
        mockTimeProvider.Setup(t => t.UtcNow).Returns(now);

        var middleware = new RateLimitMiddleware(
            Mock.Of<RequestDelegate>(),
            Options.Create(options),
            mockTimeProvider.Object);

        // 发送 2 个请求
        for (int i = 0; i < 2; i++)
        {
            var context = HttpContextHelper.CreateTestContext(path: "/api/data");
            await middleware.InvokeAsync(context);
        }

        // 模拟时间过了窗口期
        mockTimeProvider.Setup(t => t.UtcNow)
            .Returns(now.AddSeconds(2));

        // 新的请求应该被允许
        var newContext = HttpContextHelper.CreateTestContext(path: "/api/data");

        Assert.NotEqual(429, newContext.Response.StatusCode);

6.5 测试配置变更

public class ConfigurableMiddlewareTests
{
    [Fact]
    public async Task Enabled_False_SkipsMiddleware()
    {
        // Arrange
        var options = new FeatureToggleOptions { Enabled = false };
        var mockNext = new Mock<RequestDelegate>();
        var middleware = new FeatureToggleMiddleware(
            mockNext.Object, Options.Create(options));
        var context = new DefaultHttpContext();

        // Act
        await middleware.InvokeAsync(context);

        // Assert — 中间件被跳过，直接调用 next
        mockNext.Verify(n => n(context), Times.Once);
    }

    [Fact]
    public async Task Enabled_True_ExecutesMiddleware()
    {
        // Arrange
        var options = new FeatureToggleOptions { Enabled = true };
        var mockNext = new Mock<RequestDelegate>();
        var middleware = new FeatureToggleMiddleware(
            mockNext.Object, Options.Create(options));
        var context = new DefaultHttpContext();
        context.Response.Body = new MemoryStream();

        // Act
        await middleware.InvokeAsync(context);

        // Assert — 中间件逻辑被执行
        Assert.True(context.Items.ContainsKey("FeatureProcessed"));
    }

    [Fact]
    public async Task DifferentConfigs_DifferentBehavior()
    {
        // 同一个中间件，不同配置，行为不同
        var strictOptions = new ValidationOptions
        {
            Mode = ValidationMode.Strict,
            AllowEmptyBody = false
        };
        var looseOptions = new ValidationOptions
        {
            Mode = ValidationMode.Loose,
            AllowEmptyBody = true
        };

        var context1 = HttpContextHelper.CreateTestContext(
            method: "POST",
            path: "/api/data");
        context1.Response.Body = new MemoryStream();

        var context2 = HttpContextHelper.CreateTestContext(
            method: "POST",
            path: "/api/data");
        context2.Response.Body = new MemoryStream();

        var strictMiddleware = new ValidationMiddleware(
            Mock.Of<RequestDelegate>(), Options.Create(strictOptions));
        var looseMiddleware = new ValidationMiddleware(
            Mock.Of<RequestDelegate>(), Options.Create(looseOptions));

        await strictMiddleware.InvokeAsync(context1);
        await looseMiddleware.InvokeAsync(context2);

        // Strict 模式拒绝空 Body
        Assert.Equal(400, context1.Response.StatusCode);
        // Loose 模式允许空 Body
        Assert.NotEqual(400, context2.Response.StatusCode);
    }
}

小结

  • 中间件测试分为单元测试和集成测试两个层次，单元测试使用 DefaultHttpContext + Mock RequestDelegate，集成测试使用 TestServer 或 WebApplicationFactory
  • 每个中间件应独立测试类，覆盖正常路径、异常路径和边界条件
  • 使用 HttpContextHelper 等工具类减少测试样板代码
  • 测试中间件顺序时，使用 TestServer 构建不同顺序的管道
  • 配置选项变更应通过参数化测试验证不同配置下的行为差异
