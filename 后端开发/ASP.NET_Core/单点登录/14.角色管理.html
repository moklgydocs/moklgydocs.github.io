2. RoleAppService

// Sso.Application/Services/RoleAppService.cs
using Microsoft.AspNetCore.Identity;
using Microsoft.EntityFrameworkCore;
using Sso.Application.Contracts.Roles;
using Sso.Domain.Entities;
using Sso.Infrastructure.Data;
using Sso.Shared.Paging;

namespace Sso.Application.Services;

public class RoleAppService
{
    private readonly RoleManager<SsoRole> _roleManager;
    private readonly UserManager<SsoUser> _userManager;
    private readonly SsoDbContext _dbContext;

    public RoleAppService(
        RoleManager<SsoRole> roleManager,
        UserManager<SsoUser> userManager,
        SsoDbContext dbContext)
    {
        _roleManager = roleManager;
        _userManager = userManager;
        _dbContext = dbContext;
    }

    // ============================
    // 基础 CRUD
    // ============================

    /// <summary>
    /// 创建角色
    /// </summary>
    public async Task<RoleDto> CreateAsync(CreateRoleRequest request)
    {
        if (await _roleManager.RoleExistsAsync(request.Name))
            throw new InvalidOperationException($"角色 '{request.Name}' 已存在");

        var role = new SsoRole
        {
            Name = request.Name,
            Description = request.Description,
            IsSystem = request.IsSystem,
            CreatedAt = DateTime.UtcNow
        };

        var result = await _roleManager.CreateAsync(role);
        if (!result.Succeeded)
        {
            var errors = string.Join("; ", result.Errors.Select(e => e.Description));
            throw new InvalidOperationException($"创建角色失败: {errors}");
        }

        return await MapToDtoAsync(role);
    }

    /// <summary>
    /// 更新角色
    /// </summary>
    public async Task<RoleDto> UpdateAsync(Guid id, UpdateRoleRequest request)
    {
        var role = await _roleManager.FindByIdAsync(id.ToString())
            ?? throw new KeyNotFoundException($"角色 '{id}' 不存在");

        if (role.IsSystem)
            throw new InvalidOperationException("系统角色不允许修改");

        if (request.Description != null)
            role.Description = request.Description;

        var result = await _roleManager.UpdateAsync(role);
        if (!result.Succeeded)
        {
            var errors = string.Join("; ", result.Errors.Select(e => e.Description));
            throw new InvalidOperationException($"更新角色失败: {errors}");
        }

        return await MapToDtoAsync(role);
    }

    /// <summary>
    /// 删除角色
    /// </summary>
    public async Task DeleteAsync(Guid id)
    {
        var role = await _roleManager.FindByIdAsync(id.ToString())
            ?? throw new KeyNotFoundException($"角色 '{id}' 不存在");

        if (role.IsSystem)
            throw new InvalidOperationException("系统角色不允许删除");

        // 检查是否还有用户在使用
        var usersInRole = await _userManager.GetUsersInRoleAsync(role.Name!);
        if (usersInRole.Any())
            throw new InvalidOperationException(
                $"角色 '{role.Name}' 下还有 {usersInRole.Count} 个用户，请先移除用户后再删除");

        var result = await _roleManager.DeleteAsync(role);
        if (!result.Succeeded)
        {
            var errors = string.Join("; ", result.Errors.Select(e => e.Description));
            throw new InvalidOperationException($"删除角色失败: {errors}");
        }
    }

    /// <summary>
    /// 获取单个角色
    /// </summary>
    public async Task<RoleDto?> GetByIdAsync(Guid id)
    {
        var role = await _roleManager.FindByIdAsync(id.ToString());
        return role == null ? null : await MapToDtoAsync(role);
    }

    /// <summary>
    /// 根据名称获取角色
    /// </summary>
    public async Task<RoleDto?> GetByNameAsync(string name)
    {
        var role = await _roleManager.FindByNameAsync(name);
        return role == null ? null : await MapToDtoAsync(role);
    }

    /// <summary>
    /// 分页查询
    /// </summary>
    public async Task<PagedResult<RoleDto>> GetPagedAsync(RoleQueryRequest request)
    {
        var query = _roleManager.Roles.AsNoTracking();

        // 关键字搜索
        if (!string.IsNullOrWhiteSpace(request.Keyword))
        {
            var keyword = request.Keyword.Trim().ToLower();
            query = query.Where(r =>
                r.Name!.ToLower().Contains(keyword) ||
                (r.Description != null && r.Description.ToLower().Contains(keyword)));
        }

        // 排序
        query = request.OrderBy?.ToLower() switch
        {
            "name" => request.IsDesc
                ? query.OrderByDescending(r => r.Name)
                : query.OrderBy(r => r.Name),
            "createdat" => request.IsDesc
                ? query.OrderByDescending(r => r.CreatedAt)
                : query.OrderBy(r => r.CreatedAt),
            _ => query.OrderByDescending(r => r.CreatedAt)
        };

        var total = await query.CountAsync();
        var roles = await query
            .Skip((request.Page - 1) * request.PageSize)
            .Take(request.PageSize)
            .ToListAsync();

        var dtos = new List<RoleDto>();
        foreach (var role in roles)
        {
            dtos.Add(await MapToDtoAsync(role));
        }

        return new PagedResult<RoleDto>(dtos, total, request.Page, request.PageSize);
    }

    /// <summary>
    /// 获取所有角色（下拉选择用）
    /// </summary>
    public async Task<List<RoleDto>> GetAllAsync()
    {
        var roles = await _roleManager.Roles
            .AsNoTracking()
            .OrderBy(r => r.Name)
            .ToListAsync();

        var dtos = new List<RoleDto>();
        foreach (var role in roles)
        {
            dtos.Add(await MapToDtoAsync(role));
        }
        return dtos;
    }

    // ============================
    // 角色-用户关联
    // ============================

    /// <summary>
    /// 获取角色下的所有用户
    /// </summary>
    public async Task<List<RoleUserDto>> GetUsersInRoleAsync(Guid roleId)
    {
        var role = await _roleManager.FindByIdAsync(roleId.ToString())
            ?? throw new KeyNotFoundException($"角色 '{roleId}' 不存在");

        var users = await _userManager.GetUsersInRoleAsync(role.Name!);

        return users.Select(u => new RoleUserDto
        {
            Id = u.Id,
            UserName = u.UserName!,
            DisplayName = u.DisplayName,
            Email = u.Email,
            IsActive = u.IsActive
        }).ToList();
    }

    /// <summary>
    /// 批量添加用户到角色
    /// </summary>
    public async Task AddUsersToRoleAsync(Guid roleId, AssignUsersToRoleRequest request)
    {
        var role = await _roleManager.FindByIdAsync(roleId.ToString())
            ?? throw new KeyNotFoundException($"角色 '{roleId}' 不存在");

        var errors = new List<string>();

        foreach (var userId in request.UserIds)
        {
            var user = await _userManager.FindByIdAsync(userId.ToString());
            if (user == null)
            {
                errors.Add($"用户 '{userId}' 不存在");
                continue;
            }

            if (await _userManager.IsInRoleAsync(user, role.Name!))
                continue; // 已在角色中，跳过

            var result = await _userManager.AddToRoleAsync(user, role.Name!);
            if (!result.Succeeded)
            {
                errors.Add($"用户 '{user.UserName}' 分配失败: " +
                           string.Join(", ", result.Errors.Select(e => e.Description)));
            }
        }

        if (errors.Any())
            throw new InvalidOperationException(
                $"部分用户分配失败: {string.Join("; ", errors)}");
    }

    /// <summary>
    /// 批量从角色移除用户
    /// </summary>
    public async Task RemoveUsersFromRoleAsync(Guid roleId, AssignUsersToRoleRequest request)
    {
        var role = await _roleManager.FindByIdAsync(roleId.ToString())
            ?? throw new KeyNotFoundException($"角色 '{roleId}' 不存在");

        var errors = new List<string>();

        foreach (var userId in request.UserIds)
        {
            var user = await _userManager.FindByIdAsync(userId.ToString());
            if (user == null)
            {
                errors.Add($"用户 '{userId}' 不存在");
                continue;
            }

            if (!await _userManager.IsInRoleAsync(user, role.Name!))
                continue; // 不在角色中，跳过

            var result = await _userManager.RemoveFromRoleAsync(user, role.Name!);
            if (!result.Succeeded)
            {
                errors.Add($"用户 '{user.UserName}' 移除失败: " +
                           string.Join(", ", result.Errors.Select(e => e.Description)));
            }
        }

        if (errors.Any())
            throw new InvalidOperationException(
                $"部分用户移除失败: {string.Join("; ", errors)}");
    }

    // ============================
    // 私有方法
    // ============================

    private async Task<RoleDto> MapToDtoAsync(SsoRole role)
    {
        // 统计角色下的用户数
        var userCount = await _dbContext.UserRoles
            .CountAsync(ur => ur.RoleId == role.Id);

        return new RoleDto
        {
            Id = role.Id,
            Name = role.Name!,
            Description = role.Description,
            IsSystem = role.IsSystem,
            CreatedAt = role.CreatedAt,
            UserCount = userCount
        };
    }
}

// Sso.AuthServer/Controllers/RoleManageController.cs
using Microsoft.AspNetCore.Authorization;
using Microsoft.AspNetCore.Mvc;
using Sso.Application.Contracts.Roles;
using Sso.Application.Services;
using Sso.Shared.Results;

namespace Sso.AuthServer.Controllers;

[ApiController]
[Route("api/manage/roles")]
[Authorize(Roles = "admin")]
public class RoleManageController : ControllerBase
{
    private readonly RoleAppService _roleAppService;

    public RoleManageController(RoleAppService roleAppService)
    {
        _roleAppService = roleAppService;
    }

    // ============================
    // 基础 CRUD
    // ============================

    /// <summary>
    /// 创建角色
    /// </summary>
    [HttpPost]
    public async Task<IActionResult> Create([FromBody] CreateRoleRequest request)
    {
        try
        {
            var role = await _roleAppService.CreateAsync(request);
            return Ok(ApiResult<RoleDto>.Ok(role, "创建成功"));
        }
        catch (InvalidOperationException ex)
        {
            return BadRequest(ApiResult.Fail(ex.Message));
        }
    }

    /// <summary>
    /// 更新角色
    /// </summary>
    [HttpPut("{id:guid}")]
    public async Task<IActionResult> Update(Guid id, [FromBody] UpdateRoleRequest request)
    {
        try
        {
            var role = await _roleAppService.UpdateAsync(id, request);
            return Ok(ApiResult<RoleDto>.Ok(role, "更新成功"));
        }
        catch (KeyNotFoundException ex)
        {
            return NotFound(ApiResult.Fail(ex.Message, 404));
        }
        catch (InvalidOperationException ex)
        {
            return BadRequest(ApiResult.Fail(ex.Message));
        }
    }

    /// <summary>
    /// 删除角色
    /// </summary>
    [HttpDelete("{id:guid}")]
    public async Task<IActionResult> Delete(Guid id)
    {
        try
        {
            await _roleAppService.DeleteAsync(id);
            return Ok(ApiResult.Ok("删除成功"));
        }
        catch (KeyNotFoundException ex)
        {
            return NotFound(ApiResult.Fail(ex.Message, 404));
        }
        catch (InvalidOperationException ex)
        {
            return BadRequest(ApiResult.Fail(ex.Message));
        }
    }

    /// <summary>
    /// 获取单个角色
    /// </summary>
    [HttpGet("{id:guid}")]
    public async Task<IActionResult> Get(Guid id)
    {
        var role = await _roleAppService.GetByIdAsync(id);
        return role == null
            ? NotFound(ApiResult.Fail("角色不存在", 404))
            : Ok(ApiResult<RoleDto>.Ok(role));

    /// 分页查询角色
    /// </summary>

    {
        var result = await _roleAppService.GetPagedAsync(request);
        return Ok(ApiResult<PagedResult<RoleDto>>.Ok(result));

    /// 获取所有角色（下拉选择用，不分页）
    /// </summary>

    public async Task<IActionResult> GetAll()
    {

5. PagedRequest 补全排序字段

// Sso.Shared/Paging/PagedRequest.cs
namespace Sso.Shared.Paging;

public class PagedRequest
{
    private int _page = 1;
    private int _pageSize = 20;

    public int Page
    {
        get => _page;
        set => _page = value < 1 ? 1 : value;
    }

    public int PageSize
    {
        get => _pageSize;
        set => _pageSize = value switch
        {
            < 1 => 20,
            > 100 => 100,
            _ => value
        };
    }

    /// <summary>
    /// 排序字段
    /// </summary>
    public string? OrderBy { get; set; }

    /// <summary>
    /// 是否降序
    /// </summary>
    public bool IsDesc { get; set; } = true;
}

// Program.cs SeedDataAsync 中
static async Task SeedDataAsync(WebApplication app)
{
    using var scope = app.Services.CreateScope();
    var services = scope.ServiceProvider;

    var roleManager = services.GetRequiredService<RoleManager<SsoRole>>();

    // ====== 系统角色 ======
    var systemRoles = new[]
    {
        new { Name = "admin",    Description = "系统管理员", IsSystem = true  },
        new { Name = "user",     Description = "普通用户",   IsSystem = true  },
        new { Name = "auditor",  Description = "审计员",     IsSystem = false },
    };

    foreach (var r in systemRoles)
    {
        if (!await roleManager.RoleExistsAsync(r.Name))
        {
            await roleManager.CreateAsync(new SsoRole
            {
                Name = r.Name,
                Description = r.Description,
                IsSystem = r.IsSystem,
                CreatedAt = DateTime.UtcNow
            });
        }
    }

    // ... 其余种子数据不变 ...
}

8. EF Core 配置（确保新字段有映射）

// Sso.Infrastructure/Data/Configurations/SsoRoleConfiguration.cs
using Microsoft.EntityFrameworkCore;
using Microsoft.EntityFrameworkCore.Metadata.Builders;
using Sso.Domain.Entities;

namespace Sso.Infrastructure.Data.Configurations;

public class SsoRoleConfiguration : IEntityTypeConfiguration<SsoRole>
{
    public void Configure(EntityTypeBuilder<SsoRole> builder)
    {
        builder.ToTable("SsoRoles");

        builder.Property(r => r.Description)
            .HasMaxLength(200);

        builder.Property(r => r.IsSystem)
            .HasDefaultValue(false);

        builder.Property(r => r.CreatedAt)
            .HasDefaultValueSql("GETUTCDATE()"); // SQL Server
            // .HasDefaultValueSql("NOW()");      // PostgreSQL
    }
}
