接入演示

Moklgy2025/4/9大约 18 分钟ASP.NET_Coreasp.netcoreSSOOpenIddict权限中心RBAC

业务系统怎么接入SSO和权限中心，以及自己怎么管权限。

全局视角

业务系统（比如 ERP）要做的事:

1. 接入 SSO → 用户登录/Token验证
2. 接入权限中心 → 权限校验/菜单获取
3. 定义自己的权限 → 告诉权限中心"我有哪些权限点"
4. 使用权限 → 前端控制显隐，后端控制访问

问题拆解:
  谁来定义权限？→ 业务系统自己定义
  权限存在哪？→ 权限中心的数据库
  怎么同步过去？→ 启动时自动注册
  运行时怎么校验？→ SDK 调权限中心 API

using Perm.Client.Definitions;

namespace ERP.Permissions;

/// <summary>
/// ERP 权限定义提供者
/// 
/// 作用:
///   告诉权限中心 "ERP 系统有哪些权限点"
///   包括: 权限名、显示名、父子关系、分组
///   
/// 类比 ABP:
///   ABP 的 PermissionDefinitionProvider
///   定义权限 → 启动时注册到 ABP 框架
///   我们: 定义权限 → 启动时注册到权限中心
/// </summary>
public class ErpPermissionDefinitionProvider : IPermissionDefinitionProvider
{
    public string AppCode => ErpPermissions.AppCode;

    public void Define(IPermissionDefinitionContext context)
    {
        // ============ 采购管理组 ============
        var purchaseGroup = context.AddGroup(
            "Purchase",
            "采购管理");

        var purchase = purchaseGroup.AddPermission(
            ErpPermissions.Purchase.Default,
            "采购管理");

        purchase.AddChild(ErpPermissions.Purchase.View, "查看采购单");
        purchase.AddChild(ErpPermissions.Purchase.Create, "创建采购单");
        purchase.AddChild(ErpPermissions.Purchase.Edit, "编辑采购单");
        purchase.AddChild(ErpPermissions.Purchase.Delete, "删除采购单");
        purchase.AddChild(ErpPermissions.Purchase.Approve, "审批采购单");
        purchase.AddChild(ErpPermissions.Purchase.Export, "导出");

        // ============ 库存管理组 ============
        var stockGroup = context.AddGroup(
            "Stock",
            "库存管理");

        var stock = stockGroup.AddPermission(
            ErpPermissions.Stock.Default,
            "库存管理");

        stock.AddChild(ErpPermissions.Stock.View, "查看库存");
        stock.AddChild(ErpPermissions.Stock.Inbound, "入库操作");
        stock.AddChild(ErpPermissions.Stock.Outbound, "出库操作");
        stock.AddChild(ErpPermissions.Stock.Transfer, "调拨");
        stock.AddChild(ErpPermissions.Stock.Inventory, "盘点");

        // ============ 报表中心组 ============
        var reportGroup = context.AddGroup(
            "Report",
            "报表中心");

        var report = reportGroup.AddPermission(
            ErpPermissions.Report.Default,
            "报表中心");

        report.AddChild(ErpPermissions.Report.View, "查看报表");
        report.AddChild(ErpPermissions.Report.Export, "导出报表");
        report.AddChild(ErpPermissions.Report.PurchaseReport, "采购报表");
        report.AddChild(ErpPermissions.Report.StockReport, "库存报表");

            "Settings",
            "系统设置");

            ErpPermissions.Settings.Default,

    PermissionGroupBuilder AddGroup(string code, string displayName);

public class PermissionGroupBuilder
{
    public string Code { get; }
    public string DisplayName { get; }
    internal List<PermissionBuilder> Permissions { get; } = [];

    internal PermissionGroupBuilder(string code, string displayName)
    {
        Code = code;
        DisplayName = displayName;
    }

    public PermissionBuilder AddPermission(string name, string displayName)
    {
        var builder = new PermissionBuilder(name, displayName);
        Permissions.Add(builder);
        return builder;
    }
}

public class PermissionBuilder
{
    public string Name { get; }
    public string DisplayName { get; }
    internal List<PermissionBuilder> Children { get; } = [];

    internal PermissionBuilder(string name, string displayName)
    {
        Name = name;
        DisplayName = displayName;
    }

    public PermissionBuilder AddChild(string name, string displayName)
    {
        var child = new PermissionBuilder(name, displayName);
        Children.Add(child);
        return child;
    }
}

自动注册服务（启动时同步到权限中心）

// ============================================
// Perm.Client/Registration/PermissionRegistrationService.cs
// 启动时自动将权限定义同步到权限中心
// ============================================

using Microsoft.Extensions.Hosting;
using Microsoft.Extensions.Logging;
using Perm.Client.Definitions;

namespace Perm.Client.Registration;

/// <summary>
/// 权限自动注册后台服务
/// 
/// 应用启动时:
///   1. 收集所有 IPermissionDefinitionProvider
///   2. 构建权限树
///   3. POST 到权限中心 /api/apps/register
///   4. 权限中心对比差异，自动 新增/更新/标记废弃
///   
/// 这样业务开发者只需要:
///   定义权限常量 → 实现 Provider → 启动 → 自动同步 ✅
///   不需要去权限中心后台手动创建
/// </summary>
public class PermissionRegistrationService : IHostedService
{
    private readonly IEnumerable<IPermissionDefinitionProvider> _permProviders;
    private readonly IEnumerable<IMenuDefinitionProvider> _menuProviders;
    private readonly IPermissionRegistrationClient _client;
    private readonly ILogger<PermissionRegistrationService> _logger;

    public PermissionRegistrationService(
        IEnumerable<IPermissionDefinitionProvider> permProviders,
        IEnumerable<IMenuDefinitionProvider> menuProviders,
        IPermissionRegistrationClient client,
        ILogger<PermissionRegistrationService> logger)
    {
        _permProviders = permProviders;
        _menuProviders = menuProviders;
        _client = client;
        _logger = logger;
    }

    public async Task StartAsync(CancellationToken cancellationToken)
    {
        try
        {
            foreach (var provider in _permProviders)
            {
                _logger.LogInformation(
                    "Registering permissions for app: {AppCode}",
                    provider.AppCode);

                // 构建权限树
                var context = new PermissionDefinitionContext();
                provider.Define(context);

                // 构建菜单树
                var menuContext = new MenuDefinitionContext();
                var menuProvider = _menuProviders
                    .FirstOrDefault(m => m.AppCode == provider.AppCode);
                menuProvider?.Define(menuContext);

                // 同步到权限中心
                var request = new AppRegistrationRequest
                {
                    AppCode = provider.AppCode,
                    PermissionGroups = BuildPermissionGroups(context),
                    Menus = BuildMenus(menuContext),
                };

                await _client.RegisterAsync(request, cancellationToken);

                _logger.LogInformation(
                    "Permissions registered for app: {AppCode}, " +
                    "Groups: {GroupCount}, Permissions: {PermCount}",
                    provider.AppCode,
                    request.PermissionGroups.Count,
                    request.PermissionGroups
                        .SelectMany(g => g.Permissions)
                        .Count());
            }
        }
        catch (Exception ex)
        {
            // 注册失败不阻止应用启动（降级为无权限控制）
            _logger.LogError(ex,
                "Failed to register permissions. " +
                "App will start but permission checks may fail.");
        }
    }

    public Task StopAsync(CancellationToken cancellationToken) => Task.CompletedTask;

    private List<PermissionGroupDto> BuildPermissionGroups(
        PermissionDefinitionContext context)
    {
        return context.Groups.Select(g => new PermissionGroupDto
        {
            Code = g.Code,
            DisplayName = g.DisplayName,
            Permissions = g.Permissions.Select(p => BuildPermission(p)).ToList(),
        }).ToList();
    }

    private PermissionDto BuildPermission(PermissionBuilder builder)
    {
        return new PermissionDto
        {
            Name = builder.Name,
            DisplayName = builder.DisplayName,
            Children = builder.Children.Select(c => BuildPermission(c)).ToList(),
        };
    }

    private List<MenuDto> BuildMenus(MenuDefinitionContext context)
    {
        return context.Menus.Select(m => BuildMenu(m)).ToList();
    }

    private MenuDto BuildMenu(MenuDefinitionBuilder builder)
    {
        return new MenuDto
        {
            Code = builder.Menu.Code,
            Name = builder.Menu.Name,
            Icon = builder.Menu.Icon,
            Path = builder.Menu.Path,
            RequiredPermission = builder.Menu.RequiredPermission,
            Sort = builder.Menu.Sort,
            Children = builder.Children.Select(c => BuildMenu(c)).ToList(),
        };
    }
}

namespace Perm.Client.Registration;

public interface IPermissionRegistrationClient
{
    Task RegisterAsync(
        AppRegistrationRequest request,
        CancellationToken cancellationToken = default);
}

public class PermissionRegistrationClient : IPermissionRegistrationClient
{
    private readonly HttpClient _http;

    public PermissionRegistrationClient(HttpClient http) => _http = http;

    public async Task RegisterAsync(
        AppRegistrationRequest request,
        CancellationToken cancellationToken = default)
    {
        // POST /api/apps/register
        // 权限中心接收后对比差异:
        //   新增的权限 → INSERT
        //   已有的权限 → UPDATE displayName
        //   不再定义的权限 → 标记 IsEnabled=false（不删除）

        var response = await _http.PostAsJsonAsync(
            "/api/apps/register", request, cancellationToken);

        response.EnsureSuccessStatusCode();
    }
}

// Perm.Client/IPermissionChecker.cs
// 业务系统端的权限检查接口

namespace Perm.Client;

/// <summary>
/// 权限检查器（业务系统端）
/// 
/// 和权限中心的 IPermissionChecker 不同:
///   权限中心: 直接查数据库
///   业务系统: HTTP 调权限中心 + 本地缓存
/// </summary>
public interface IPermissionChecker
{
    /// <summary>
    /// 检查当前用户是否有某权限
    /// </summary>
    Task<bool> IsGrantedAsync(string permissionName);

    /// <summary>
    /// 检查指定用户是否有某权限
    /// </summary>
    Task<bool> IsGrantedAsync(Guid userId, string permissionName);

    /// <summary>
    /// 获取当前用户的所有权限
    /// </summary>
    Task<HashSet<string>> GetGrantedPermissionsAsync();

    /// <summary>
    /// 获取当前用户的菜单树
    /// </summary>
    Task<List<MenuDto>> GetMenusAsync();
}
// ============================================
// Perm.Client/PermissionChecker.cs
// 权限检查器实现
// ============================================

using Microsoft.Extensions.Caching.Distributed;
using System.Text.Json;
using Platform.EntityFrameworkCore;

namespace Perm.Client;

public class PermissionChecker : IPermissionChecker
{
    private readonly HttpClient _http;
    private readonly IDistributedCache _cache;
    private readonly ICurrentUser _currentUser;
    private readonly PermClientOptions _options;

    public PermissionChecker(
        HttpClient http,
        IDistributedCache cache,
        ICurrentUser currentUser,
        Microsoft.Extensions.Options.IOptions<PermClientOptions> options)
    {
        _http = http;
        _cache = cache;
        _currentUser = currentUser;
        _options = options.Value;
    }

    public async Task<bool> IsGrantedAsync(string permissionName)
    {
        if (!_currentUser.IsAuthenticated)
            return false;

        var permissions = await GetGrantedPermissionsAsync();
        return permissions.Contains(permissionName);
    }

    public async Task<bool> IsGrantedAsync(Guid userId, string permissionName)
    {
        var permissions = await GetGrantedPermissionsForUserAsync(userId);
        return permissions.Contains(permissionName);
    }

    public async Task<HashSet<string>> GetGrantedPermissionsAsync()
    {
        if (_currentUser.Id == null)
            return [];

        return await GetGrantedPermissionsForUserAsync(_currentUser.Id.Value);
    }

    public async Task<List<MenuDto>> GetMenusAsync()
    {
        if (_currentUser.Id == null)
            return [];

        var cacheKey = $"perm:menus:{_options.AppCode}:{_currentUser.Id}";
        var cached = await _cache.GetStringAsync(cacheKey);

        if (cached != null)
            return JsonSerializer.Deserialize<List<MenuDto>>(cached) ?? [];

        // 调权限中心
        var response = await _http.GetAsync(
            $"/api/menus/current?appCode={_options.AppCode}");
        response.EnsureSuccessStatusCode();

        var result = await response.Content
            .ReadFromJsonAsync<ApiResult<List<MenuDto>>>();
        var menus = result?.Data ?? [];

        await _cache.SetStringAsync(cacheKey,
            JsonSerializer.Serialize(menus),
            new DistributedCacheEntryOptions
            {
                AbsoluteExpirationRelativeToNow =
                    TimeSpan.FromMinutes(_options.CacheMinutes)
            });

        return menus;
    }

    private async Task<HashSet<string>> GetGrantedPermissionsForUserAsync(
        Guid userId)
    {
        var cacheKey = $"perm:granted:{_options.AppCode}:{userId}";
        var cached = await _cache.GetStringAsync(cacheKey);

        if (cached != null)
            return JsonSerializer.Deserialize<HashSet<string>>(cached) ?? [];

        // 调权限中心
        var response = await _http.GetAsync(
            $"/api/permissions/granted?appCode={_options.AppCode}&userId={userId}");
        response.EnsureSuccessStatusCode();

        var result = await response.Content
            .ReadFromJsonAsync<ApiResult<List<string>>>();
        var permissions = new HashSet<string>(result?.Data ?? []);

        await _cache.SetStringAsync(cacheKey,
            JsonSerializer.Serialize(permissions),
            new DistributedCacheEntryOptions
            {
                AbsoluteExpirationRelativeToNow =
                    TimeSpan.FromMinutes(_options.CacheMinutes)
            });

        return permissions;
    }

    private class ApiResult<T>
    {
        public bool Success { get; set; }
        public T? Data { get; set; }
    }
}

/// <summary>
/// 声明式权限校验
/// 
/// 用法:
///   [RequirePermission(ErpPermissions.Purchase.Create)]
///   public async Task<IActionResult> CreatePurchase() { ... }
///   
///   // 多个权限（默认全部满足）
///   [RequirePermission(ErpPermissions.Purchase.Create)]
///   [RequirePermission(ErpPermissions.Purchase.Edit)]
///   
///   // 多个权限满足一个即可
///   [RequirePermission(ErpPermissions.Purchase.Create, RequireAll = false)]
///   [RequirePermission(ErpPermissions.Purchase.Edit)]
/// </summary>
[AttributeUsage(
    AttributeTargets.Class | AttributeTargets.Method,
    AllowMultiple = true)]
public class RequirePermissionAttribute : Attribute
{
    public string[] Permissions { get; }
    public bool RequireAll { get; set; } = true;

    public RequirePermissionAttribute(params string[] permissions)
    {
        Permissions = permissions;
    }
}
// ============================================
// Perm.Client/Authorization/PermissionAuthorizationFilter.cs
// ============================================

using Microsoft.AspNetCore.Mvc;
using Microsoft.AspNetCore.Mvc.Filters;
using Platform.AspNetCore.ApiResult;

namespace Perm.Client.Authorization;

/// <summary>
/// 权限校验过滤器
/// 
/// 执行顺序:
///   1. ASP.NET Core 的 [Authorize] → 检查是否登录
///   2. 本过滤器 → 检查是否有权限
///   3. Controller Action → 执行业务
/// </summary>
public class PermissionAuthorizationFilter : IAsyncAuthorizationFilter
{
    private readonly IPermissionChecker _checker;

    public PermissionAuthorizationFilter(IPermissionChecker checker)
    {
        _checker = checker;
    }

    public async Task OnAuthorizationAsync(AuthorizationFilterContext context)
    {
        // 收集所有 RequirePermission 特性
        var attributes = context.ActionDescriptor.EndpointMetadata
            .OfType<RequirePermissionAttribute>()
            .ToList();

        if (attributes.Count == 0)
            return;

        // 必须已认证
        if (context.HttpContext.User.Identity?.IsAuthenticated != true)
        {
            context.Result = new ObjectResult(
                ApiResult.Fail("未登录", 401))
            { StatusCode = 401 };
            return;
        }

        // 获取用户所有权限（一次性拉取，有缓存）
        var grantedPermissions = await _checker.GetGrantedPermissionsAsync();

        foreach (var attr in attributes)
        {
            bool hasPermission;

            if (attr.RequireAll)
            {
                // 所有权限都要满足
                hasPermission = attr.Permissions
                    .All(p => grantedPermissions.Contains(p));
            }
            else
            {
                // 满足任一即可
                hasPermission = attr.Permissions
                    .Any(p => grantedPermissions.Contains(p));
            }

            if (!hasPermission)
            {
                var required = string.Join(", ", attr.Permissions);

                context.Result = new ObjectResult(
                    ApiResult.Fail($"没有权限: {required}", 403,
                        "Platform:Forbidden"))
                { StatusCode = 403 };
                return;
            }
        }
    }
}

    /// <summary>
    /// 接入权限中心
    /// 
    /// 用法:
    ///   builder.Services.AddPermissionCenter(options =>
    ///   {
    ///       options.AppCode = "erp";
    ///       options.BaseUrl = "https://perm.example.com";
    ///   });
    /// </summary>
    public static IServiceCollection AddPermissionCenter(
        this IServiceCollection services,
        Action<PermClientOptions> configure)
    {
        var options = new PermClientOptions();
        configure(options);

        services.Configure(configure);

        // HTTP Client
        services.AddHttpClient<IPermissionChecker, PermissionChecker>(client =>
        {
            client.BaseAddress = new Uri(options.BaseUrl);
            client.Timeout = TimeSpan.FromSeconds(5);
        });

        services.AddHttpClient<IPermissionRegistrationClient,
            PermissionRegistrationClient>(client =>
        {
            client.BaseAddress = new Uri(options.BaseUrl);
            client.Timeout = TimeSpan.FromSeconds(30);
        });

        // 权限校验 Filter
        services.AddControllers(mvcOptions =>
        {
            mvcOptions.Filters.Add<PermissionAuthorizationFilter>();
        });

        // 启动时自动注册权限
        if (options.AutoRegisterPermissions)
        {
            services.AddHostedService<PermissionRegistrationService>();
        }

        return services;
    }

    /// <summary>
    /// 注册权限定义提供者
    /// 
    /// 用法:
    ///   builder.Services.AddPermissionDefinition<ErpPermissionDefinitionProvider>();
    ///   builder.Services.AddMenuDefinition<ErpMenuDefinitionProvider>();
    /// </summary>
    public static IServiceCollection AddPermissionDefinition<T>(
        this IServiceCollection services)
        where T : class, IPermissionDefinitionProvider
    {
        services.AddSingleton<IPermissionDefinitionProvider, T>();
        return services;
    }

    public static IServiceCollection AddMenuDefinition<T>(
        this IServiceCollection services)
        where T : class, IMenuDefinitionProvider
    {
        services.AddSingleton<IMenuDefinitionProvider, T>();

    }

                .IsGrantedAsync(ErpPermissions.Purchase.Approve);

            if (!hasApprovePermission)
            {
                throw new BusinessException(
                    "ERP:Stock:TransferNeedApproval",
                    "金额超过1万需要审批权限");
            }
        }

        await _service.TransferAsync(request);
    }

    /// <summary>

            .IsGrantedAsync("ERP.Stock.ViewCost");

        if (!canViewCost)
        {
            dto.UnitCost = null;
            dto.TotalCost = null;

export const usePermissionStore = create<PermissionStore>((set, get) => ({

// 权限包裹组件
interface AuthorizedProps {
  permission: string | string[];
  requireAll?: boolean;
  fallback?: React.ReactNode;
  children: React.ReactNode;
}

export const Authorized: React.FC<AuthorizedProps> = ({
  permission,
  requireAll = true,
  fallback = null,
  children,
}) => {
  const { hasPermission } = usePermissionStore();

  const permissions = Array.isArray(permission) ? permission : [permission];
  
  const authorized = requireAll
    ? permissions.every(p => hasPermission(p))
    : permissions.some(p => hasPermission(p));

  return authorized ? <>{children}</> : <>{fallback}</>;
};

业务开发者日常操作清单

新增一个功能模块（如"供应商管理"）:

  Step 1: 在 ErpPermissions.cs 加常量
  ─────────────────────────────────
  public static class Supplier
  {
      public const string Default = "ERP.Supplier";
      public const string View = "ERP.Supplier.View";
      public const string Create = "ERP.Supplier.Create";
      public const string Edit = "ERP.Supplier.Edit";
      public const string Delete = "ERP.Supplier.Delete";
  }

  Step 2: 在 Provider 加定义
  ──────────────────────────
  var supplierGroup = context.AddGroup("Supplier", "供应商管理");
  var supplier = supplierGroup.AddPermission(ErpPermissions.Supplier.Default, "供应商管理");
  supplier.AddChild(ErpPermissions.Supplier.View, "查看供应商");
  supplier.AddChild(ErpPermissions.Supplier.Create, "新建供应商");
  supplier.AddChild(ErpPermissions.Supplier.Edit, "编辑供应商");
  supplier.AddChild(ErpPermissions.Supplier.Delete, "删除供应商");

  Step 3: Controller 加特性
  ────────────────────────
  [RequirePermission(ErpPermissions.Supplier.Create)]
  public async Task<SupplierDto> Create(...) { }

  Step 4: 前端加权限控制
  ────────────────────
  <Authorized permission="ERP.Supplier.Create">
    <Button>新建供应商</Button>
  </Authorized>

  Step 5: 部署 → 自动注册 → 管理员配权限 → 完成
  
  总结: 加 4 处代码 → 部署 → 完成 ✅
