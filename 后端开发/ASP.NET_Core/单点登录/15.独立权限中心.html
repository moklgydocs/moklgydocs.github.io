独立权限中心

Moklgy2025/4/9大约 22 分钟单点登录asp.netcoreSSOOpenIddict权限中心RBAC

从SSO出发，实现一个独立的独立权限中心

整体定位

权限中心 = 独立微服务
  
  项目名: Perm（Permission Center）
  
  对标: ABP 的 PermissionManagement + Identity 模块
  但是: 不依赖 ABP 框架，纯手写
  目标: 轻量、可扩展、兼容 ABAC 演进

核心设计 — 模仿 ABP 权限框架

1. ABP 的核心概念（我们要复刻的）

ABP 权限框架的精髓:

1. PermissionDefinition（权限定义）
   → 代码里声明有哪些权限
   → 启动时自动注册
   → 也可以从数据库加载（我们要做的）

2. PermissionGrant（权限授予）
   → 不直接绑定"角色-权限"
   → 而是 ProviderName + ProviderKey + PermissionName
   → ProviderName = "Role" / "User" / "Client" / "OrganizationUnit"
   → ProviderKey = "manager" / "user-guid" / "client-id" / "ou-id"
   
   这就是 ABP 最精妙的设计 ✅
   这也是兼容 ABAC 的关键

3. IPermissionValueProvider（权限值提供者）
   → 策略模式
   → RolePermissionValueProvider: 检查角色有没有这个权限
   → UserPermissionValueProvider: 检查用户是否被直接授权
   → ClientPermissionValueProvider: 检查客户端权限
   → 未来: AttributePermissionValueProvider → ABAC
   → 按顺序遍历所有 Provider，任何一个说"有权限"就通过

4. IPermissionChecker
   → 最终对外暴露的接口
   → IsGrantedAsync(userId, permissionName)
   → 内部遍历所有 ValueProvider
我们的复刻:

ABP 的:
  PermissionDefinitionProvider → 我们: IPermissionDefinitionProvider
  PermissionGrant             → 我们: PermissionGrant（同名）
  IPermissionValueProvider    → 我们: IPermissionValueProvider
  PermissionChecker           → 我们: PermissionChecker

  核心区别: 
    ABP 权限定义在代码里（编译时确定）
    我们: 权限定义在数据库里（运行时动态，因为是独立服务）
    我们也支持代码定义（业务系统SDK里）

PermissionGrant — ABP 精髓的复刻 ⭐

// ============================================
// Perm.Domain/Entities/PermissionGrant.cs
// 权限授予（ABP 核心设计的复刻）
// ============================================

namespace Perm.Domain.Entities;

/// <summary>
/// 权限授予
/// 
/// ABP 的精髓: 不直接做 RolePermission / UserPermission 表
/// 而是用 ProviderName + ProviderKey 抽象
/// 
/// 这样一张表就能存所有类型的授权:
///   Role 授权:   ProviderName="R", ProviderKey="manager"
///   User 授权:   ProviderName="U", ProviderKey="user-guid"
///   Client 授权: ProviderName="C", ProviderKey="erp"
///   OU 授权:     ProviderName="O", ProviderKey="ou-guid"
///   
/// 未来 ABAC:
///   Attribute:   ProviderName="A", ProviderKey="age>=18&dept=IT"
///   
/// 无需改表结构，只加新的 Provider 就行 ✅
/// </summary>
public class PermissionGrant : CreationAuditedEntity
{
    /// <summary>
    /// 所属应用ID
    /// </summary>
    public Guid AppId { get; set; }

    /// <summary>
    /// 权限名称
    /// "ERP.Purchase.Approve"
    /// </summary>
    public string PermissionName { get; set; } = default!;

    /// <summary>
    /// 授权提供者名称
    /// "R" = Role
    /// "U" = User
    /// "C" = Client
    /// "O" = OrganizationUnit
    /// "A" = Attribute (ABAC 预留)
    /// </summary>
    public string ProviderName { get; set; } = default!;

    /// <summary>
    /// 授权提供者 Key
    /// 当 ProviderName = "R" 时: "manager"（角色编码）
    /// 当 ProviderName = "U" 时: "user-guid"（用户ID）
    /// 当 ProviderName = "C" 时: "erp"（客户端编码）
    /// 当 ProviderName = "A" 时: JSON 条件
    /// </summary>
    public string ProviderKey { get; set; } = default!;

    /// <summary>
    /// 是否授予（true = 授予, false = 显式拒绝）
    /// 显式拒绝优先级高于授予
    /// </summary>
    public bool IsGranted { get; set; } = true;

    // 导航
    public App App { get; set; } = default!;
}

// Perm.Application/Authorization/IPermissionValueProvider.cs
// 权限值提供者接口（ABP 核心抽象）
// ============================================

namespace Perm.Application.Authorization;

/// <summary>
/// 权限值提供者
/// 
/// ABP 的核心设计: 每种授权来源是一个 Provider
/// 系统遍历所有 Provider，任何一个返回 Granted 就通过
/// 
/// 扩展 ABAC 时: 只需新增一个 AttributePermissionValueProvider
/// 无需修改任何现有代码 ✅
/// </summary>
public interface IPermissionValueProvider
{
    /// <summary>
    /// 提供者名称: "R" / "U" / "C" / "O" / "A"
    /// </summary>
    string Name { get; }

    /// <summary>
    /// 优先级（越小越先执行）
    /// 显式拒绝类的 Provider 应优先
    /// </summary>
    int Order { get; }

    /// <summary>

namespace Perm.Application.Authorization;

/// <summary>
/// 权限检查器
/// 
/// 遍历所有 IPermissionValueProvider
/// 执行优先级逻辑:
///   Denied > Granted > Undefined
///   
/// 如果所有 Provider 都返回 Undefined → 拒绝
/// </summary>
public class PermissionChecker : IPermissionChecker
{
    private readonly IEnumerable<IPermissionValueProvider> _providers;
    private readonly PermDbContext _db;
    private readonly IDistributedCache _cache;
    private readonly ILogger<PermissionChecker> _logger;

    public PermissionChecker(
        IEnumerable<IPermissionValueProvider> providers,
        PermDbContext db,
        IDistributedCache cache,
        ILogger<PermissionChecker> logger)
    {
        _providers = providers.OrderBy(p => p.Order);
        _db = db;
        _cache = cache;
        _logger = logger;
    }

    /// <summary>
    /// 检查单个权限
    /// </summary>
    public async Task<bool> IsGrantedAsync(PermissionCheckContext context)
    {
        // 填充用户角色
        if (context.Roles.Length == 0)
        {
            context.Roles = await GetUserRolesAsync(
                context.UserId, context.AppCode);
        }

        // 遍历所有 Provider
        foreach (var provider in _providers)
        {
            var result = await provider.CheckAsync(context);

            switch (result)
            {
                case PermissionValueProviderResult.Denied:
                    // 显式拒绝，立即返回
                    _logger.LogDebug(
                        "Permission {Permission} denied by {Provider} for user {User}",
                        context.PermissionName, provider.Name, context.UserId);
                    return false;

                case PermissionValueProviderResult.Granted:
                    // 授予，立即返回
                    _logger.LogDebug(
                        "Permission {Permission} granted by {Provider} for user {User}",
                        context.PermissionName, provider.Name, context.UserId);
                    return true;

                case PermissionValueProviderResult.Undefined:
                    // 该 Provider 不处理，继续下一个
                    continue;
            }
        }

        // 所有 Provider 都 Undefined → 拒绝
        _logger.LogDebug(
            "Permission {Permission} not granted for user {User} (no provider matched)",
            context.PermissionName, context.UserId);
        return false;
    }

    /// <summary>
    /// 批量检查权限
    /// </summary>
    public async Task<Dictionary<string, bool>> IsGrantedAsync(
        Guid userId, string appCode, string[] permissionNames)
    {
        var roles = await GetUserRolesAsync(userId, appCode);
        var result = new Dictionary<string, bool>();

        foreach (var permName in permissionNames)
        {
            var context = new PermissionCheckContext
            {
                AppCode = appCode,
                UserId = userId,
                PermissionName = permName,
                Roles = roles,
            };

            result[permName] = await IsGrantedAsync(context);
        }

        return result;
    }

    /// <summary>
    /// 获取用户所有已授权的权限
    /// </summary>
    public async Task<List<string>> GetGrantedPermissionsAsync(
        Guid userId, string appCode)
    {
        var cacheKey = $"perm:granted:{appCode}:{userId}";

        var cached = await _cache.GetStringAsync(cacheKey);
        if (cached != null)
            return JsonSerializer.Deserialize<List<string>>(cached) ?? [];

        var roles = await GetUserRolesAsync(userId, appCode);

        // 获取该应用所有权限定义
        var allPermissions = await _db.PermissionDefinitions
            .Include(x => x.Group)
            .Where(x => x.Group.App.Code == appCode && x.IsEnabled)
            .Select(x => x.Name)
            .ToListAsync();

        var granted = new List<string>();

        foreach (var permName in allPermissions)
        {
            var context = new PermissionCheckContext
            {
                AppCode = appCode,
                UserId = userId,
                PermissionName = permName,
                Roles = roles,
            };

            if (await IsGrantedAsync(context))
            {
                granted.Add(permName);
            }
        }

        // 缓存 5 分钟
        await _cache.SetStringAsync(cacheKey,
            JsonSerializer.Serialize(granted),
            new DistributedCacheEntryOptions
            {
                AbsoluteExpirationRelativeToNow = TimeSpan.FromMinutes(5)
            });

        return granted;
    }

    /// <summary>
    /// 获取用户在某应用的角色
    /// </summary>
    private async Task<string[]> GetUserRolesAsync(Guid userId, string appCode)
    {
        var cacheKey = $"perm:roles:{appCode}:{userId}";

        var cached = await _cache.GetStringAsync(cacheKey);
        if (cached != null)
            return JsonSerializer.Deserialize<string[]>(cached) ?? [];

        var roles = await _db.UserRoles
            .Include(x => x.Role)
            .Where(x =>
                x.UserId == userId &&
                x.App.Code == appCode &&
                x.Role.IsEnabled)
            .Select(x => x.Role.Code)
            .ToArrayAsync();

        await _cache.SetStringAsync(cacheKey,
            JsonSerializer.Serialize(roles),
            new DistributedCacheEntryOptions
            {
                AbsoluteExpirationRelativeToNow = TimeSpan.FromMinutes(10)
            });

        return roles;
    }
}

权限判定流程

请求: 张三在 ERP 是否有 Purchase.Approve 权限

PermissionChecker.IsGrantedAsync:

  Step 1: 获取张三在 ERP 的角色 → ["manager", "purchaser"]

  Step 2: 遍历 Provider（按 Order 排序）

    Provider 1: UserPermissionValueProvider (Order=10)
      → 查 PermissionGrant: ProviderName="U", ProviderKey="张三guid", Perm="Purchase.Approve"
      → 没有记录 → Undefined
      → 继续

    Provider 2: RolePermissionValueProvider (Order=20)
      → 查 PermissionGrant: ProviderName="R", ProviderKey in ["manager","purchaser"], Perm="Purchase.Approve"
      → 找到: R + manager + Purchase.Approve + IsGranted=true
      → Granted ✅
      → 立即返回 true

  结果: true（张三可以审批采购单）


另一个例子: 张三在 ERP 是否有 Purchase.Delete 权限

  Provider 1: UserPermissionValueProvider
    → 查到: U + 张三guid + Purchase.Delete + IsGranted=false
    → Denied ❌
    → 立即返回 false

  结果: false（张三被显式拒绝删除权限）
  即使 manager 角色有 Delete 权限也没用（显式拒绝优先）

namespace Perm.Client;

/// <summary>
/// 权限中心 HTTP 客户端
/// </summary>
public class PermissionClient : IPermissionClient
{
    private readonly HttpClient _http;
    private readonly IDistributedCache _cache;
    private readonly PermClientOptions _options;

    public PermissionClient(
        HttpClient http,
        IDistributedCache cache,
        IOptions<PermClientOptions> options)
    {
        _http = http;
        _cache = cache;
        _options = options.Value;
    }

    /// <summary>
    /// 检查权限
    /// </summary>
    public async Task<bool> IsGrantedAsync(
        string token, string permission)
    {
        // 先查本地缓存
        var cacheKey = $"perm:{_options.AppCode}:{HashToken(token)}";
        var cached = await _cache.GetStringAsync(cacheKey);

        List<string>? grantedPerms;

        if (cached != null)
        {
            grantedPerms = JsonSerializer.Deserialize<List<string>>(cached);
        }
        else
        {
            // 请求权限中心
            var request = new HttpRequestMessage(
                HttpMethod.Get,
                $"/api/permissions/granted?appCode={_options.AppCode}");
            request.Headers.Authorization =
                new AuthenticationHeaderValue("Bearer", token);

            var response = await _http.SendAsync(request);
            response.EnsureSuccessStatusCode();

            var result = await response.Content
                .ReadFromJsonAsync<ApiResult<List<string>>>();
            grantedPerms = result?.Data ?? [];

            // 缓存 5 分钟
            await _cache.SetStringAsync(cacheKey,
                JsonSerializer.Serialize(grantedPerms),
                new DistributedCacheEntryOptions
                {
                    AbsoluteExpirationRelativeToNow = TimeSpan.FromMinutes(5)
                });
        }

        return grantedPerms?.Contains(permission) ?? false;
    }

    /// <summary>
    /// 获取所有权限（前端用）
    /// </summary>
    public async Task<List<string>> GetGrantedPermissionsAsync(string token)
    {
        var request = new HttpRequestMessage(
            HttpMethod.Get,
            $"/api/permissions/granted?appCode={_options.AppCode}");
        request.Headers.Authorization =
            new AuthenticationHeaderValue("Bearer", token);

        var response = await _http.SendAsync(request);
        response.EnsureSuccessStatusCode();

        var result = await response.Content
            .ReadFromJsonAsync<ApiResult<List<string>>>();
        return result?.Data ?? [];
    }

    /// <summary>
    /// 获取用户菜单树

    public async Task<List<MenuDto>> GetMenusAsync(string token)

        var request = new HttpRequestMessage(

        response.EnsureSuccessStatusCode();

        var result = await response.Content
            .ReadFromJsonAsync<ApiResult<List<MenuDto>>>();
        return result?.Data ?? [];
    }

    private static string HashToken(string token)
    {
        // 用 token 的后16位做缓存key（避免完整token做key太长）

public class PermissionAuthorizationFilter : IAsyncAuthorizationFilter
{
    private readonly IPermissionClient _client;

    public PermissionAuthorizationFilter(IPermissionClient client)
        => _client = client;

    public async Task OnAuthorizationAsync(AuthorizationFilterContext context)
    {
        var attrs = context.ActionDescriptor.EndpointMetadata
            .OfType<RequirePermissionAttribute>()
            .ToList();

        if (attrs.Count == 0) return;

        var user = context.HttpContext.User;
        if (!user.Identity?.IsAuthenticated ?? true)
        {
            context.Result = new UnauthorizedResult();
            return;
        }

        // 从请求头拿 Token
        var token = context.HttpContext.Request.Headers
            .Authorization.ToString().Replace("Bearer ", "");

        if (string.IsNullOrEmpty(token))
        {
            context.Result = new UnauthorizedResult();
            return;
        }

        foreach (var attr in attrs)
        {
            var granted = await _client.IsGrantedAsync(token, attr.Permission);
            if (!granted)
            {
                context.Result = new JsonResult(new
                {
                    code = 403,
                    message = $"没有权限: {attr.Permission}"
                })
                { StatusCode = 403 };
                return;
            }
        }
    }
}

namespace Perm.Infrastructure.Caching;

/// <summary>
/// 权限缓存管理
/// 当权限数据变更时清除相关缓存
/// </summary>
public class PermissionCacheManager
{
    private readonly IDistributedCache _cache;
    private readonly ILogger<PermissionCacheManager> _logger;

    public PermissionCacheManager(
        IDistributedCache cache,
        ILogger<PermissionCacheManager> logger)
    {
        _cache = cache;
        _logger = logger;
    }

    /// <summary>
    /// 用户角色变更时清除缓存
    /// </summary>
    public async Task OnUserRoleChangedAsync(Guid userId, string appCode)
    {
        await _cache.RemoveAsync($"perm:roles:{appCode}:{userId}");
        await _cache.RemoveAsync($"perm:granted:{appCode}:{userId}");
        _logger.LogInformation(
            "Cache cleared for user {UserId} in app {App}", userId, appCode);
    }

    /// <summary>
    /// 角色权限变更时清除缓存
    /// 需要清除该角色下所有用户的缓存
    /// </summary>
    public async Task OnRolePermissionChangedAsync(
        Guid roleId, string appCode, PermDbContext db)
    {
        var userIds = await db.UserRoles
            .Where(x => x.RoleId == roleId)
            .Select(x => x.UserId)
            .ToListAsync();

        foreach (var userId in userIds)
        {
            await _cache.RemoveAsync($"perm:granted:{appCode}:{userId}");
        }

        _logger.LogInformation(
            "Cache cleared for role {RoleId} in app {App}, affected {Count} users",
            roleId, appCode, userIds.Count);
    }
}
