Token 失效与登出安全

登出要让token失效

问题分析

当前登出流程的问题:

用户调用 /api/account/logout
  → SignInManager.SignOutAsync()   ← 只是清除了 Cookie
  → 但已发放的 Access Token / Refresh Token 还在有效期内
  → 拿着旧 Token 依然能调用 API ❌

根本原因:
  JWT Token 是无状态的，服务端默认不存储、不检查是否已撤销
  Cookie 登出 ≠ Token 失效

解决方案全景

┌──────────────────────────────────────────────────────┐
│              Token 失效策略                            │
├───────────────┬──────────────────────────────────────┤
│ 方案          │ 适用场景                              │
├───────────────┼──────────────────────────────────────┤
│ Token 撤销    │ 主动登出、踢人、改密码                 │
│ (Revocation)  │ OpenIddict 原生支持 ✅                │
├───────────────┼──────────────────────────────────────┤
│ 引用Token     │ 每次请求都查库验证                     │
│ (Reference)   │ 安全性最高，性能有损耗                 │
├───────────────┼──────────────────────────────────────┤
│ 短过期+刷新   │ AccessToken 5分钟过期                  │
│               │ 配合 Refresh Token 续期                │
├───────────────┼──────────────────────────────────────┤
│ 黑名单缓存    │ Redis 存已撤销Token的jti               │
│               │ 高性能，适合分布式                     │
└───────────────┴──────────────────────────────────────┘

生产推荐: 引用Token + Token撤销 + 短过期 组合使用

完整实现

第一步：启用引用 Token（Reference Token）

// Sso.Infrastructure/OpenIddict/OpenIddictServiceExtensions.cs

.AddServer(options =>
{
    // ... 端点、授权流程等配置不变 ...

    // ====== 关键：使用引用 Token 而非自包含 JWT ======
    // 引用 Token = 数据库里存完整 Token，客户端只拿到一个不透明的引用ID
    // 每次验证都查库 → 撤销后立即失效
    options.UseReferenceAccessTokens();
    options.UseReferenceRefreshTokens();

    // 如果你只想对 Access Token 用引用模式，Refresh Token 默认就是引用的
    // options.UseReferenceAccessTokens();  // 至少开这个

    // ... 签名、加密等配置不变 ...
})
引用 Token vs 自包含 JWT 对比:

自包含 JWT:
  eyJhbGciOiJSUzI1NiIs...  (很长，包含所有 Claims)
  → 不查库，本地验证签名即可
  → 无法即时撤销 ❌

引用 Token:
  3F2504E0-4F89-11D3-9A0C-0305E82C3301  (短字符串，只是一个引用)
  → 每次验证查 SsoTokens 表
  → 删除记录 = 立即失效 ✅
  → 有数据库开销，但可以加缓存

第二步：完善登出逻辑 — 撤销所有 Token

// Sso.Application/Services/TokenRevocationService.cs
using System.Security.Claims;
using OpenIddict.Abstractions;
using static OpenIddict.Abstractions.OpenIddictConstants;

namespace Sso.Application.Services;

public class TokenRevocationService
{
    private readonly IOpenIddictTokenManager _tokenManager;
    private readonly IOpenIddictAuthorizationManager _authorizationManager;

    public TokenRevocationService(
        IOpenIddictTokenManager tokenManager,
        IOpenIddictAuthorizationManager authorizationManager)
    {
        _tokenManager = tokenManager;
        _authorizationManager = authorizationManager;
    }

    /// <summary>
    /// 撤销用户的所有 Token（登出时调用）
    /// </summary>
    public async Task RevokeAllUserTokensAsync(string userId, CancellationToken ct = default)
    {
        // 1. 找到该用户所有的授权记录
        await foreach (var authorization in _authorizationManager.FindBySubjectAsync(userId, ct))
        {
            var authId = await _authorizationManager.GetIdAsync(authorization, ct);

            // 2. 撤销该授权下的所有 Token
            await foreach (var token in _tokenManager.FindByAuthorizationIdAsync(authId!, ct))
            {
                var status = await _tokenManager.GetStatusAsync(token, ct);

                // 只撤销还有效的
                if (status is Statuses.Valid or Statuses.Inactive)
                {
                    await _tokenManager.TryRevokeAsync(token, ct);
                }
            }

            // 3. 同时撤销授权记录本身
            var authStatus = await _authorizationManager.GetStatusAsync(authorization, ct);
            if (authStatus == Statuses.Valid)
            {
                await _authorizationManager.TryRevokeAsync(authorization, ct);
            }
        }
    }

    /// <summary>
    /// 撤销用户在指定客户端上的 Token（单点登出某个应用）
    /// </summary>
    public async Task RevokeUserTokensForClientAsync(
        string userId, string clientId, CancellationToken ct = default)
    {
        await foreach (var authorization in _authorizationManager.FindBySubjectAsync(userId, ct))
        {
            // 检查是否属于指定客户端
            var application = await _authorizationManager.GetApplicationIdAsync(authorization, ct);
            // 这里需要判断 application 是否是目标客户端
            // 简化处理：通过 authorizationId 撤销 Token

            var authId = await _authorizationManager.GetIdAsync(authorization, ct);
            await foreach (var token in _tokenManager.FindByAuthorizationIdAsync(authId!, ct))
            {
                var status = await _tokenManager.GetStatusAsync(token, ct);
                if (status is Statuses.Valid or Statuses.Inactive)
                {
                    await _tokenManager.TryRevokeAsync(token, ct);
                }
            }
        }
    }

    /// <summary>
    /// 撤销指定的单个 Token（通过 Token ID）
    /// </summary>
    public async Task RevokeTokenByIdAsync(string tokenId, CancellationToken ct = default)
    {
        var token = await _tokenManager.FindByIdAsync(tokenId, ct);
        if (token != null)
        {
            await _tokenManager.TryRevokeAsync(token, ct);
        }
    }

    /// <summary>
    /// 清理用户所有过期和已撤销的 Token 记录
    /// </summary>
    public async Task PruneUserTokensAsync(string userId, CancellationToken ct = default)
    {
        await foreach (var authorization in _authorizationManager.FindBySubjectAsync(userId, ct))
        {
            var authId = await _authorizationManager.GetIdAsync(authorization, ct);
            await foreach (var token in _tokenManager.FindByAuthorizationIdAsync(authId!, ct))
            {
                var status = await _tokenManager.GetStatusAsync(token, ct);
                if (status is Statuses.Revoked or Statuses.Expired)
                {
                    await _tokenManager.DeleteAsync(token, ct);
                }
            }
        }
    }
}

第三步：修改 AccountController

// Sso.AuthServer/Controllers/AccountController.cs
using Microsoft.AspNetCore.Authorization;
using Microsoft.AspNetCore.Identity;
using Microsoft.AspNetCore.Mvc;
using OpenIddict.Abstractions;
using OpenIddict.Validation.AspNetCore;
using Sso.Application.Services;
using Sso.Domain.Entities;
using Sso.Shared.Results;

namespace Sso.AuthServer.Controllers;

[ApiController]
[Route("api/account")]
public class AccountController : ControllerBase
{
    private readonly SignInManager<SsoUser> _signInManager;
    private readonly UserManager<SsoUser> _userManager;
    private readonly TokenRevocationService _tokenRevocation;

    public AccountController(
        SignInManager<SsoUser> signInManager,
        UserManager<SsoUser> userManager,
        TokenRevocationService tokenRevocation)
    {
        _signInManager = signInManager;
        _userManager = userManager;
        _tokenRevocation = tokenRevocation;
    }

    [HttpPost("login")]
    [AllowAnonymous]
    public async Task<IActionResult> Login([FromBody] LoginRequest request)
    {
        if (string.IsNullOrEmpty(request.Username) || string.IsNullOrEmpty(request.Password))
            return BadRequest(ApiResult.Fail("用户名和密码不能为空"));

        var user = await _userManager.FindByNameAsync(request.Username)
                   ?? await _userManager.FindByEmailAsync(request.Username);

        if (user == null)
            return BadRequest(ApiResult.Fail("用户名或密码错误"));

        if (!user.IsActive)
            return BadRequest(ApiResult.Fail("用户已被禁用"));

        var result = await _signInManager.PasswordSignInAsync(
            user, request.Password,
            isPersistent: request.RememberMe,
            lockoutOnFailure: true);

        if (result.IsLockedOut)
            return BadRequest(ApiResult.Fail("账户已被锁定，请15分钟后再试"));

        if (!result.Succeeded)
            return BadRequest(ApiResult.Fail("用户名或密码错误"));

        user.LastLoginAt = DateTime.UtcNow;
        await _userManager.UpdateAsync(user);

        return Ok(ApiResult<object>.Ok(new
        {
            user.Id,
            user.UserName,
            user.DisplayName,
            user.Email,
            returnUrl = request.ReturnUrl
        }, "登录成功"));
    }

    /// <summary>
    /// 登出 — Cookie 清除 + Token 全部撤销
    /// </summary>
    [HttpPost("logout")]
    public async Task<IActionResult> Logout()
    {
        // 1. 尝试从 Cookie 认证获取用户
        var user = await _userManager.GetUserAsync(User);

        // 2. 清除 Cookie
        await _signInManager.SignOutAsync();

        // 3. 撤销该用户所有 Token
        if (user != null)
        {
            await _tokenRevocation.RevokeAllUserTokensAsync(user.Id.ToString());
        }

        return Ok(ApiResult.Ok("已登出，所有令牌已撤销"));
    }

    /// <summary>
    /// 通过 Bearer Token 调用的登出（API 客户端场景）
    /// 前端拿着 access_token 来登出
    /// </summary>
    [HttpPost("logout-token")]
    [Authorize(AuthenticationSchemes = OpenIddictValidationAspNetCoreDefaults.AuthenticationScheme)]
    public async Task<IActionResult> LogoutByToken()
    {
        var userId = User.FindFirst(OpenIddictConstants.Claims.Subject)?.Value;

        if (string.IsNullOrEmpty(userId))
            return BadRequest(ApiResult.Fail("无法识别用户"));

        // 撤销该用户所有 Token
        await _tokenRevocation.RevokeAllUserTokensAsync(userId);

        return Ok(ApiResult.Ok("所有令牌已撤销"));
    }

    /// <summary>
    /// 修改密码后撤销所有 Token（强制重新登录）
    /// </summary>
    [HttpPost("change-password")]
    [Authorize]
    public async Task<IActionResult> ChangePassword([FromBody] ChangePasswordRequest request)
    {
        var user = await _userManager.GetUserAsync(User);
        if (user == null)
            return Unauthorized(ApiResult.Fail("未登录", 401));

        var result = await _userManager.ChangePasswordAsync(
            user, request.CurrentPassword, request.NewPassword);

        if (!result.Succeeded)
        {
            var errors = string.Join("; ", result.Errors.Select(e => e.Description));
            return BadRequest(ApiResult.Fail(errors));
        }

        // ▼▼▼ 改完密码 → 撤销所有 Token → 强制重新登录 ▼▼▼
        await _tokenRevocation.RevokeAllUserTokensAsync(user.Id.ToString());

        // 同时更新 Security Stamp（让所有现有 Cookie 也失效）
        await _userManager.UpdateSecurityStampAsync(user);

        return Ok(ApiResult.Ok("密码修改成功，请重新登录"));
    }

    [HttpGet("current")]
    [Authorize]
    public async Task<IActionResult> GetCurrentUser()
    {
        var user = await _userManager.GetUserAsync(User);
        if (user == null)
            return Unauthorized(ApiResult.Fail("未登录", 401));

        var roles = await _userManager.GetRolesAsync(user);

        return Ok(ApiResult<object>.Ok(new
        {
            user.Id,
            user.UserName,
            user.DisplayName,
            user.Email,
            user.Avatar,
            Roles = roles
        }));
    }
}

public record LoginRequest(
    string Username,
    string Password,
    bool RememberMe = false,
    string? ReturnUrl = null);

public record ChangePasswordRequest(
    string CurrentPassword,
    string NewPassword);

第四步：管理端强制踢人

// Sso.AuthServer/Controllers/UserManageController.cs 添加以下端点

/// <summary>
/// 强制踢出用户（撤销所有 Token）
/// </summary>
[HttpPost("{id:guid}/kick")]
public async Task<IActionResult> KickUser(Guid id)
{
    var user = await _userAppService.GetByIdAsync(id);
    if (user == null)
        return NotFound(ApiResult.Fail("用户不存在", 404));

    await _tokenRevocation.RevokeAllUserTokensAsync(id.ToString());
    return Ok(ApiResult.Ok($"用户 '{user.UserName}' 已被踢出"));
}

/// <summary>
/// 禁用用户（禁用 + 撤销 Token）
/// </summary>
[HttpPost("{id:guid}/disable")]
public async Task<IActionResult> DisableUser(Guid id)
{
    await _userAppService.UpdateAsync(id, new UpdateUserRequest { IsActive = false });
    await _tokenRevocation.RevokeAllUserTokensAsync(id.ToString());
    return Ok(ApiResult.Ok("用户已禁用，所有令牌已撤销"));
}

/// <summary>
/// 重置密码（管理员操作后也撤销 Token）
/// </summary>
[HttpPost("{id:guid}/reset-password")]
public async Task<IActionResult> ResetPassword(Guid id, [FromBody] ResetPasswordRequest request)
{
    try
    {
        await _userAppService.ResetPasswordAsync(id, request.NewPassword);
        await _tokenRevocation.RevokeAllUserTokensAsync(id.ToString());
        return Ok(ApiResult.Ok("密码重置成功，用户需要重新登录"));
    }
    catch (Exception ex)
    {
        return BadRequest(ApiResult.Fail(ex.Message));
    }
}

第六步：配置短过期 Token（纵深防御）

// appsettings.json
{
  "Sso": {
    "OpenIddict": {
      "AccessTokenLifetimeMinutes": 5,
      "RefreshTokenLifetimeDays": 7,
      "UseDevelopmentCertificates": true
    }
  }
}
为什么短过期也重要？

即使用了引用Token + 撤销，也建议短过期:
  - 减少 Token 表膨胀
  - Quartz 自动清理更高效
  - 即使某种极端情况撤销失败，5分钟后也自动失效

第七步：引用Token的性能优化（可选）

引用 Token 每次验证都查库，高并发场景需要加缓存：

// Sso.Shared/Options/CacheOptions.cs
namespace Sso.Shared.Options;

[ConfigSection("Sso:Cache")]
public class SsoCacheOptions
{
    public bool UseRedis { get; set; } = false;
    public string? RedisConnection { get; set; }

    /// <summary>
    /// Token 验证结果缓存时间（秒）
    /// 不要太长，否则撤销后还有缓存延迟
    /// </summary>
    public int TokenCacheSeconds { get; set; } = 30;
}
// Sso.Infrastructure/InfrastructureServiceExtensions.cs 添加缓存配置
public static IServiceCollection AddSsoInfrastructure(
    this IServiceCollection services,
    IConfiguration configuration)
{
    // ... EF Core, Identity 配置 ...

    // 缓存
    var cacheOptions = configuration
        .GetSection("Sso:Cache")
        .Get<SsoCacheOptions>() ?? new SsoCacheOptions();

    if (cacheOptions.UseRedis && !string.IsNullOrEmpty(cacheOptions.RedisConnection))
    {
        services.AddStackExchangeRedisCache(options =>
        {
            options.Configuration = cacheOptions.RedisConnection;
            options.InstanceName = "sso:";
        });
    }
    else
    {
        services.AddDistributedMemoryCache();
    }

    return services;
}
// appsettings.json
{
  "Sso": {
    "Cache": {
      "UseRedis": false,
      "RedisConnection": "localhost:6379",
      "TokenCacheSeconds": 30
    }
  }
}

完整验证流程

# 1. 获取 Token
curl -X POST https://localhost:7008/connect/token \
  -d "grant_type=password&username=admin&password=Admin@123456\
&scope=openid profile email api&client_id=react-app"

# 返回 (注意：引用 Token 是短字符串，不再是长 JWT)
# {"access_token":"3F2504E0...","refresh_token":"8A7B3C...","expires_in":300}

# 2. 用 Token 访问 API — 正常 ✅
curl -X GET https://localhost:7008/connect/userinfo \
  -H "Authorization: Bearer 3F2504E0..."
# 返回用户信息

# 3. 登出
curl -X POST https://localhost:7008/api/account/logout-token \
  -H "Authorization: Bearer 3F2504E0..."
# {"success":true,"message":"所有令牌已撤销"}

# 4. 再次用同一 Token — 失败 ✅
curl -X GET https://localhost:7008/connect/userinfo \
  -H "Authorization: Bearer 3F2504E0..."
# 401 Unauthorized

# 5. Refresh Token 也失效 ✅
curl -X POST https://localhost:7008/connect/token \
  -d "grant_type=refresh_token&refresh_token=8A7B3C...&client_id=react-app"
# {"error":"invalid_grant","error_description":"令牌已失效"}
