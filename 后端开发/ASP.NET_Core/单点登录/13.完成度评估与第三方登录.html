二、第三方登录兼容性分析

完全可以扩展，而且架构已经天然支持。 原因如下：

当前架构:

  React SPA → SSO 鉴权中心 → 数据库(用户/Token)
                  ↑
                  │ 目前只有这一种登录方式
                  │
              本地账号密码


扩展后:

  React SPA → SSO 鉴权中心 → 数据库(用户/Token)
                  ↑
                  │ 多种登录方式
                  ├── 本地账号密码
                  ├── GitHub
                  ├── Google
                  ├── 微信
                  ├── 钉钉
                  ├── 企业微信
                  └── SAML/LDAP/AD

  关键点: 对接入的业务系统来说完全透明！
  业务系统只认 SSO 发的 Token，不关心用户怎么登录的

第三步：注册外部登录 Provider

// Sso.Infrastructure/Authentication/ExternalLoginExtensions.cs
using Microsoft.AspNetCore.Authentication;
using Microsoft.Extensions.Configuration;
using Microsoft.Extensions.DependencyInjection;
using Sso.Shared.Options;

namespace Sso.Infrastructure.Authentication;

public static class ExternalLoginExtensions
{
    public static AuthenticationBuilder AddSsoExternalLogins(
        this AuthenticationBuilder builder,
        IConfiguration configuration)
    {
        var options = configuration
            .GetSection("Sso:ExternalLogin")
            .Get<ExternalLoginOptions>();

        if (options == null) return builder;

        // ====== GitHub ======
        if (options.GitHub is { Enabled: true })
        {
            builder.AddGitHub("GitHub", "GitHub 登录", o =>
            {
                o.ClientId = options.GitHub.ClientId;
                o.ClientSecret = options.GitHub.ClientSecret;
                o.Scope.Add("user:email");
                o.Scope.Add("read:user");

                // 回调路径（ASP.NET Core 自动处理）
                o.CallbackPath = "/signin-github";

                // 映射 Claims
                o.ClaimActions.MapJsonKey("urn:github:avatar", "avatar_url");
                o.ClaimActions.MapJsonKey("urn:github:url", "html_url");

                // 保存 Token（可选，用于后续调用 GitHub API）
                o.SaveTokens = true;
            });
        }

        // ====== Google ======
        if (options.Google is { Enabled: true })
        {
            builder.AddGoogle("Google", "Google 登录", o =>
            {
                o.ClientId = options.Google.ClientId;
                o.ClientSecret = options.Google.ClientSecret;
                o.CallbackPath = "/signin-google";
            });
        }

        // ====== 微信开放平台 ======
        if (options.WeChat is { Enabled: true })
        {
            builder.AddWeChatOpen("WeChat", "微信登录", o =>
            {
                o.ClientId = options.WeChat.AppId;
                o.ClientSecret = options.WeChat.AppSecret;
                o.CallbackPath = "/signin-wechat";

                o.ClaimActions.MapJsonKey("urn:wechat:avatar", "headimgurl");
                o.ClaimActions.MapJsonKey("urn:wechat:nickname", "nickname");
                o.ClaimActions.MapJsonKey("urn:wechat:unionid", "unionid");
            });
        }

        // ====== 钉钉 ======
        if (options.DingTalk is { Enabled: true })
        {
            builder.AddDingTalk("DingTalk", "钉钉登录", o =>
            {
                o.ClientId = options.DingTalk.AppKey;
                o.ClientSecret = options.DingTalk.AppSecret;
                o.CallbackPath = "/signin-dingtalk";
            });
        }

        return builder;
    }
}

第五步：外部登录 Controller — 核心逻辑

// Sso.AuthServer/Controllers/ExternalLoginController.cs
using System.Security.Claims;
using Microsoft.AspNetCore.Authentication;
using Microsoft.AspNetCore.Identity;
using Microsoft.AspNetCore.Mvc;
using Sso.Domain.Entities;
using Sso.Shared.Results;

namespace Sso.AuthServer.Controllers;

[ApiController]
[Route("api/external")]
public class ExternalLoginController : ControllerBase
{
    private readonly SignInManager<SsoUser> _signInManager;
    private readonly UserManager<SsoUser> _userManager;

    public ExternalLoginController(
        SignInManager<SsoUser> signInManager,
        UserManager<SsoUser> userManager)
    {
        _signInManager = signInManager;
        _userManager = userManager;
    }

    /// <summary>
    /// 获取已启用的外部登录提供商列表
    /// React 前端根据这个列表渲染登录按钮
    /// </summary>
    [HttpGet("providers")]
    public async Task<IActionResult> GetProviders()
    {
        var schemes = await _signInManager.GetExternalAuthenticationSchemesAsync();
        var providers = schemes.Select(s => new
        {
            s.Name,
            s.DisplayName
        });

        return Ok(ApiResult<object>.Ok(providers));
    }

    /// <summary>
    /// 发起外部登录
    /// React 前端: window.location.href = '/api/external/challenge/GitHub?returnUrl=xxx'
    /// </summary>
    [HttpGet("challenge/{provider}")]
    public IActionResult Challenge(
        string provider,
        [FromQuery] string? returnUrl = null)
    {
        // returnUrl 是用户登录成功后要跳回的地址
        // 通常是 OAuth 的 authorize 端点 URL 或前端页面
        var callbackUrl = Url.Action(
            nameof(Callback),
            new { provider, returnUrl });

        var properties = _signInManager.ConfigureExternalAuthenticationProperties(
            provider, callbackUrl);

        return Challenge(properties, provider);
    }

    /// <summary>
    /// 外部登录回调 — 核心关联逻辑
    /// 外部 Provider 认证成功后回调到这里
    /// </summary>
    [HttpGet("callback")]
    public async Task<IActionResult> Callback(
        string provider,
        string? returnUrl = null)
    {
        // 1. 获取外部登录信息
        var info = await _signInManager.GetExternalLoginInfoAsync();
        if (info == null)
        {
            return Redirect($"/login?error=external_login_failed&provider={provider}");
        }

        // 2. 尝试用外部登录信息直接登录（已关联的用户）
        var signInResult = await _signInManager.ExternalLoginSignInAsync(
            info.LoginProvider,
            info.ProviderKey,
            isPersistent: true,
            bypassTwoFactor: true);

        if (signInResult.Succeeded)
        {
            // 已关联，直接登录成功，更新外部 Token
            await UpdateExternalTokensAsync(info);

            return Redirect(returnUrl ?? "/");
        }

        // 3. 未关联 → 自动创建或关联用户
        var user = await FindOrCreateUserFromExternalAsync(info);
        if (user == null)
        {
            return Redirect("/login?error=external_user_creation_failed");
        }

        // 4. 添加外部登录关联
        var addLoginResult = await _userManager.AddLoginAsync(user, info);
        if (!addLoginResult.Succeeded)
        {
            return Redirect("/login?error=external_link_failed");
        }

        // 5. 登录
        await _signInManager.SignInAsync(user, isPersistent: true);
        await UpdateExternalTokensAsync(info);

        // 更新最后登录时间
        user.LastLoginAt = DateTime.UtcNow;
        await _userManager.UpdateAsync(user);

        return Redirect(returnUrl ?? "/");
    }

    /// <summary>
    /// 已登录用户关联外部账号
    /// </summary>
    [HttpPost("link/{provider}")]
    public IActionResult LinkExternalLogin(
        string provider,
        [FromQuery] string? returnUrl = null)
    {
        var userId = _userManager.GetUserId(User);
        if (string.IsNullOrEmpty(userId))
            return Unauthorized(ApiResult.Fail("请先登录"));

        var callbackUrl = Url.Action(nameof(LinkCallback), new { returnUrl });
        var properties = _signInManager.ConfigureExternalAuthenticationProperties(
            provider, callbackUrl);

        // 传递当前用户ID到回调
        properties.Items["LoginProvider"] = provider;
        properties.Items["UserId"] = userId;

        return Challenge(properties, provider);
    }

    /// <summary>
    /// 关联回调
    /// </summary>
    [HttpGet("link-callback")]
    public async Task<IActionResult> LinkCallback(string? returnUrl = null)
    {
        var info = await _signInManager.GetExternalLoginInfoAsync(
            await _userManager.GetUserIdAsync(
                (await _userManager.GetUserAsync(User))!));

        if (info == null)
            return Redirect("/account/settings?error=link_failed");

        var user = await _userManager.GetUserAsync(User);
        if (user == null)
            return Redirect("/login");

        var result = await _userManager.AddLoginAsync(user, info);

        return result.Succeeded
            ? Redirect(returnUrl ?? "/account/settings?linked=true")
            : Redirect("/account/settings?error=link_failed");
    }

    /// <summary>
    /// 取消关联外部账号
    /// </summary>
    [HttpPost("unlink")]
    public async Task<IActionResult> UnlinkExternalLogin([FromBody] UnlinkRequest request)
    {
        var user = await _userManager.GetUserAsync(User);
        if (user == null)
            return Unauthorized(ApiResult.Fail("请先登录"));

        // 安全检查：不能取消所有登录方式
        var logins = await _userManager.GetLoginsAsync(user);
        var hasPassword = await _userManager.HasPasswordAsync(user);

        if (logins.Count <= 1 && !hasPassword)
            return BadRequest(ApiResult.Fail("至少保留一种登录方式"));

        var result = await _userManager.RemoveLoginAsync(
            user, request.Provider, request.ProviderKey);

        return result.Succeeded
            ? Ok(ApiResult.Ok("取消关联成功"))
            : BadRequest(ApiResult.Fail("取消关联失败"));
    }

    /// <summary>
    /// 获取当前用户已关联的外部登录
    /// </summary>
    [HttpGet("linked")]
    public async Task<IActionResult> GetLinkedLogins()
    {
        var user = await _userManager.GetUserAsync(User);
        if (user == null)
            return Unauthorized(ApiResult.Fail("请先登录"));

        var logins = await _userManager.GetLoginsAsync(user);
        var allProviders = await _signInManager.GetExternalAuthenticationSchemesAsync();

        var result = allProviders.Select(p => new
        {
            Provider = p.Name,
            p.DisplayName,
            IsLinked = logins.Any(l => l.LoginProvider == p.Name),
            ProviderKey = logins.FirstOrDefault(l => l.LoginProvider == p.Name)?.ProviderKey
        });

        return Ok(ApiResult<object>.Ok(result));
    }

    // ======================
    // 私有方法
    // ======================

    /// <summary>
    /// 根据外部登录信息查找或创建本地用户
    /// </summary>
    private async Task<SsoUser?> FindOrCreateUserFromExternalAsync(ExternalLoginInfo info)
    {
        var email = info.Principal.FindFirstValue(ClaimTypes.Email);
        var name = info.Principal.FindFirstValue(ClaimTypes.Name);
        var avatar = info.Principal.FindFirstValue("urn:github:avatar")
                     ?? info.Principal.FindFirstValue("urn:wechat:avatar");

        // 策略1: 如果有 Email，先找已有用户（支持多个外部账号关联同一用户）
        SsoUser? user = null;
        if (!string.IsNullOrEmpty(email))
        {
            user = await _userManager.FindByEmailAsync(email);
        }

        // 策略2: 创建新用户
        if (user == null)
        {
            var username = await GenerateUniqueUsernameAsync(
                name ?? info.ProviderKey,
                info.LoginProvider);

            user = new SsoUser
            {
                UserName = username,
                Email = email,
                EmailConfirmed = !string.IsNullOrEmpty(email), // 外部 Provider 已验证
                DisplayName = name ?? username,
                Avatar = avatar,
                IsActive = true,
                LastLoginAt = DateTime.UtcNow
            };

            var createResult = await _userManager.CreateAsync(user);
            if (!createResult.Succeeded)
                return null;

            // 给外部登录用户一个默认角色
            await _userManager.AddToRoleAsync(user, "user");
        }
        else
        {
            // 更新已有用户的信息
            if (!string.IsNullOrEmpty(avatar) && string.IsNullOrEmpty(user.Avatar))
                user.Avatar = avatar;

            if (!string.IsNullOrEmpty(name) && string.IsNullOrEmpty(user.DisplayName))
                user.DisplayName = name;

            user.LastLoginAt = DateTime.UtcNow;
            await _userManager.UpdateAsync(user);
        }

        return user;
    }

    /// <summary>
    /// 生成唯一用户名
    /// </summary>
    private async Task<string> GenerateUniqueUsernameAsync(string baseName, string provider)
    {
        // 清理非法字符
        var clean = new string(baseName
            .Where(c => char.IsLetterOrDigit(c) || c == '_' || c == '-')
            .ToArray());

        if (string.IsNullOrEmpty(clean))
            clean = provider.ToLower();

        var username = clean;
        var counter = 1;

        while (await _userManager.FindByNameAsync(username) != null)
        {
            username = $"{clean}_{counter++}";
        }

        return username;
    }

    /// <summary>
    /// 保存外部登录的 Token（可选，用于后续调用外部 API）
    /// </summary>
    private async Task UpdateExternalTokensAsync(ExternalLoginInfo info)
    {
        if (info.AuthenticationTokens?.Any() == true)
        {
            var user = await _userManager.FindByLoginAsync(
                info.LoginProvider, info.ProviderKey);

            if (user != null)
            {
                foreach (var token in info.AuthenticationTokens)
                {
                    await _userManager.SetAuthenticationTokenAsync(
                        user, info.LoginProvider, token.Name, token.Value);
                }
            }
        }
    }
}

public record UnlinkRequest(string Provider, string ProviderKey);

      .then(data => setProviders(data.data));
  }, []);

  const handleExternalLogin = (provider: string) => {
    const url = `/api/external/challenge/${provider}?returnUrl=${encodeURIComponent(returnUrl || '/')}`;
    window.location.href = url;
  };

  const iconMap: Record<string, string> = {
    GitHub: '🐙',
    Google: '🔍',
    WeChat: '💬',
    DingTalk: '🔔',
  };

  if (providers.length === 0) return null;

  return (
    <div className="external-logins">
      <div className="divider">其他登录方式</div>
      <div className="provider-buttons">
        {providers.map(p => (
          <button
            key={p.name}
            onClick={() => handleExternalLogin(p.name)}
            className={`btn-external btn-${p.name.toLowerCase()}`}
          >
            {iconMap[p.name] || '🔗'} {p.displayName}
          </button>
        ))}
      </div>
    </div>
  );
};

export default ExternalLoginButtons;

总结：当前架构的扩展能力非常好，第三方登录、双因素认证、多租户等都可以在不改动核心代码的前提下扩展进去。对下游业务系统完全透明。

你想先实现哪个模块？我建议先做 P0 的资源服务器接入示例，这样整个 SSO 的价值才能真正体现出来。
