如何在真实项目中使用

10 - 如何在真实项目中使用

本章提供实战指南——如何在真实的 ASP.NET Core Web 项目中使用 MokFramework。

快速开始：Web API 项目

项目结构

MyApp/
├── MyApp.sln
├── MyApp.Core/                      # 领域层
│   ├── MyApp.Core.csproj
│   └── Services/
│       ├── IUserService.cs
│       └── UserService.cs           : IScopedDependency
├── MyApp.Infrastructure/            # 基础设施层
│   ├── MyApp.Infrastructure.csproj
│   └── InfrastructureModule.cs      : MokFrameworkModule
├── MyApp.Web/                       # Web 层
│   ├── MyApp.Web.csproj
│   ├── Program.cs
│   ├── WebModule.cs                 : MokFrameworkModule
│   └── Controllers/
│       └── UserController.cs
└── packages/
    ├── MokFramework.Core
    ├── MokFramework.AspNetCore
    ├── MokFramework.Autofac
    └── MokFramework.Serilog

Program.cs

using Serilog;

var builder = WebApplication.CreateBuilder(args);

// ① 加载 secrets 配置
builder.Host.AddAppSettingsSecretsJson();

// ② 使用 Autofac 替换 DI 容器（可选）
builder.Host.UseAutofac();

// ③ 使用 Serilog 替换日志系统（可选）
builder.Host.UseMokSerilog((context, config) =>
{
    config.ReadFrom.Configuration(context.Configuration);
});

// ④ 注册 MokFramework 模块化应用
await builder.AddApplicationAsync<WebModule>();

var app = builder.Build();

// ⑤ Serilog 请求日志（可选）
app.UseSerilogRequestLogging();

// ⑥ 中间件管道
app.UseRouting();
app.UseAuthentication();
app.UseAuthorization();
app.MapControllers();

// ⑦ 初始化模块
await app.InitializeApplicationAsync();

// ⑧ 运行
await app.RunAsync();

WebModule.cs

using MokFramework.AspNetCore;
using MokFramework.Core.Modularity;

[DependsOn(typeof(MokFrameworkAspNetCoreModule))]  // ASP.NET Core 基础
[DependsOn(typeof(InfrastructureModule))]           // 基础设施
public class WebModule : MokFrameworkModule
{
    public override void ConfigureServices(ServiceConfigurationContext context)
    {
        // 注册 Controllers
        context.Services.AddControllers();

        // 注册 Swagger
        context.Services.AddEndpointsApiExplorer();
        context.Services.AddSwaggerGen();

        // 配置 CORS
        context.Services.AddCors(options =>
        {
            options.AddDefaultPolicy(policy =>
            {
                policy.AllowAnyOrigin()
                      .AllowAnyMethod()
                      .AllowAnyHeader();
            });
        });
    }

    public override void OnApplicationInitialization(ApplicationInitializationContext context)
    {
        var logger = context.ServiceProvider
            .GetRequiredService<ILogger<WebModule>>();
        logger.LogInformation("WebModule initialized successfully");
    }
}

InfrastructureModule.cs

using MokFramework.Core.Modularity;

public class InfrastructureModule : MokFrameworkModule
{
    public override void ConfigureServices(ServiceConfigurationContext context)
    {
        // 注册 EF Core
        context.Services.AddDbContext<AppDbContext>(options =>
        {
            options.UseNpgsql(
                context.Services.GetConfiguration()
                    .GetConnectionString("Default"));
        });

        // 注册 Redis 缓存
        context.Services.AddStackExchangeRedisCache(options =>
        {
            options.Configuration = "localhost:6379";
        });
    }
}

UserService.cs

using MokFramework.Core.DependencyInjection;

// 实现 IScopedDependency → 自动注册为 Scoped
// 实现 IUserService → 按 I{ClassName} 约定匹配
public class UserService : IUserService, IScopedDependency
{
    private readonly AppDbContext _db;

    public UserService(AppDbContext db)
    {
        _db = db;
    }

    public async Task<User?> GetByIdAsync(int id)
    {
        return await _db.Users.FindAsync(id);
    }
}

UserController.cs

[ApiController]
[Route("api/[controller]")]
public class UserController : ControllerBase
{
    private readonly IUserService _userService;

    // IUserService 由框架自动注册，直接注入
    public UserController(IUserService userService)
    {
        _userService = userService;
    }

    [HttpGet("{id}")]
    public async Task<IActionResult> Get(int id)
    {
        var user = await _userService.GetByIdAsync(id);
        return user is null ? NotFound() : Ok(user);
    }
}

模块设计指南

模块粒度

推荐：一个 NuGet 包/项目 = 一个模块

// ✅ 好的模块划分
MyApp.Core                 → CoreModule
MyApp.Infrastructure       → InfrastructureModule
MyApp.Web                  → WebModule
MyApp.BackgroundJobs       → BackgroundJobsModule

// ❌ 不好的模块划分
MyApp.Everything           → 一个巨大的 EverythingModule
MyApp.UserModule           → 一个功能 = 一个模块（太细）

模块依赖原则

WebModule (顶层，面向用户)
  ├── [DependsOn] MokFrameworkAspNetCoreModule
  ├── [DependsOn] InfrastructureModule
  └── [DependsOn] CoreModule

InfrastructureModule (中间层，技术实现)
  └── [DependsOn] CoreModule

CoreModule (底层，领域模型和接口)
  └── 无依赖

原则：依赖方向只能向下（高层依赖低层），禁止循环依赖。

选择注册方式

需要自动注册？
├── 是 → 大部分业务服务使用标记接口（最简洁）
│   ├── Transient (无状态服务) → ITransientDependency
│   ├── Scoped (有请求状态) → IScopedDependency
│   └── Singleton (全局状态) → ISingletonDependency
│
├── 需要精确控制暴露类型？ → [Dependency] + [ExposeServices]
│
└── 不需要自动注册（第三方库、特殊配置）
    → 在模块的 ConfigureServices 中手动注册

常见模式

模式 1：Options 配置

// 定义 Options
public class EmailOptions
{
    public string SmtpHost { get; set; } = "localhost";
    public int SmtpPort { get; set; } = 25;
    public string? ApiKey { get; set; }
}

// 在模块中配置
public override void ConfigureServices(ServiceConfigurationContext context)
{
    var config = context.Services.GetConfiguration();

    context.Services.Configure<EmailOptions>(
        config.GetSection("Email"));
}

// 在服务中使用
public class EmailService : IEmailService, ITransientDependency
{
    private readonly EmailOptions _options;

    public EmailService(IOptions<EmailOptions> options)
    {
        _options = options.Value;
    }
}

模式 2：模块间数据共享

// Module A：在 PreConfigureServices 中共享数据
public override void PreConfigureServices(ServiceConfigurationContext context)
{
    context["ConnectionStrings:Default"] = "Host=localhost;Database=mydb";
}

// Module B：在 ConfigureServices 中读取
public override void ConfigureServices(ServiceConfigurationContext context)
{
    var connStr = (string?)context["ConnectionStrings:Default"];
    if (connStr != null)
    {
        context.Services.AddDbContext<AppDbContext>(o => o.UseNpgsql(connStr));
    }
}

模式 3：使用 Autofac 高级功能

[DependsOn(typeof(MokFrameworkAutofacModule))]
public class AopModule : MokFrameworkModule
{
    public override void ConfigureServices(ServiceConfigurationContext context)
    {
        var containerBuilder = context.Services
            .GetSingletonInstance<ObjectAccessor<ContainerBuilder>>()
            .Value!;

        // 注册带拦截器的服务
        containerBuilder.RegisterType<AuditLogInterceptor>();

        containerBuilder.RegisterType<OrderService>()
            .As<IOrderService>()
            .EnableInterfaceInterceptors()
            .InterceptedBy(typeof(AuditLogInterceptor))
            .InstancePerLifetimeScope();
    }
}

模式 4：后台服务

public class HealthCheckWorker : BackgroundService, ISingletonDependency
{
    private readonly IServiceScopeFactory _scopeFactory;

    public HealthCheckWorker(IServiceScopeFactory scopeFactory)
    {
        _scopeFactory = scopeFactory;
    }

    protected override async Task ExecuteAsync(CancellationToken stoppingToken)
    {
        while (!stoppingToken.IsCancellationRequested)
        {
            using var scope = _scopeFactory.CreateScope();
            var healthService = scope.ServiceProvider
                .GetRequiredService<IHealthCheckService>();

            await healthService.CheckAsync();
            await Task.Delay(TimeSpan.FromMinutes(1), stoppingToken);
        }
    }
}
