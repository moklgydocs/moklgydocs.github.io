存储过程与企业级规范

存储过程是 SQL Server 应用中最核心的服务端编程单元。 写好存储过程不只是语法问题，更是事务、错误处理、安全和规范的系统工程。

与现有内容的关系

本篇聚焦存储过程的企业级规范与最佳实践。更多存储过程编写规范，参考 。

一、存储过程基础

1.1 创建与执行

-- 创建存储过程
CREATE PROCEDURE dbo.usp_GetCustomerOrders
    @CustomerId INT,
    @StartDate DATETIME2(3) = '2024-01-01',  -- 默认参数
    @MaxRows INT = 100
AS
BEGIN
    SET NOCOUNT ON;  -- 减少网络流量，不返回 "受影响的行数"

    SELECT TOP (@MaxRows)
        o.OrderId, o.OrderDate, o.Amount,
        c.CustomerName
    FROM Orders o
    JOIN Customers c ON o.CustomerId = c.CustomerId
    WHERE o.CustomerId = @CustomerId
      AND o.OrderDate >= @StartDate
    ORDER BY o.OrderDate DESC;
END

-- 执行
EXEC dbo.usp_GetCustomerOrders @CustomerId = 1001;
EXEC dbo.usp_GetCustomerOrders @CustomerId = 1001, @StartDate = '2024-06-01';
EXEC dbo.usp_GetCustomerOrders 1001, '2024-06-01', 50;  -- 位置参数（不推荐）

1.2 OUTPUT 参数和 RETURN

CREATE PROCEDURE dbo.usp_TransferFunds
    @FromAccountId INT,
    @ToAccountId INT,
    @Amount DECIMAL(18,2),
    @NewBalance DECIMAL(18,2) OUTPUT,  -- 输出参数
    @ErrorCode INT = 0 OUTPUT
AS
BEGIN
    SET NOCOUNT ON;

    BEGIN TRY
        BEGIN TRANSACTION;

        UPDATE Accounts SET Balance = Balance - @Amount
        WHERE AccountId = @FromAccountId;

        IF @@ROWCOUNT = 0
        BEGIN
            ROLLBACK TRANSACTION;
            SET @ErrorCode = 1;  -- 源账户不存在
            RETURN 1;
        END

        UPDATE Accounts SET Balance = Balance + @Amount
        WHERE AccountId = @ToAccountId;

        IF @@ROWCOUNT = 0
        BEGIN
            ROLLBACK TRANSACTION;
            SET @ErrorCode = 2;  -- 目标账户不存在
            RETURN 2;
        END

        SELECT @NewBalance = Balance
        FROM Accounts WHERE AccountId = @FromAccountId;

        COMMIT TRANSACTION;
        RETURN 0;  -- 成功
    END TRY
    BEGIN CATCH
        IF @@TRANCOUNT > 0 ROLLBACK TRANSACTION;
        SET @ErrorCode = ERROR_NUMBER();
        RETURN ERROR_NUMBER();
    END CATCH
END

-- 调用带 OUTPUT 参数
DECLARE @Balance DECIMAL(18,2), @ErrCode INT;
EXEC dbo.usp_TransferFunds
    @FromAccountId = 1,
    @ToAccountId = 2,
    @Amount = 500.00,
    @NewBalance = @Balance OUTPUT,
    @ErrorCode = @ErrCode OUTPUT;

PRINT '新余额: ' + CAST(@Balance AS NVARCHAR(20));
PRINT '错误码: ' + CAST(@ErrCode AS NVARCHAR(10));

二、错误处理

2.1 TRY...CATCH 完整模式

CREATE PROCEDURE dbo.usp_CreateOrder
    @CustomerId INT,
    @Items OrderItemTable READONLY  -- 表值参数
AS
BEGIN
    SET NOCOUNT ON;
    SET XACT_ABORT ON;  -- ✅ 运行时错误自动回滚事务

    DECLARE @OrderId INT;
    DECLARE @ErrorMessage NVARCHAR(4000);
    DECLARE @ErrorSeverity INT;
    DECLARE @ErrorState INT;
    DECLARE @ErrorProcedure NVARCHAR(128);
    DECLARE @ErrorLine INT;
    DECLARE @ErrorNumber INT;

    BEGIN TRY
        BEGIN TRANSACTION;

        -- 创建订单
        INSERT INTO Orders (CustomerId, OrderDate, Status)
        VALUES (@CustomerId, SYSDATETIME(), 'Pending');

        SET @OrderId = SCOPE_IDENTITY();

        -- 插入订单项
        INSERT INTO OrderItems (OrderId, ProductId, Quantity, UnitPrice)
        SELECT @OrderId, ProductId, Quantity, UnitPrice
        FROM @Items;

        -- 更新库存
        UPDATE Inventory
        SET Stock = Stock - i.Quantity
        FROM Inventory inv
        JOIN @Items i ON inv.ProductId = i.ProductId;

        COMMIT TRANSACTION;
        SELECT @OrderId AS OrderId;
    END TRY
    BEGIN CATCH
        IF @@TRANCOUNT > 0
            ROLLBACK TRANSACTION;

        -- 记录错误日志
        SET @ErrorNumber = ERROR_NUMBER();
        SET @ErrorMessage = ERROR_MESSAGE();
        SET @ErrorSeverity = ERROR_SEVERITY();
        SET @ErrorState = ERROR_STATE();
        SET @ErrorProcedure = ERROR_PROCEDURE();
        SET @ErrorLine = ERROR_LINE();

        INSERT INTO ErrorLog (ErrorNumber, ErrorMessage, ErrorSeverity,
            ErrorState, ErrorProcedure, ErrorLine, ErrorTime)
        VALUES (@ErrorNumber, @ErrorMessage, @ErrorSeverity,
            @ErrorState, @ErrorProcedure, @ErrorLine, SYSDATETIME());

        -- 重新抛出错误（2012+）
        THROW;
    END CATCH
END

2.2 RAISERROR vs THROW

-- RAISERROR（旧方式，仍支持）
RAISERROR('库存不足: 产品ID %d, 库存 %d', 16, 1, @ProductId, @Stock);
-- 可格式化消息，可指定严重级别和状态

-- THROW（2012+ 推荐）
THROW 50001, '库存不足，无法创建订单', 1;
-- 错误号必须 >= 50000
-- 无格式化，但直接在 CATCH 中使用可保留原始错误信息

-- ✅ 最佳实践：CATCH 中直接 THROW（保留原始调用栈）
BEGIN CATCH
    IF @@TRANCOUNT > 0 ROLLBACK TRANSACTION;
    THROW;  -- 不带参数，重新抛出原始错误
END CATCH

三、事务嵌套

-- 事务嵌套的正确处理
CREATE PROCEDURE dbo.usp_ParentProc
AS
BEGIN
    SET NOCOUNT ON;
    SET XACT_ABORT ON;

    BEGIN TRANSACTION;
    SAVE TRANSACTION SavePoint1;  -- ✅ 保存点

    -- 调用子存储过程（内部也有事务）
    EXEC dbo.usp_ChildProc;  -- 如果失败，只回滚到保存点

    COMMIT TRANSACTION;
END

-- 子存储过程的事务处理模式
CREATE PROCEDURE dbo.usp_ChildProc
AS
BEGIN
    SET NOCOUNT ON;

    -- ✅ 正确：检查是否已在事务中
    DECLARE @TranCount INT = @@TRANCOUNT;

    IF @TranCount = 0
        BEGIN TRANSACTION usp_ChildProc;
    ELSE
        SAVE TRANSACTION usp_ChildProc;

    BEGIN TRY
        -- 业务逻辑
        UPDATE Accounts SET Balance = Balance - 100 WHERE AccountId = 1;

        IF @TranCount = 0
            COMMIT TRANSACTION usp_ChildProc;
    END TRY
    BEGIN CATCH
        IF @TranCount = 0
            ROLLBACK TRANSACTION usp_ChildProc;
        ELSE
            ROLLBACK TRANSACTION usp_ChildProc;  -- 回滚到保存点

        THROW;  -- 向上传播错误
    END CATCH
END

事务嵌套陷阱

  • SQL Server 的事务没有真正的嵌套——@@TRANCOUNT 只是计数器
  • COMMIT 只减少计数，只有最外层 COMMIT 才真正提交
  • ROLLBACK 回滚所有事务（不管嵌套层级），无法只回滚内层
  • 使用 SAVE TRANSACTION 实现部分回滚
  • 最佳实践：SET XACT_ABORT ON + 顶层统一事务管理

四、重新编译

-- 方式 1：存储过程级别重新编译
CREATE PROCEDURE dbo.usp_SearchOrders
    @CustomerId INT
WITH RECOMPILE  -- 每次执行都重新编译
AS
    SELECT * FROM Orders WHERE CustomerId = @CustomerId;

-- 方式 2：单次执行重新编译
EXEC dbo.usp_SearchOrders @CustomerId = 1001 WITH RECOMPILE;

-- 方式 3：语句级别 OPTION (RECOMPILE)
CREATE PROCEDURE dbo.usp_SearchOrders
    @CustomerId INT,
    @StartDate DATETIME2(3) = NULL
AS
BEGIN
    SET NOCOUNT ON;

    SELECT * FROM Orders
    WHERE CustomerId = @CustomerId
      AND (@StartDate IS NULL OR OrderDate >= @StartDate)
    OPTION (RECOMPILE);  -- 只重新编译这条语句
END

-- 方式 4：标记存储过程下次执行时重编译
EXEC sp_recompile 'dbo.usp_SearchOrders';

-- 方式 5：清除整个计划缓存（慎用！）
DBCC FREEPROCCACHE;

何时使用 RECOMPILE

  • ✅ 参数值选择性差异大（参数嗅探问题）
  • ✅ 临时表数据量变化大
  • ❌ 高频执行的存储过程（编译开销不可接受）
  • 折中方案：只在问题语句加 OPTION (RECOMPILE)

五、安全：所有权链与 EXECUTE AS

5.1 所有权链（Ownership Chaining）

-- 当存储过程和表属于同一所有者时，调用者无需直接权限
-- 用户只需 EXEC 存储过程的权限，不需要 SELECT 表的权限

GRANT EXECUTE ON dbo.usp_GetCustomerOrders TO AppUser;
-- AppUser 可以通过存储过程读取 Orders 表
-- 但不能直接 SELECT * FROM Orders

5.2 EXECUTE AS

-- 存储过程以指定身份执行
CREATE PROCEDURE dbo.usp_SensitiveReport
WITH EXECUTE AS OWNER  -- 以存储过程所有者身份执行
AS
BEGIN
    SELECT * FROM SensitiveData;
END

-- EXECUTE AS 选项：
-- CALLER：以调用者身份（默认）
-- SELF：以创建者身份
-- OWNER：以所有者身份
-- 'user_name'：以指定用户身份

六、ABP Framework 的存储过程模式

ABP Framework 虽然主要通过 EF Core 操作数据库，但也支持存储过程：

// ABP 执行存储过程的方式
public class OrderRepository : EfCoreRepository<MyDbContext, Order, Guid>, IOrderRepository
{
    public OrderRepository(IDbContextProvider<MyDbContext> dbContextProvider)
        : base(dbContextProvider)
    {
    }

    public async Task<List<OrderReportDto>> GetOrderReportAsync(DateTime startDate, DateTime endDate)
    {
        var dbContext = await GetDbContextAsync();

        // 方式 1：FromSqlRaw（EF Core）
        var results = await dbContext.OrderReports
            .FromSqlRaw("EXEC dbo.usp_GetOrderReport @p0, @p1", startDate, endDate)
            .ToListAsync();

        return results;
    }

    public async Task<decimal> CalculateTotalAsync(int customerId)
    {
        var dbContext = await GetDbContextAsync();
        var connection = dbContext.Database.GetDbConnection();

        // 方式 2：ADO.NET 直接调用
        await using var command = connection.CreateCommand();
        command.CommandText = "dbo.usp_CalculateCustomerTotal";
        command.CommandType = CommandType.StoredProcedure;

        var param = command.CreateParameter();
        param.ParameterName = "@CustomerId";
        param.Value = customerId;
        command.Parameters.Add(param);

        var result = await command.ExecuteScalarAsync();
        return Convert.ToDecimal(result);
    }
}

7.2 代码审查清单

## 存储过程代码审查清单

### 必须项
- [ ] SET NOCOUNT ON
- [ ] SET XACT_ABORT ON（有事务时）
- [ ] TRY...CATCH 错误处理
- [ ] 事务中检查 @@TRANCOUNT
- [ ] 参数化查询（无字符串拼接 SQL）
- [ ] 避免SELECT *（明确列出列名）
- [ ] 使用 SCOPE_IDENTITY() 而非 @@IDENTITY

### 性能项
- [ ] 临时表 vs 表变量选择合理
- [ ] 大数据量操作使用分批处理
- [ ] 避免 WHERE 中的非 SARGable 条件
- [ ] 关键查询有覆盖索引支持

### 安全项
- [ ] 无动态 SQL 拼接（或使用 QUOTENAME/QUOTEIDENTIFIER）
- [ ] 最小权限原则
- [ ] EXECUTE AS 使用合理
- [ ] 敏感数据不记录到日志表

八、面试技巧

面试高频考点

  1. SET NOCOUNT ON：减少网络流量，存储过程必备
  2. SET XACT_ABORT ON：运行时错误自动回滚，避免事务悬挂
  3. TRY...CATCH + THROW：2012+ 推荐的错误处理模式
  4. @@TRANCOUNT 事务嵌套：嵌套事务的正确处理方式
  5. 所有权链：存储过程的安全优势——调用者无需直接表权限
  6. RECOMPILE 场景：参数嗅探问题时选择性使用
  7. SCOPE_IDENTITY vs @@IDENTITY：前者只返回当前作用域，后者包括触发器

易错点

  • "ROLLBACK 内层事务只回滚内层"——❌，ROLLBACK 回滚所有层级，用 SAVE TRANSACTION 实现部分回滚
  • "THROW 和 RAISERROR 一样"——❌，THROW 在 CATCH 中不带参数可保留原始调用栈
  • "存储过程比动态 SQL 快因为预编译"——❌，存储过程首次执行也需要编译，优势在于计划缓存复用和安全封装
  • "@@IDENTITY 和 SCOPE_IDENTITY() 一样"——❌，@@IDENTITY 包含触发器产生的 ID
