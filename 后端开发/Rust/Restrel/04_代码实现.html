//! 统一错误类型
//! 对应 C# 中分散在各层的异常，这里统一为一个 enum

use thiserror::Error;

#[derive(Debug, Error)]
pub enum ServerError {
    // ── 解析层（parser.rs 产生）──────────────────────────────────────
    /// 对端主动关闭连接（EOF）
    #[error("连接已关闭")]
    ConnectionClosed,

    /// headers 超过配置的最大字节数（对应 Kestrel 返回 431）
    #[error("请求头过大")]
    RequestHeadersTooLarge,

    /// 请求格式错误（对应 Kestrel 返回 400）
    #[error("请求格式错误：{0}")]
    BadRequest(String),

    /// TCP IO 错误
    #[error("IO 错误：{0}")]
    Io(String),

    // ── 路由层（router.rs 产生）──────────────────────────────────────
    #[error("路由未找到：{0}")]
    NotFound(String),

    #[error("方法不允许：{0}")]
    MethodNotAllowed(String),

    #[error("路由冲突：{path}（{cause}）")]
    RouteConflict { path: String, cause: String },

    // ── 鉴权层（auth_mw.rs 产生）─────────────────────────────────────
    #[error("未授权：{0}")]
    Unauthorized(String),

    // ── 数据库层（service_repo.rs 产生）──────────────────────────────
    #[error("数据库错误：{0}")]
    Database(#[from] sqlx::Error),

    // ── 序列化（handlers 产生）───────────────────────────────────────
    #[error("JSON 错误：{0}")]
    Json(#[from] serde_json::Error),
}

impl ServerError {
    /// 将错误映射到 HTTP 状态码
    pub fn status_code(&self) -> u16 {
        match self {
            Self::BadRequest(_)           => 400,
            Self::Unauthorized(_)         => 401,
            Self::NotFound(_)             => 404,
            Self::MethodNotAllowed(_)     => 405,
            Self::RequestHeadersTooLarge  => 431,
            _                             => 500,
        }
    }
}

1.2 parser.rs ★ 核心：手写 HTTP/1.1 解析

//! HTTP/1.1 请求解析器
//!
//! 对应 Kestrel 的两个核心组件：
//!   - `Http1Connection<TTransport>`：调用 llhttp 解析字节流
//!   - `System.IO.Pipelines.PipeReader`：管理缓冲区
//!
//! # 解析算法（Partial 状态机）
//! ```text
//! loop:
//!   ┌─ 尝试在 buf 中解析（httparse::parse）
//!   │    Complete(n) → n = headers 区字节数（含 \r\n\r\n）
//!   │    Partial     → 数据不足，继续读 socket
//!   │    Err         → 400 Bad Request
//!   └─ AsyncReadExt::read_buf(&mut buf)
//! ```

use std::collections::HashMap;

use bytes::{Bytes, BytesMut};
use tokio::io::AsyncReadExt;

use crate::error::ServerError;

const MAX_HEADERS: usize = 64;

/// 触发 431 的 headers 区最大字节数
/// 对应 KestrelServerLimits.MaxRequestHeadersTotalSize（Kestrel 默认 32 KB）
pub const MAX_HEADER_BYTES: usize = 32 * 1024;

/// 从 TCP 流解析得到的原始 HTTP/1.1 请求
#[derive(Debug)]
pub struct RawRequest {
    pub method:  String,
    /// 完整 request-target（含 query string），例如 "/services?page=1"
    pub path:    String,
    /// 0 = HTTP/1.0   1 = HTTP/1.1
    pub version: u8,
    /// 所有请求头，key 已 to_ascii_lowercase()
    pub headers: HashMap<String, String>,
    /// Content-Length 字节已完整读取；无 body 时为 Bytes::new()
    pub body:    Bytes,
}

/// 从 `reader` 读取并解析一个完整的 HTTP/1.1 请求
///
/// # 参数
/// - `reader` : 底层 TCP 流（OwnedReadHalf）
/// - `buf`    : 连接级持久缓冲区（keep-alive 时跨请求保留剩余字节）
///
/// # 等价 C# 代码
/// ```csharp
/// // Kestrel Http1Connection.ParseRequestAsync()
/// while (true) {
///     var result = await _reader.ReadAsync();
///     var n = _parser.Execute(result.Buffer); // llhttp
///     if (n.IsCompleted) { _reader.AdvanceTo(result.Buffer.GetPosition(n)); break; }
///     _reader.AdvanceTo(result.Buffer.Start, result.Buffer.End);
/// }
/// ```
pub async fn read_request<R>(
    reader: &mut R,
    buf:    &mut BytesMut,
) -> Result<RawRequest, ServerError>
where
    R: AsyncReadExt + Unpin,
{
    if buf.capacity() == 0 {
        buf.reserve(4096); // 首次按需分配（对应 PipeOptions.MinimumSegmentSize）
    }

    loop {
        // ── 步骤 1：用现有数据尝试解析 headers ───────────────────────
        if !buf.is_empty() {
            // headers_storage 放循环内，生命周期与 req 对齐，避免借用冲突
            let mut storage = [httparse::EMPTY_HEADER; MAX_HEADERS];
            let mut req     = httparse::Request::new(&mut storage);

            match req.parse(buf) {
                // n = status-line + headers + \r\n\r\n 总字节数
                Ok(httparse::Status::Complete(n)) => {
                    // 立即提取字段，结束对 storage/req 的借用
                    let method  = req.method.unwrap_or("GET").to_string();
                    let path    = req.path.unwrap_or("/").to_string();
                    let version = req.version.unwrap_or(1);

                    let mut headers        = HashMap::new();
                    let mut content_length = None::<usize>;

                    for h in req.headers.iter() {
                        let name  = h.name.to_ascii_lowercase(); // 统一 lowercase
                        let value = String::from_utf8_lossy(h.value).into_owned();
                        if name == "content-length" {
                            content_length = value.trim().parse().ok();
                        }
                        headers.insert(name, value);
                    }
                    // 借用结束 ───────────────────────────────────────────

                    // ── 步骤 2：消费 headers 区 ─────────────────────
                    // split_to(n)：buf 变为 [n..]（等价 reader.AdvanceTo(consumed)）
                    let _ = buf.split_to(n);

                    // ── 步骤 3：读取 body ────────────────────────────
                    let body = read_body(reader, buf, content_length).await?;

                    return Ok(RawRequest { method, path, version, headers, body });
                }

                // 数据不足：继续 loop，在底部 read_buf
                Ok(httparse::Status::Partial) => {}

                // 格式错误 → 400
                Err(e) => return Err(ServerError::BadRequest(e.to_string())),
            }
        }

        // ── 步骤 4：从 socket 读取更多数据 ──────────────────────────
        if buf.len() >= MAX_HEADER_BYTES {
            return Err(ServerError::RequestHeadersTooLarge);
        }

        // read_buf：把 TCP 内核缓冲区数据追加到 buf（零额外拷贝）
        // 对应 await pipeReader.ReadAsync()
        let n = reader
            .read_buf(buf)
            .await
            .map_err(|e| ServerError::Io(e.to_string()))?;

        if n == 0 {
            return Err(ServerError::ConnectionClosed); // 对端 TCP FIN
        }
        // 继续 loop 重试解析
    }
}

/// 读取请求体
///
/// 对应 Kestrel `Http1MessageBody.ConsumeAsync()`。
/// buf 里可能已经有部分 body（TCP 粘包），先检查再读。
async fn read_body<R>(
    reader:         &mut R,
    buf:            &mut BytesMut,
    content_length: Option<usize>,
) -> Result<Bytes, ServerError>
where
    R: AsyncReadExt + Unpin,
{
    let len = match content_length {
        None | Some(0) => return Ok(Bytes::new()),
        Some(n)        => n,
    };

    while buf.len() < len {
        let n = reader
            .read_buf(buf)
            .await
            .map_err(|e| ServerError::Io(e.to_string()))?;
        if n == 0 {
            return Err(ServerError::ConnectionClosed);
        }
    }

    // split_to + freeze：零拷贝切出 len 字节
    // buf 继续保留 [len..] 部分（可能是下一个请求的起始数据）
    Ok(buf.split_to(len).freeze())
}

1.3 serializer.rs ★ 核心：手写 HTTP/1.1 响应序列化

//! HTTP/1.1 响应序列化器
//!
//! 对应 Kestrel 的 `Http1OutputProducer`：
//!   WriteResponseHeaders() → body 追加 → PipeWriter.FlushAsync()
//!
//! Restrel 把三步合并为一个函数，直接产出 Bytes，
//! 由 connection.rs 一次 write_all() 写入 socket。

use bytes::{BufMut, Bytes, BytesMut};

/// 序列化结果
pub struct SerializedResponse {
    /// 完整 HTTP/1.1 字节流（直接 write_all 到 socket）
    pub bytes: Bytes,
    /// connection loop 是否继续等待下一个请求
    pub keep_alive: bool,
}

/// 把 HTTP 响应序列化为 TCP 字节流
///
/// # 输出格式（RFC 7230）
/// ```text
/// HTTP/1.1 200 OK\r\n
/// server: mo-server/0.1.0\r\n
/// content-length: 42\r\n
/// connection: keep-alive\r\n
/// keep-alive: timeout=75, max=1000\r\n
/// content-type: application/json; charset=utf-8\r\n
/// \r\n
/// {"status":"ok"}
/// ```
pub fn serialize_response(
    status:             u16,
    headers:            &[(String, String)],
    body:               &Bytes,
    http_version:       u8,
    request_keep_alive: bool,
) -> SerializedResponse {
    let keep_alive = http_version >= 1 && request_keep_alive;

    let estimated = 256 + headers.len() * 64 + body.len();
    let mut buf   = BytesMut::with_capacity(estimated);

    // 状态行
    let ver    = if http_version >= 1 { "HTTP/1.1" } else { "HTTP/1.0" };
    let reason = status_reason(status);
    buf.put(format!("{ver} {status} {reason}\r\n").as_bytes());

    // 固定响应头
    buf.put(b"server: mo-server/0.1.0\r\n".as_ref());
    buf.put(format!("content-length: {}\r\n", body.len()).as_bytes());

    if keep_alive {
        buf.put(b"connection: keep-alive\r\nkeep-alive: timeout=75, max=1000\r\n".as_ref());
    } else {
        buf.put(b"connection: close\r\n".as_ref());
    }

    // 自定义响应头
    for (name, value) in headers {
        buf.put(format!("{name}: {value}\r\n").as_bytes());
    }

    // 空行 + body
    buf.put(b"\r\n".as_ref());
    buf.put(body.as_ref());

    // freeze()：BytesMut → 不可变 Bytes（同一块内存，不拷贝）
    SerializedResponse { bytes: buf.freeze(), keep_alive }
}

fn status_reason(status: u16) -> &'static str {
    match status {
        200 => "OK",
        201 => "Created",
        204 => "No Content",
        400 => "Bad Request",
        401 => "Unauthorized",
        403 => "Forbidden",
        404 => "Not Found",
        405 => "Method Not Allowed",
        409 => "Conflict",
        413 => "Payload Too Large",
        422 => "Unprocessable Entity",
        431 => "Request Header Fields Too Large",
        500 => "Internal Server Error",
        503 => "Service Unavailable",
        _   => "Unknown",
    }
}

//! HttpContext：在中间件管道中传递的请求上下文
//! 对应 ASP.NET Core `Microsoft.AspNetCore.Http.HttpContext`

use std::{
    any::{Any, TypeId},
    collections::HashMap,
    sync::Arc,
};
use bytes::Bytes;
use crate::parser::RawRequest;

/// 类型安全的扩展包（对应 C# HttpContext.Items，但用 TypeId 作 key）
#[derive(Default)]
pub struct Extensions(HashMap<TypeId, Box<dyn Any + Send + Sync>>);

impl Extensions {
    /// 插入值（类型即 key）
    /// 对应 C#: `httpContext.Items["Claims"] = claims;`
    pub fn insert<T: Send + Sync + 'static>(&mut self, val: T) {
        self.0.insert(TypeId::of::<T>(), Box::new(val));
    }

    /// 取出类型化引用
    /// 对应 C#: `httpContext.Items["Claims"] as Claims`
    pub fn get<T: 'static>(&self) -> Option<&T> {
        self.0.get(&TypeId::of::<T>())?.downcast_ref::<T>()
    }
}

/// 请求上下文（中间件链中逐层传递）
pub struct HttpContext {
    raw:              RawRequest,
    pub route_params: HashMap<String, String>,
    pub extensions:   Extensions,
    /// 类型擦除的应用状态（handler 通过 state_arc::<T>() 取回具体类型）
    state: Arc<dyn Any + Send + Sync>,
}

impl HttpContext {
    pub fn new(raw: RawRequest, state: Arc<dyn Any + Send + Sync>) -> Self {
        Self { raw, route_params: HashMap::new(), extensions: Extensions::default(), state }
    }

    pub fn method(&self) -> &str  { &self.raw.method }

    /// 请求路径（不含 query string）
    pub fn path(&self) -> &str {
        self.raw.path.split_once('?').map(|(p, _)| p).unwrap_or(&self.raw.path)
    }

    pub fn uri(&self) -> &str { &self.raw.path }

    /// 请求头（key 已 lowercase）
    pub fn header(&self, name: &str) -> Option<&str> {
        self.raw.headers.get(name).map(String::as_str)
    }

    pub fn body(&self) -> &Bytes { &self.raw.body }

    pub fn version(&self) -> u8 { self.raw.version }

    /// 路由路径参数（Router 匹配后注入）
    /// 对应 C#: `[FromRoute] Guid id` — Rust 需手动 `.parse::<Uuid>()?`
    pub fn param(&self, name: &str) -> Option<&str> {
        self.route_params.get(name).map(String::as_str)
    }

    /// Query string 参数（例如 ?page=1&size=10）
    pub fn query(&self, name: &str) -> Option<&str> {
        let qs = self.raw.path.split_once('?').map(|(_, q)| q)?;
        qs.split('&').find_map(|kv| {
            let (k, v) = kv.split_once('=')?;
            (k == name).then_some(v)
        })
    }

    /// 获取 Arc<T> 形式的应用状态
    /// 对应 C#: `context.RequestServices.GetRequiredService<AppState>()`
    ///
    /// # 典型用法
    /// ```rust,ignore
    /// let state = ctx.state_arc::<AppState>().unwrap();
    /// let rows = state.db.fetch_all(...).await?;
    /// ```
    pub fn state_arc<T: Any + Send + Sync + 'static>(&self) -> Option<Arc<T>> {
        Arc::clone(&self.state).downcast::<T>().ok()
    }
}

//! HTTP 响应类型及便利构造函数

use bytes::Bytes;

/// 中间件链产出的 HTTP 响应（序列化前）
#[derive(Debug)]
pub struct Response {
    pub status:  u16,
    pub headers: Vec<(String, String)>,
    pub body:    Bytes,
}

impl Response {
    pub fn new(status: u16) -> Self {
        Self { status, headers: Vec::new(), body: Bytes::new() }
    }

    pub fn with_json(mut self, data: &impl serde::Serialize) -> Self {
        self.body = serde_json::to_vec(data).unwrap_or_default().into();
        self.headers.push((
            "content-type".into(),
            "application/json; charset=utf-8".into(),
        ));
        self
    }

    pub fn with_header(mut self, name: impl Into<String>, value: impl Into<String>) -> Self {
        self.headers.push((name.into(), value.into()));
        self
    }
}

// 便利构造函数（等同 C# Results.Ok / Results.Created 等）
pub fn ok(data: &impl serde::Serialize) -> Response     { Response::new(200).with_json(data) }
pub fn created(data: &impl serde::Serialize) -> Response{ Response::new(201).with_json(data) }
pub fn no_content() -> Response                          { Response::new(204) }
pub fn not_found(msg: &str) -> Response {
    Response::new(404).with_json(&serde_json::json!({ "error": msg }))
}
pub fn bad_request(msg: &str) -> Response {
    Response::new(400).with_json(&serde_json::json!({ "error": msg }))
}

1.6 middleware.rs

//! 中间件管道（对应 ASP.NET Core IApplicationBuilder + RequestDelegate）

use std::{future::Future, sync::Arc};
use crate::{context::HttpContext, error::ServerError, response::Response};

pub type BoxFuture<T> = std::pin::Pin<Box<dyn Future<Output = T> + Send + 'static>>;

/// 调用下一个中间件的函数类型（对应 C# RequestDelegate）
pub type Next = Arc<
    dyn Fn(HttpContext) -> BoxFuture<Result<Response, ServerError>> + Send + Sync,
>;

/// 中间件 trait
pub trait Middleware: Send + Sync + 'static {
    fn handle(&self, ctx: HttpContext, next: Next)
        -> BoxFuture<Result<Response, ServerError>>;
}

/// 把 `async fn(ctx, next)` 包装为 `Middleware`
pub struct FnMiddleware<F>(pub F);

impl<F, Fut> Middleware for FnMiddleware<F>
where
    F:   Fn(HttpContext, Next) -> Fut + Send + Sync + 'static,
    Fut: Future<Output = Result<Response, ServerError>> + Send + 'static,
{
    fn handle(&self, ctx: HttpContext, next: Next)
        -> BoxFuture<Result<Response, ServerError>>
    {
        Box::pin((self.0)(ctx, next))
    }
}

/// 中间件管道（对应 C# ApplicationBuilder）
#[derive(Default)]
pub struct Pipeline {
    middlewares: Vec<Box<dyn Middleware>>,
}

impl Pipeline {
    pub fn new() -> Self { Self::default() }

    pub fn add(&mut self, mw: impl Middleware) {
        self.middlewares.push(Box::new(mw));
    }

    /// 仅供 server.rs 内部使用（接受已装箱的中间件）
    pub fn add_boxed(&mut self, mw: Box<dyn Middleware>) {
        self.middlewares.push(mw);
    }

    /// 将中间件列表 fold 为单个 Next 闭包（对应 C# ApplicationBuilder.Build()）
    ///
    /// # 洋葱模型
    /// 从末尾向前 fold，第一个 add 的中间件在最外层（先进先出）。
    /// 编译期单态化：每层是一个 Box::pin 闭包，无虚表开销。
    pub fn build(self, terminal: Next) -> Next {
        self.middlewares
            .into_iter()
            .rev()
            .fold(terminal, |next, mw| {
                let mw = Arc::new(mw);
                Arc::new(move |ctx: HttpContext| -> BoxFuture<_> {
                    let next = Arc::clone(&next);
                    let mw   = Arc::clone(&mw);
                    Box::pin(async move { mw.handle(ctx, next).await })
                })
            })
    }
}

        let l2  = Arc::clone(&log);
        let l3  = Arc::clone(&log);

        let mut p = Pipeline::new();

        p.add(FnMiddleware(move |ctx, next| {
            let l = Arc::clone(&l1);
            Box::pin(async move {
                l.lock().unwrap().push("mw1-before");
                let r = next(ctx).await;
                l.lock().unwrap().push("mw1-after");
                r
            })
        }));

        p.add(FnMiddleware(move |ctx, next| {
            let l = Arc::clone(&l2);
            Box::pin(async move {
                l.lock().unwrap().push("mw2-before");
                let r = next(ctx).await;
                l.lock().unwrap().push("mw2-after");
                r
            })
        }));

        let terminal: Next = Arc::new(move |_ctx| {
            let l = Arc::clone(&l3);
            Box::pin(async move {
                l.lock().unwrap().push("handler");
                Ok(no_content())
            })
        });

        let composed = p.build(terminal);
        let _ = composed(test_ctx()).await;

        let order = log.lock().unwrap().clone();

        assert!(result.is_ok());
    }
}

//! 路径路由（对应 ASP.NET Core IRouter + MapGet/MapPost）
//! 使用 matchit：axum 内部使用的同款基数树，O(log n) 路径匹配。

use std::{collections::HashMap, sync::Arc};
use crate::{context::HttpContext, error::ServerError, middleware::{BoxFuture, Next}, response::Response};

pub type BoxHandler = Arc<
    dyn Fn(HttpContext) -> BoxFuture<Result<Response, ServerError>> + Send + Sync,
>;

/// HTTP 路由表
pub struct Router {
    routes: HashMap<String, matchit::Router<BoxHandler>>,
}

impl Router {
    pub fn new() -> Self { Self { routes: HashMap::new() } }

    /// 注册路由（对应 C# app.MapGet("/services/{id}", handler)）
    pub fn insert(
        &mut self,
        method:  &str,
        path:    &str,
        handler: BoxHandler,
    ) -> Result<(), ServerError> {
        self.routes
            .entry(method.to_ascii_uppercase())
            .or_insert_with(matchit::Router::new)
            .insert(path, handler)
            .map_err(|e| ServerError::RouteConflict { path: path.into(), cause: e.to_string() })
    }

    /// 包装为 `Next`（Pipeline 的最后一个节点）
    pub fn into_next(self: Arc<Self>) -> Next {
        Arc::new(move |mut ctx: HttpContext| -> BoxFuture<_> {
            let router = Arc::clone(&self);
            Box::pin(async move {
                let method = ctx.method().to_ascii_uppercase();

                let method_router = router.routes.get(&method)
                    .ok_or_else(|| ServerError::MethodNotAllowed(method.clone()))?;

                let matched = method_router.at(ctx.path())
                    .map_err(|_| ServerError::NotFound(ctx.path().to_string()))?;

                // 注入路由参数（对应 C# RouteData.Values）
                for (k, v) in matched.params.iter() {
                    ctx.route_params.insert(k.to_string(), v.to_string());
                }

                let handler = Arc::clone(matched.value);
                handler(ctx).await
            })
        })
    }
}

//! per-connection keep-alive 请求处理循环
//!
//! 对应 Kestrel 的 `Http1Connection<TTransport>.ProcessRequestsAsync()`。
//!
//! # 等价 C# 代码
//! ```csharp
//! // Kestrel Http1Connection.ProcessRequestsAsync()
//! while (true) {
//!     if (!await ParseRequestAsync()) break;       // 读 + 解析
//!     await application.ProcessRequestAsync(ctx); // 中间件管道
//!     await output.WriteResponseAsync(response);  // 序列化 + 写出
//!     if (!keepAlive || cancel.IsCancellationRequested) break;

use std::sync::Arc;

use bytes::BytesMut;
use tokio::io::{AsyncWriteExt, BufWriter};
use tokio::net::TcpStream;
use tokio_util::sync::CancellationToken;

use mo_server_core::{

    error::ServerError,

    serializer::serialize_response,

use crate::{limits::ServerLimits, state::AppState};

/// 处理单个 TCP 连接上的所有 HTTP 请求（keep-alive 循环）
///
/// 函数返回时，TcpStream 自动 drop（连接关闭）。
/// 调用方（transport.rs）在此之后 drop Semaphore permit，归还并发槽位。
pub async fn handle_connection(
    stream:   TcpStream,
    pipeline: Arc<Next>,      // 编译好的中间件链
    state:    Arc<AppState>,
    limits:   ServerLimits,
    cancel:   CancellationToken,
) {
    let (mut reader, writer) = stream.into_split();
    // BufWriter：把多次小 write 合并为一次系统调用（等同 PipeWriter 批量 flush）
    let mut writer = BufWriter::with_capacity(16 * 1024, writer);

    // ★ 连接级持久缓冲区
    //   生命周期 = 连接生命周期（对应 Kestrel Pipe 的生命周期）
    //   TCP 粘包时，上一请求结束后 buf 里可能还有下一请求的起始字节，不会丢失。
    let mut buf = BytesMut::with_capacity(4096);

    loop {
        // ── 读取并解析下一个请求 ─────────────────────────────────────
        // tokio::select! 同时等待两件事：
        //   1. 读取请求（可能阻塞，等客户端发送数据）
        //   2. CancellationToken 触发（SIGTERM 优雅关闭）
        let raw_req = tokio::select! {
            biased; // 优先检查 cancel（避免已收到 SIGTERM 后还处理新请求）

            _ = cancel.cancelled() => {
                // 优雅关闭：当前是空闲 keep-alive 等待，直接关闭连接
                break;
            }

            result = read_request(&mut reader, &mut buf) => {
                match result {
                    Ok(req) => req,
                    Err(ServerError::ConnectionClosed) => break,
                    Err(ServerError::RequestHeadersTooLarge) => {
                        let _ = writer.write_all(
                            b"HTTP/1.1 431 Request Header Fields Too Large\r\nconnection: close\r\ncontent-length: 0\r\n\r\n"
                        ).await;
                        let _ = writer.flush().await;
                        break;
                    }
                    Err(e) => {
                        tracing::debug!("请求解析错误: {e}");
                        break;
                    }
                }
            }
        };

        // body 大小检查（对应 KestrelServerLimits.MaxRequestBodySize）
        if raw_req.body.len() > limits.max_request_body_size {
            let _ = writer.write_all(
                b"HTTP/1.1 413 Payload Too Large\r\nconnection: close\r\ncontent-length: 0\r\n\r\n"
            ).await;
            let _ = writer.flush().await;
            break;
        }

        // 判断客户端是否希望 keep-alive（HTTP/1.1 默认 keep-alive）
        let client_keep_alive = raw_req.version >= 1
            && raw_req.headers.get("connection")
                .map(|v| !v.eq_ignore_ascii_case("close"))
                .unwrap_or(true);

        let http_version = raw_req.version;

        // ── 运行中间件管道（对应 application.ProcessRequestAsync）──────
        let state_any: Arc<dyn std::any::Any + Send + Sync> = state.clone();
        let ctx  = HttpContext::new(raw_req, state_any);
        let resp = pipeline(ctx).await;

        // ── 序列化响应为 HTTP/1.1 字节流 ────────────────────────────────
        let (status, headers, body) = match resp {
            Ok(r)  => (r.status, r.headers, r.body),
            Err(e) => {
                let code  = e.status_code();
                let bytes = serde_json::to_vec(
                    &serde_json::json!({ "error": e.to_string() })
                ).unwrap_or_default();
                let hdrs = vec![(
                    "content-type".to_string(),
                    "application/json; charset=utf-8".to_string(),
                )];
                (code, hdrs, bytes::Bytes::from(bytes))
            }
        };

        let serialized = serialize_response(status, &headers, &body, http_version, client_keep_alive);

        // ── 写入 TCP socket ────────────────────────────────────────────
        if writer.write_all(&serialized.bytes).await.is_err() { break; }
        if writer.flush().await.is_err()                       { break; }

        // keep-alive 判断
        if !serialized.keep_alive || cancel.is_cancelled() { break; }
        // 继续 loop：等待同一连接上的下一个请求
    }
    // 函数返回：stream 自动 drop（TCP 连接关闭）
}

//! TCP 监听 + 连接生命周期管理
//! 对应 Kestrel 的 ConnectionManager + IConnectionListenerFactory

use std::sync::Arc;

use tokio::{net::TcpListener, sync::Semaphore};
use tokio_util::sync::CancellationToken;

use mo_server_core::middleware::Next;
use crate::{connection::handle_connection, limits::ServerLimits, state::AppState};

/// 启动 TCP 监听循环（对应 Kestrel KestrelServer.StartAsync）
///
/// # 并发控制
/// Semaphore 限制最大并发连接数（对应 KestrelServerLimits.MaxConcurrentConnections）：
/// - accept 前 acquire_owned() 取 permit（超限则阻塞，不丢弃连接）
/// - 连接关闭时 drop(permit)，信号量 +1
///
/// # 优雅关闭（对应 StopAsync + ConnectionManager.CloseAllConnectionsAsync）
/// 1. cancel.cancelled() → 停止 accept
/// 2. acquire_many(max)  → 等待所有 permit 归还（= 所有连接已关闭）
pub async fn run_transport(
    listener:  TcpListener,
    pipeline:  Arc<Next>,
    state:     Arc<AppState>,
    limits:    ServerLimits,
    cancel:    CancellationToken,
) -> anyhow::Result<()> {
    let max       = limits.max_concurrent_connections;
    let semaphore = Arc::new(Semaphore::new(max));

    tracing::info!("TCP 监听开始");

    loop {
        tokio::select! {
            biased;

            _ = cancel.cancelled() => {
                tracing::info!("收到关闭信号，停止 accept");
                break;
            }

            result = listener.accept() => {
                let (stream, addr) = result?;
                tracing::debug!(?addr, "新 TCP 连接");

                // acquire_owned：返回 'static 的 OwnedSemaphorePermit，可 move 进 spawn
                let permit = Arc::clone(&semaphore)
                    .acquire_owned()
                    .await
                    .expect("semaphore 已关闭");

                tokio::spawn({
                    let pipeline = Arc::clone(&pipeline);
                    let state    = Arc::clone(&state);
                    let limits   = limits.clone();
                    let cancel   = cancel.clone();
                    async move {
                        handle_connection(stream, pipeline, state, limits, cancel).await;
                        drop(permit); // RAII：连接关闭后释放槽位
                    }
                });
            }
        }
    }

    // 优雅关闭：等待所有 in-flight 连接关闭
    tracing::info!("等待 {max} 个连接全部完成…");
    let _ = semaphore.acquire_many(max as u32).await;
    tracing::info!("服务器优雅退出");

    Ok(())
}

use std::sync::Arc;

use tokio::net::TcpListener;
use tokio_util::sync::CancellationToken;

use mo_server_core::{
    middleware::{FnMiddleware, Next, Pipeline},
    router::{BoxHandler, Router},
};

use crate::{limits::ServerLimits, state::AppState, transport::run_transport};

pub struct ServerBuilder {
    addr:        String,
    limits:      ServerLimits,
    state:       Arc<AppState>,
    middlewares: Vec<Box<dyn mo_server_core::middleware::Middleware>>,
    routes:      Vec<(String, String, BoxHandler)>,
}

impl ServerBuilder {
    pub fn new(addr: impl Into<String>, state: Arc<AppState>) -> Self {
        Self { addr: addr.into(), limits: ServerLimits::default(), state,
               middlewares: Vec::new(), routes: Vec::new() }
    }

    pub fn with_limits(mut self, limits: ServerLimits) -> Self {
        self.limits = limits; self
    }

    pub fn use_middleware<F, Fut>(mut self, f: F) -> Self
    where
        F: Fn(mo_server_core::context::HttpContext, Next) -> Fut + Send + Sync + 'static,
        Fut: std::future::Future<
            Output = Result<mo_server_core::response::Response, mo_server_core::error::ServerError>
        > + Send + 'static,
    {
        self.middlewares.push(Box::new(FnMiddleware(f)));
        self
    }

    pub fn map_get(mut self, path: &str, h: BoxHandler) -> Self {
        self.routes.push(("GET".into(), path.into(), h)); self
    }
    pub fn map_post(mut self, path: &str, h: BoxHandler) -> Self {
        self.routes.push(("POST".into(), path.into(), h)); self
    }
    pub fn map_put(mut self, path: &str, h: BoxHandler) -> Self {
        self.routes.push(("PUT".into(), path.into(), h)); self
    }
    pub fn map_delete(mut self, path: &str, h: BoxHandler) -> Self {
        self.routes.push(("DELETE".into(), path.into(), h)); self
    }

    pub async fn run(self) -> anyhow::Result<()> {
        // 构建路由
        let mut router = Router::new();
        for (method, path, handler) in self.routes {
            router.insert(&method, &path, handler)?;
        }

        // 编译中间件链（Router 作为 terminal）
        let terminal = Arc::new(router).into_next();
        let mut pipeline = Pipeline::new();
        for mw in self.middlewares { pipeline.add_boxed(mw); }
        let compiled = Arc::new(pipeline.build(terminal));

        // 绑定端口
        let listener = TcpListener::bind(&self.addr).await?;
        tracing::info!(addr = %self.addr, "服务器启动");

        // SIGTERM / Ctrl-C → CancellationToken
        let cancel = CancellationToken::new();
        {
            let cancel = cancel.clone();
            tokio::spawn(async move {
                #[cfg(unix)] {
                    use tokio::signal::unix::{signal, SignalKind};
                    let mut sig = signal(SignalKind::terminate()).unwrap();
                    tokio::select! {
                        _ = sig.recv()              => {}
                        _ = tokio::signal::ctrl_c() => {}
                    }
                }
                #[cfg(not(unix))] {
                    let _ = tokio::signal::ctrl_c().await;
                }
                tracing::info!("收到关闭信号");
                cancel.cancel();
            });
        }

        run_transport(listener, compiled, self.state, self.limits, cancel).await
    }
}

//! 请求日志中间件（对应 ASP.NET Core UseHttpLogging）
//!
//! 注入 tracing span，记录：method, path, status, latency_ms, request_id

use std::time::Instant;

use mo_server_core::{
    context::HttpContext,
    error::ServerError,
    middleware::Next,
    response::Response,
};
use uuid::Uuid;

/// 链路追踪中间件
///
/// 在请求开始时创建 span，在请求结束后记录结果。
/// 对应 C#:
/// ```csharp
/// app.Use(async (ctx, next) => {
///     var sw = Stopwatch.StartNew();
///     await next();
///     logger.LogInformation("{Method} {Path} → {Status} in {Ms}ms",
///         ctx.Request.Method, ctx.Request.Path, ctx.Response.StatusCode, sw.ElapsedMilliseconds);
/// });
/// ```
pub async fn tracing_mw(ctx: HttpContext, next: Next) -> Result<Response, ServerError> {
    let method     = ctx.method().to_string();
    let path       = ctx.path().to_string();
    let request_id = Uuid::new_v4().to_string();
    let start      = Instant::now();

    let span = tracing::info_span!(
        "http_request",
        method   = %method,
        path     = %path,
        req_id   = %request_id,
    );
    let _enter = span.enter();

    tracing::debug!("请求开始");

    let result = next(ctx).await;
    let latency_ms = start.elapsed().as_millis();

    match &result {
        Ok(resp) => tracing::info!(
            status     = resp.status,
            latency_ms = %latency_ms,
            "请求完成"
        ),
        Err(e) => tracing::error!(
            error      = %e,
            latency_ms = %latency_ms,
            "请求失败"
        ),
    }

    result
}

//! JWT 鉴权中间件（对应 ASP.NET Core UseAuthentication + UseAuthorization）
//!
//! 读取 Authorization: Bearer <token>，验证 JWT，将 Claims 注入 HttpContext.extensions

use jsonwebtoken::{decode, encode, DecodingKey, EncodingKey, Header, Validation};
use serde::{Deserialize, Serialize};

use mo_server_core::{
    context::HttpContext,
    error::ServerError,
    middleware::Next,
    response::Response,
};

use crate::state::AppState;

/// JWT payload（对应 C# ClaimsPrincipal 中的 Claims）
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct Claims {
    /// Subject：用户 ID（对应 C# ClaimTypes.NameIdentifier）
    pub sub: String,
    /// Expiry：Unix 时间戳（对应 C# ClaimTypes.Expiration 隐含在 JwtSecurityToken.ValidTo）
    pub exp: u64,
}

/// JWT 鉴权中间件
///
/// 白名单路由（`/health`）放行；其余路由验证 Bearer token。
/// 验证成功后将 `Claims` 注入 `ctx.extensions`，后续 handler 可通过
/// `ctx.extensions.get::<Claims>()` 取回。
///
/// 对应 C#:
/// ```csharp
/// [Authorize]
/// public async Task<IActionResult> ListServices() { ... }
/// ```
pub async fn auth_mw(mut ctx: HttpContext, next: Next) -> Result<Response, ServerError> {
    // 白名单：健康检查不需要鉴权
    if ctx.path() == "/health" {
        return next(ctx).await;
    }

    let state = ctx
        .state_arc::<AppState>()
        .ok_or_else(|| ServerError::Unauthorized("state unavailable".into()))?;

    let header_val = ctx
        .header("authorization")
        .ok_or_else(|| ServerError::Unauthorized("missing Authorization header".into()))?
        .to_string();

    let token = header_val
        .strip_prefix("Bearer ")
        .ok_or_else(|| ServerError::Unauthorized("Authorization value must start with 'Bearer '".into()))?;

    let key        = DecodingKey::from_secret(state.config.jwt_secret.as_bytes());
    let validation = Validation::default(); // 默认：验证 exp, iat, alg

    let token_data = decode::<Claims>(token, &key, &validation)
        .map_err(|e| ServerError::Unauthorized(format!("invalid token: {e}")))?;

    // 将 Claims 注入 context extensions（对应 C# HttpContext.User）
    ctx.extensions.insert(token_data.claims);

    next(ctx).await
}

/// 测试辅助：生成合法的测试用 JWT
///
/// # 注意
/// 仅用于测试，不要在生产代码中硬编码密钥。
pub fn make_test_token(secret: &str, sub: &str) -> String {
    let claims = Claims {
        sub: sub.to_string(),
        exp: u64::MAX, // 永不过期（测试专用）
    };
    encode(
        &Header::default(),
        &claims,
        &EncodingKey::from_secret(secret.as_bytes()),
    )
    .expect("test token generation failed")
}

    /// 等价场景：Kestrel 的 PipeReader.ReadAsync() 返回 NeedMoreData 多次

    async fn test_partial_request_eventually_parses() {
        // duplex 返回一对内存管道（等同 MemoryStream，无真实 socket）
        let (mut server_side, mut client_side) = duplex(128);

        // 模拟"慢客户端"：分 3 次写入一个完整请求
        let task = tokio::spawn(async move {
            use tokio::io::AsyncWriteExt;
            client_side.write_all(b"GET /health HTTP").await.unwrap();
            tokio::time::sleep(std::time::Duration::from_millis(5)).await;
            client_side.write_all(b"/1.1\r\nhost: localhost").await.unwrap();
            tokio::time::sleep(std::time::Duration::from_millis(5)).await;
            client_side.write_all(b"\r\n\r\n").await.unwrap();
        });

        let mut buf = BytesMut::new();
        let req = read_request(&mut server_side, &mut buf).await.unwrap();

        task.await.unwrap();

        assert_eq!(req.method,  "GET");
        assert_eq!(req.path,    "/health");
        assert_eq!(req.version, 1);
    }

    /// 验证 TCP 粘包：第一个请求结束后 buf 里有第二个请求的起始字节
    ///
    /// 等价场景：Kestrel keep-alive 时 PipeReader 不 Reset，保留余量数据
    #[tokio::test]
    async fn test_buffer_persists_across_requests() {
        let (mut server_side, mut client_side) = duplex(256);

        // 写入两个连续请求（不含 body）
        let req1 = b"GET /a HTTP/1.1\r\nhost: x\r\n\r\n";
        let req2 = b"GET /b HTTP/1.1\r\nhost: x\r\n\r\n";

        tokio::spawn(async move {
            use tokio::io::AsyncWriteExt;
            // 一次性写入两个请求（模拟 TCP 粘包）
            client_side.write_all(req1).await.unwrap();
            client_side.write_all(req2).await.unwrap();

        let r1 = read_request(&mut server_side, &mut buf).await.unwrap();
        assert_eq!(r1.path, "/a");

        // buf 里应该残留第二个请求的字节（粘包）
        // 第二次解析直接命中 buf，不用再从 socket 读
        let r2 = read_request(&mut server_side, &mut buf).await.unwrap();
        assert_eq!(r2.path, "/b");

const TEST_SECRET: &str = "integration-test-secret";

// ── 测试基础设施 ────────────────────────────────────────────────────────────

struct TestServer {
    addr:  SocketAddr,
    token: String,
}

impl TestServer {
    /// 启动完整 MoServer（真实 DB + 真实 TCP）
    async fn start(db: PgPool) -> Self {
        let token = make_test_token(TEST_SECRET, "test_user");

        let config = AppConfig {
            database_url: "unused".into(),
            port:         0, // 随机端口
            jwt_secret:   TEST_SECRET.into(),
        };

        let state = Arc::new(AppState::new(db, config));

        // 绑定随机端口（port=0，OS 分配）
        let listener = TcpListener::bind("127.0.0.1:0").await.unwrap();
        let addr     = listener.local_addr().unwrap();

        use mo_server_core::router::BoxHandler;
        fn h<F, Fut>(f: F) -> BoxHandler
        where
            F:   Fn(mo_server_core::context::HttpContext) -> Fut + Send + Sync + 'static,
            Fut: std::future::Future<
                Output = Result<
                    mo_server_core::response::Response,
                    mo_server_core::error::ServerError,
                >
            > + Send + 'static,
        {
            Arc::new(move |ctx| Box::pin(f(ctx)))
        }

        // 在后台任务中运行服务器
        tokio::spawn(
            ServerBuilder::new_with_listener(listener, state)
                .use_middleware(tracing_mw::tracing_mw)
                .use_middleware(mo_server::middleware::auth_mw::auth_mw)
                .map_get("/health",          h(mo_server::handlers::health::handle))
                .map_get("/services",        h(mo_server::handlers::services::list))
                .map_post("/services",       h(mo_server::handlers::services::create))
                .map_get("/services/:id",    h(mo_server::handlers::services::get))
                .map_delete("/services/:id", h(mo_server::handlers::services::delete))
                .run()
        );

        // 等待服务器就绪（简单重试 accept）
        for _ in 0..20 {
            if TcpStream::connect(addr).await.is_ok() { break; }
            tokio::time::sleep(Duration::from_millis(50)).await;
        }

        TestServer { addr, token }
    }
}

// ── HTTP over TCP 辅助 ──────────────────────────────────────────────────────

/// 发送原始 HTTP/1.1 请求并读取完整响应
async fn send(addr: SocketAddr, request: &[u8]) -> (u16, String) {
    let mut stream = TcpStream::connect(addr).await.unwrap();
    stream.write_all(request).await.unwrap();

    let mut buf = Vec::new();
    loop {
        let mut tmp = [0u8; 4096];
        let n = timeout(Duration::from_secs(5), stream.read(&mut tmp))
            .await
            .expect("read timed out")
            .expect("read error");
        if n == 0 { break; }
        buf.extend_from_slice(&tmp[..n]);

        // 找到 header 结束标志
        if let Some(hdr_end) = buf.windows(4).position(|w| w == b"\r\n\r\n") {
            let status_line = std::str::from_utf8(&buf[..hdr_end]).unwrap_or("");
            let status: u16 = status_line
                .split_whitespace().nth(1).unwrap_or("500")
                .parse().unwrap_or(500);

            let content_length: usize = status_line.lines()
                .find(|l| l.to_lowercase().starts_with("content-length:"))
                .and_then(|l| l.split(':').nth(1))
                .and_then(|v| v.trim().parse().ok())
                .unwrap_or(0);

            let body_start = hdr_end + 4;
            let mut body   = buf[body_start..].to_vec();

            while body.len() < content_length {
                let n = timeout(Duration::from_secs(5), stream.read(&mut tmp))
                    .await.unwrap().unwrap();
                if n == 0 { break; }
                body.extend_from_slice(&tmp[..n]);
            }

            let body_str = String::from_utf8_lossy(&body).to_string();
            return (status, body_str);
        }
    }
    (0, String::new())
}

fn get(path: &str, token: Option<&str>) -> Vec<u8> {
    let auth = token
        .map(|t| format!("authorization: Bearer {t}\r\n"))
        .unwrap_or_default();
    format!("GET {path} HTTP/1.1\r\nhost: localhost\r\n{auth}\r\n").into_bytes()
}

fn post_json(path: &str, token: Option<&str>, body: &str) -> Vec<u8> {
    let auth = token.map(|t| format!("authorization: Bearer {t}\r\n")).unwrap_or_default();
    format!(
        "POST {path} HTTP/1.1\r\nhost: localhost\r\n{auth}content-type: application/json\r\ncontent-length: {}\r\n\r\n{body}",
        body.len()
    )
    .into_bytes()
}

fn delete(path: &str, token: Option<&str>) -> Vec<u8> {
    let auth = token.map(|t| format!("authorization: Bearer {t}\r\n")).unwrap_or_default();
    format!("DELETE {path} HTTP/1.1\r\nhost: localhost\r\n{auth}\r\n").into_bytes()
}

// ── 测试夹具（testcontainers + 迁移）──────────────────────────────────────

async fn setup() -> (TestServer, testcontainers::Container<'static, Postgres>) {
    let docker = Box::leak(Box::new(Cli::default()));
    let container = docker.run(Postgres::default());
    let port = container.get_host_port_ipv4(5432);

    let db_url = format!("postgres://postgres:postgres@127.0.0.1:{port}/postgres");
    let pool   = PgPoolOptions::new()
        .max_connections(5)
        .connect(&db_url)
        .await
        .expect("testcontainers postgres connect failed");

    sqlx::migrate!("../../migrations").run(&pool).await.expect("migrate failed");

    let server = TestServer::start(pool).await;
    (server, container)
}

#[tokio::test]
async fn test_health_returns_200() {
    let (srv, _c) = setup().await;
    let (status, body) = send(srv.addr, &get("/health", None)).await;
    assert_eq!(status, 200);
    let json: serde_json::Value = serde_json::from_str(&body).unwrap();
    assert_eq!(json["status"], "ok");
}

#[tokio::test]
async fn test_keepalive_reuses_connection() {
    let (srv, _c) = setup().await;
    let mut stream = TcpStream::connect(srv.addr).await.unwrap();

    // 在同一 TCP 连接上发送两个请求（keep-alive 验证）
    let req = get("/health", None);
    for _ in 0..2 {
        stream.write_all(&req).await.unwrap();
        let mut buf = [0u8; 1024];
        let n = timeout(Duration::from_secs(3), stream.read(&mut buf))
            .await.unwrap().unwrap();
        let resp = std::str::from_utf8(&buf[..n]).unwrap();
        assert!(resp.starts_with("HTTP/1.1 200"));
        // 服务器返回 connection: keep-alive（TCP 不关闭）
        assert!(resp.to_lowercase().contains("keep-alive"));
    }
    // 连接仍然有效（未被服务器关闭）
    assert!(stream.peer_addr().is_ok());
}
