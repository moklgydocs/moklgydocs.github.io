二叉树遍历

1. 算法定位

二叉树遍历是树与图算法的基石，几乎所有树相关问题都建立在遍历之上。掌握前序、中序、后序（递归+迭代）以及层序遍历，是解决二叉树问题的必备能力。

2. 一句话理解

按照不同的访问根节点的时机，系统地访问二叉树中的每一个节点恰好一次。

3. 生活化比喻

想象你参观一栋大楼：

  • 前序遍历：先在大厅签到（访问根），再参观左翼，最后参观右翼
  • 中序遍历：先参观左翼，回到大厅签到（访问根），再参观右翼
  • 后序遍历：先参观左翼，再参观右翼，最后在大厅签到（访问根）
  • 层序遍历：从一楼开始，逐层往上参观

4. 核心思想

遍历方式访问顺序特点
前序 (Preorder)根 → 左 → 右第一个元素是根
中序 (Inorder)左 → 根 → 右BST中得到有序序列
后序 (Postorder)左 → 右 → 根最后一个元素是根
层序 (Level Order)逐层从左到右使用队列实现

5. 图解推演

示例二叉树结构

         1
        / \
       2   3
      / \   \
     4   5   6

各遍历结果

前序遍历: 1 → 2 → 4 → 5 → 3 → 6
中序遍历: 4 → 2 → 5 → 1 → 3 → 6
后序遍历: 4 → 5 → 2 → 6 → 3 → 1
层序遍历: 1 → 2 → 3 → 4 → 5 → 6

前序遍历递归调用栈

│   │   └── preorder(null) → 返回
│   └── preorder(5)
│       ├── 访问 5
│       ├── preorder(null) → 返回
│       └── preorder(null) → 返回
└── preorder(3)
    ├── 访问 3
    ├── preorder(null) → 返回
    └── preorder(6)
        ├── 访问 6
        ├── preorder(null) → 返回
        └── preorder(null) → 返回

迭代前序遍历（栈模拟）

步骤  栈状态        输出
1    [1]           
2    [3,2]         1      ← 弹出1，压入右3、左2
3    [3,5,4]       1,2    ← 弹出2，压入右5、左4
4    [3,5]         1,2,4  ← 弹出4，无子节点
5    [3]           1,2,4,5 ← 弹出5，无子节点
6    [6]           1,2,4,5,3 ← 弹出3，压入右6
7    []            1,2,4,5,3,6 ← 弹出6

6. 执行步骤

递归遍历步骤

  1. 判断当前节点是否为空（基线条件）
  2. 根据遍历类型决定访问顺序
  3. 递归处理左子树和右子树

迭代遍历步骤（以前序为例）

  1. 将根节点压入栈
  2. 循环：弹出栈顶节点并访问
  3. 先压右子节点，再压左子节点（保证左先被访问）
  4. 直到栈为空

层序遍历步骤

  1. 将根节点加入队列
  2. 循环：取出队首节点并访问
  3. 将其左右子节点加入队列
  4. 直到队列为空

7. C# 实现

using System;
using System.Collections.Generic;

/// <summary>
/// 二叉树节点定义
/// </summary>
public class TreeNode
{
    public int Val;          // 节点值
    public TreeNode Left;    // 左子节点
    public TreeNode Right;   // 右子节点

    public TreeNode(int val, TreeNode left = null, TreeNode right = null)
    {
        Val = val;
        Left = left;
        Right = right;
    }
}

/// <summary>
/// 二叉树遍历算法集合
/// </summary>
public class BinaryTreeTraversal
{
    // ========== 递归实现 ==========

    /// <summary>
    /// 前序遍历（递归）：根 → 左 → 右
    /// </summary>
    public List<int> PreorderRecursive(TreeNode root)
    {
        var result = new List<int>();
        PreorderHelper(root, result);
        return result;
    }

    private void PreorderHelper(TreeNode node, List<int> result)
    {
        if (node == null) return;   // 基线条件：空节点直接返回
        result.Add(node.Val);       // 先访问根节点
        PreorderHelper(node.Left, result);   // 再遍历左子树
        PreorderHelper(node.Right, result);  // 最后遍历右子树
    }

    /// <summary>
    /// 中序遍历（递归）：左 → 根 → 右
    /// </summary>
    public List<int> InorderRecursive(TreeNode root)
    {
        var result = new List<int>();
        InorderHelper(root, result);
        return result;
    }

    private void InorderHelper(TreeNode node, List<int> result)
    {
        if (node == null) return;
        InorderHelper(node.Left, result);    // 先遍历左子树
        result.Add(node.Val);                // 再访问根节点
        InorderHelper(node.Right, result);   // 最后遍历右子树
    }

    /// <summary>
    /// 后序遍历（递归）：左 → 右 → 根
    /// </summary>
    public List<int> PostorderRecursive(TreeNode root)
    {
        var result = new List<int>();
        PostorderHelper(root, result);
        return result;
    }

    private void PostorderHelper(TreeNode node, List<int> result)
    {
        if (node == null) return;
        PostorderHelper(node.Left, result);  // 先遍历左子树
        PostorderHelper(node.Right, result); // 再遍历右子树
        result.Add(node.Val);                // 最后访问根节点
    }

    // ========== 迭代实现 ==========

    /// <summary>
    /// 前序遍历（迭代）：使用栈模拟递归
    /// </summary>
    public List<int> PreorderIterative(TreeNode root)
    {
        var result = new List<int>();
        if (root == null) return result;

        var stack = new Stack<TreeNode>();
        stack.Push(root);

        while (stack.Count > 0)
        {
            var node = stack.Pop();      // 弹出栈顶节点
            result.Add(node.Val);        // 访问该节点

            // 注意：先压右后压左，保证左子节点先被弹出
            if (node.Right != null) stack.Push(node.Right);
            if (node.Left != null) stack.Push(node.Left);
        }

        return result;
    }

    /// <summary>
    /// 中序遍历（迭代）：一路向左压栈，回溯时访问
    /// </summary>
    public List<int> InorderIterative(TreeNode root)
    {
        var result = new List<int>();
        var stack = new Stack<TreeNode>();
        var current = root;

        while (current != null || stack.Count > 0)
        {
            // 一路向左，将所有左子节点压入栈
            while (current != null)
            {
                stack.Push(current);
                current = current.Left;
            }

            // 弹出栈顶（最左节点），访问它
            current = stack.Pop();
            result.Add(current.Val);

            // 转向右子树
            current = current.Right;
        }

        return result;
    }

    /// <summary>
    /// 后序遍历（迭代）：双栈法
    /// 思路：前序是 根→左→右，将其改为 根→右→左，然后反转即得 左→右→根
    /// </summary>
    public List<int> PostorderIterative(TreeNode root)
    {
        var result = new List<int>();
        if (root == null) return result;

        var stack = new Stack<TreeNode>();
        stack.Push(root);

        while (stack.Count > 0)
        {
            var node = stack.Pop();
            result.Add(node.Val);   // 根→右→左 的顺序添加

            // 先压左后压右（与前序相反）
            if (node.Left != null) stack.Push(node.Left);
            if (node.Right != null) stack.Push(node.Right);
        }

        result.Reverse();   // 反转得到 左→右→根
        return result;
    }

    /// <summary>
    /// 层序遍历：使用队列逐层访问
    /// </summary>
    public List<List<int>> LevelOrder(TreeNode root)
    {
        var result = new List<List<int>>();
        if (root == null) return result;

        var queue = new Queue<TreeNode>();
        queue.Enqueue(root);

        while (queue.Count > 0)
        {
            int levelSize = queue.Count;    // 当前层的节点数
            var levelList = new List<int>();

            for (int i = 0; i < levelSize; i++)
            {
                var node = queue.Dequeue();
                levelList.Add(node.Val);

                // 将下一层的节点加入队列
                if (node.Left != null) queue.Enqueue(node.Left);
                if (node.Right != null) queue.Enqueue(node.Right);
            }

            result.Add(levelList);
        }

        return result;
    }

    // ========== Morris 遍历（进阶） ==========

    /// <summary>
    /// Morris 中序遍历：O(1) 空间复杂度
    /// 核心思想：利用叶节点的空指针指向后继节点，遍历完后恢复树结构
    /// </summary>
    public List<int> MorrisInorder(TreeNode root)
    {
        var result = new List<int>();
        var current = root;

        while (current != null)
        {
            if (current.Left == null)
            {
                // 无左子树，直接访问当前节点，移向右子树
                result.Add(current.Val);
                current = current.Right;
            }
            else
            {
                // 找到左子树的最右节点（中序前驱）
                var predecessor = current.Left;
                while (predecessor.Right != null && predecessor.Right != current)
                {
                    predecessor = predecessor.Right;
                }

                if (predecessor.Right == null)
                {
                    // 建立线索：前驱的右指针指向当前节点
                    predecessor.Right = current;
                    current = current.Left;
                }
                else
                {
                    // 已经遍历完左子树，恢复树结构
                    predecessor.Right = null;
                    result.Add(current.Val);
                    current = current.Right;
                }
            }
        }

        return result;
    }
}

/// <summary>
/// 测试程序
/// </summary>
public class Program
{
    public static void Main()
    {
        // 构建示例二叉树
        //       1
        //      / \
        //     2   3
        //    / \   \
        //   4   5   6
        var root = new TreeNode(1,
            new TreeNode(2, new TreeNode(4), new TreeNode(5)),
            new TreeNode(3, null, new TreeNode(6))
        );

        var traversal = new BinaryTreeTraversal();

        Console.WriteLine("前序遍历(递归): " + string.Join(", ", traversal.PreorderRecursive(root)));
        Console.WriteLine("前序遍历(迭代): " + string.Join(", ", traversal.PreorderIterative(root)));
        Console.WriteLine("中序遍历(递归): " + string.Join(", ", traversal.InorderRecursive(root)));
        Console.WriteLine("中序遍历(迭代): " + string.Join(", ", traversal.InorderIterative(root)));
        Console.WriteLine("后序遍历(递归): " + string.Join(", ", traversal.PostorderRecursive(root)));
        Console.WriteLine("后序遍历(迭代): " + string.Join(", ", traversal.PostorderIterative(root)));
        Console.WriteLine("Morris中序:     " + string.Join(", ", traversal.MorrisInorder(root)));

        Console.WriteLine("\n层序遍历:");
        var levels = traversal.LevelOrder(root);
        for (int i = 0; i < levels.Count; i++)
        {
            Console.WriteLine($"  第{i + 1}层: [{string.Join(", ", levels[i])}]");
        }
    }
}

9. 复杂度分析

遍历方式时间复杂度空间复杂度说明
递归遍历O(n)O(h)h为树高，最差O(n)
迭代遍历O(n)O(h)栈空间与树高相关
层序遍历O(n)O(w)w为最大层宽度
Morris遍历O(n)O(1)不需要额外空间

其中 n 为节点总数，h 为树高（平衡树为 log n，退化为链表时为 n）

10. 易错点

  1. 迭代中序遍历：容易忘记 current != null 的条件，导致无法处理右子树
  2. 后序迭代：不要尝试直接模拟，用"前序变体+反转"更简洁
  3. 层序遍历：必须在循环开始时记录 queue.Count，因为循环中会添加新节点
  4. Morris遍历：遍历过程中临时修改了树结构，多线程环境下不安全
  5. 递归深度：极端情况（链式树）递归可能导致栈溢出，需要改用迭代

13. 记忆口诀

前中后序看根位，根在前中后分别对。
前序：根左右，先根后子树；
中序：左根右，BST变有序；
后序：左右根，删除安全用；
层序：用队列，逐层从左右。
迭代记住栈，Morris空间省。

14. 面试常问点

  4. 不用递归和栈如何遍历？ → Morris遍历
  5. 递归遍历的时间和空间复杂度？ → O(n)时间，O(h)空间
