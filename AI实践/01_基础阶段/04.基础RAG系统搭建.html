基础 RAG 系统从零搭建

RAG（Retrieval-Augmented Generation）是把知识库"喂给" LLM 的核心技术。本文手把手搭建一个能处理 PDF 文档的问答系统，不依赖 LangChain，每一步都透明可控。

1. RAG 整体流程

文档输入
  ↓
文档加载（PDF/Word/网页）
  ↓
文本分块（Chunking）
  ↓
Embedding 向量化
  ↓
存入向量数据库
  ↓
              ← 用户提问
              ↓
          相似度检索
              ↓
          上下文组装
              ↓
          LLM 生成回答

2. 文档加载

PDF 加载

pip install pypdf httpx openai chromadb
from pathlib import Path
import pypdf

def load_pdf(file_path: str) -> list[dict]:
    """加载 PDF，返回带页码的文本块列表"""
    reader = pypdf.PdfReader(file_path)
    pages = []
    
    for page_num, page in enumerate(reader.pages, 1):
        text = page.extract_text()
        if text.strip():  # 跳过空页
            pages.append({
                "content": text,
                "metadata": {
                    "source": Path(file_path).name,
                    "page": page_num,
                }
            })
    
    return pages

# 使用
pages = load_pdf("技术文档.pdf")
print(f"加载了 {len(pages)} 页")

网页加载

import httpx
from bs4 import BeautifulSoup

async def load_webpage(url: str) -> dict:
    async with httpx.AsyncClient() as client:
        resp = await client.get(url, follow_redirects=True)
        resp.raise_for_status()
    
    soup = BeautifulSoup(resp.text, "html.parser")
    
    # 去掉 script/style 标签
    for tag in soup(["script", "style", "nav", "footer"]):
        tag.decompose()
    
    text = soup.get_text(separator="\n", strip=True)
    
    return {
        "content": text,
        "metadata": {"source": url, "type": "webpage"}
    }

3. 文本分块（Chunking）

分块是 RAG 效果的关键，块太大检索不精准，块太小上下文不足。

固定大小分块

def chunk_by_size(
    text: str,
    chunk_size: int = 500,
    chunk_overlap: int = 50,
) -> list[str]:
    """按固定字符数分块，带重叠"""
    chunks = []
    start = 0
    
    while start < len(text):
        end = start + chunk_size
        chunk = text[start:end]
        
        # 尽量在句号/换行处断开
        if end < len(text):
            # 找最后一个句号或换行
            last_break = max(
                chunk.rfind("。"),
                chunk.rfind("\n"),
                chunk.rfind(". "),
            )
            if last_break > chunk_size * 0.6:  # 找到了且不会太短
                chunk = chunk[:last_break + 1]
                end = start + last_break + 1
        
        if chunk.strip():
            chunks.append(chunk.strip())
        
        start = end - chunk_overlap  # 带重叠
    
    return chunks

按标题分块（Markdown 文档）

import re

def chunk_by_markdown_headers(text: str) -> list[dict]:
    """按 Markdown 标题分块，保留标题层级信息"""
    # 匹配 # ## ### 开头的行
    header_pattern = re.compile(r"^(#{1,3})\s+(.+)$", re.MULTILINE)
    
    sections = []
    matches = list(header_pattern.finditer(text))
    
    for i, match in enumerate(matches):
        level = len(match.group(1))
        title = match.group(2)
        start = match.end()
        end = matches[i + 1].start() if i + 1 < len(matches) else len(text)
        
        content = text[start:end].strip()
        if content:
            sections.append({
                "title": title,
                "level": level,
                "content": f"{title}\n\n{content}",
            })
    
    return sections

分块策略选择

文档类型推荐策略chunk_size
PDF 技术文档固定大小 + 重叠500-800 字
Markdown 文档按标题分块—
代码文件按函数/类分块—
新闻/长文固定大小 + 重叠300-500 字

4. Embedding 向量化

from openai import AsyncOpenAI
import asyncio

client = AsyncOpenAI()

async def embed_texts(texts: list[str], batch_size: int = 100) -> list[list[float]]:
    """批量生成 embedding，处理大量文本时分批"""
    all_embeddings = []
    
    for i in range(0, len(texts), batch_size):
        batch = texts[i:i + batch_size]
        response = await client.embeddings.create(
            model="text-embedding-3-small",
            input=batch,
        )
        batch_embeddings = [item.embedding for item in response.data]
        all_embeddings.extend(batch_embeddings)
        
        print(f"已处理 {min(i + batch_size, len(texts))}/{len(texts)} 个文本块")
    
    return all_embeddings

5. 向量数据库（ChromaDB）

ChromaDB 是最适合本地开发的向量数据库，零配置启动。

import chromadb
from chromadb.config import Settings

# 持久化存储
chroma_client = chromadb.PersistentClient(path="./chroma_db")

class VectorStore:
    def __init__(self, collection_name: str):
        self.collection = chroma_client.get_or_create_collection(
            name=collection_name,
            metadata={"hnsw:space": "cosine"},  # 余弦相似度
        )
    
    def add_documents(
        self,
        chunks: list[str],
        embeddings: list[list[float]],
        metadatas: list[dict] | None = None,
    ):
        """批量添加文档"""
        ids = [f"id_{i}" for i in range(len(chunks))]
        self.collection.add(
            documents=chunks,
            embeddings=embeddings,
            metadatas=metadatas or [{}] * len(chunks),
            ids=ids,
        )
        print(f"已添加 {len(chunks)} 个文档块")
    
    def search(
        self,
        query_embedding: list[float],
        n_results: int = 5,
        where: dict | None = None,  # 元数据过滤
    ) -> list[dict]:
        """语义检索"""
        results = self.collection.query(
            query_embeddings=[query_embedding],
            n_results=n_results,
            where=where,
        )
        
        return [
            {
                "content": doc,
                "metadata": meta,
                "distance": dist,
            }
            for doc, meta, dist in zip(
                results["documents"][0],
                results["metadatas"][0],
                results["distances"][0],
            )
        ]

6. 完整 RAG 问答系统

把以上模块串联起来：

import asyncio
from pathlib import Path
from openai import AsyncOpenAI

client = AsyncOpenAI()
vector_store = VectorStore("my_knowledge_base")

async def index_pdf(file_path: str):
    """把 PDF 文档建立索引"""
    print(f"正在处理: {file_path}")
    
    # 1. 加载文档
    pages = load_pdf(file_path)
    
    # 2. 分块
    all_chunks = []
    all_metadata = []
    
    for page in pages:
        chunks = chunk_by_size(page["content"], chunk_size=500, chunk_overlap=50)
        for chunk in chunks:
            all_chunks.append(chunk)
            all_metadata.append({
                **page["metadata"],
                "chunk_index": len(all_chunks),
            })
    
    print(f"共 {len(all_chunks)} 个文本块")
    
    # 3. 向量化
    embeddings = await embed_texts(all_chunks)
    
    # 4. 入库
    vector_store.add_documents(all_chunks, embeddings, all_metadata)
    print("索引建立完成！")

async def ask(question: str) -> str:
    """问答"""
    # 1. 问题向量化
    q_emb_resp = await client.embeddings.create(
        model="text-embedding-3-small",
        input=[question],
    )
    q_embedding = q_emb_resp.data[0].embedding
    
    # 2. 检索最相关的文档块
    results = vector_store.search(q_embedding, n_results=5)
    
    if not results:
        return "知识库中没有相关信息。"
    
    # 3. 组装上下文
    context_parts = []
    for r in results:
        source = r["metadata"].get("source", "未知")
        page = r["metadata"].get("page", "")
        page_info = f"（{source} 第{page}页）" if page else f"（{source}）"
        context_parts.append(f"{r['content']}{page_info}")
    
    context = "\n\n---\n\n".join(context_parts)
    
    # 4. 生成回答
    response = await client.chat.completions.create(
        model="gpt-4o",
        messages=[
            {
                "role": "system",
                "content": """你是一个知识库问答助手。
请仅根据提供的参考资料回答问题。
如果参考资料中没有相关信息，请明确说明"我在知识库中没有找到相关信息"。
回答时引用信息来源（文档名和页码）。"""
            },
            {
                "role": "user",
                "content": f"""参考资料：

{context}

问题：{question}"""
            }
        ],
        temperature=0.3,
    )
    
    return response.choices[0].message.content

# 主程序
async def main():
    # 建立索引（只需要一次）
    if Path("技术文档.pdf").exists():
        await index_pdf("技术文档.pdf")
    
    # 交互问答
    while True:
        question = input("\n请提问（输入 q 退出）：")
        if question.lower() == "q":
            break
        
        answer = await ask(question)
        print(f"\n回答：{answer}")

asyncio.run(main())

7. 效果优化

问题：检索到不相关的内容

原因：相似度阈值太低，把低相关度的内容也返回了

解法：

def search_with_threshold(
    query_embedding: list[float],
    threshold: float = 0.7,  # 余弦相似度阈值
    n_results: int = 5,
) -> list[dict]:
    results = vector_store.search(query_embedding, n_results=n_results * 2)
    
    # ChromaDB 的 distance 是 1 - cosine_similarity
    # similarity = 1 - distance，越高越相关
    filtered = [r for r in results if (1 - r["distance"]) >= threshold]
    return filtered[:n_results]

问题：中文分词效果差

对于中文文档，在分块前进行清理：

import re

def clean_chinese_text(text: str) -> str:
    # 去掉多余空行
    text = re.sub(r"\n{3,}", "\n\n", text)
    # 去掉行内多余空格（中文一般不用空格分词）
    text = re.sub(r"[ \t]+", " ", text)
    # 去掉可能影响分块的莫名字符
    text = re.sub(r"[^\u4e00-\u9fff\w\s\n\r.,!?;:，。！？；：「」『』【】《》—…]", "", text)
    return text.strip()
