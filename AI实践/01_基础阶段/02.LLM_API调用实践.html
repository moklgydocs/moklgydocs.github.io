LLM API 调用实践

多厂商 LLM 接入是 AI 工程师的基本功。本文覆盖 OpenAI、通义千问、DeepSeek 三大厂商，并给出统一封装方案，让业务代码与具体厂商解耦。

1. OpenAI SDK 使用

安装与基础调用

pip install openai
from openai import AsyncOpenAI

client = AsyncOpenAI(api_key="sk-...")

async def chat(message: str) -> str:
    response = await client.chat.completions.create(
        model="gpt-4o",
        messages=[{"role": "user", "content": message}],
        temperature=0.7,
        max_tokens=2048,
    )
    return response.choices[0].message.content

流式输出

async def chat_stream(message: str):
    stream = await client.chat.completions.create(
        model="gpt-4o",
        messages=[{"role": "user", "content": message}],
        stream=True,
    )
    async for chunk in stream:
        delta = chunk.choices[0].delta.content
        if delta:
            yield delta

多轮对话

from openai.types.chat import ChatCompletionMessageParam

async def multi_turn_chat(history: list[ChatCompletionMessageParam], new_msg: str) -> str:
    history.append({"role": "user", "content": new_msg})
    
    response = await client.chat.completions.create(
        model="gpt-4o",
        messages=history,
    )
    
    assistant_msg = response.choices[0].message.content
    history.append({"role": "assistant", "content": assistant_msg})
    return assistant_msg

# 使用
history = [{"role": "system", "content": "你是一个专业的 Python 导师"}]
await multi_turn_chat(history, "什么是装饰器？")
await multi_turn_chat(history, "给我一个实际的例子")  # 有上下文

结构化输出（Structured Output）

from pydantic import BaseModel

class ProductInfo(BaseModel):
    name: str
    price: float
    category: str
    in_stock: bool

async def extract_product(text: str) -> ProductInfo:
    response = await client.beta.chat.completions.parse(
        model="gpt-4o",
        messages=[
            {"role": "system", "content": "从文本中提取产品信息"},
            {"role": "user", "content": text},
        ],
        response_format=ProductInfo,
    )
    return response.choices[0].message.parsed

from openai import AsyncOpenAI

)

async def deepseek_chat(message: str, use_reasoner: bool = False) -> str:

DeepSeek Reasoner 获取推理过程

async def deepseek_with_reasoning(problem: str):

4. 统一封装方案

生产环境中需要能灵活切换模型，业务代码不应感知具体厂商。

接口定义

# llm/base.py
from abc import ABC, abstractmethod
from typing import AsyncIterator
from dataclasses import dataclass

@dataclass
class LLMMessage:
    role: str  # "system" | "user" | "assistant"
    content: str

@dataclass 
class LLMResponse:
    content: str
    model: str
    prompt_tokens: int
    completion_tokens: int

class BaseLLM(ABC):
    @abstractmethod
    async def complete(
        self,
        messages: list[LLMMessage],
        **kwargs,
    ) -> LLMResponse:
        ...

    @abstractmethod
    async def stream(
        self,
        messages: list[LLMMessage],
        **kwargs,
    ) -> AsyncIterator[str]:
        ...

OpenAI 实现

# llm/openai_llm.py
from openai import AsyncOpenAI
from .base import BaseLLM, LLMMessage, LLMResponse
from typing import AsyncIterator

class OpenAILLM(BaseLLM):
    def __init__(self, api_key: str, model: str = "gpt-4o", base_url: str | None = None):
        self.client = AsyncOpenAI(api_key=api_key, base_url=base_url)
        self.model = model

    def _to_openai_msgs(self, messages: list[LLMMessage]) -> list[dict]:
        return [{"role": m.role, "content": m.content} for m in messages]

    async def complete(self, messages: list[LLMMessage], **kwargs) -> LLMResponse:
        resp = await self.client.chat.completions.create(
            model=self.model,
            messages=self._to_openai_msgs(messages),
            **kwargs,
        )
        return LLMResponse(
            content=resp.choices[0].message.content,
            model=resp.model,
            prompt_tokens=resp.usage.prompt_tokens,
            completion_tokens=resp.usage.completion_tokens,
        )

    async def stream(self, messages: list[LLMMessage], **kwargs) -> AsyncIterator[str]:
        s = await self.client.chat.completions.create(
            model=self.model,
            messages=self._to_openai_msgs(messages),
            stream=True,
            **kwargs,
        )
        async for chunk in s:
            delta = chunk.choices[0].delta.content
            if delta:
                yield delta

工厂函数

# llm/factory.py
from .base import BaseLLM
from .openai_llm import OpenAILLM
import os

def create_llm(provider: str = "openai", **kwargs) -> BaseLLM:
    match provider:
        case "openai":
            return OpenAILLM(
                api_key=os.getenv("OPENAI_API_KEY"),
                model=kwargs.get("model", "gpt-4o"),
            )
        case "qwen":
            return OpenAILLM(
                api_key=os.getenv("DASHSCOPE_API_KEY"),
                model=kwargs.get("model", "qwen-max"),
                base_url="https://dashscope.aliyuncs.com/compatible-mode/v1",
            )
        case "deepseek":
            return OpenAILLM(
                api_key=os.getenv("DEEPSEEK_API_KEY"),
                model=kwargs.get("model", "deepseek-chat"),
                base_url="https://api.deepseek.com",
            )
        case _:
            raise ValueError(f"Unknown provider: {provider}")

使用示例

from llm.factory import create_llm
from llm.base import LLMMessage

# 只改这一行就能切换厂商
llm = create_llm("deepseek")

messages = [
    LLMMessage(role="system", content="你是一个代码审查专家"),
    LLMMessage(role="user", content="帮我审查这段 Python 代码：\n```python\ndef add(a, b): return a+b\n```"),
]

response = await llm.complete(messages)
print(response.content)

6. 常见模型对比

模型适用场景优势注意
GPT-4o复杂推理、多模态能力最强，接口稳定贵，国内需代理
GPT-4o-mini一般任务性价比高复杂任务能力弱些
Qwen-Max中文任务中文理解强，不需代理价格适中
DeepSeek-Chat代码/逻辑极低价，代码能力强并发限制
DeepSeek-Reasoner复杂推理推理能力强，R1 开源版响应较慢
