Python 异步编程与 FastAPI

Python 异步编程与 FastAPI

LLM 服务天然适合异步：等模型推理时 CPU 几乎空闲，异步并发可以极大提升吞吐量。本文从 asyncio 基础出发，直接落到 FastAPI 流式 LLM 接口的完整实现。

1. asyncio 核心概念

协程 vs 线程

import asyncio
import time

# 同步版本：串行等待，总耗时 = 各任务之和
def sync_fetch(url: str) -> str:
    time.sleep(1)  # 模拟 IO 等待
    return f"result:{url}"

# 异步版本：并发等待，总耗时 ≈ 最慢任务
async def async_fetch(url: str) -> str:
    await asyncio.sleep(1)  # 挂起，让出控制权
    return f"result:{url}"

async def main():
    # 并发执行 3 个任务，总耗时约 1 秒而非 3 秒
    results = await asyncio.gather(
        async_fetch("url1"),
        async_fetch("url2"),
        async_fetch("url3"),
    )
    print(results)

asyncio.run(main())

关键 API 速查

API用途注意
await coroutine等待协程完成只能在 async 函数内用
asyncio.gather(*coros)并发执行多个协程任一报错默认取消其余
asyncio.create_task(coro)后台执行不阻塞当前需持有引用防止 GC
asyncio.sleep(n)非阻塞等待替代 time.sleep
asyncio.timeout(n)超时控制（3.11+）旧版用 wait_for

避坑：同步阻塞不能放在 async 函数里

import asyncio
from concurrent.futures import ThreadPoolExecutor

executor = ThreadPoolExecutor()

async def run_sync_in_thread(sync_fn, *args):
    """把同步阻塞函数放到线程池，不阻塞事件循环"""
    loop = asyncio.get_event_loop()
    return await loop.run_in_executor(executor, sync_fn, *args)

# 错误示范：直接在 async 函数里调用 requests（会阻塞事件循环）
async def bad_fetch(url):
    import requests
    return requests.get(url).text  # ❌ 阻塞！

# 正确：用 httpx.AsyncClient
async def good_fetch(url):
    import httpx
    async with httpx.AsyncClient() as client:
        resp = await client.get(url)
        return resp.text

2. httpx 异步 HTTP 客户端

httpx 是 LLM 调用场景的首选库，支持异步、流式读取、HTTP/2。

基础用法

import httpx
import asyncio

async def call_llm_api():
    async with httpx.AsyncClient(timeout=60.0) as client:
        response = await client.post(
            "https://api.openai.com/v1/chat/completions",
            headers={"Authorization": "Bearer sk-..."},
            json={
                "model": "gpt-4o",
                "messages": [{"role": "user", "content": "你好"}],
            }
        )
        response.raise_for_status()
        return response.json()

流式读取（SSE）

async def stream_llm():
    async with httpx.AsyncClient(timeout=None) as client:
        async with client.stream(

4. 流式输出接口（核心）

流式输出是 LLM 接口的标配，用户体验从"等待 10 秒"变为"看着字一个个出现"。

StreamingResponse + SSE

# routers/chat.py (流式版)
from fastapi import APIRouter
from fastapi.responses import StreamingResponse
from schemas.chat import ChatRequest
from services.llm import LLMService
import json

router = APIRouter(tags=["chat"])
llm = LLMService()

@router.post("/chat/stream")
async def chat_stream(req: ChatRequest):
    async def event_generator():
        async for chunk in llm.stream(req):
            # SSE 格式：每条消息以 "data: " 开头，两个换行结束
            yield f"data: {json.dumps({'content': chunk}, ensure_ascii=False)}\n\n"
        yield "data: [DONE]\n\n"

    return StreamingResponse(
        event_generator(),
        media_type="text/event-stream",
        headers={
            "Cache-Control": "no-cache",
            "X-Accel-Buffering": "no",  # 禁用 nginx 缓冲
        }
    )

LLM Service 实现

# services/llm.py
import httpx
import json
from typing import AsyncIterator
from schemas.chat import ChatRequest, ChatResponse

class LLMService:
    def __init__(self):
        self.api_key = "sk-..."  # 实际用环境变量
        self.base_url = "https://api.openai.com/v1"

    async def complete(self, req: ChatRequest) -> ChatResponse:
        async with httpx.AsyncClient(timeout=60.0) as client:
            resp = await client.post(
                f"{self.base_url}/chat/completions",
                headers={"Authorization": f"Bearer {self.api_key}"},
                json={
                    "model": req.model,
                    "messages": [{"role": "user", "content": req.message}],
                    "temperature": req.temperature,
                }
            )
            resp.raise_for_status()
            data = resp.json()
            return ChatResponse(
                content=data["choices"][0]["message"]["content"],
                model=data["model"],
                usage=data["usage"],
            )

    async def stream(self, req: ChatRequest) -> AsyncIterator[str]:
        async with httpx.AsyncClient(timeout=None) as client:
            async with client.stream(
                "POST",
                f"{self.base_url}/chat/completions",
                headers={"Authorization": f"Bearer {self.api_key}"},
                json={
                    "model": req.model,
                    "messages": [{"role": "user", "content": req.message}],
                    "temperature": req.temperature,
                    "stream": True,
                }
            ) as response:
                response.raise_for_status()
                async for line in response.aiter_lines():
                    if line.startswith("data: ") and line != "data: [DONE]":
                        chunk = json.loads(line[6:])
                        delta = chunk["choices"][0]["delta"].get("content")
                        if delta:
                            yield delta

5. 完整可运行示例

# app.py — 单文件完整示例，直接运行
import asyncio
import json
import os
import httpx
from fastapi import FastAPI
from fastapi.responses import StreamingResponse
from pydantic import BaseModel

app = FastAPI()

class ChatReq(BaseModel):
    message: str
    model: str = "gpt-4o-mini"

OPENAI_KEY = os.getenv("OPENAI_API_KEY", "")

@app.post("/chat/stream")
async def chat_stream(req: ChatReq):
    async def gen():
        async with httpx.AsyncClient(timeout=None) as client:
            async with client.stream(
                "POST",
                "https://api.openai.com/v1/chat/completions",
                headers={"Authorization": f"Bearer {OPENAI_KEY}"},
                json={
                    "model": req.model,
                    "messages": [{"role": "user", "content": req.message}],
                    "stream": True,
                }
            ) as resp:
                async for line in resp.aiter_lines():
                    if line.startswith("data: ") and "[DONE]" not in line:
                        data = json.loads(line[6:])
                        content = data["choices"][0]["delta"].get("content", "")
                        if content:
                            yield f"data: {json.dumps({'content': content})}\n\n"
        yield "data: [DONE]\n\n"

    return StreamingResponse(gen(), media_type="text/event-stream")

# 测试用健康检查
@app.get("/health")
async def health():
    return {"status": "ok"}

# 运行：uvicorn app:app --reload

6. 避坑指南

坑原因解法
响应不流式，一次性返回nginx 缓冲或缺少 Cache-Control加 X-Accel-Buffering: no header
await 在普通函数里报错忘记加 async def将函数改为 async def
连接超时LLM 推理慢，默认超时 5s流式接口设 timeout=None
并发量大时内存暴涨每次创建新 AsyncClient用 lifespan 共享一个 client
文字截断（中文乱码）SSE 按行分割时 utf-8 被截断用 aiter_text() + 手动解析

共享 AsyncClient（生产推荐）

from contextlib import asynccontextmanager
from fastapi import FastAPI
import httpx

http_client: httpx.AsyncClient = None

@asynccontextmanager
async def lifespan(app: FastAPI):
    global http_client
    http_client = httpx.AsyncClient(timeout=None)
    yield
    await http_client.aclose()

app = FastAPI(lifespan=lifespan)
