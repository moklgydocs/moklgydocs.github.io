生产可观测性建设

生产可观测性建设

AI 系统的可观测性不只是"记日志"。本文建立覆盖链路追踪、成本监控、质量指标的完整体系。

1. 可观测性三支柱

支柱对 LLM 系统的含义
日志每次 LLM 调用的 Prompt/Response、工具调用详情
指标延迟、Token 用量、成本、错误率、召回质量
链路追踪一个用户问题触发的多个 LLM/工具调用的完整链路

2. LangSmith 集成（最简方案）

LangSmith 是 LangChain 官方追踪平台，5 分钟接入。

手动追踪（非 LangChain）

from langsmith import traceable, Client

client = Client()

@traceable(name="rag_query")  # 函数级追踪
async def rag_query(question: str) -> str:
    # 这里的一切调用都会被记录
    contexts = await retrieve(question)
    answer = await generate(question, contexts)
    return answer

# 追加评估分数
@traceable(name="evaluate_answer")
def evaluate(run_id: str, score: float, comment: str):
    client.create_feedback(
        run_id=run_id,
        key="user_rating",
        score=score,
        comment=comment,
    )

from opentelemetry import trace
from opentelemetry.sdk.trace import TracerProvider
from opentelemetry.sdk.trace.export import BatchSpanProcessor
from opentelemetry.exporter.otlp.proto.grpc.trace_exporter import OTLPSpanExporter

    BatchSpanProcessor(
        OTLPSpanExporter(endpoint="http://jaeger:4317")  # 发送到 Jaeger
    )
)
trace.set_tracer_provider(provider)

# 自动为 OpenAI 调用添加追踪
OpenAIInstrumentor().instrument()

# 之后使用 OpenAI 会自动产生 Span
tracer = trace.get_tracer("my-app")

with tracer.start_as_current_span("rag_pipeline") as span:
    span.set_attribute("user.id", user_id)
    span.set_attribute("question", question)
    
    result = await rag_query(question)
    
    span.set_attribute("answer_length", len(result))

4. 自定义监控中间件

import time
import logging
from dataclasses import dataclass, field
from datetime import datetime

logger = logging.getLogger("llm_monitor")

@dataclass
class LLMCallRecord:
    timestamp: str = field(default_factory=lambda: datetime.now().isoformat())
    model: str = ""
    prompt_tokens: int = 0
    completion_tokens: int = 0
    latency_ms: float = 0.0
    cost_usd: float = 0.0
    success: bool = True
    error: str = ""
    session_id: str = ""

# 模型定价（每 1K tokens）
MODEL_PRICING = {
    "gpt-4o": {"input": 0.0025, "output": 0.01},
    "gpt-4o-mini": {"input": 0.00015, "output": 0.0006},
    "deepseek-chat": {"input": 0.00014, "output": 0.00028},
}

class LLMMonitor:
    def __init__(self):
        self.records: list[LLMCallRecord] = []
        self.daily_cost = 0.0

    async def track_call(self, model: str, prompt_tokens: int, func, *args, **kwargs):
        record = LLMCallRecord(model=model, prompt_tokens=prompt_tokens)
        start = time.time()
        
        try:
            result = await func(*args, **kwargs)
            record.latency_ms = (time.time() - start) * 1000
            record.completion_tokens = result.usage.completion_tokens
            record.cost_usd = self._calc_cost(model, prompt_tokens, record.completion_tokens)
            self.daily_cost += record.cost_usd
            return result
        except Exception as e:
            record.success = False
            record.error = str(e)
            record.latency_ms = (time.time() - start) * 1000
            raise
        finally:
            self.records.append(record)
            self._log(record)

    def _calc_cost(self, model: str, input_tokens: int, output_tokens: int) -> float:
        pricing = MODEL_PRICING.get(model, {"input": 0.001, "output": 0.002})
        return (input_tokens * pricing["input"] + output_tokens * pricing["output"]) / 1000

    def _log(self, record: LLMCallRecord):
        logger.info(
            "llm_call",
            extra={
                "model": record.model,
                "latency_ms": record.latency_ms,
                "tokens": record.prompt_tokens + record.completion_tokens,
                "cost_usd": record.cost_usd,
                "success": record.success,
            }
        )
    
    def get_stats(self) -> dict:
        if not self.records:
            return {}
        
        latencies = [r.latency_ms for r in self.records if r.success]
        return {
            "total_calls": len(self.records),
            "success_rate": sum(1 for r in self.records if r.success) / len(self.records),
            "avg_latency_ms": sum(latencies) / len(latencies) if latencies else 0,
            "p95_latency_ms": sorted(latencies)[int(len(latencies) * 0.95)] if latencies else 0,
            "total_cost_usd": sum(r.cost_usd for r in self.records),
            "total_tokens": sum(r.prompt_tokens + r.completion_tokens for r in self.records),
        }

monitor = LLMMonitor()

5. 关键指标告警

# 用 Prometheus + Grafana（生产推荐）
from prometheus_client import Counter, Histogram, Gauge, start_http_server

# 定义指标
llm_calls_total = Counter(
    "llm_calls_total",
    "LLM 调用总次数",
    ["model", "status"]
)
llm_latency = Histogram(
    "llm_latency_seconds",
    "LLM 调用延迟",
    ["model"],
    buckets=[0.5, 1, 2, 5, 10, 30]
)
llm_cost_total = Counter(
    "llm_cost_dollars_total",
    "LLM 总费用（美元）",
    ["model"]
)
daily_budget_remaining = Gauge(
    "llm_daily_budget_remaining_dollars",
    "今日剩余预算"
)

def record_llm_call(model: str, latency: float, cost: float, success: bool):
    status = "success" if success else "error"
    llm_calls_total.labels(model=model, status=status).inc()
    llm_latency.labels(model=model).observe(latency)
    llm_cost_total.labels(model=model).inc(cost)

# 启动 metrics 端点（供 Prometheus 抓取）
start_http_server(8080)

Grafana 告警规则（示例）

# grafana-alerts.yaml
groups:
  - name: llm_alerts
    rules:
      - alert: HighLLMErrorRate
        expr: rate(llm_calls_total{status="error"}[5m]) > 0.1
        annotations:
          summary: "LLM 错误率超过 10%"
      
      - alert: DailyBudgetExceeded
        expr: llm_cost_dollars_total > 50  # $50/天预算
        annotations:
          summary: "今日 LLM 费用超过预算限制"
      
      - alert: HighLatency
        expr: histogram_quantile(0.95, llm_latency_seconds) > 10
        annotations:
          summary: "LLM P95 延迟超过 10 秒"

@app.post("/feedback")
async def submit_feedback(req: FeedbackRequest):
    feedback_store.append(req.model_dump())
    
    # 低评分触发告警

        logger.warning(
            "low_rating_feedback",
            extra={"session_id": req.session_id, "rating": req.rating}

    return {"status": "ok"}
