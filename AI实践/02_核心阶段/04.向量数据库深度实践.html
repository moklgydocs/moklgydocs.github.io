向量数据库深度实践

向量数据库是 RAG 系统的存储核心。本文对比主流选型，深入讲解索引调优、过滤查询和生产运维实践。

1. 主流选型对比

数据库适用场景部署方式性能备注
ChromaDB本地开发、原型嵌入式/服务一般零配置，适合快速验证
Qdrant中小规模生产Docker/云优秀Rust 编写，内存效率高
Milvus大规模生产Kubernetes最强10亿+ 向量，运维复杂
Weaviate混合搜索Docker/云优秀内置 BM25 + 向量混合
PgVector已有 PostgreSQL插件中等规模不大时首选，减少运维

2. ChromaDB 深度使用

集合配置与索引

import chromadb
from chromadb.config import Settings

# 持久化客户端
client = chromadb.PersistentClient(
    path="./chroma_data",
    settings=Settings(
        anonymized_telemetry=False,  # 关闭遥测
    )
)

# 创建集合时指定距离函数
collection = client.get_or_create_collection(
    name="documents",
    metadata={
        "hnsw:space": "cosine",          # 余弦相似度（文本推荐）
        "hnsw:construction_ef": 200,      # 建索引精度，越大越准但越慢
        "hnsw:search_ef": 100,            # 查询精度
        "hnsw:M": 16,                     # 每个节点的连接数，影响内存和精度
    }
)

元数据过滤查询

# 结构化过滤 + 语义搜索组合
results = collection.query(

支持的过滤操作符

操作符含义示例
$eq等于{"status": {"$eq": "active"}}
$ne不等于{"deleted": {"$ne": True}}
$gt/$gte大于/大于等于{"score": {"$gte": 0.8}}
$lt/$lte小于/小于等于{"age": {"$lt": 30}}
$in在列表中{"tag": {"$in": ["AI", "ML"]}}
$and/$or逻辑组合{"$and": [...]}

批量插入

import uuid

def upsert_documents(
    chunks: list[str],
    embeddings: list[list[float]],
    metadatas: list[dict],
):
    points = [
        PointStruct(
            id=str(uuid.uuid4()),
            vector=emb,
            payload={
                "content": chunk,
                **metadata,
            }
        )
        for chunk, emb, metadata in zip(chunks, embeddings, metadatas)
    ]
    
    # 批量插入（每批 100 个）
    batch_size = 100
    for i in range(0, len(points), batch_size):
        client.upsert(
            collection_name="knowledge_base",
            points=points[i:i + batch_size],
        )
        print(f"已插入 {min(i + batch_size, len(points))}/{len(points)}")

语义搜索 + 过滤

def search(
    query_vector: list[float],
    filters: dict | None = None,
    top_k: int = 5,
    score_threshold: float = 0.7,
) -> list[dict]:
    
    # 构建过滤条件
    qdrant_filter = None
    if filters:
        conditions = []
        for key, value in filters.items():
            if isinstance(value, list):
                conditions.append(
                    FieldCondition(key=key, match=MatchAny(any=value))
                )
            elif isinstance(value, dict) and "$gte" in value:
                conditions.append(
                    FieldCondition(key=key, range=Range(gte=value["$gte"]))
                )
            else:
                conditions.append(
                    FieldCondition(key=key, match=MatchValue(value=value))
                )
        qdrant_filter = Filter(must=conditions)
    
    results = client.search(
        collection_name="knowledge_base",
        query_vector=query_vector,
        query_filter=qdrant_filter,
        limit=top_k,
        score_threshold=score_threshold,
        with_payload=True,
    )
    
    return [
        {
            "content": r.payload.get("content", ""),
            "score": r.score,
            "metadata": {k: v for k, v in r.payload.items() if k != "content"},
        }
        for r in results
    ]

4. PgVector：PostgreSQL 插件方案

适合已有 PostgreSQL 基础设施，不想再维护一个新系统。

-- 安装扩展
CREATE EXTENSION vector;

-- 创建表
CREATE TABLE documents (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    content TEXT NOT NULL,
    embedding VECTOR(1536),  -- text-embedding-3-small 维度
    source VARCHAR(255),
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 创建 IVFFlat 索引（必须！否则全表扫描）
CREATE INDEX ON documents 
USING ivfflat (embedding vector_cosine_ops)
WITH (lists = 100);  -- lists ≈ sqrt(行数)

-- 语义搜索
SELECT content, 1 - (embedding <=> $1::vector) AS similarity
FROM documents
WHERE source = 'tech-docs'  -- 普通字段过滤
ORDER BY embedding <=> $1::vector
LIMIT 5;
# Python 使用
import asyncpg
import numpy as np

async def search_pgvector(
    query_embedding: list[float],
    filter_source: str | None = None,
    top_k: int = 5,
) -> list[dict]:
    conn = await asyncpg.connect("postgresql://...")
    
    vector_str = "[" + ",".join(str(x) for x in query_embedding) + "]"
    
    if filter_source:
        rows = await conn.fetch("""
            SELECT content, source, 1 - (embedding <=> $1::vector) AS similarity
            FROM documents
            WHERE source = $2
            ORDER BY embedding <=> $1::vector
            LIMIT $3
        """, vector_str, filter_source, top_k)
    else:
        rows = await conn.fetch("""
            SELECT content, source, 1 - (embedding <=> $1::vector) AS similarity
            FROM documents
            ORDER BY embedding <=> $1::vector
            LIMIT $2
        """, vector_str, top_k)
    
    return [dict(row) for row in rows]

5. 索引调优

HNSW vs IVFFlat

指标HNSWIVFFlat
查询速度极快快
内存占用高（图结构）低
建索引时间慢快
精度高中（取决于 nprobe）
适用内存充足、查询频繁内存有限、数据量大

关键参数调优

# ChromaDB HNSW 参数
# M (16-64)：每个节点的双向连接数
# 越大 → 精度越高，内存越多，建索引越慢
# 推荐：16（平衡）到 48（高精度）

# ef_construction (100-500)：建索引搜索宽度
# 越大 → 精度越高，建索引越慢
# 推荐：200

# ef_search (50-300)：查询宽度
# 越大 → 精度越高，查询越慢
# 推荐：50（低延迟）到 200（高精度）

6. 混合搜索（向量 + 关键词）

纯向量搜索对精确词匹配效果差（如产品编号、人名），混合搜索更全面。

from qdrant_client.models import SparseVector, NamedSparseVector

# 稀疏向量（BM25 关键词）+ 稠密向量（语义）
async def hybrid_search(
    query: str,
    dense_vector: list[float],
    sparse_vector: dict,  # {token_id: weight}
    top_k: int = 10,
) -> list[dict]:
    results = client.query_points(
        collection_name="hybrid_collection",
        prefetch=[
            # 语义搜索：100 个候选
            models.Prefetch(
                query=dense_vector,
                using="dense",
                limit=100,
            ),
            # 关键词搜索：100 个候选
            models.Prefetch(
                query=models.SparseVector(**sparse_vector),
                using="sparse",
                limit=100,
            ),
        ],
        query=models.FusionQuery(
            fusion=models.Fusion.RRF  # Reciprocal Rank Fusion 重排
        ),
        limit=top_k,
    )
    return results
