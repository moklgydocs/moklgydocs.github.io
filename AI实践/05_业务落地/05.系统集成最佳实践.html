系统集成最佳实践

系统集成最佳实践

把 AI 功能嵌入到员工已在用的工具里，才能真正被使用。本文覆盖钉钉机器人、企业微信应用、统一 SSE 接口和鉴权方案。

1. 通用 SSE 流式接口规范

所有 AI 服务对外统一提供 SSE（Server-Sent Events）流式接口，前端体验"边想边说"。

1.1 接口协议

POST /api/ai/chat
Content-Type: application/json
Authorization: Bearer <token>

{
  "session_id": "uuid",           // 会话 ID，用于多轮对话
  "message": "用户输入",
  "context": {                    // 可选：业务上下文
    "page": "order_detail",
    "order_id": "ORD001"
  }
}

响应流（SSE）：

data: {"type": "status", "content": "正在检索..."}

data: {"type": "token", "content": "根"}
data: {"type": "token", "content": "据"}
data: {"type": "token", "content": "您"}
...

data: {"type": "source", "content": {"title": "退款政策", "url": "..."}}

data: {"type": "done", "full_content": "完整回答"}

data: [DONE]

消息类型说明：

type含义前端处理
status中间状态文字（如"正在检索"）显示加载提示
tokenLLM 流式 token，逐字追加append 到对话气泡
source来源引用（文档、订单等）在回答下方显示引用
error错误信息显示错误提示
done回答完成，含完整文本停止加载动画，记录历史

1.2 前端 SSE 客户端（TypeScript）

// lib/ai-client.ts
interface AIMessage {
  type: 'status' | 'token' | 'source' | 'done' | 'error';
  content: string | object;
}

interface AIStreamCallbacks {
  onStatus?: (text: string) => void;
  onToken?: (token: string) => void;
  onSource?: (source: object) => void;
  onDone?: (fullText: string) => void;
  onError?: (error: string) => void;
}

export async function streamChat(
  message: string,
  sessionId: string,
  callbacks: AIStreamCallbacks,
  signal?: AbortSignal,
) {
  const token = localStorage.getItem("auth_token");
  
  const response = await fetch("/api/ai/chat", {
    method: "POST",
    headers: {
      "Content-Type": "application/json",
      "Authorization": `Bearer ${token}`,
    },
    body: JSON.stringify({ message, session_id: sessionId }),
    signal,
  });
  
  if (!response.ok) {
    callbacks.onError?.(`请求失败：${response.status}`);
    return;
  }
  
  const reader = response.body!.getReader();
  const decoder = new TextDecoder();
  let buffer = "";
  
  while (true) {
    const { done, value } = await reader.read();
    if (done) break;
    
    buffer += decoder.decode(value, { stream: true });
    const lines = buffer.split("\n\n");
    buffer = lines.pop() || "";
    
    for (const line of lines) {
      if (!line.startsWith("data: ")) continue;
      const data = line.slice(6);
      
      if (data === "[DONE]") return;
      
      try {
        const msg: AIMessage = JSON.parse(data);
        
        switch (msg.type) {
          case "status": callbacks.onStatus?.(msg.content as string); break;
          case "token": callbacks.onToken?.(msg.content as string); break;
          case "source": callbacks.onSource?.(msg.content); break;
          case "done": callbacks.onDone?.(msg.content as string); break;
          case "error": callbacks.onError?.(msg.content as string); break;
        }
      } catch {}
    }
  }
}

// React 组件使用示例

# integrations/dingtalk.py
import hashlib
import hmac
import base64
import time
import httpx
from fastapi import FastAPI, Request

app = FastAPI()

DINGTALK_APP_ID = os.getenv("DINGTALK_APP_ID")
DINGTALK_APP_SECRET = os.getenv("DINGTALK_APP_SECRET")
DINGTALK_ROBOT_CODE = os.getenv("DINGTALK_ROBOT_CODE")

# === 获取访问令牌 ===

_token_cache = {"token": None, "expire": 0}

async def get_access_token() -> str:
    if _token_cache["token"] and time.time() < _token_cache["expire"]:
        return _token_cache["token"]
    
    timestamp = str(int(time.time() * 1000))
    sign_str = f"{timestamp}\n{DINGTALK_APP_SECRET}"
    sign = base64.b64encode(
        hmac.new(DINGTALK_APP_SECRET.encode(), sign_str.encode(), hashlib.sha256).digest()
    ).decode()
    
    async with httpx.AsyncClient() as client:
        resp = await client.post(
            "https://api.dingtalk.com/v1.0/oauth2/accessToken",
            json={
                "appKey": DINGTALK_APP_ID,
                "appSecret": DINGTALK_APP_SECRET,
            }
        )
    data = resp.json()
    _token_cache["token"] = data["accessToken"]
    _token_cache["expire"] = time.time() + data["expireIn"] - 60
    return _token_cache["token"]

# === 接收消息 WebHook ===

@app.post("/webhook/dingtalk")
async def receive_dingtalk_message(request: Request):
    """接收钉钉机器人消息"""
    # 验证签名（防止伪造）
    timestamp = request.headers.get("timestamp")
    sign = request.headers.get("sign")
    
    expected_sign = base64.b64encode(
        hmac.new(
            DINGTALK_APP_SECRET.encode(),
            f"{timestamp}\n{DINGTALK_APP_SECRET}".encode(),
            hashlib.sha256
        ).digest()
    ).decode()
    
    if sign != expected_sign:
        return {"code": 403, "message": "签名验证失败"}
    
    payload = await request.json()
    
    # 提取消息内容
    msg_type = payload.get("msgtype")
    sender_id = payload["senderStaffId"]
    conversation_id = payload.get("conversationId")
    
    if msg_type == "text":
        user_message = payload["text"]["content"].strip()
        
        # 异步处理，先快速返回 200
        import asyncio
        asyncio.create_task(
            process_and_reply_dingtalk(
                user_message=user_message,
                sender_id=sender_id,
                conversation_id=conversation_id,
                session_id=f"dd:{sender_id}",
            )
        )
    
    return {"code": 0}

async def process_and_reply_dingtalk(
    user_message: str,
    sender_id: str,
    conversation_id: str,
    session_id: str,
):
    """处理消息并发送回复"""
    # 调用 AI 服务
    answer = await ai_service.chat(user_message, session_id)
    
    # 发送消息
    await send_dingtalk_message(
        conversation_id=conversation_id,
        content=answer,
    )

async def send_dingtalk_message(conversation_id: str, content: str):
    """发送钉钉消息"""
    token = await get_access_token()
    
    async with httpx.AsyncClient() as client:
        await client.post(
            "https://api.dingtalk.com/v1.0/robot/messages/send",
            headers={"x-acs-dingtalk-access-token": token},
            json={
                "robotCode": DINGTALK_ROBOT_CODE,
                "conversationId": conversation_id,
                "msgKey": "sampleText",
                "msgParam": f'{{"content": "{content}"}}',
            }
        )

3. 企业微信应用接入

# integrations/wecom.py
import hashlib
import xml.etree.ElementTree as ET
from Crypto.Cipher import AES  # pip install pycryptodome
import base64
import struct

CORP_ID = os.getenv("WECOM_CORP_ID")
AGENT_ID = os.getenv("WECOM_AGENT_ID")
CORP_SECRET = os.getenv("WECOM_CORP_SECRET")
TOKEN = os.getenv("WECOM_TOKEN")
ENCODING_AES_KEY = os.getenv("WECOM_ENCODING_AES_KEY")

_wecom_token_cache = {"token": None, "expire": 0}

async def get_wecom_token() -> str:
    if _wecom_token_cache["token"] and time.time() < _wecom_token_cache["expire"]:
        return _wecom_token_cache["token"]
    
    async with httpx.AsyncClient() as client:
        resp = await client.get(
            "https://qyapi.weixin.qq.com/cgi-bin/gettoken",
            params={"corpid": CORP_ID, "corpsecret": CORP_SECRET},
        )
    data = resp.json()
    _wecom_token_cache["token"] = data["access_token"]
    _wecom_token_cache["expire"] = time.time() + data["expires_in"] - 60
    return _wecom_token_cache["token"]

@app.get("/webhook/wecom")
async def wecom_verify(
    msg_signature: str,
    timestamp: str,
    nonce: str,
    echostr: str,
):
    """企业微信接入验证"""
    # 验证签名
    sign_list = sorted([TOKEN, timestamp, nonce, echostr])
    sign_str = "".join(sign_list)
    expected = hashlib.sha1(sign_str.encode()).hexdigest()
    
    if msg_signature != expected:
        raise HTTPException(403, "签名验证失败")
    
    # 解密并返回 echostr
    decrypted = decrypt_wecom_message(echostr)
    return Response(content=decrypted)

@app.post("/webhook/wecom")
async def wecom_receive(
    msg_signature: str,
    timestamp: str,
    nonce: str,
    request: Request,
    background_tasks: BackgroundTasks,
):
    """接收企业微信消息"""
    body = await request.body()
    root = ET.fromstring(body.decode())
    
    # 解密消息
    encrypted_content = root.find("Encrypt").text
    decrypted_xml = decrypt_wecom_message(encrypted_content)
    msg_root = ET.fromstring(decrypted_xml)
    
    msg_type = msg_root.find("MsgType").text
    from_user = msg_root.find("FromUserName").text
    
    if msg_type == "text":
        content = msg_root.find("Content").text
        background_tasks.add_task(
            process_wecom_message,
            user_id=from_user,
            content=content,
        )
    
    return Response(content="success")

async def send_wecom_message(user_id: str, content: str):
    """向指定员工发送企业微信消息"""
    token = await get_wecom_token()
    
    async with httpx.AsyncClient() as client:
        await client.post(
            "https://qyapi.weixin.qq.com/cgi-bin/message/send",
            params={"access_token": token},
            json={
                "touser": user_id,
                "msgtype": "text",
                "agentid": AGENT_ID,
                "text": {"content": content},
                "safe": 0,
            }
        )

def decrypt_wecom_message(encrypted: str) -> str:
    """AES 解密企业微信消息"""
    aes_key = base64.b64decode(ENCODING_AES_KEY + "=")
    ciphertext = base64.b64decode(encrypted)
    cipher = AES.new(aes_key, AES.MODE_CBC, aes_key[:16])
    decrypted = cipher.decrypt(ciphertext)
    
    # 去除 padding
    pad_size = decrypted[-1]
    decrypted = decrypted[:-pad_size]
    
    # 解析 content（前 16 字节随机，后 4 字节长度，再后是内容）
    msg_len = struct.unpack(">I", decrypted[16:20])[0]
    return decrypted[20:20 + msg_len].decode("utf-8")

4. 统一鉴权方案

4.1 双模式鉴权（API Key + JWT）

# auth/middleware.py
from fastapi import Request, HTTPException, Depends
from fastapi.security import HTTPBearer, HTTPAuthorizationCredentials
import jwt
import hashlib

security = HTTPBearer()

# API Key 哈希存储（数据库）
# key_hash → {name, permissions, rate_limit, owner}

async def get_current_user(
    request: Request,
    creds: HTTPAuthorizationCredentials = Depends(security),
) -> dict:
    token = creds.credentials
    
    # 模式一：API Key 鉴权（以 "sk-" 开头）
    if token.startswith("sk-"):
        return await validate_api_key(token)
    
    # 模式二：JWT 鉴权（SSO 场景）
    return validate_jwt(token)

async def validate_api_key(api_key: str) -> dict:
    # 存储 Key 的哈希，不存明文
    key_hash = hashlib.sha256(api_key.encode()).hexdigest()
    
    record = await db.fetch_one(
        "SELECT * FROM api_keys WHERE key_hash = :hash AND is_active = true",
        {"hash": key_hash}
    )
    
    if not record:
        raise HTTPException(401, "无效的 API Key")
    
    return {
        "type": "api_key",
        "name": record["name"],
        "permissions": record["permissions"],
        "owner": record["owner"],
    }

def validate_jwt(token: str) -> dict:
    try:
        payload = jwt.decode(
            token,
            os.getenv("JWT_SECRET"),
            algorithms=["HS256"]
        )
        return {
            "type": "user",
            "user_id": payload["sub"],
            "departments": payload.get("departments", []),
            "name": payload.get("name"),
        }
    except jwt.ExpiredSignatureError:
        raise HTTPException(401, "Token 已过期")
    except jwt.InvalidTokenError:
        raise HTTPException(401, "无效的 Token")

oauth.register(
    name="company_sso",
    client_id=os.getenv("SSO_CLIENT_ID"),
    client_secret=os.getenv("SSO_CLIENT_SECRET"),
    server_metadata_url=os.getenv("SSO_METADATA_URL"),  # .well-known/openid-configuration

)

@app.get("/auth/login")
async def login(request: Request):
    redirect_uri = "https://ai-tools.example.com/auth/callback"
    return await oauth.company_sso.authorize_redirect(request, redirect_uri)

@app.get("/auth/callback")
async def callback(request: Request):
    token = await oauth.company_sso.authorize_access_token(request)
    user_info = token.get("userinfo")
    
    # 用公司 SSO 的用户信息生成内部 JWT
    internal_jwt = jwt.encode(
        {
            "sub": user_info["sub"],
            "name": user_info.get("name"),
            "email": user_info.get("email"),
            "departments": user_info.get("departments", []),
            "exp": int(time.time()) + 8 * 3600,  # 8 小时有效期
        },
        os.getenv("JWT_SECRET"),
        algorithm="HS256",
    )
    
    # 重定向到前端，携带 token
    return RedirectResponse(f"/app?token={internal_jwt}")

6. 故障排查速查

现象排查方向
SSE 不流式，一次返回检查 nginx proxy_buffering off 配置
钉钉机器人不回复检查 WebHook 签名验证日志、IP 白名单
JWT 失效超快检查服务器时钟是否同步（NTP）
API Key 泄露立即在 DB 设 is_active=false，颁发新 Key
向量检索很慢Qdrant 集合是否创建了 HNSW 索引
LLM 响应很慢检查是否走了代理，测试直连延迟
