from sqlalchemy.orm import DeclarativeBase
from app.config import settings

engine = create_async_engine(
    settings.DATABASE_URL,
    echo=False,
    pool_size=20,
    max_overflow=10,
    pool_pre_ping=True,
)

async_session = async_sessionmaker(engine, class_=AsyncSession, expire_on_commit=False)


class Base(DeclarativeBase):
    pass


async def get_db() -> AsyncSession:
    async with async_session() as session:
        try:
            yield session
            await session.commit()
        except Exception:
            await session.rollback()
            raise
        finally:
            await session.close()

from sqlalchemy import Column, Integer, String, Text, Numeric, DateTime, Boolean, JSON, Enum as SAEnum
from sqlalchemy.orm import relationship

from app.database import Base
import enum


class ProductStatus(str, enum.Enum):
    DRAFT = "draft"
    ACTIVE = "active"
    INACTIVE = "inactive"
    DISCONTINUED = "discontinued"


class Product(Base):
    """商品/SPU模型"""
    __tablename__ = "products"

    id = Column(Integer, primary_key=True, autoincrement=True)
    spu_code = Column(String(64), unique=True, nullable=False, index=True, comment="SPU编码")
    name = Column(String(256), nullable=False, comment="商品名称")
    brand = Column(String(128), comment="品牌")
    category_id = Column(Integer, nullable=False, index=True, comment="品类ID")
    description = Column(Text, comment="商品描述")
    key_attrs = Column(JSON, default=dict, comment="关键属性")
    desc_attrs = Column(JSON, default=dict, comment="描述属性")
    status = Column(SAEnum(ProductStatus), default=ProductStatus.DRAFT, comment="状态")
    created_at = Column(DateTime(timezone=True), server_default=func.now(), comment="创建时间")
    updated_at = Column(DateTime(timezone=True), server_default=func.now(), onupdate=func.now(), comment="更新时间")

    # 关联
    skus = relationship("SKU", back_populates="product", lazy="selectin")

    def __repr__(self):
        return f"<Product(spu_code={self.spu_code}, name={self.name})>"


class SKU(Base):
    """SKU模型 - 最小库存单元"""
    __tablename__ = "skus"

    id = Column(Integer, primary_key=True, autoincrement=True)

    barcode = Column(String(64), comment="条形码")
    status = Column(SAEnum(ProductStatus), default=ProductStatus.ACTIVE, comment="状态")
    created_at = Column(DateTime(timezone=True), server_default=func.now())
    updated_at = Column(DateTime(timezone=True), server_default=func.now(), onupdate=func.now())

    def __repr__(self):
        return f"<SKU(sku_code={self.sku_code})>"

class Inventory(Base):
    """库存模型"""
    __tablename__ = "inventories"

    id = Column(Integer, primary_key=True, autoincrement=True)
    sku_id = Column(Integer, nullable=False, index=True, comment="SKU ID")
    warehouse_id = Column(Integer, nullable=False, index=True, comment="仓库ID")
    available_qty = Column(Integer, default=0, comment="可用库存")
    frozen_qty = Column(Integer, default=0, comment="冻结库存")
    in_transit_qty = Column(Integer, default=0, comment="在途库存")
    pre_sale_qty = Column(Integer, default=0, comment="预售占用量")
    avg_cost = Column(Numeric(12, 2), default=0, comment="加权平均成本")
    safety_stock = Column(Integer, default=0, comment="安全库存")
    reorder_point = Column(Integer, default=0, comment="再订货点")
    last_replenish_date = Column(DateTime(timezone=True), comment="最近补货日期")
    created_at = Column(DateTime(timezone=True), server_default=func.now())
    updated_at = Column(DateTime(timezone=True), server_default=func.now(), onupdate=func.now())

    # 关联
    sku = relationship("SKU", back_populates="inventories")
    warehouse = relationship("Warehouse", back_populates="inventories")
    logs = relationship("InventoryLog", back_populates="inventory", lazy="selectin")

    # 唯一约束
    __table_args__ = (
        {"comment": "库存表，每个SKU在每个仓库一条记录"},
    )

    @property
    def total_qty(self) -> int:
        return self.available_qty + self.frozen_qty + self.in_transit_qty

    @property
    def net_available(self) -> int:
        return self.available_qty - self.pre_sale_qty

    def __repr__(self):
        return f"<Inventory(sku_id={self.sku_id}, wh_id={self.warehouse_id}, qty={self.available_qty})>"

缓存操作

import json
import redis.asyncio as redis
from app.config import settings

redis_client = redis.from_url(settings.REDIS_URL, decode_responses=True)


async def get_inventory_cache(sku_id: int, warehouse_id: int) -> dict | None:
    """获取库存缓存"""
    key = f"inventory:{sku_id}:{warehouse_id}"
    data = await redis_client.hgetall(key)
    return data if data else None


async def set_inventory_cache(sku_id: int, warehouse_id: int, data: dict) -> None:
    """设置库存缓存"""
    key = f"inventory:{sku_id}:{warehouse_id}"
    await redis_client.hset(key, mapping=data)
    await redis_client.expire(key, 300)  # 5分钟过期


async def get_price_cache(sku_id: int) -> float | None:
    """获取价格缓存"""
    key = f"price:{sku_id}"
    data = await redis_client.get(key)
    return float(data) if data else None


async def set_price_cache(sku_id: int, price: float) -> None:
    """设置价格缓存"""
    key = f"price:{sku_id}"
    await redis_client.set(key, str(price), ex=600)  # 10分钟过期


async def invalidate_inventory_cache(sku_id: int, warehouse_id: int) -> None:
    """使库存缓存失效"""
    key = f"inventory:{sku_id}:{warehouse_id}"
    await redis_client.delete(key)

from asyncio import run
from sqlalchemy.ext.asyncio import create_async_engine
from app.database import Base
from app.config import settings
from app.models import *  # noqa: 确保所有模型被导入

target_metadata = Base.metadata


def run_migrations_online():
    connectable = create_async_engine(settings.DATABASE_URL)
    async def do_run():
        async with connectable.connect() as connection:
            await connection.run_sync(target_metadata.create_all)
    run(do_run())

async def seed_sales_history(session: AsyncSession, days: int = 365):
    """生成模拟销售历史数据"""
    np.random.seed(42)
    sku_ids = [1, 2, 3, 4, 5]
    base_demands = [50, 30, 20, 25, 200]

    for sku_id, base_demand in zip(sku_ids, base_demands):
        for day_offset in range(days):
            date = datetime.now() - timedelta(days=days - day_offset)
            # 加入趋势、季节性和噪声
            trend = day_offset * 0.02
            seasonality = 20 * np.sin(2 * np.pi * day_offset / 365)
            promo_boost = 50 if (date.month == 6 and date.day <= 18) or (date.month == 11 and date.day <= 11) else 0
            noise = np.random.normal(0, base_demand * 0.15)
            qty = max(int(base_demand + trend + seasonality + promo_boost + noise), 1)

            order = SalesOrder(
                order_no=f"SO-{sku_id:03d}-{day_offset:05d}",
                status=SalesOrderStatus.DELIVERED,
                total_amount=qty * 100,
                payment_amount=qty * 100,
                warehouse_id=1,
                created_at=date,
            )
            session.add(order)
            await session.flush()

            item = SalesItem(
                order_id=order.id,
                sku_id=sku_id,
                quantity=qty,
                unit_price=100,
                amount=qty * 100,
            )
            session.add(item)


async def main():
    """主初始化函数"""
    async with async_session() as session:
        await seed_products(session)
        await seed_warehouses(session)
        await seed_suppliers(session)
        await seed_sales_history(session, days=180)  # 生成180天历史数据
        await session.commit()
        print("数据初始化完成")


if __name__ == "__main__":
    asyncio.run(main())
