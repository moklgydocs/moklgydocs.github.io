药品库存管理

from pydantic import BaseModel, Field
from datetime import date, timedelta
from typing import Optional
from enum import Enum

class DrugStorageStatus(str, Enum):
    normal = "正常"
    low_stock = "低库存"
    out_of_stock = "缺货"
    near_expiry = "近效期"
    expired = "过期"

class DrugInventory(BaseModel):
    """药品库存记录"""
    drug_code: str
    drug_name: str
    batch_no: str = Field(..., description="批号")
    production_date: date
    expiry_date: date
    quantity: int = Field(..., description="库存数量")
    unit: str = Field(..., description="最小包装单位")
    location: str = Field(..., description="货位号")
    purchase_price: float = Field(..., description="采购价")
    retail_price: float = Field(..., description="零售价")
    supplier: str = Field(..., description="供应商")

    @property
    def days_to_expiry(self) -> int:
        return (self.expiry_date - date.today()).days

    @property
    def storage_status(self) -> DrugStorageStatus:
        if self.days_to_expiry <= 0:
            return DrugStorageStatus.expired
        elif self.days_to_expiry <= 90:
            return DrugStorageStatus.near_expiry
        elif self.quantity == 0:
            return DrugStorageStatus.out_of_stock
        elif self.quantity <= 10:  # 简化的低库存阈值
            return DrugStorageStatus.low_stock
        return DrugStorageStatus.normal

# 库存预警规则

from pydantic import BaseModel, Field
from typing import Optional
from datetime import datetime
from enum import Enum

class PrescriptionType(str, Enum):
    ordinary = "普通处方"
    emergency = "急诊处方"
    pediatric = "儿科处方"
    anesthetic = "麻醉药品处方"
    psychotropic = "精神药品处方"

class DosageForm(str, Enum):
    tablet = "片剂"
    capsule = "胶囊"
    injection = "注射液"
    syrup = "糖浆"
    ointment = "软膏"
    drop = "滴剂"
    aerosol = "气雾剂"

class RouteOfAdministration(str, Enum):
    PO = "口服"       # Per Os
    IV = "静脉注射"   # Intravenous
    IM = "肌肉注射"   # Intramuscular
    SC = "皮下注射"   # Subcutaneous
    TOP = "外用"      # Topical
    INH = "吸入"      # Inhalation
    PR = "直肠给药"   # Per Rectum
    SL = "舌下含服"   # Sublingual

class Frequency(str, Enum):
    QD = "每天一次"
    BID = "每天两次"
    TID = "每天三次"
    QID = "每天四次"
    Q8H = "每8小时一次"
    Q12H = "每12小时一次"
    QN = "每晚一次"
    PRN = "必要时"
    STAT = "立即"
    QOD = "隔天一次"

class PrescriptionItem(BaseModel):
    """处方项目"""
    drug_name: str = Field(..., description="药品通用名")
    specification: str = Field(..., description="规格")
    quantity: str = Field(..., description="数量，如：24粒")
    dosage_form: DosageForm
    route: RouteOfAdministration
    frequency: Frequency
    dosage_per_time: str = Field(..., description="单次剂量，如：0.5g")
    duration_days: int = Field(..., description="用药天数")
    sig: str = Field(..., description="用法简写")
    remark: Optional[str] = Field(None, description="备注")

class Prescription(BaseModel):
    """处方"""
    prescription_id: str
    prescription_type: PrescriptionType
    patient_id: str
    patient_name: str
    patient_age: int
    patient_gender: str
    patient_weight: Optional[float] = Field(None, description="体重(kg)，儿童必填")
    doctor_id: str
    doctor_name: str
    department: str
    diagnosis: str = Field(..., description="临床诊断")
    icd10_code: Optional[str] = Field(None, description="ICD-10编码")
    items: list[PrescriptionItem]
    is_reviewed: bool = False
    reviewer: Optional[str] = None
    created_at: datetime = Field(default_factory=datetime.now)

    @property
    def item_count(self) -> int:
        """处方药品数(一般不超过5种)"""
        return len(self.items)

    @property
    def is_valid_item_count(self) -> bool:
        """处方药品数是否合规"""
        return self.item_count <= 5

配伍禁忌

配伍禁忌是指两种或以上药物合用时产生的不良相互作用。

from pydantic import BaseModel, Field
from enum import Enum

class InteractionSeverity(str, Enum):
    contraindicated = "禁忌"          # 严禁合用
    major = "严重"                     # 应避免合用
    moderate = "中等"                   # 需谨慎
    minor = "轻微"                     # 临床意义不大

class DrugInteraction(BaseModel):
    """药物相互作用"""
    drug_a: str
    drug_b: str
    severity: InteractionSeverity
    mechanism: str
    clinical_effect: str
    management: str

# 常见药物配伍禁忌

        clinical_effect="地高辛血药浓度升高，洋地黄中毒风险",
        management="合用时地高辛减量50%，监测血药浓度",
    ),
]

def check_drug_interaction(drug_list: list[str]) -> list[DrugInteraction]:
    """检查药品列表中的配伍禁忌"""
    found_interactions = []
    for interaction in DRUG_INTERACTIONS:
        if interaction.drug_a in drug_list and interaction.drug_b in drug_list:
            found_interactions.append(interaction)
    return found_interactions

用药审查规则

from pydantic import BaseModel, Field
from typing import Optional
from enum import Enum

class AlertLevel(str, Enum):
    info = "提示"
    warning = "警告"
    error = "禁止"

class MedicationAlert(BaseModel):
    """用药审查提示"""
    alert_type: str
    alert_level: AlertLevel
    drug_name: str
    message: str
    suggestion: Optional[str] = None

def review_prescription(
    prescription: Prescription,
    patient_allergies: list[str],
    patient_conditions: list[str],
) -> list[MedicationAlert]:
    """处方审查：过敏+剂量+配伍+适应症"""
    alerts = []

    # 1. 过敏检查
    ALLERGY_MAPPING = {
        "青霉素": ["阿莫西林", "氨苄西林", "哌拉西林", "青霉素"],
        "磺胺": ["磺胺甲噁唑", "甲氧苄啶"],
        "碘": ["碘海醇", "碘佛醇"],
    }
    for allergen in patient_allergies:
        if allergen in ALLERGY_MAPPING:
            for item in prescription.items:
                if item.drug_name in ALLERGY_MAPPING[allergen]:
                    alerts.append(MedicationAlert(
                        alert_type="过敏禁忌",
                        alert_level=AlertLevel.error,
                        drug_name=item.drug_name,
                        message=f"患者对{allergen}过敏，禁用{item.drug_name}",
                        suggestion="更换为替代药物",
                    ))

    # 2. 配伍禁忌检查
    drug_names = [item.drug_name for item in prescription.items]
    interactions = check_drug_interaction(drug_names)
    for interaction in interactions:
        level = AlertLevel.error if interaction.severity == InteractionSeverity.contraindicated else AlertLevel.warning
        alerts.append(MedicationAlert(
            alert_type="配伍禁忌",
            alert_level=level,
            drug_name=f"{interaction.drug_a}+{interaction.drug_b}",
            message=interaction.clinical_effect,
            suggestion=interaction.management,
        ))

    # 3. 儿童用药禁忌
    if prescription.patient_age < 18:
        PEDIATRIC_CONTRAINDICATIONS = {
            "喹诺酮类": ["左氧氟沙星", "环丙沙星", "莫西沙星"],
            "四环素类": ["多西环素", "米诺环素"],
            "氨基糖苷类": ["庆大霉素", "阿米卡星"],
        }
        for category, drugs in PEDIATRIC_CONTRAINDICATIONS.items():
            for item in prescription.items:
                if item.drug_name in drugs:
                    alerts.append(MedicationAlert(
                        alert_type="儿童用药禁忌",
                        alert_level=AlertLevel.error,
                        drug_name=item.drug_name,
                        message=f"{item.drug_name}({category})禁用于18岁以下患者",
                        suggestion="更换为儿童适用药物",
                    ))

    # 4. 处方药品数审查
    if not prescription.is_valid_item_count:
        alerts.append(MedicationAlert(
            alert_type="处方规范性",
            alert_level=AlertLevel.warning,
            drug_name="",
            message=f"处方含{prescription.item_count}种药品，超过5种上限",
            suggestion="减少用药种类，必要时分次开具",
        ))

    # 5. 孕妇用药禁忌
    if "妊娠" in patient_conditions:
        PREGNANCY_CONTRAINDICATIONS = [
            "异维A酸", "华法林", "甲氨蝶呤", "利巴韦林",
            "ACEI", "ARB", "喹诺酮类",
        ]
        for item in prescription.items:
            for contrain in PREGNANCY_CONTRAINDICATIONS:
                if contrain in item.drug_name:
                    alerts.append(MedicationAlert(
                        alert_type="孕妇用药禁忌",
                        alert_level=AlertLevel.error,
                        drug_name=item.drug_name,
                        message=f"{item.drug_name}禁用于妊娠期患者",
                        suggestion="更换为妊娠期安全药物",
                    ))

    return alerts
