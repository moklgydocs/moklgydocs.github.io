// src/pages/PatientDashboard/index.tsx
import React, { useState, useEffect } from 'react';
import {
  Card, Table, Button, Input, Modal, Form, Space,
  Tag, message, Popconfirm, Row, Col, Statistic,
} from 'antd';
import { UserAddOutlined, SearchOutlined } from '@ant-design/icons';
import type { ColumnsType } from 'antd/es/table';

interface Patient {
  id: string;
  name: string;
  gender: string;
  birth_date: string;
  id_card: string;
  phone: string;
  allergies: string;
  insurance_type: string;
  created_at: string;
}

const PatientDashboard: React.FC = () => {
  const [patients, setPatients] = useState<Patient[]>([]);
  const [loading, setLoading] = useState(false);
  const [total, setTotal] = useState(0);
  const [page, setPage] = useState(1);
  const [search, setSearch] = useState('');
  const [modalVisible, setModalVisible] = useState(false);
  const [form] = Form.useForm();

  const fetchPatients = async (pageNum: number = 1, keyword: string = '') => {
    setLoading(true);
    try {
      const params = new URLSearchParams({
        page: String(pageNum),
        page_size: '20',
        ...(keyword ? { search: keyword } : {}),
      });
      const response = await fetch(`/api/v1/patients?${params}`);
      const data = await response.json();
      setPatients(data.items);
      setTotal(data.total);
    } catch (error) {
      message.error('获取患者列表失败');
    } finally {
      setLoading(false);
    }
  };

  useEffect(() => {
    fetchPatients();
  }, []);

  const handleCreatePatient = async (values: any) => {
    try {
      const response = await fetch('/api/v1/patients', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(values),
      });
      if (response.ok) {
        message.success('患者注册成功');
        setModalVisible(false);
        form.resetFields();
        fetchPatients(page, search);
      } else {
        const err = await response.json();
        message.error(err.detail || '注册失败');
      }
    } catch (error) {
      message.error('网络错误');
    }
  };

  const columns: ColumnsType<Patient> = [
    { title: '姓名', dataIndex: 'name', width: 100 },
    { title: '性别', dataIndex: 'gender', width: 60 },
    { title: '出生日期', dataIndex: 'birth_date', width: 110 },
    { title: '身份证号', dataIndex: 'id_card', width: 180 },
    { title: '手机号', dataIndex: 'phone', width: 130 },
    {
      title: '过敏史', dataIndex: 'allergies', width: 150,
      render: (text: string) => text
        ? text.split(',').map((a: string) => <Tag key={a} color="red">{a}</Tag>)
        : <Tag color="green">无</Tag>,
    },
    { title: '医保类型', dataIndex: 'insurance_type', width: 120 },
    {
      title: '操作', width: 150, fixed: 'right',
      render: (_, record) => (
        <Space>
          <Button type="link" size="small" onClick={() => window.open(`/patients/${record.id}/records`, '_self')}>
            病历
          </Button>
          <Button type="link" size="small" onClick={() => window.open(`/diagnosis?patient_id=${record.id}`, '_self')}>
            AI诊断
          </Button>
        </Space>
      ),
    },
  ];

  return (
    <div>
      <Card title="患者管理" extra={
        <Space>
          <Input
            placeholder="搜索姓名/身份证号"
            prefix={<SearchOutlined />}
            value={search}
            onChange={e => setSearch(e.target.value)}
            onPressEnter={() => fetchPatients(1, search)}
            style={{ width: 250 }}
          />
          <Button type="primary" icon={<UserAddOutlined />} onClick={() => setModalVisible(true)}>
            新增患者
          </Button>
        </Space>
      }>
        <Table
          columns={columns}
          dataSource={patients}
          rowKey="id"
          loading={loading}
          pagination={{
            current: page,
            total,
            pageSize: 20,
            onChange: (p) => { setPage(p); fetchPatients(p, search); },
          }}
          scroll={{ x: 1200 }}
        />
      </Card>

      <Modal
        title="新增患者"
        open={modalVisible}
        onCancel={() => setModalVisible(false)}
        onOk={() => form.submit()}
        width={600}
      >
        <Form form={form} layout="vertical" onFinish={handleCreatePatient}>
          <Row gutter={16}>
            <Col span={12}>
              <Form.Item name="name" label="姓名" rules={[{ required: true }]}>
                <Input />
              </Form.Item>
            </Col>
            <Col span={12}>
              <Form.Item name="gender" label="性别" rules={[{ required: true }]}>
                <Input />
              </Form.Item>
            </Col>
          </Row>
          <Row gutter={16}>
            <Col span={12}>
              <Form.Item name="birth_date" label="出生日期" rules={[{ required: true }]}>
                <Input type="date" />
              </Form.Item>
            </Col>
            <Col span={12}>
              <Form.Item name="id_card" label="身份证号" rules={[{ required: true, len: 18 }]}>

            <Col span={12}>

// src/pages/MedicalRecordEditor/index.tsx
import React, { useState, useEffect } from 'react';
import {
  Card, Form, Input, Select, Button, Space, Divider,
  Tabs, Table, Modal, message, Tag, Row, Col,
} from 'antd';
import { SaveOutlined, PlusOutlined } from '@ant-design/icons';

const { TextArea } = Input;

interface DiagnosisItem {
  id: string;
  disease_name: string;
  icd10_code: string;
  is_primary: boolean;
  diagnosis_type: string;
}

const MedicalRecordEditor: React.FC<{ patientId: string }> = ({ patientId }) => {
  const [form] = Form.useForm();
  const [diagnoses, setDiagnoses] = useState<DiagnosisItem[]>([]);
  const [diagModalVisible, setDiagModalVisible] = useState(false);
  const [diagForm] = Form.useForm();

  const handleSave = async () => {
    try {
      const values = await form.validateFields();
      const response = await fetch(`/api/v1/patients/${patientId}/records`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(values),
      });
      if (response.ok) {
        message.success('病历保存成功');
      }
    } catch (error) {
      message.error('保存失败');
    }
  };

  const handleAddDiagnosis = async () => {
    try {
      const values = await diagForm.validateFields();
      // 如果是主诊断，取消其他主诊断
      if (values.is_primary) {
        setDiagnoses(prev => prev.map(d => ({ ...d, is_primary: false })));
      }
      setDiagnoses(prev => [...prev, {
        id: `diag-${Date.now()}`,
        ...values,
      }]);
      setDiagModalVisible(false);
      diagForm.resetFields();
    } catch (error) {
      // validation error
    }
  };

  const diagnosisColumns = [
    { title: '诊断', dataIndex: 'disease_name', key: 'disease' },
    { title: 'ICD-10', dataIndex: 'icd10_code', key: 'icd10', width: 100 },
    {
      title: '主诊断', dataIndex: 'is_primary', key: 'primary', width: 80,
      render: (v: boolean) => v ? <Tag color="red">主</Tag> : null,
    },
    { title: '类型', dataIndex: 'diagnosis_type', key: 'type', width: 100 },
    {
      title: '操作', key: 'action', width: 80,
      render: (_: any, record: DiagnosisItem) => (
        <Button type="link" danger size="small" onClick={() => {
          setDiagnoses(prev => prev.filter(d => d.id !== record.id));
        }}>删除</Button>
      ),
    },
  ];

  return (
    <div>
      <Card title="病历编辑" extra={
        <Space>
          <Button icon={<SaveOutlined />} type="primary" onClick={handleSave}>保存</Button>
        </Space>
      }>
        <Form form={form} layout="vertical">
          <Form.Item name="encounter_type" label="就诊类型" rules={[{ required: true }]}>
            <Select style={{ width: 200 }}>
              <Select.Option value="门诊">门诊</Select.Option>
              <Select.Option value="急诊">急诊</Select.Option>
              <Select.Option value="住院">住院</Select.Option>
            </Select>
          </Form.Item>

          <Form.Item name="chief_complaint" label="主诉" rules={[{ required: true }]}>
            <Input placeholder="主要症状+持续时间，如：反复上腹痛3天" />
          </Form.Item>

          <Form.Item name="present_illness" label="现病史">
            <TextArea rows={4} placeholder="详细描述发病经过、伴随症状、诊治经过" />
          </Form.Item>

          <Row gutter={16}>
            <Col span={12}>
              <Form.Item name="past_history" label="既往史">
                <TextArea rows={3} />
              </Form.Item>
            </Col>
            <Col span={12}>
              <Form.Item name="allergy_history" label="过敏史">
                <TextArea rows={3} />
              </Form.Item>
            </Col>
          </Row>

          <Form.Item name="physical_exam" label="体格检查">
            <TextArea rows={3} placeholder="T: P: R: BP:\n专科检查..." />
          </Form.Item>
        </Form>

        <Divider>诊断</Divider>

        <Table
          columns={diagnosisColumns}
          dataSource={diagnoses}
          rowKey="id"
          pagination={false}
          size="small"
          footer={() => (
            <Button type="dashed" block icon={<PlusOutlined />} onClick={() => setDiagModalVisible(true)}>
              添加诊断
            </Button>
          )}
        />
      </Card>

      <Modal
        title="添加诊断"
        open={diagModalVisible}
        onCancel={() => setDiagModalVisible(false)}
        onOk={handleAddDiagnosis}
      >
        <Form form={diagForm} layout="vertical">
          <Form.Item name="disease_name" label="疾病名称" rules={[{ required: true }]}>
            <Input placeholder="输入疾病名称" />
          </Form.Item>
          <Form.Item name="icd10_code" label="ICD-10编码">
            <Input placeholder="自动填充或手动输入" />
          </Form.Item>
          <Form.Item name="diagnosis_type" label="诊断类型" rules={[{ required: true }]}>
            <Select>
              <Select.Option value="初步诊断">初步诊断</Select.Option>

              <Select.Option value="出院诊断">出院诊断</Select.Option>

          </Form.Item>

AIDiagnosisPanel - AI诊断面板

// src/pages/AIDiagnosisPanel/index.tsx
import React, { useState } from 'react';
import {
  Card, Input, Button, Tag, List, Progress, Alert,
  Descriptions, Spin, Steps, Select, Space, Divider, Tooltip,
} from 'antd';
import { RobotOutlined, WarningOutlined, CheckCircleOutlined } from '@ant-design/icons';

interface DifferentialDiagnosis {
  disease_name: string;
  icd10_code: string;
  probability: number;
  confidence_level: string;
  supporting_evidence: string[];
  against_evidence: string[];
  key_differentiator: string;
}

interface ExamSuggestion {
  exam_name: string;
  urgency: string;
  reason: string;
  target_disease: string;
}

const AIDiagnosisPanel: React.FC = () => {
  const [symptoms, setSymptoms] = useState<string[]>([]);
  const [symptomInput, setSymptomInput] = useState('');
  const [loading, setLoading] = useState(false);
  const [diagnoses, setDiagnoses] = useState<DifferentialDiagnosis[]>([]);
  const [exams, setExams] = useState<ExamSuggestion[]>([]);
  const [reasoning, setReasoning] = useState('');

  const addSymptom = () => {
    if (symptomInput.trim() && !symptoms.includes(symptomInput.trim())) {
      setSymptoms(prev => [...prev, symptomInput.trim()]);
      setSymptomInput('');
    }
  };

  const removeSymptom = (symptom: string) => {
    setSymptoms(prev => prev.filter(s => s !== symptom));
  };

  const analyzeSymptoms = async () => {
    if (symptoms.length === 0) return;
    setLoading(true);
    try {
      const response = await fetch('/api/v1/diagnosis/symptoms?patient_id=temp', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          symptoms,
          patient_age: 30,
          patient_gender: '未知',
        }),
      });
      const data = await response.json();
      setDiagnoses(data.differential_diagnoses || []);
      setExams(data.exam_suggestions || []);
      setReasoning(data.reasoning_summary || '');
    } catch (error) {
      // handle error
    } finally {
      setLoading(false);
    }
  };

  const getConfidenceColor = (level: string) => {
    switch (level) {
      case '高': return 'green';
      case '中': return 'orange';
      case '低': return 'red';
      default: return 'default';
    }
  };

  return (
    <div>
      <Card title={
        <Space>
          <RobotOutlined />
          <span>AI辅助诊断</span>
        </Space>
      }>
        <Alert
          message="AI辅助声明"
          description="本系统提供AI辅助诊断分析，所有结果仅供参考，不构成诊断依据。最终诊断须由执业医师确认。"
          type="warning"
          showIcon
          icon={<WarningOutlined />}
          style={{ marginBottom: 16 }}
        />

        <div style={{ marginBottom: 16 }}>
          <Space style={{ width: '100%' }}>
            <Input
              placeholder="输入症状，如：发热、咳嗽、胸痛"
              value={symptomInput}
              onChange={e => setSymptomInput(e.target.value)}
              onPressEnter={addSymptom}
              style={{ width: 300 }}
            />
            <Button type="primary" onClick={addSymptom}>添加</Button>
            <Button type="primary" onClick={analyzeSymptoms} loading={loading}
              disabled={symptoms.length === 0}>
              开始分析
            </Button>
          </Space>

          <div style={{ marginTop: 8 }}>
            {symptoms.map(s => (
              <Tag key={s} closable onClose={() => removeSymptom(s)} color="blue" style={{ marginBottom: 4 }}>
                {s}
              </Tag>
            ))}
          </div>
        </div>

        {loading && (
          <div style={{ textAlign: 'center', padding: 40 }}>
            <Spin size="large" tip="AI分析中..." />
          </div>
        )}

        {!loading && diagnoses.length > 0 && (
          <>
            <Divider>鉴别诊断</Divider>

            <List
              dataSource={diagnoses}
              renderItem={(item, index) => (
                <List.Item>
                  <Card size="small" style={{ width: '100%' }}
                    title={
                      <Space>
                        <span style={{ fontWeight: index === 0 ? 'bold' : 'normal', fontSize: index === 0 ? 16 : 14 }}>
                          {index + 1}. {item.disease_name}
                        </span>
                        <Tag color={getConfidenceColor(item.confidence_level)}>
                          置信度: {item.confidence_level}
                        </Tag>
                        {item.icd10_code && <Tag>{item.icd10_code}</Tag>}
                      </Space>
                    }
                  >
                    <Progress
                      percent={Math.round(item.probability * 100)}
                      strokeColor={item.probability >= 0.7 ? '#52c41a' : item.probability >= 0.4 ? '#faad14' : '#ff4d4f'}
                      format={percent => `${percent}%`}
                    />
                    <Descriptions column={1} size="small">
                      <Descriptions.Item label="支持依据">
                        {item.supporting_evidence.map(e => (
                          <Tag key={e} color="green">{e}</Tag>
                        ))}
                      </Descriptions.Item>
                      {item.against_evidence.length > 0 && (
                        <Descriptions.Item label="不支持">
                          {item.against_evidence.map(e => (
                            <Tag key={e} color="red">{e}</Tag>
                          ))}
                        </Descriptions.Item>
                      )}
                      <Descriptions.Item label="鉴别要点">
                        {item.key_differentiator}
                      </Descriptions.Item>
                    </Descriptions>
                  </Card>
                </List.Item>
              )}
            />

            {exams.length > 0 && (
              <>
                <Divider>推荐检查</Divider>
                <List
                  size="small"
                  dataSource={exams}
                  renderItem={item => (
                    <List.Item>
                      <Space>
                        <Tag color={item.urgency === '紧急' ? 'red' : 'blue'}>{item.urgency}</Tag>
                        <span>{item.exam_name}</span>
                        <span style={{ color: '#888' }}>- {item.reason}</span>
                      </Space>
                    </List.Item>
                  )}
                />
              </>
            )}

            {reasoning && (
              <>
                <Divider>推理过程</Divider>
                <Card size="small" style={{ background: '#f5f5f5' }}>
                  <pre style={{ whiteSpace: 'pre-wrap', margin: 0 }}>{reasoning}</pre>
                </Card>
              </>
            )}

            <Divider />
            <Alert
              message="重要提示：以上AI辅助分析结果仅供临床参考，不构成诊断依据。请结合患者实际情况，由执业医师做出最终诊断。"
              type="info"
              showIcon
              icon={<CheckCircleOutlined />}
            />
          </>
        )}
      </Card>
    </div>
  );
};

export default AIDiagnosisPanel;

// src/pages/PrescriptionEditor/index.tsx
import React, { useState } from 'react';
import {
  Card, Form, Input, Select, Button, Table, Space,
  Tag, Alert, message, Modal, Row, Col, Divider,
} from 'antd';
import { PlusOutlined, CheckCircleOutlined, WarningOutlined } from '@ant-design/icons';

interface PrescriptionItem {
  key: string;
  drug_name: string;
  specification: string;
  dosage: string;
  frequency: string;
  route: string;
  duration_days: number;
  sig: string;
}

interface MedicationAlert {
  alert_type: string;
  severity: string;
  drug_name: string;
  message: string;
  suggestion: string;
}

const PrescriptionEditor: React.FC<{ patientId: string }> = ({ patientId }) => {
  const [items, setItems] = useState<PrescriptionItem[]>([]);
  const [alerts, setAlerts] = useState<MedicationAlert[]>([]);
  const [checking, setChecking] = useState(false);
  const [itemForm] = Form.useForm();

  const addItem = () => {
    const values = itemForm.getFieldsValue();
    const sig = `${values.dosage} ${values.route} ${values.frequency}`;
    setItems(prev => [...prev, {
      key: `item-${Date.now()}`,
      ...values,
      sig,
    }]);
    itemForm.resetFields();
  };

  const removeItem = (key: string) => {
    setItems(prev => prev.filter(i => i.key !== key));
  };

  const checkPrescription = async () => {
    if (items.length === 0) {
      message.warning('请先添加药品');
      return;
    }
    setChecking(true);
    try {
      const response = await fetch('/api/v1/prescriptions/check', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          patient_id: patientId,
          drugs: items.map(i => ({
            drug_name: i.drug_name,
            dosage: i.dosage,
            frequency: i.frequency,
          })),
          patient_allergies: [],
          patient_conditions: [],
          patient_age: 30,
        }),
      });
      const data = await response.json();
      setAlerts(data.alerts || []);
      if (data.is_safe) {
        message.success('审查通过，无安全隐患');
      } else {
        message.warning('发现用药安全隐患，请查看详情');
      }
    } catch (error) {
      message.error('审查失败');
    } finally {
      setChecking(false);
    }
  };

  const submitPrescription = async () => {
    // 提交处方逻辑
    message.success('处方已提交');
  };

  const columns = [
    { title: '药品', dataIndex: 'drug_name', key: 'drug' },
    { title: '规格', dataIndex: 'specification', key: 'spec', width: 100 },
    { title: '剂量', dataIndex: 'dosage', key: 'dose', width: 100 },
    { title: '频次', dataIndex: 'frequency', key: 'freq', width: 80 },
    { title: '途径', dataIndex: 'route', key: 'route', width: 80 },
    { title: '天数', dataIndex: 'duration_days', key: 'days', width: 60 },
    { title: '用法', dataIndex: 'sig', key: 'sig' },
    {
      title: '操作', key: 'action', width: 80,
      render: (_: any, record: PrescriptionItem) => (
        <Button type="link" danger size="small" onClick={() => removeItem(record.key)}>
          删除
        </Button>
      ),
