Pydantic Schema定义

# app/schemas/prescription.py
from pydantic import BaseModel, Field, field_validator
from typing import Optional
from datetime import datetime
from enum import Enum

class PrescriptionType(str, Enum):
    ORDINARY = "普通处方"
    EMERGENCY = "急诊处方"
    PEDIATRIC = "儿科处方"
    ANESTHETIC = "麻醉药品处方"

class PrescriptionItemCreate(BaseModel):
    """处方项目创建"""
    drug_name: str = Field(..., min_length=2, max_length=100, description="药品通用名")
    brand_name: Optional[str] = Field(None, description="商品名")
    specification: str = Field(..., description="规格")
    dosage_form: Optional[str] = Field(None, description="剂型")
    quantity: str = Field(..., description="数量")
    dosage: str = Field(..., description="单次剂量")
    frequency: str = Field(..., description="频次: qd/bid/tid/qid/q8h/q12h/prn")
    route: str = Field("口服", description="给药途径")
    duration_days: Optional[int] = Field(None, ge=1, le=90, description="用药天数")
    sig: Optional[str] = Field(None, description="用法简写")
    remark: Optional[str] = Field(None, description="备注")

    @field_validator("frequency")
    @classmethod
    def validate_frequency(cls, v):
        valid = ["qd", "bid", "tid", "qid", "q8h", "q12h", "qn", "prn", "stat", "qod"]
        if v.lower() not in valid:
            raise ValueError(f"不支持的频次: {v}，有效值: {', '.join(valid)}")
        return v.lower()

class PrescriptionCreate(BaseModel):
    """开具处方请求"""
    patient_id: str = Field(..., description="患者ID")
    record_id: Optional[str] = Field(None, description="关联病历ID")
    prescription_type: PrescriptionType = Field(PrescriptionType.ORDINARY, description="处方类型")
    diagnosis: str = Field(..., min_length=2, description="临床诊断")
    icd10_code: Optional[str] = Field(None, description="诊断ICD-10编码")
    items: list[PrescriptionItemCreate] = Field(..., min_length=1, max_length=5, description="处方药品列表")

class PrescriptionResponse(BaseModel):
    """处方响应"""
    id: str
    patient_id: str
    doctor_id: str
    prescription_type: str
    diagnosis: str
    icd10_code: Optional[str] = None
    items: list[dict]
    is_reviewed: bool = False
    status: str = "active"
    created_at: str
    ai_disclaimer: str = "AI辅助审查结果仅供参考，最终由药师审核确认"

    model_config = {"from_attributes": True}

class PrescriptionCheckRequest(BaseModel):
    """合理用药审查请求"""
    patient_id: str
    drugs: list[dict] = Field(..., description="药品列表")
    patient_allergies: list[str] = Field(default_factory=list, description="过敏史")
    patient_conditions: list[str] = Field(default_factory=list, description="合并疾病")
    patient_age: Optional[int] = Field(None, ge=0, le=150)
    patient_gender: Optional[str] = None
    patient_egfr: Optional[float] = Field(None, ge=0, description="eGFR")

class AlertSeverity(str, Enum):
    INFO = "提示"
    WARNING = "警告"
    CRITICAL = "严重"
    BLOCKED = "禁止"

class MedicationAlertResponse(BaseModel):
    """用药审查提示"""
    alert_type: str
    severity: AlertSeverity
    drug_name: str
    message: str
    suggestion: Optional[str] = None
    reference: Optional[str] = None

class PrescriptionCheckResponse(BaseModel):
    """合理用药审查响应"""
    is_safe: bool
    alerts: list[MedicationAlertResponse]
    checked_at: str
    ai_disclaimer: str = "AI辅助审查结果仅供参考，最终由药师审核确认"

# app/routers/prescriptions.py
from fastapi import APIRouter, Depends, HTTPException
from sqlalchemy.ext.asyncio import AsyncSession
from sqlalchemy import select
from datetime import datetime
from typing import Optional

from app.database import get_db
from app.models.prescription import Prescription, PrescriptionItem
from app.models.patient import Patient
from app.schemas.prescription import (
    PrescriptionCreate, PrescriptionResponse,
    PrescriptionCheckRequest, PrescriptionCheckResponse,
    MedicationAlertResponse, AlertSeverity,
)
from app.services.prescription_service import PrescriptionService, MedicationReviewEngine
from app.core.security import get_current_user, require_role, UserRole, User

router = APIRouter(prefix="/api/v1/prescriptions", tags=["处方与药品管理"])

@router.post("", response_model=PrescriptionResponse, status_code=201)
async def create_prescription(
    data: PrescriptionCreate,
    db: AsyncSession = Depends(get_db),
    current_user: User = Depends(require_role(UserRole.DOCTOR)),
):
    """开具处方"""
    # 验证患者
    patient = await db.get(Patient, data.patient_id)
    if not patient:
        raise HTTPException(status_code=404, detail="患者不存在")

    now = datetime.now().isoformat()

    # 创建处方
    prescription = Prescription(
        patient_id=data.patient_id,
        doctor_id=current_user.id,
        record_id=data.record_id,
        prescription_type=data.prescription_type.value,
        diagnosis=data.diagnosis,
        icd10_code=data.icd10_code,
        is_reviewed=False,
        status="active",
        created_at=now,
    )
    db.add(prescription)
    await db.flush()

    # 创建处方项目
    for idx, item_data in enumerate(data.items):
        sig = item_data.sig or f"{item_data.dosage} {item_data.route} {item_data.frequency}"
        item = PrescriptionItem(
            prescription_id=prescription.id,
            drug_name=item_data.drug_name,
            brand_name=item_data.brand_name,
            specification=item_data.specification,
            dosage_form=item_data.dosage_form,
            quantity=item_data.quantity,
            dosage=item_data.dosage,
            frequency=item_data.frequency,
            route=item_data.route,
            duration_days=item_data.duration_days,
            sig=sig,
            remark=item_data.remark,
            sort_order=idx,
        )
        db.add(item)

    await db.flush()
    await db.refresh(prescription)

    # 自动触发合理用药审查
    service = PrescriptionService(db)
    patient_allergies = [a.strip() for a in (patient.allergies or "").split(",") if a.strip()]
    check_result = service.review_prescription(
        prescription=prescription,
        patient_allergies=patient_allergies,
        patient_conditions=[],
        patient_egfr=None,
    )

    # 如果有严重或禁止级别警告，标记处方状态
    if any(a.severity in [AlertSeverity.CRITICAL, AlertSeverity.BLOCKED] for a in check_result["alerts"]):
        prescription.status = "pending_review"

    return PrescriptionResponse(
        id=prescription.id,
        patient_id=prescription.patient_id,
        doctor_id=prescription.doctor_id,
        prescription_type=prescription.prescription_type,
        diagnosis=prescription.diagnosis,
        icd10_code=prescription.icd10_code,
        items=[
            {
                "drug_name": item.drug_name,
                "specification": item.specification,
                "dosage": item.dosage,
                "frequency": item.frequency,
                "route": item.route,
                "duration_days": item.duration_days,
                "sig": item.sig,
            }
            for item in prescription.items
        ],
        is_reviewed=prescription.is_reviewed,
        status=prescription.status,
        created_at=prescription.created_at,
    )

@router.get("/{prescription_id}", response_model=PrescriptionResponse)
async def get_prescription(
    prescription_id: str,
    db: AsyncSession = Depends(get_db),
    current_user: User = Depends(get_current_user),
):
    """获取处方详情"""
    prescription = await db.get(Prescription, prescription_id)
    if not prescription:
        raise HTTPException(status_code=404, detail="处方不存在")

    # 加载处方项目
    result = await db.execute(
        select(PrescriptionItem)
        .where(PrescriptionItem.prescription_id == prescription_id)
        .order_by(PrescriptionItem.sort_order)
    )
    items = result.scalars().all()

    return PrescriptionResponse(
        id=prescription.id,
        patient_id=prescription.patient_id,
        doctor_id=prescription.doctor_id,
        prescription_type=prescription.prescription_type,
        diagnosis=prescription.diagnosis,
        icd10_code=prescription.icd10_code,
        items=[
            {
                "drug_name": item.drug_name,
                "specification": item.specification,
                "quantity": item.quantity,
                "dosage": item.dosage,
                "frequency": item.frequency,
                "route": item.route,
                "duration_days": item.duration_days,
                "sig": item.sig,
                "remark": item.remark,
            }
            for item in items
        ],
        is_reviewed=prescription.is_reviewed,
        status=prescription.status,
        created_at=prescription.created_at,
    )

@router.post("/check", response_model=PrescriptionCheckResponse)
async def check_prescription(
    request: PrescriptionCheckRequest,
    current_user: User = Depends(require_role(UserRole.DOCTOR, UserRole.PHARMACIST)),
):
    """合理用药审查"""
    engine = MedicationReviewEngine()
    result = engine.check(
        drugs=request.drugs,
        patient_allergies=request.patient_allergies,
        patient_conditions=request.patient_conditions,
        patient_age=request.patient_age or 30,
        patient_gender=request.patient_gender or "未知",
        patient_egfr=request.patient_egfr,
    )

    is_safe = not any(a.severity in [AlertSeverity.CRITICAL, AlertSeverity.BLOCKED] for a in result)

    return PrescriptionCheckResponse(
        is_safe=is_safe,
        alerts=result,
        checked_at=datetime.now().isoformat(),
    )

from sqlalchemy.ext.asyncio import AsyncSession
from typing import Optional
from datetime import datetime

from app.models.prescription import Prescription, PrescriptionItem
from app.models.patient import Patient
from app.schemas.prescription import (
    MedicationAlertResponse, AlertSeverity,

class MedicationReviewEngine:

    DRUG_INTERACTIONS = {
        frozenset(["华法林", "阿司匹林"]): {

            "suggestion": "避免联用，或密切监测INR和出血征象",
            "reference": "ACCP抗栓指南",
        },
        frozenset(["华法林", "布洛芬"]): {
            "severity": AlertSeverity.CRITICAL,
            "message": "华法林与NSAIDs联用增加出血风险",
            "suggestion": "如需镇痛，优先使用对乙酰氨基酚",
            "reference": "药品说明书",
        },
        frozenset(["甲氨蝶呤", "布洛芬"]): {
            "severity": AlertSeverity.BLOCKED,
            "message": "布洛芬减少甲氨蝶呤排泄，可致甲氨蝶呤中毒",
            "suggestion": "禁止合用，选用对乙酰氨基酚",
            "reference": "甲氨蝶呤药品说明书",
        },
        frozenset(["ACEI", "螺内酯"]): {
            "severity": AlertSeverity.WARNING,
            "message": "ACEI与螺内酯联用增加高钾血症风险",
            "suggestion": "合用时监测血钾，避免补钾",
            "reference": "ACEI临床应用专家共识",
        },
        frozenset(["地高辛", "胺碘酮"]): {
            "severity": AlertSeverity.WARNING,
            "message": "胺碘酮升高地高辛血药浓度，增加洋地黄中毒风险",
            "suggestion": "合用时地高辛减量50%，监测血药浓度",
            "reference": "地高辛临床应用指南",
        },
        frozenset(["二甲双胍", "造影剂"]): {
            "severity": AlertSeverity.WARNING,
            "message": "使用碘造影剂前后需停用二甲双胍，防止乳酸酸中毒",
            "suggestion": "造影前48h停用，造影后48h复查肾功能后恢复",
            "reference": "二甲双胍临床应用专家共识",
        },
        frozenset(["头孢曲松", "含钙输液"]): {
            "severity": AlertSeverity.BLOCKED,
            "message": "头孢曲松与钙离子可形成致命性沉淀",
            "suggestion": "禁止在同一输液管路中混合使用",
            "reference": "头孢曲松药品说明书",
        },
    }

    # 儿童用药禁忌
    PEDIATRIC_CONTRAINDICATIONS = {
        "喹诺酮类": ["左氧氟沙星", "环丙沙星", "莫西沙星"],
        "四环素类": ["多西环素", "米诺环素"],
        "氨基糖苷类": ["庆大霉素", "阿米卡星"],
    }

    # 孕妇用药禁忌
    PREGNANCY_CONTRAINDICATIONS = [
        "异维A酸", "华法林", "甲氨蝶呤", "利巴韦林",
        "ACEI", "ARB", "喹诺酮类", "四环素类",
    ]

    # 最大日剂量
    MAX_DAILY_DOSAGE = {
        "阿莫西林": {"adult_mg": 4000, "unit": "mg/day"},
        "二甲双胍": {"adult_mg": 2550, "unit": "mg/day"},
        "阿托伐他汀": {"adult_mg": 80, "unit": "mg/day"},
        "布洛芬": {"adult_mg": 2400, "unit": "mg/day"},
        "对乙酰氨基酚": {"adult_mg": 2000, "unit": "mg/day"},
        "奥美拉唑": {"adult_mg": 40, "unit": "mg/day"},
        "氨氯地平": {"adult_mg": 10, "unit": "mg/day"},
        "阿司匹林": {"adult_mg": 300, "unit": "mg/day(抗血小板)"},
    }

    # 肾功能不全禁忌
    RENAL_CONTRAINDICATIONS = {
        "二甲双胍": {"egfr_threshold": 30, "message": "eGFR<30时二甲双胍禁忌"},
        "华法林": {"egfr_threshold": 15, "message": "严重肾功能不全时需调整剂量"},
    }

    def check(
        self,
        drugs: list[dict],
        patient_allergies: list[str],
        patient_conditions: list[str],
        patient_age: int = 30,
        patient_gender: str = "未知",
        patient_egfr: Optional[float] = None,
    ) -> list[MedicationAlertResponse]:
        """完整合理用药审查"""
        alerts = []

        # 1. 过敏检查
        alerts.extend(self._check_allergies(drugs, patient_allergies))

        # 2. 配伍禁忌
        alerts.extend(self._check_interactions(drugs))

        # 3. 剂量审查
        alerts.extend(self._check_dosage(drugs, patient_age))

        # 4. 重复用药
        alerts.extend(self._check_duplicates(drugs))

        # 5. 儿童用药禁忌
        if patient_age < 18:
            alerts.extend(self._check_pediatric(drugs))

        # 6. 孕妇用药禁忌
        if "妊娠" in patient_conditions or "孕妇" in patient_conditions:
            alerts.extend(self._check_pregnancy(drugs))

        # 7. 肾功能不全
        if patient_egfr is not None:
            alerts.extend(self._check_renal(drugs, patient_egfr))

        # 8. 处方药品数量
        if len(drugs) > 5:
            alerts.append(MedicationAlertResponse(
                alert_type="处方规范性",
                severity=AlertSeverity.WARNING,
                drug_name="",
                message=f"处方含{len(drugs)}种药品，超过5种上限",
                suggestion="减少用药种类或分次开具",
            ))

        return alerts

    def _check_allergies(self, drugs: list[dict], allergies: list[str]) -> list[MedicationAlertResponse]:
        """过敏检查"""
        alerts = []
        for allergen in allergies:
            if allergen in self.ALLERGY_CROSS_REACTIVITY:
                cross_drugs = self.ALLERGY_CROSS_REACTIVITY[allergen]
                for drug in drugs:
                    drug_name = drug.get("drug_name", "")
                    if drug_name in cross_drugs:
                        alerts.append(MedicationAlertResponse(
                            alert_type="过敏禁忌",
                            severity=AlertSeverity.BLOCKED,
                            drug_name=drug_name,
                            message=f"患者对{allergen}过敏，禁用{drug_name}",
                            suggestion="更换为替代药物",
                            reference="过敏史记录",
                        ))
        return alerts

    def _check_interactions(self, drugs: list[dict]) -> list[MedicationAlertResponse]:
        """配伍禁忌检查"""
        alerts = []
        drug_names = [d.get("drug_name", "") for d in drugs]

        for i in range(len(drug_names)):
            for j in range(i + 1, len(drug_names)):
                pair = frozenset([drug_names[i], drug_names[j]])
                if pair in self.DRUG_INTERACTIONS:
                    interaction = self.DRUG_INTERACTIONS[pair]
                    alerts.append(MedicationAlertResponse(
                        alert_type="配伍禁忌",
                        severity=interaction["severity"],
                        drug_name=f"{drug_names[i]} + {drug_names[j]}",
                        message=interaction["message"],
                        suggestion=interaction.get("suggestion"),
                        reference=interaction.get("reference"),
                    ))
        return alerts

    def _check_dosage(self, drugs: list[dict], patient_age: int) -> list[MedicationAlertResponse]:
        """剂量审查"""
        alerts = []
        for drug in drugs:
            drug_name = drug.get("drug_name", "")
            daily_dose = drug.get("daily_dose_mg", 0)

            if drug_name in self.MAX_DAILY_DOSAGE and daily_dose:
                max_dose = self.MAX_DAILY_DOSAGE[drug_name]["adult_mg"]
                unit = self.MAX_DAILY_DOSAGE[drug_name]["unit"]

                if patient_age >= 65:
                    max_dose = int(max_dose * 0.75)  # 老年人减量

                try:
                    dose_val = float(daily_dose)
                    if dose_val > max_dose:
                        alerts.append(MedicationAlertResponse(
                            alert_type="剂量异常",
                            severity=AlertSeverity.CRITICAL,
                            drug_name=drug_name,
                            message=f"日剂量{daily_dose}{unit}超过推荐最大剂量{max_dose}{unit}",
                            suggestion=f"建议调整至不超过{max_dose}{unit}",
                        ))
                except (ValueError, TypeError):
                    pass
        return alerts

    def _check_duplicates(self, drugs: list[dict]) -> list[MedicationAlertResponse]:
        """重复用药检查"""
        alerts = []
        DRUG_CLASS = {
            "阿莫西林": "青霉素类", "阿莫西林克拉维酸钾": "青霉素类",
            "头孢呋辛": "头孢菌素类", "头孢曲松": "头孢菌素类",
            "布洛芬": "NSAIDs", "双氯芬酸": "NSAIDs",
            "奥美拉唑": "PPI", "兰索拉唑": "PPI", "泮托拉唑": "PPI",
            "氨氯地平": "CCB", "硝苯地平": "CCB",
            "依那普利": "ACEI", "卡托普利": "ACEI",
        }

        class_drugs: dict[str, list[str]] = {}
        for drug in drugs:
            name = drug.get("drug_name", "")
            drug_class = DRUG_CLASS.get(name)
            if drug_class:
                class_drugs.setdefault(drug_class, []).append(name)

        for drug_class, drug_list in class_drugs.items():
            if len(drug_list) > 1:
                alerts.append(MedicationAlertResponse(
                    alert_type="重复用药",
                    severity=AlertSeverity.WARNING,
                    drug_name=" + ".join(drug_list),
                    message=f"同类别({drug_class})药物重复使用",
                    suggestion="保留一种即可",
                ))
        return alerts

    def _check_pediatric(self, drugs: list[dict]) -> list[MedicationAlertResponse]:
        """儿童用药禁忌"""
        alerts = []
        for category, drug_list in self.PEDIATRIC_CONTRAINDICATIONS.items():
            for drug in drugs:
                if drug.get("drug_name", "") in drug_list:
                    alerts.append(MedicationAlertResponse(
                        alert_type="儿童用药禁忌",
                        severity=AlertSeverity.BLOCKED,
                        drug_name=drug["drug_name"],
                        message=f"{drug['drug_name']}({category})禁用于18岁以下患者",
                        suggestion="更换为儿童适用药物",
                    ))
        return alerts

    def _check_pregnancy(self, drugs: list[dict]) -> list[MedicationAlertResponse]:
        """孕妇用药禁忌"""
        alerts = []
        for drug in drugs:
            drug_name = drug.get("drug_name", "")
            if drug_name in self.PREGNANCY_CONTRAINDICATIONS:
                alerts.append(MedicationAlertResponse(
                    alert_type="孕妇用药禁忌",
                    severity=AlertSeverity.BLOCKED,
                    drug_name=drug_name,
                    message=f"妊娠期禁用{drug_name}",
                    suggestion="更换为妊娠期安全药物",
                    reference="妊娠期用药指南",
                ))
        return alerts

    def _check_renal(self, drugs: list[dict], egfr: float) -> list[MedicationAlertResponse]:
        """肾功能不全审查"""
        alerts = []
        for drug in drugs:
            drug_name = drug.get("drug_name", "")
            if drug_name in self.RENAL_CONTRAINDICATIONS:
                rule = self.RENAL_CONTRAINDICATIONS[drug_name]
                if egfr < rule["egfr_threshold"]:
                    alerts.append(MedicationAlertResponse(
                        alert_type="肾功能禁忌",
                        severity=AlertSeverity.BLOCKED,
                        drug_name=drug_name,
                        message=f"eGFR={egfr:.0f}，{rule['message']}",
                        suggestion="更换为肾安全性药物或调整剂量",
                    ))
        return alerts


class PrescriptionService:
    """处方服务"""

    def __init__(self, db: AsyncSession):
        self.db = db
        self.review_engine = MedicationReviewEngine()

    def review_prescription(
        self,
        prescription: Prescription,
        patient_allergies: list[str],
        patient_conditions: list[str],
        patient_egfr: Optional[float] = None,
    ) -> dict:
        """审查处方"""
        drugs = [
            {
                "drug_name": item.drug_name,
                "dosage": item.dosage,
                "frequency": item.frequency,
            }
            for item in prescription.items
        ]

        alerts = self.review_engine.check(
            drugs=drugs,
            patient_allergies=patient_allergies,
            patient_conditions=patient_conditions,
            patient_egfr=patient_egfr,
        )

        is_safe = not any(a.severity in [AlertSeverity.CRITICAL, AlertSeverity.BLOCKED] for a in alerts)

        return {
            "prescription_id": prescription.id,
            "is_safe": is_safe,
            "alerts": alerts,
        }
