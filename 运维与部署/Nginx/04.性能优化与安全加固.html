性能优化与安全加固

性能优化与安全加固

Nginx 默认配置就能用，但生产环境需要调优。压缩、缓存、限流、安全头，每个都能让你的服务更快更安全。

Gzip 压缩

http {
    gzip on;
    gzip_vary on;
    gzip_proxied any;
    gzip_comp_level 6;           # 压缩级别 1-9，6 是性价比最高的
    gzip_min_length 1000;        # 小于 1KB 不压缩
    gzip_types
        text/plain
        text/css
        text/xml
        text/javascript
        application/json
        application/javascript
        application/xml
        application/rss+xml
        image/svg+xml;
}

效果：JSON 响应体积减少 60-80%，页面加载快很多。

静态文件缓存

# 图片、字体、CSS、JS 设长缓存
location ~* \.(jpg|jpeg|png|gif|ico|svg|woff2|woff|ttf|css|js)$ {
    expires 30d;
    add_header Cache-Control "public, immutable";
    access_log off;     # 静态文件不记日志
}

# HTML 不缓存（SPA 入口文件）
location ~* \.html$ {
    expires -1;
    add_header Cache-Control "no-cache, no-store, must-revalidate";
}

连接优化

http {
    # Keep-alive
    keepalive_timeout 65;
    keepalive_requests 1000;

    # 与后端的 Keep-alive
    upstream erp_backend {
        server 127.0.0.1:5000;
        keepalive 32;            # 保持 32 个空闲长连接
    }

    server {
        location / {
            proxy_pass http://erp_backend;
            proxy_http_version 1.1;
            proxy_set_header Connection "";    # 清空 Connection 头，启用 keepalive
        }
    }

    # 文件传输
    sendfile on;
    tcp_nopush on;
    tcp_nodelay on;

    # 缓冲区
    client_body_buffer_size 16k;
    client_header_buffer_size 1k;
    large_client_header_buffers 4 8k;

    # 超时
    client_body_timeout 12;
    client_header_timeout 12;
    send_timeout 10;
}

Worker 调优

# 全局
worker_processes auto;         # 自动检测 CPU 核心数
worker_rlimit_nofile 65535;    # 最大打开文件数

events {
    worker_connections 4096;   # 每个 worker 最大并发连接
    multi_accept on;           # 一次接受多个连接
    use epoll;                 # Linux 高性能事件模型
}

请求限流

限制请求频率

http {
    # 定义限流区域：按 IP，每秒 10 个请求
    limit_req_zone $binary_remote_addr zone=api_limit:10m rate=10r/s;

    server {
        location /api/ {
            # burst=20：允许突发 20 个
            # nodelay：突发请求不排队立即处理
            limit_req zone=api_limit burst=20 nodelay;
            limit_req_status 429;

            proxy_pass http://127.0.0.1:5000;
        }
    }
}

限制并发连接

http {
    limit_conn_zone $binary_remote_addr zone=conn_limit:10m;

    server {
        # 每个 IP 最多 50 个并发连接
        limit_conn conn_limit 50;
        limit_conn_status 429;
    }
}

限制上传大小

server {
    client_max_body_size 50m;    # 最大上传 50MB

    # 针对特定路径
    location /api/upload {
        client_max_body_size 200m;
    }
}

安全 Header

server {
    # 防止 MIME 类型嗅探
    add_header X-Content-Type-Options "nosniff" always;

    # 防止点击劫持
    add_header X-Frame-Options "SAMEORIGIN" always;

    # XSS 保护
    add_header X-XSS-Protection "1; mode=block" always;

    # HSTS
    add_header Strict-Transport-Security "max-age=31536000; includeSubDomains" always;

    # 隐藏 Nginx 版本号
    server_tokens off;

    # 禁止 .git 等敏感目录访问
    location ~ /\. {
        deny all;
        access_log off;
        log_not_found off;
    }
}

IP 黑白名单

# 只允许内网访问管理后台
location /admin/ {
    allow 192.168.1.0/24;
    allow 10.0.0.0/8;
    deny all;

    proxy_pass http://127.0.0.1:5000;
}

# 屏蔽恶意 IP
# /etc/nginx/conf.d/blocklist.conf
deny 1.2.3.4;
deny 5.6.7.0/24;

# 在 http 或 server 块里 include
include /etc/nginx/conf.d/blocklist.conf;

日志优化

http {
    # 自定义日志格式（含响应时间）
    log_format detailed '$remote_addr - [$time_local] '
                        '"$request" $status $body_bytes_sent '
                        'rt=$request_time '
                        '"$http_user_agent"';

    # 按站点分日志
    server {
        access_log /var/log/nginx/erp-access.log detailed;
        error_log /var/log/nginx/erp-error.log warn;
    }

    # 静态文件不记日志
    location ~* \.(js|css|png|jpg|gif|ico)$ {
        access_log off;
    }

    # 健康检查不记日志
    location = /health {
        access_log off;
        return 200 'OK';
    }
}

日志轮转

# /etc/logrotate.d/nginx
/var/log/nginx/*.log {
    daily
    missingok
    rotate 30
    compress
    delaycompress
    notifempty
    sharedscripts
    postrotate
        [ -f /run/nginx.pid ] && kill -USR1 $(cat /run/nginx.pid)
    endscript
}

生产环境完整配置模板

user nginx;
worker_processes auto;
worker_rlimit_nofile 65535;
error_log /var/log/nginx/error.log warn;
pid /run/nginx.pid;

events {
    worker_connections 4096;
    multi_accept on;
    use epoll;
}

http {
    include /etc/nginx/mime.types;
    default_type application/octet-stream;
    server_tokens off;

    # 日志
    log_format main '$remote_addr - [$time_local] "$request" $status '
                    '$body_bytes_sent rt=$request_time';
    access_log /var/log/nginx/access.log main;

    # 性能
    sendfile on;
    tcp_nopush on;
    tcp_nodelay on;
    keepalive_timeout 65;
    keepalive_requests 1000;

    # Gzip
    gzip on;
    gzip_vary on;
    gzip_comp_level 6;
    gzip_min_length 1000;
    gzip_types text/plain text/css application/json application/javascript text/xml image/svg+xml;

    # 限流
    limit_req_zone $binary_remote_addr zone=api:10m rate=20r/s;
    limit_conn_zone $binary_remote_addr zone=conn:10m;

    # 安全
    add_header X-Content-Type-Options "nosniff" always;
    add_header X-Frame-Options "SAMEORIGIN" always;

    # 站点
    include /etc/nginx/conf.d/*.conf;
}
