Shell 脚本

不会写 Shell 脚本的后端，每次部署都得手敲一堆命令。学会写脚本，一键部署、定时清日志、批量操作全搞定。

基础语法

第一个脚本

#!/bin/bash
# deploy.sh - 我的第一个部署脚本

echo "开始部署..."
echo "当前时间: $(date '+%Y-%m-%d %H:%M:%S')"
echo "当前用户: $(whoami)"
echo "部署完成"
# 给执行权限
chmod +x deploy.sh

# 运行
./deploy.sh
# 或
bash deploy.sh

变量

#!/bin/bash

# 定义变量（等号两边不能有空格）
APP_NAME="erp-api"
APP_DIR="/opt/apps/${APP_NAME}"
PORT=5000

# 使用变量
echo "应用名: ${APP_NAME}"
echo "目录: ${APP_DIR}"

# 命令结果赋值给变量
CURRENT_DATE=$(date '+%Y%m%d')
IP_ADDR=$(hostname -I | awk '{print $1}')

echo "日期: ${CURRENT_DATE}"
echo "IP: ${IP_ADDR}"

# 接收命令行参数
# ./deploy.sh v1.2.3 production
VERSION=$1        # 第一个参数
ENV=$2            # 第二个参数
SCRIPT_NAME=$0    # 脚本自身名称
PARAM_COUNT=$#    # 参数个数

echo "版本: ${VERSION}, 环境: ${ENV}"

条件判断

#!/bin/bash

APP_DIR="/opt/apps/erp-api"

# 文件/目录判断
if [ -d "$APP_DIR" ]; then
    echo "目录存在"
fi

if [ -f "$APP_DIR/appsettings.json" ]; then
    echo "配置文件存在"
fi

if [ ! -f "$APP_DIR/ERP.Host.dll" ]; then
    echo "DLL 不存在，部署可能失败了"
    exit 1
fi

# 字符串判断
if [ -z "$1" ]; then
    echo "用法: $0 <版本号>"
    exit 1
fi

# 数字比较
PORT=5000
if [ $PORT -eq 5000 ]; then
    echo "端口是 5000"
fi
# -eq 等于 / -ne 不等于 / -gt 大于 / -lt 小于

# 命令是否成功
if systemctl is-active --quiet erp-api; then
    echo "服务正在运行"
else
    echo "服务未运行"
fi

函数

#!/bin/bash

# 定义函数
log_info() {
    echo "[$(date '+%H:%M:%S')] INFO: $1"
}

log_error() {
    echo "[$(date '+%H:%M:%S')] ERROR: $1" >&2
}

check_service() {
    local service_name=$1
    if systemctl is-active --quiet $service_name; then
        log_info "${service_name} 运行中 ✓"
        return 0
    else
        log_error "${service_name} 未运行 ✗"
        return 1
    fi
}

# 调用
log_info "开始检查服务"
check_service "erp-api"
check_service "nginx"

实战脚本

一键部署脚本

#!/bin/bash
set -e  # 遇到错误立即退出

# ====== 配置 ======
APP_NAME="erp-api"
APP_DIR="/opt/apps/${APP_NAME}"
BACKUP_DIR="/opt/backups/${APP_NAME}"
PUBLISH_FILE="$1"

# ====== 函数 ======
log() { echo "[$(date '+%Y-%m-%d %H:%M:%S')] $1"; }
die() { log "ERROR: $1"; exit 1; }

# ====== 检查 ======
[ -z "$PUBLISH_FILE" ] && die "用法: $0 <发布包.tar.gz>"
[ -f "$PUBLISH_FILE" ] || die "文件不存在: $PUBLISH_FILE"

# ====== 部署流程 ======
log "========== 开始部署 ${APP_NAME} =========="

# 1. 备份
mkdir -p $BACKUP_DIR
BACKUP_FILE="${BACKUP_DIR}/${APP_NAME}_$(date +%Y%m%d_%H%M%S).tar.gz"
log "备份到 ${BACKUP_FILE}"
tar -czf $BACKUP_FILE -C $APP_DIR .

# 2. 停服务
log "停止服务"
sudo systemctl stop $APP_NAME

# 3. 部署
log "部署新版本"
tar -xzf $PUBLISH_FILE -C $APP_DIR --strip-components=1

# 4. 修复权限
chown -R deploy:deploy $APP_DIR

# 5. 启服务
log "启动服务"
sudo systemctl start $APP_NAME
sleep 3

# 6. 健康检查
log "健康检查..."
HTTP_CODE=$(curl -o /dev/null -s -w "%{http_code}" http://localhost:5000/health || echo "000")

if [ "$HTTP_CODE" = "200" ]; then
    log "✅ 部署成功！健康检查通过"
else
    log "❌ 健康检查失败 (HTTP ${HTTP_CODE})，回滚..."
    sudo systemctl stop $APP_NAME
    tar -xzf $BACKUP_FILE -C $APP_DIR
    chown -R deploy:deploy $APP_DIR
    sudo systemctl start $APP_NAME
    die "已回滚到上一版本"
fi

日志清理定时任务

#!/bin/bash
# clean-logs.sh - 清理过期日志

LOG_DIRS=(
    "/opt/apps/erp-api/logs"
    "/opt/apps/sso-server/logs"
    "/var/log/nginx"
)
KEEP_DAYS=30

for dir in "${LOG_DIRS[@]}"; do
    if [ -d "$dir" ]; then
        count=$(find "$dir" -name "*.log" -mtime +$KEEP_DAYS | wc -l)
        if [ $count -gt 0 ]; then
            echo "清理 ${dir}: ${count} 个文件(>${KEEP_DAYS}天)"
            find "$dir" -name "*.log" -mtime +$KEEP_DAYS -delete
        fi
    fi
done

echo "清理完成: $(date)"

配合 crontab 定时执行：

# 编辑定时任务
crontab -e

# 每天凌晨 3 点执行日志清理
0 3 * * * /opt/scripts/clean-logs.sh >> /var/log/clean-logs.log 2>&1

# 查看已有的定时任务
crontab -l

# crontab 时间格式：
# 分 时 日 月 周
# 0  3  *  *  *   → 每天凌晨3点
# */5 * * * *     → 每5分钟
# 0 */2 * * *     → 每2小时
# 0 9 * * 1       → 每周一早上9点

批量检查服务状态

#!/bin/bash
# check-services.sh

SERVICES=("erp-api" "sso-server" "nginx" "docker" "postgresql-14")

echo "======= 服务状态检查 $(date '+%Y-%m-%d %H:%M') ======="
printf "%-20s %-10s %-10s\n" "服务" "状态" "端口"
echo "------------------------------------------------"

for svc in "${SERVICES[@]}"; do
    if systemctl is-active --quiet $svc 2>/dev/null; then
        status="✅ 运行"
    else
        status="❌ 停止"
    fi
    printf "%-20s %-10s\n" "$svc" "$status"
done

echo ""
echo "======= 磁盘使用 ======="
df -h | grep -E "/$|/data"

echo ""
echo "======= 内存使用 ======="
free -h | head -2

常用技巧

# set -e: 任何命令失败就退出（部署脚本必加）
set -e

# set -x: 打印每条命令（调试用）
set -x

# 管道和重定向
echo "hello" > file.txt      # 覆盖写入
echo "world" >> file.txt     # 追加写入
command 2>&1                 # stderr 合并到 stdout
command > /dev/null 2>&1     # 丢弃所有输出

# 字符串操作
filename="erp-api-v1.2.3.tar.gz"
echo "${filename%.tar.gz}"   # erp-api-v1.2.3（去掉后缀）
echo "${filename##*-}"       # v1.2.3.tar.gz（去掉最后一个-前面的）

# 默认值
PORT=${1:-5000}              # 如果 $1 为空，用 5000
