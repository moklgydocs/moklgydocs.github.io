Compose 语法全解

Compose 语法全解

一台服务器上跑 API + 数据库 + Redis + Nginx，用 docker run 要敲一堆命令。Compose 把所有服务写到一个 yml 文件里，docker compose up 一键全起。

最小示例

# docker-compose.yml
services:
  erp-api:
    image: erp-api:v1.0
    ports:
      - "5000:5000"
docker compose up -d      # 启动
docker compose down        # 停止并删除

就这么简单。下面看完整语法。

完整结构

# docker-compose.yml 的顶层结构

services:     # 服务定义（必须）
  erp-api:
    ...
  postgres:
    ...

volumes:      # 数据卷定义（可选）
  pgdata:
    ...

networks:     # 网络定义（可选）
  app-net:
    ...

services 详解

基础配置

services:
  erp-api:
    # 用现成镜像
    image: registry.company.com/erp-api:v1.0

    # 或者用 Dockerfile 构建
    build:
      context: ./src/ERP
      dockerfile: Dockerfile

    # 容器名（不指定的话会自动生成）
    container_name: erp-api

    # 端口映射
    ports:
      - "5000:5000"         # 宿主机5000 → 容器5000
      - "5001:5001"

    # 环境变量
    environment:
      - ASPNETCORE_ENVIRONMENT=Production
      - ASPNETCORE_URLS=http://+:5000
      - ConnectionStrings__Default=Host=postgres;Port=5432;Database=erp;Username=postgres;Password=secret

    # 或者用 env_file
    env_file:
      - .env
      - .env.production

    # 数据卷挂载
    volumes:
      - ./config/appsettings.Production.json:/app/appsettings.Production.json:ro
      - erp-logs:/app/logs

    # 加入网络
    networks:
      - app-net

    # 重启策略
    restart: unless-stopped

    # 依赖（先启动 postgres 和 redis）
    depends_on:
      - postgres
      - redis

资源限制

services:
  erp-api:
    image: erp-api:v1.0
    deploy:
      resources:
        limits:
          cpus: '1.5'
          memory: 512M
        reservations:
          cpus: '0.5'
          memory: 256M

日志配置

services:
  erp-api:
    image: erp-api:v1.0
    logging:
      driver: json-file
      options:
        max-size: "50m"
        max-file: "3"

健康检查

services:
  erp-api:
    image: erp-api:v1.0
    healthcheck:
      test: ["CMD", "curl", "-f", "http://localhost:5000/health"]
      interval: 30s
      timeout: 5s
      retries: 3
      start_period: 10s

启动命令

services:
  erp-api:
    image: erp-api:v1.0
    # 覆盖默认启动命令
    command: ["dotnet", "ERP.Host.dll", "--urls", "http://+:5000"]

    # 或者覆盖 entrypoint
    entrypoint: ["/app/entrypoint.sh"]

volumes 详解

services:
  postgres:
    volumes:
      # 命名卷（Docker 管理）
      - pgdata:/var/lib/postgresql/data
      # 绑定挂载（指定宿主机路径）
      - ./init-scripts:/docker-entrypoint-initdb.d
      # 只读挂载
      - ./config/pg_hba.conf:/etc/postgresql/pg_hba.conf:ro

# 顶层声明命名卷
volumes:
  pgdata:
    driver: local
  erp-logs:

networks 详解

services:
  erp-api:
    networks:
      - frontend    # 接 Nginx
      - backend     # 接数据库

  postgres:
    networks:
      - backend     # 只在后端网络

  nginx:
    networks:
      - frontend    # 只在前端网络

networks:
  frontend:
  backend:

# erp-api 能同时访问 nginx 和 postgres
# nginx 访问不了 postgres（不在同一个网络）
# 实现了网络隔离

depends_on 与启动顺序

services:
  erp-api:
    depends_on:
      # 简单写法：等容器启动了就行
      - postgres
      - redis

      # 完整写法：等健康检查通过再启动（推荐）
      postgres:
        condition: service_healthy
      redis:
        condition: service_started

  postgres:
    image: postgres:16
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U postgres"]
      interval: 5s
      timeout: 3s
      retries: 5

注意

depends_on 只控制启动顺序，不等应用就绪。数据库容器启动了 ≠ 数据库已经可以接受连接。用 condition: service_healthy + healthcheck 才靠谱。

常用指令速查

指令类型说明
image字符串使用的镜像
build对象构建上下文和 Dockerfile
container_name字符串容器名
ports列表端口映射
environment列表/字典环境变量
env_file列表环境变量文件
volumes列表挂载卷
networks列表加入的网络
depends_on列表/字典依赖的服务
restart字符串重启策略
healthcheck对象健康检查
logging对象日志配置
command字符串/列表覆盖CMD
working_dir字符串工作目录
user字符串运行用户
extra_hosts列表添加hosts记录
