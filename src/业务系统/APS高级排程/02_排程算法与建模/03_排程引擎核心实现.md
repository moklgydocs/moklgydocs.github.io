# 排程引擎核心实现

## 数据结构设计

排程引擎的核心数据结构围绕Job、Operation、Resource和Constraint四个核心实体展开。

### 核心类定义

```python
from dataclasses import dataclass, field
from datetime import datetime, timedelta
from enum import Enum
from typing import List, Optional, Dict

class ConstraintType(Enum):
    HARD = "HARD"    # 必须满足，违反则解不可行
    SOFT = "SOFT"    # 尽量满足，违反则扣分

class OperationStatus(Enum):
    UNSCHEDULED = "UNSCHEDULED"
    SCHEDULED = "SCHEDULED"
    LOCKED = "LOCKED"         # 冻结区，不可移动
    IN_PROGRESS = "IN_PROGRESS"
    COMPLETED = "COMPLETED"

@dataclass
class Job:
    """排程任务：对应一个生产订单"""
    job_id: str
    order_id: str
    product_id: str
    qty: float
    due_date: datetime
    priority: int                    # 1=紧急, 5=普通
    operations: List['Operation'] = field(default_factory=list)
    weight: float = 1.0             # 优化目标权重

@dataclass
class Operation:
    """工序：Job的一个加工步骤"""
    op_id: str
    job_id: str
    sequence: int                    # 工序序号
    duration: timedelta              # 加工时长
    required_resources: List[str]    # 所需资源ID列表
    next_op_id: Optional[str] = None # 后续工序
    prev_op_id: Optional[str] = None # 前置工序

    # 排程结果（求解后填充）
    start_time: Optional[datetime] = None
    end_time: Optional[datetime] = None
    assigned_resource: Optional[str] = None
    status: OperationStatus = OperationStatus.UNSCHEDULED

    # 换型时间
    setup_duration: Optional[timedelta] = None

@dataclass
class Resource:
    """资源：机器/人员/模具等"""
    resource_id: str
    resource_name: str
    resource_type: str               # MACHINE/WORKER/MOLD

    calendar_id: str = ""            # 工作日历

@dataclass
class Constraint:
    """约束条件"""
    constraint_id: str
    constraint_type: ConstraintType
    constraint_name: str

    params: Dict = field(default_factory=dict)
```

### 排程状态表示

排程引擎需要维护两种核心状态：资源时间线和操作链。

```python
class ResourceTimeline:
    """资源时间线：记录资源上已分配的操作"""

    def __init__(self, resource_id: str):
        self.resource_id = resource_id
        self.segments = []  # [(start, end, op_id), ...]

    def find_available_slot(self, duration, earliest_start, latest_end):
        """在时间线上找到满足时长要求的空闲时段"""
        # 按开始时间排序
        sorted_segments = sorted(self.segments, key=lambda s: s[0])

        # 检查各空闲区间
        current_time = earliest_start
        for seg_start, seg_end, op_id in sorted_segments:
            if seg_start >= current_time + duration:
                return current_time  # 找到足够长的空闲区间
            current_time = max(current_time, seg_end)

        # 检查最后一个操作之后的区间
        if current_time + duration <= latest_end:
            return current_time

        return None  # 无可用时段

    def assign(self, op_id, start, end):
        """将操作分配到时间线上"""
        self.segments.append((start, end, op_id))

    def remove(self, op_id):
        """从时间线上移除操作"""
        self.segments = [s for s in self.segments if s[2] != op_id]

class OperationChain:
    """操作链：记录Job内工序的前后依赖关系"""

    def __init__(self, job: Job):
        self.job_id = job.job_id
        self.chain = {}  # {op_id: Operation}
        for op in job.operations:
            self.chain[op.op_id] = op

    def get_earliest_start(self, op_id, resource_timelines):
        """计算工序的最早开始时间"""
        op = self.chain[op_id]
        earliest = datetime.min

        # 前置工序约束
        if op.prev_op_id:
            prev_op = self.chain[op.prev_op_id]
            if prev_op.end_time:
                earliest = max(earliest, prev_op.end_time)

        # 物料约束（前置工序完工后物料才可用）

        return earliest
```

## 约束检查算法

### 硬约束校验

硬约束必须全部满足，否则排程方案不可行：

```python
class HardConstraintChecker:
    def check_all(self, schedule):
        violations = []
        violations.extend(self.check_capacity(schedule))
        violations.extend(self.check_precedence(schedule))
        violations.extend(self.check_material(schedule))
        violations.extend(self.check_calendar(schedule))
        return violations

    def check_capacity(self, schedule):
        """产能约束：同一资源同一时间只能执行一个工序"""
        violations = []
        for resource_id, timeline in schedule.resource_timelines.items():
            sorted_segments = sorted(timeline.segments, key=lambda s: s[0])
            for i in range(len(sorted_segments) - 1):
                current_end = sorted_segments[i][1]
                next_start = sorted_segments[i + 1][0]
                if next_start < current_end:
                    violations.append(ConstraintViolation(
                        type='CAPACITY',
                        resource_id=resource_id,
                        op1=sorted_segments[i][2],
                        op2=sorted_segments[i + 1][2],
                        message=f'资源 {resource_id} 上操作重叠'
                    ))
        return violations

    def check_precedence(self, schedule):
        """工序先后约束：前工序完成前，后工序不能开始"""
        violations = []
        for job in schedule.jobs:
            for op in job.operations:
                if op.prev_op_id and op.start_time:
                    prev_op = schedule.get_operation(op.prev_op_id)
                    if prev_op.end_time and op.start_time < prev_op.end_time:
                        violations.append(ConstraintViolation(
                            type='PRECEDENCE',
                            job_id=job.job_id,
                            op_id=op.op_id,
                            message=f'工序 {op.op_id} 在前工序 {op.prev_op_id} 完成前开始'
                        ))
        return violations

    def check_calendar(self, schedule):
        """日历约束：操作时间必须在资源工作时间内"""
        violations = []
        for job in schedule.jobs:
            for op in job.operations:
                if op.assigned_resource and op.start_time:
                    calendar = schedule.get_calendar(op.assigned_resource)
                    if not calendar.is_working_time(op.start_time, op.end_time):
                        violations.append(ConstraintViolation(
                            type='CALENDAR',
                            resource_id=op.assigned_resource,
                            op_id=op.op_id,
                            message=f'工序 {op.op_id} 跨越非工作时间'
                        ))
        return violations
```

### 软约束评分

软约束用于评估排程方案的质量，违反则扣分：

```python
class SoftConstraintScorer:
    def calculate_score(self, schedule):
        """计算软约束总评分（负分制，0为最优）"""
        score = 0.0
        score += self.delay_penalty(schedule)
        score += self.setup_time_penalty(schedule)
        score += self.utilization_penalty(schedule)
        score += self.preference_penalty(schedule)
        return score

    def delay_penalty(self, schedule):
        """交期延迟惩罚：每延迟1小时扣1分"""
        penalty = 0.0
        for job in schedule.jobs:
            last_op = job.operations[-1]
            if last_op.end_time and last_op.end_time > job.due_date:
                delay_hours = (last_op.end_time - job.due_date).total_seconds() / 3600
                penalty -= delay_hours * job.priority  # 优先级越高惩罚越重
        return penalty

    def setup_time_penalty(self, schedule):
        """换型时间惩罚：总换型时间越长扣分越多"""
        total_setup = timedelta(0)
        for timeline in schedule.resource_timelines.values():
            sorted_ops = sorted(timeline.segments, key=lambda s: s[0])
            for i in range(len(sorted_ops) - 1):
                setup = schedule.get_setup_time(
                    sorted_ops[i][2], sorted_ops[i + 1][2]
                )
                total_setup += setup
        penalty = -total_setup.total_seconds() / 3600 * 0.5  # 每小时换型扣0.5分
        return penalty

    def utilization_penalty(self, schedule):
        """利用率惩罚：利用率低于80%扣分"""
        penalty = 0.0
        for resource_id, timeline in schedule.resource_timelines.items():
            total_working = sum(
                (end - start).total_seconds() for start, end, _ in timeline.segments
            )
            total_available = schedule.get_available_time(resource_id)
            utilization = total_working / total_available if total_available > 0 else 0
            if utilization < 0.8:
                penalty -= (0.8 - utilization) * 10
        return penalty
```

## Job-Shop排程的贪心构造算法

```python
class GreedyScheduler:
    """贪心构造算法：按优先级顺序逐个分配工序"""

    def schedule(self, jobs, resources, calendars, setup_matrix):
        schedule = Schedule()

        # 1. 初始化资源时间线
        for res in resources:
            schedule.resource_timelines[res.resource_id] = ResourceTimeline(res.resource_id)

        # 2. 按优先级+交期排序Job
        sorted_jobs = sorted(jobs, key=lambda j: (j.priority, j.due_date))

        # 3. 逐Job逐工序分配
        for job in sorted_jobs:
            for op in job.operations:
                best_assignment = self.find_best_assignment(
                    op, schedule, resources, calendars, setup_matrix
                )

                if best_assignment:
                    op.assigned_resource = best_assignment.resource_id
                    op.start_time = best_assignment.start_time
                    op.end_time = best_assignment.end_time
                    op.setup_duration = best_assignment.setup_duration
                    op.status = OperationStatus.SCHEDULED

                    schedule.resource_timelines[best_assignment.resource_id].assign(
                        op.op_id, op.start_time, op.end_time
                    )
                else:
                    # 无法找到可行分配，标记为未排程
                    op.status = OperationStatus.UNSCHEDULED

        return schedule

    def find_best_assignment(self, op, schedule, resources, calendars, setup_matrix):
        """为工序寻找最优分配"""
        candidates = []

        # 计算最早开始时间（受前置工序约束）
        earliest_start = schedule.get_earliest_start(op)

        # 遍历所有可用资源
        for res_id in op.required_resources:
            res = next(r for r in resources if r.resource_id == res_id)
            calendar = calendars[res.calendar_id]

            # 在资源时间线上找空闲时段
            timeline = schedule.resource_timelines[res_id]
            slot_start = timeline.find_available_slot(
                op.duration,
                earliest_start,
                op.job.due_date  # 不超过交期
            )

            if slot_start and calendar.is_working_time(slot_start, slot_start + op.duration):
                # 计算换型时间
                setup = self.calculate_setup(timeline, op, setup_matrix)

                candidates.append(AssignmentCandidate(
                    resource_id=res_id,
                    start_time=slot_start,
                    end_time=slot_start + op.duration,
                    setup_duration=setup,
                    score=self.evaluate_candidate(slot_start, op, setup)
                ))

        # 也考虑替代资源
        for res_id in op.required_resources:
            res = next(r for r in resources if r.resource_id == res_id)
            for alt_id in res.alternate_ids:
                # ... 同上逻辑，但评分略低（替代资源非最优）

        if candidates:
            return min(candidates, key=lambda c: c.score)
        return None
```

## 约束满足传播算法

约束传播通过缩小变量的取值域来加速搜索：

```python
class ConstraintPropagator:
    """约束传播：缩小工序时间窗"""

    def propagate(self, schedule):
        changed = True
        iterations = 0
        max_iterations = 100

        while changed and iterations < max_iterations:
            changed = False
            iterations += 1

            for job in schedule.jobs:
                for op in job.operations:
                    # 正向传播：前工序 → 后工序
                    if op.prev_op_id:
                        prev_op = schedule.get_operation(op.prev_op_id)
                        new_earliest = prev_op.earliest_end
                        if new_earliest > op.earliest_start:
                            op.earliest_start = new_earliest
                            changed = True

                    # 反向传播：后工序 → 前工序
                    if op.next_op_id:
                        next_op = schedule.get_operation(op.next_op_id)
                        new_latest = next_op.latest_start - op.duration
                        if new_latest < op.latest_start:
                            op.latest_start = new_latest
                            changed = True

                    # 资源容量传播
                    resource = schedule.get_resource(op.assigned_resource)
                    if resource.capacity == 1:
                        # 单能力资源：与其他操作互斥
                        for other_op in schedule.get_operations_on_resource(op.assigned_resource):
                            if other_op.op_id != op.op_id:
                                # 如果时间窗不重叠，可缩小域
                                if op.earliest_start >= other_op.latest_start:
                                    pass  # op在other之后，无冲突
                                elif op.latest_end <= other_op.earliest_start:
                                    pass  # op在other之前，无冲突
                                else:
                                    # 时间窗重叠，需要排序决策
                                    pass
```

## 时间线数据结构

### Interval Tree实现

Interval Tree用于高效查询时间线上的重叠操作：

```python
class IntervalNode:
    """区间树节点"""
    def __init__(self, interval, op_id):
        self.interval = interval  # (start, end)
        self.op_id = op_id
        self.max_end = interval[1]
        self.left = None
        self.right = None

class IntervalTree:
    """区间树：高效查询时间重叠的操作"""

    def __init__(self):
        self.root = None

    def insert(self, start, end, op_id):
        """插入一个操作区间"""
        self.root = self._insert(self.root, (start, end), op_id)

    def _insert(self, node, interval, op_id):
        if node is None:
            return IntervalNode(interval, op_id)

        if interval[0] < node.interval[0]:
            node.left = self._insert(node.left, interval, op_id)
        else:
            node.right = self._insert(node.right, interval, op_id)

        node.max_end = max(node.max_end, interval[1])
        return node

    def find_overlaps(self, start, end):
        """查找与指定时间段重叠的所有操作"""
        results = []
        self._find_overlaps(self.root, (start, end), results)
        return results

    def _find_overlaps(self, node, query, results):
        if node is None:
            return

        # 检查当前节点是否重叠
        if node.interval[0] < query[1] and node.interval[1] > query[0]:
            results.append(node.op_id)

        # 检查左子树
        if node.left and node.left.max_end > query[0]:
            self._find_overlaps(node.left, query, results)

        # 检查右子树
        if node.right and node.interval[0] < query[1]:
            self._find_overlaps(node.right, query, results)
```

## 性能优化

### 剪枝策略

在搜索过程中提前剪掉不可能产生更优解的分支：

```python
class PruningStrategy:
    def should_prune(self, current_score, best_score, remaining_ops):
        """判断是否应该剪枝"""
        # 1. 当前评分已经比已知最优解差
        if current_score < best_score:
            # 估算剩余操作的最佳可能评分
            optimistic_bound = self.estimate_optimistic_bound(remaining_ops)
            if current_score + optimistic_bound < best_score:
                return True  # 剪枝

        # 2. 硬约束已违反
        if self.has_hard_violation(current_score):
            return True

        return False

    def estimate_optimistic_bound(self, remaining_ops):
        """乐观估计：假设剩余操作都能零延迟排程"""
        return 0  # 最乐观估计
```

```python
    def evaluate_population(self, population, max_workers=4):
```

### 增量计算

当只有少量操作变更时，采用增量计算而非全量重算：

```python
class IncrementalRescheduler:
    def reschedule_affected(self, changed_ops, schedule):
        """只重排受影响的操作"""
        affected = set()

        # BFS查找受影响的操作
        queue = list(changed_ops)
        while queue:
            op_id = queue.pop(0)
            if op_id in affected:
                continue
            affected.add(op_id)

            # 后续工序受影响
            op = schedule.get_operation(op_id)
            if op.next_op_id:
                queue.append(op.next_op_id)

            # 同一资源上的后续操作受影响
            if op.assigned_resource:
                timeline = schedule.resource_timelines[op.assigned_resource]
                for start, end, tid in timeline.segments:
                    if start >= (op.end_time or datetime.max):
                        queue.append(tid)

        # 仅重排受影响的操作
        affected_ops = [schedule.get_operation(oid) for oid in affected]
        return self.local_reschedule(affected_ops, schedule)
```

## 内存模型设计

大规模排程场景下的内存管理策略：

| 数据规模 | 内存估算 | 策略 |
|----------|----------|------|
| <500操作 | <100MB | 全部内存 |
| 500~5000操作 | 100MB~1GB | 内存+分页 |
| >5000操作 | >1GB | 分片+流式处理 |

```python
class MemoryAwareScheduler:
    def schedule_large_problem(self, jobs, resources):
        """大规模排程的内存感知策略"""

        total_ops = sum(len(j.operations) for j in jobs)

        if total_ops <= 5000:
            # 小规模：全部在内存中计算
            return self.in_memory_schedule(jobs, resources)

        # 大规模：按资源组分片
        resource_groups = self.partition_resources(resources)
        partial_schedules = []

        for group in resource_groups:
            # 筛选只使用该资源组的Job
            group_jobs = self.filter_jobs_by_resource(jobs, group)

            # 分片排程
            partial = self.in_memory_schedule(group_jobs, group)
            partial_schedules.append(partial)

            # 显式释放内存
            del group_jobs
            gc.collect()

        # 合并分片结果，处理跨组依赖
        return self.merge_partial_schedules(partial_schedules)
```

## 开发者实战：从零实现一个简单排程引擎

以下是一个最小可用的排程引擎实现，涵盖核心逻辑：

```python
class SimpleScheduler:
    """最小排程引擎：支持工序先后约束和资源产能约束"""

    def __init__(self):
        self.timelines = {}  # resource_id -> [(start, end, op_id)]

    def schedule(self, jobs, resources):
        # 按优先级+交期排序
        sorted_jobs = sorted(jobs, key=lambda j: (j.priority, j.due_date))

        for job in sorted_jobs:
            for op in job.operations:
                self.assign_operation(op, job)

        return self.build_result(jobs)

    def assign_operation(self, op, job):
        earliest = self.get_earliest_start(op, job)

        best_resource = None
        best_start = None
        best_end = None

        for res_id in op.required_resources:
            start = self.find_slot(res_id, op.duration, earliest)
            if start and (best_start is None or start < best_start):
                best_resource = res_id
                best_start = start
                best_end = start + op.duration

        if best_resource:
            op.assigned_resource = best_resource
            op.start_time = best_start
            op.end_time = best_end
            op.status = OperationStatus.SCHEDULED
            self.timelines.setdefault(best_resource, []).append(
                (best_start, best_end, op.op_id)
            )

    def get_earliest_start(self, op, job):
        earliest = datetime.now()
        if op.prev_op_id:
            prev = next(o for o in job.operations if o.op_id == op.prev_op_id)
            if prev.end_time:
                earliest = max(earliest, prev.end_time)
        return earliest

    def find_slot(self, resource_id, duration, earliest):
        if resource_id not in self.timelines or not self.timelines[resource_id]:
            return earliest

        sorted_slots = sorted(self.timelines[resource_id], key=lambda s: s[0])
        current = earliest

        for slot_start, slot_end, _ in sorted_slots:
            if current + duration <= slot_start:
                return current
            current = max(current, slot_end)

        return current if current + duration <= current + duration else None
```

## 开发者实战Tips

1. **先贪心后优化**：先用贪心算法快速生成可行解，再用局部搜索优化，避免一开始就追求最优解
2. **时间用datetime**：排程时间务必使用绝对时间（datetime），避免时区和夏令时问题
3. **区间树加速**：资源时间线的重叠查询使用Interval Tree，O(log n)查询替代O(n)遍历
4. **内存池复用**：排程引擎频繁创建Operation对象时，使用对象池（Object Pool）减少GC压力
5. **断点续算**：长时间排程任务支持保存中间状态，故障恢复后从断点继续而非重新开始
6. **可视化调试**：将排程中间结果导出为甘特图JSON，在浏览器中可视化调试，比日志更直观
