# LangChain 入门实战

LangChain 是 LLM 应用开发最流行的框架，提供统一的抽象层，减少重复代码。本文聚焦实用部分：LCEL 链式调用、Memory 记忆管理和常用 Chain 模式。

## 1. 安装与配置

```bash
pip install langchain langchain-openai langchain-community python-dotenv
```

```python
# .env
OPENAI_API_KEY=sk-...
```

```python

# DeepSeek（兼容 OpenAI）

```

## 2. 基础用法：Messages

```python
from langchain_core.messages import SystemMessage, HumanMessage, AIMessage

messages = [
    SystemMessage(content="你是一个专业的代码审查员"),
    HumanMessage(content="帮我看这段代码有没有问题：\ndef div(a, b): return a/b"),
]

response = llm.invoke(messages)
print(response.content)
```

## 3. LCEL：LangChain Expression Language

LCEL 是 LangChain 的核心，用 `|` 操作符把各组件串联，清晰可读。

### 基础链

```python
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.output_parsers import StrOutputParser

# 定义各组件
prompt = ChatPromptTemplate.from_messages([
    ("system", "你是一名 {role}"),
    ("human", "{question}"),
])

parser = StrOutputParser()  # 提取 .content 字符串

# 用 | 串联成链
chain = prompt | llm | parser

# 调用
result = chain.invoke({
    "role": "Python 专家",
    "question": "解释一下 GIL 是什么",
})
print(result)
```

### 流式输出

```python
# LCEL 链天然支持流式
async for chunk in chain.astream({
    "role": "讲故事的人",
    "question": "讲一个关于程序员的短故事",
}):
    print(chunk, end="", flush=True)
```

### 并行执行（RunnableParallel）

```python
from langchain_core.runnables import RunnableParallel

parallel_chain = RunnableParallel({
    "summary": prompt_summary | llm | parser,
    "keywords": prompt_keywords | llm | parser,
    "sentiment": prompt_sentiment | llm | parser,
})

# 三个任务并发执行
result = parallel_chain.invoke({"text": "..."})
print(result["summary"])
print(result["keywords"])
```

### 链式传递（RunnablePassthrough）

```python
from langchain_core.runnables import RunnablePassthrough

# 把问题原样传下去，同时附加检索到的上下文
rag_chain = (
    {"context": retriever, "question": RunnablePassthrough()}
    | answer_prompt
    | llm
    | parser
)

answer = rag_chain.invoke("Python 的 GIL 是什么？")
```

## 4. 对话记忆（Memory）

### 方式一：手动管理历史（推荐，最透明）

```python
from langchain_core.messages import HumanMessage, AIMessage
from langchain_core.prompts import ChatPromptTemplate, MessagesPlaceholder

prompt = ChatPromptTemplate.from_messages([
    ("system", "你是一个乐于助人的助手"),
    MessagesPlaceholder(variable_name="history"),  # 插入历史消息
    ("human", "{input}"),
])

chain = prompt | llm | StrOutputParser()

# 维护历史
history = []

def chat(user_input: str) -> str:
    response = chain.invoke({
        "history": history,
        "input": user_input,
    })
    
    # 更新历史
    history.append(HumanMessage(content=user_input))
    history.append(AIMessage(content=response))
    
    return response

print(chat("我叫张三"))
print(chat("我叫什么名字？"))  # 应该能回答"张三"
```

### 方式二：InMemoryChatMessageHistory（封装版）

```python
from langchain_core.chat_history import InMemoryChatMessageHistory
from langchain_core.runnables.history import RunnableWithMessageHistory

chain = prompt | llm | parser

# 每个 session_id 对应一段对话历史
store = {}

def get_history(session_id: str) -> InMemoryChatMessageHistory:
    if session_id not in store:
        store[session_id] = InMemoryChatMessageHistory()
    return store[session_id]

# 为链附加历史管理
chain_with_history = RunnableWithMessageHistory(
    chain,
    get_history,
    input_messages_key="input",
    history_messages_key="history",
)

# 使用（指定 session_id 实现多用户隔离）
chain_with_history.invoke(
    {"input": "我的名字是张三"},
    config={"configurable": {"session_id": "user_001"}},
)
```

### 历史压缩（防止超出 token 限制）

```python
from langchain_core.messages import trim_messages

# 保留最近 N 个 token 的历史
trimmer = trim_messages(
    max_tokens=4000,
    strategy="last",  # 保留最近的
    token_counter=llm,  # 用 llm 计算 token 数
    include_system=True,  # 保留 system prompt
)

chain = (
    RunnablePassthrough.assign(history=lambda x: trimmer.invoke(x["history"]))
    | prompt
    | llm
    | parser
)
```

## 5. 常用 Chain 模式

### RAG Chain

```python
from langchain_community.vectorstores import Chroma
from langchain_openai import OpenAIEmbeddings
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.runnables import RunnablePassthrough

# 初始化向量库
embeddings = OpenAIEmbeddings(model="text-embedding-3-small")
vectorstore = Chroma(persist_directory="./chroma_db", embedding_function=embeddings)
retriever = vectorstore.as_retriever(search_kwargs={"k": 5})

# RAG Prompt
rag_prompt = ChatPromptTemplate.from_template("""
你是知识库问答助手。仅根据以下参考资料回答问题，如果资料中没有相关信息，请说"我在知识库中没有找到相关信息"。

参考资料：
{context}

问题：{question}
""")

def format_docs(docs):
    return "\n\n".join(doc.page_content for doc in docs)

rag_chain = (
    {"context": retriever | format_docs, "question": RunnablePassthrough()}
    | rag_prompt
    | llm
    | StrOutputParser()
)

answer = rag_chain.invoke("什么是向量数据库？")
```

### 文档摘要链

```python
from langchain.chains.summarize import load_summarize_chain
from langchain_community.document_loaders import PyPDFLoader
from langchain.text_splitter import RecursiveCharacterTextSplitter

loader = PyPDFLoader("长文档.pdf")
docs = loader.load()

splitter = RecursiveCharacterTextSplitter(chunk_size=2000, chunk_overlap=200)
split_docs = splitter.split_documents(docs)

# Map-Reduce 摘要（适合长文档）
chain = load_summarize_chain(llm, chain_type="map_reduce")
summary = chain.invoke(split_docs)
print(summary["output_text"])
```

### Sequential Chain（顺序处理）

```python
# 第一步：将代码翻译成英文注释
translate_prompt = ChatPromptTemplate.from_template(
    "为以下代码添加英文注释：\n{code}"
)

# 第二步：代码审查
review_prompt = ChatPromptTemplate.from_template(
    "审查以下带注释的代码，找出潜在问题：\n{annotated_code}"
)

# 串联两步
pipeline = (
    translate_prompt | llm | parser
    | (lambda text: {"annotated_code": text})  # 转换输出格式
    | review_prompt | llm | parser
)

result = pipeline.invoke({"code": "def f(x): return x**2 if x else None"})
```

## 6. 工具调用（Tools）

```python
from langchain_core.tools import tool
from langchain_openai import ChatOpenAI

@tool
def get_weather(city: str) -> str:
    """获取指定城市的当前天气"""
    # 实际中调用天气 API
    return f"{city} 今天晴，气温 25°C"

@tool
def calculate(expression: str) -> float:
    """计算数学表达式"""
    return eval(expression)  # 生产环境用安全的计算库

# 绑定工具到模型
llm_with_tools = llm.bind_tools([get_weather, calculate])

# 会自动判断是否调用工具
messages = [HumanMessage(content="北京现在天气怎么样？")]
response = llm_with_tools.invoke(messages)

# 处理工具调用
if response.tool_calls:
    for tool_call in response.tool_calls:
        if tool_call["name"] == "get_weather":
            result = get_weather.invoke(tool_call["args"])
            print(f"工具返回: {result}")
```

## 7. 避坑与最佳实践

| 坑 | 解法 |
|----|------|
| `invoke` 调用很慢 | 改用 `ainvoke` 异步调用 |
| 历史消息 token 溢出 | 用 `trim_messages` 或摘要压缩 |
| LangChain 版本升级 API 变化 | 锁定版本，用 `langchain-core` 替代 `langchain` |
| DeepSeek 不支持 Structured Output | 用 `JsonOutputParser` + Prompt 约束 |
| 链复杂后难以调试 | 用 `LangSmith` 追踪每一步的输入输出 |

### 启用 LangSmith 调试追踪

```python
import os
os.environ["LANGCHAIN_TRACING_V2"] = "true"
os.environ["LANGCHAIN_API_KEY"] = "ls-..."  # LangSmith API Key

# 之后的所有链调用都会自动上传到 LangSmith 可视化
result = chain.invoke({"question": "..."})
```
