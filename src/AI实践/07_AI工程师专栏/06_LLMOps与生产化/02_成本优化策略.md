# 成本优化：Token 经济学与缓存策略

## Token 成本分析

LLM 应用的运营成本主要由 Token 消耗决定。优化成本的第一步是搞清楚钱花在哪里。

### 成本构成

```
总成本 = 输入 Token 成本 + 输出 Token 成本 + 基础设施成本
```

| 成本项 | 占比 | 优化空间 |
|--------|------|----------|
| 输入 Token | 40-60% | 高（缓存、压缩） |
| 输出 Token | 30-40% | 中（控制长度） |
| 基础设施 | 10-20% | 低（弹性伸缩） |

### 成本审计

```python
from dataclasses import dataclass
from datetime import datetime
import json

@dataclass
class TokenCostConfig:
    """Token 定价配置（单位：美元/百万 Token）"""
    input_price_per_m: float
    output_price_per_m: float

# 主流模型定价参考
# ⚠️ 注意：模型定价随时间变化，以下为 2025 年参考值
# 生产环境应从 API Provider 获取实时定价
MODEL_PRICING = {
    "gpt-4o": TokenCostConfig(2.5, 10.0),
    "gpt-4o-mini": TokenCostConfig(0.15, 0.60),
    "claude-3.5-sonnet": TokenCostConfig(3.0, 15.0),
    "claude-3-haiku": TokenCostConfig(0.25, 1.25),
    "deepseek-chat": TokenCostConfig(0.14, 0.28),
    "qwen-max": TokenCostConfig(1.6, 6.4),
}

class CostAnalyzer:
    """Token 成本分析器"""

    def __init__(self):
        self.records: list[dict] = []

    def record_request(self, model: str, prompt_tokens: int,
                       completion_tokens: int, endpoint: str = ""):
        pricing = MODEL_PRICING.get(model)
        if not pricing:
            return

        input_cost = prompt_tokens * pricing.input_price_per_m / 1_000_000
        output_cost = completion_tokens * pricing.output_price_per_m / 1_000_000

        self.records.append({
            "timestamp": datetime.now().isoformat(),
            "model": model,
            "endpoint": endpoint,
            "prompt_tokens": prompt_tokens,
            "completion_tokens": completion_tokens,
            "input_cost": input_cost,
            "output_cost": output_cost,
            "total_cost": input_cost + output_cost,
        })

    def report(self, group_by: str = "model") -> dict:
        """生成成本报告"""
        if not self.records:
            return {}

        total_cost = sum(r["total_cost"] for r in self.records)
        total_input = sum(r["prompt_tokens"] for r in self.records)
        total_output = sum(r["completion_tokens"] for r in self.records)

        # 按维度分组
        groups: dict[str, list] = {}
        for r in self.records:
            key = r.get(group_by, "unknown")
            groups.setdefault(key, []).append(r)

        breakdown = {}
        for key, records in groups.items():
            breakdown[key] = {
                "total_cost": sum(r["total_cost"] for r in records),
                "input_tokens": sum(r["prompt_tokens"] for r in records),
                "output_tokens": sum(r["completion_tokens"] for r in records),
                "request_count": len(records),
                "avg_cost_per_request": sum(r["total_cost"] for r in records) / len(records),
            }

        return {
            "total_cost": total_cost,
            "total_input_tokens": total_input,
            "total_output_tokens": total_output,
            "total_requests": len(self.records),
            "breakdown": breakdown,
        }
```

## 语义缓存

重复或相似的问题消耗大量 Token。语义缓存通过向量匹配复用历史回答。

### 向量检索语义缓存

# ❌ 旧方案：redis.keys("sem_cache:*") + 逐条比较
#    时间复杂度 O(N)，10000 条缓存时每次查询需遍历全部
# ✅ 新方案：Chroma 向量索引 + Redis 元数据
#    时间复杂度 O(log N)，通过 HNSW 近似最近邻检索

```python
import hashlib
import redis

class SemanticCache:
    """基于向量检索的语义缓存，避免 O(N) 全表扫描"""

    def __init__(self, similarity_threshold: float = 0.95,
                 ttl: int = 3600):
        self.threshold = similarity_threshold
        self.ttl = ttl
        # 使用 Chroma 作为向量索引，实现 O(log N) 检索
        import chromadb
        self.chroma = chromadb.Client()
        self.collection = self.chroma.get_or_create_collection(
            name="semantic_cache",
            metadata={"hnsw:space": "cosine"}
        )
        # Redis 用于存储完整响应（向量库只存 embedding + ID）
        self.redis = redis.Redis(decode_responses=True)

    def get(self, query: str, embed_fn) -> str | None:
        """O(log N) 语义检索，而非 O(N) 全表扫描"""
        query_embedding = embed_fn(query)

        # 向量相似度检索，返回最近的候选
        results = self.collection.query(
            query_embeddings=[query_embedding],
            n_results=1  # 只需最近邻
        )

        if results["distances"] and results["distances"][0]:
            distance = results["distances"][0][0]
            # cosine distance: 0 = identical, 2 = opposite
            similarity = 1 - distance

            if similarity >= self.threshold:
                cache_id = results["ids"][0][0]
                cached = self.redis.getex(
                    f"cache:{cache_id}",
                    ex=self.ttl
                )
                if cached:
                    return cached

        return None

    def set(self, query: str, response: str, embed_fn):
        """缓存查询-响应对"""
        query_embedding = embed_fn(query)
        cache_id = f"q_{hashlib.md5(query.encode()).hexdigest()}"

        # 存入向量索引
        self.collection.add(
            ids=[cache_id],
            embeddings=[query_embedding],
            documents=[query]
        )

        # 存入 Redis（含 TTL 自动过期）
        self.redis.setex(
            f"cache:{cache_id}",
            self.ttl,
            response
        )
```

### 集成到 LLM 调用

```python
class CachedLLMClient:
    """带缓存的 LLM 客户端"""

    def __init__(self, cache: SemanticCache, model: str = "gpt-4o"):
        self.cache = cache
        self.model = model
        from openai import OpenAI
        self.client = OpenAI()
        self.stats = {"cache_hits": 0, "cache_misses": 0}

    def chat(self, messages: list[dict], **kwargs) -> str:
        # 提取用户查询作为缓存键
        user_query = next(
            (m["content"] for m in messages if m["role"] == "user"), ""
        )

        # 查缓存
        cached = self.cache.get(user_query)
        if cached:
            self.stats["cache_hits"] += 1
            return cached

        # 调用 LLM
        self.stats["cache_misses"] += 1
        response = self.client.chat.completions.create(
            model=self.model,
            messages=messages,
            **kwargs,
        )
        result = response.choices[0].message.content

        # 写缓存
        self.cache.set(user_query, result)

        return result

    def cache_hit_rate(self) -> float:
        total = self.stats["cache_hits"] + self.stats["cache_misses"]
        return self.stats["cache_hits"] / total if total else 0
```

## 模型路由

不是所有请求都需要最强模型。根据任务复杂度路由到不同模型，可大幅降低成本。

```python
import re

class ModelRouter:
    """基于规则的模型路由器"""

    # 模型成本倍数（相对于最便宜模型）
    MODEL_COST = {
        "gpt-4o-mini": 1.0,
        "gpt-4o": 16.7,
        "deepseek-chat": 0.9,
    }

    def __init__(self):
        self.rules = [
            # (匹配函数, 推荐模型)
            (self._is_simple_qa, "gpt-4o-mini"),
            (self._is_translation, "gpt-4o-mini"),
            (self._is_summarization, "gpt-4o-mini"),
            (self._is_code_generation, "deepseek-chat"),
            (self._is_complex_reasoning, "gpt-4o"),
            (self._is_creative_writing, "gpt-4o"),
        ]

    def route(self, query: str, context: dict = None) -> str:
        """根据查询内容路由到合适的模型"""
        for matcher, model in self.rules:
            if matcher(query, context):
                return model
        return "gpt-4o-mini"  # 默认使用便宜模型

    def _is_simple_qa(self, query: str, context: dict = None) -> bool:
        """简单问答：短问题，无复杂推理"""
        if len(query) < 50:
            simple_patterns = [
                r"什么是", r"定义", r"意思", r"how to",
                r"what is", r"define",
            ]
            return any(re.search(p, query.lower()) for p in simple_patterns)
        return False

    def _is_translation(self, query: str, context: dict = None) -> bool:
        """翻译任务"""
        patterns = [r"翻译", r"translate", r"译为"]
        return any(re.search(p, query.lower()) for p in patterns)

    def _is_summarization(self, query: str, context: dict = None) -> bool:
        """摘要任务"""
        patterns = [r"总结", r"摘要", r"summarize", r"summarise"]
        return any(re.search(p, query.lower()) for p in patterns)

    def _is_code_generation(self, query: str, context: dict = None) -> bool:
        """代码生成"""
        patterns = [r"写.*代码", r"实现.*函数", r"write.*code", r"implement"]
        return any(re.search(p, query.lower()) for p in patterns)

    def _is_complex_reasoning(self, query: str, context: dict = None) -> bool:
        """复杂推理"""
        patterns = [r"分析", r"比较.*区别", r"为什么", r"analyze", r"compare"]
        return any(re.search(p, query.lower()) for p in patterns) and len(query) > 100

    def _is_creative_writing(self, query: str, context: dict = None) -> bool:
        """创意写作"""
        patterns = [r"写.*文章", r"创作", r"write.*essay", r"creative"]
        return any(re.search(p, query.lower()) for p in patterns)
```

### 进阶：基于分类器的路由

```python
class ClassifierModelRouter:
    """基于轻量分类器的动态路由"""

    def __init__(self, classifier_model: str = "gpt-4o-mini"):
        self.classifier_model = classifier_model
        from openai import OpenAI
        self.client = OpenAI()

    def route(self, query: str) -> tuple[str, float]:
        """分类并路由"""
        response = self.client.chat.completions.create(
            model=self.classifier_model,
            messages=[{
                "role": "user",
                "content": f"""对以下用户查询进行复杂度分类。

查询: {query}

请输出 JSON:
{{"complexity": "simple|medium|complex", "confidence": 0.0-1.0}}

分类标准:
- simple: 简单问答、翻译、格式转换
- medium: 摘要、代码生成、数据分析
- complex: 深度分析、创意写作、多步推理"""
            }],
            response_format={"type": "json_object"},
            temperature=0,
        )

        result = json.loads(response.choices[0].message.content)
        complexity = result["complexity"]
        confidence = result["confidence"]

        model_map = {
            "simple": "gpt-4o-mini",
            "medium": "deepseek-chat",
            "complex": "gpt-4o",
        }

        return model_map.get(complexity, "gpt-4o"), confidence
```

## Prompt 长度优化

### 系统提示压缩

```python
class PromptOptimizer:
    """Prompt 压缩与优化"""

    def compress_system_prompt(self, prompt: str, max_tokens: int = 500) -> str:
        """压缩系统提示，保留关键信息"""
        # 1. 去除冗余空白
        compressed = re.sub(r'\n{3,}', '\n\n', prompt)
        compressed = re.sub(r' {2,}', ' ', compressed)

        # 2. 合并重复指令
        lines = compressed.split('\n')
        seen = set()
        deduped = []
        for line in lines:
            normalized = line.strip().lower()
            if normalized not in seen:
                seen.add(normalized)
                deduped.append(line)
        compressed = '\n'.join(deduped)

        # 3. 精简表述
        replacements = {
            "请务必确保": "确保",
            "你需要注意的是": "注意",
            "请按照以下步骤": "步骤",
            "please make sure that": "ensure",
            "it is important to note that": "note:",
        }
        for old, new in replacements.items():
            compressed = compressed.replace(old, new)

        return compressed

    def estimate_tokens(self, text: str) -> int:
        """估算 Token 数（中文约 1.5 字/Token，英文约 4 字符/Token）"""
        chinese_chars = sum(1 for c in text if '\u4e00' <= c <= '\u9fff')
        english_chars = len(text) - chinese_chars
        return int(chinese_chars / 1.5 + english_chars / 4)
```

### Few-Shot 示例动态选择

```python
class DynamicFewShot:
    """动态 Few-Shot：只选择最相关的示例"""

    def __init__(self, examples: list[dict], embedder, max_examples: int = 3):
        self.examples = examples
        self.embedder = embedder
        self.max_examples = max_examples
        # 预计算示例的 embedding
        self.example_embeddings = [
            self.embedder.embed_query(ex["input"]) for ex in examples
        ]

    def select_examples(self, query: str) -> list[dict]:
        """根据查询选择最相关的示例"""
        query_embedding = self.embedder.embed_query(query)

        scores = []
        for i, ex_emb in enumerate(self.example_embeddings):
            similarity = np.dot(query_embedding, ex_emb) / (
                np.linalg.norm(query_embedding) * np.linalg.norm(ex_emb)
            )
            scores.append((similarity, i))

        # 按相似度排序，取 top-k
        scores.sort(reverse=True)
        selected = [self.examples[idx] for _, idx in scores[:self.max_examples]]
        return selected
```

## ROI 计算

```python
def calculate_llm_roi(
    monthly_token_cost: float,
    monthly_infra_cost: float,
    monthly_labor_cost: float,
    revenue_attribution: float,
    cost_savings: float,
    time_savings_hours: float,
    avg_hourly_rate: float = 200,
) -> dict:
    """计算 LLM 应用的投资回报率"""
    total_investment = monthly_token_cost + monthly_infra_cost + monthly_labor_cost
    total_return = revenue_attribution + cost_savings + (time_savings_hours * avg_hourly_rate)

    roi = (total_return - total_investment) / total_investment * 100
    payback_months = total_investment / total_return if total_return > 0 else float('inf')

    return {
        "monthly_investment": total_investment,
        "monthly_return": total_return,
        "monthly_net": total_return - total_investment,
        "roi_percent": roi,
        "payback_months": round(payback_months, 1),
        "breakdown": {
            "token_cost": monthly_token_cost,
            "infra_cost": monthly_infra_cost,
            "labor_cost": monthly_labor_cost,
            "revenue": revenue_attribution,
            "cost_savings": cost_savings,
            "time_savings_value": time_savings_hours * avg_hourly_rate,
        }
    }
```

## 优化效果对比

| 优化手段 | 成本降低 | 实施难度 | 风险 |
|----------|----------|----------|------|
| 语义缓存 | 30-50% | 中 | 缓存过期导致回答过时 |
| 模型路由 | 40-60% | 中 | 简单模型误判复杂问题 |
| Prompt 压缩 | 10-20% | 低 | 关键信息被压缩掉 |
| 输出长度控制 | 5-15% | 低 | 回答不完整 |
| 自建模型 | 60-80% | 高 | 初期投入大，维护成本高 |

## 实施路径

1. **第一周**：建立成本监控，搞清楚钱花在哪里
2. **第二周**：实施语义缓存，这是 ROI 最高的优化
3. **第三周**：上线模型路由，简单请求用便宜模型
4. **第四周**：Prompt 优化，压缩 Token 消耗
5. **持续**：定期审查成本报告，发现新的优化点
