### 定价引擎核心实现

```python
import numpy as np
from pydantic import BaseModel, Field
from typing import Optional, List
from enum import Enum


class PricingStrategy(str, Enum):
    COST_PLUS = "cost_plus"
    COMPETITIVE = "competitive"
    ELASTICITY = "elasticity"
    DYNAMIC = "dynamic"


class PricingRequest(BaseModel):
    sku_id: str
    current_price: float = Field(..., gt=0)
    cost: float = Field(..., gt=0)
    strategy: PricingStrategy = PricingStrategy.ELASTICITY
    competitor_prices: Optional[List[float]] = None
    elasticity: Optional[float] = None
    current_demand: Optional[float] = None
    target_margin: float = Field(default=0.15, gt=0, lt=1)
    max_price_change: float = Field(default=0.10, gt=0, lt=0.5)


class PricingResult(BaseModel):
    sku_id: str
    suggested_price: float
    current_price: float
    price_change: float
    price_change_pct: float
    expected_margin: float
    expected_demand_change: float
    strategy_used: str
    confidence: float
    reasoning: str


class DynamicPricingEngine:
    """动态定价引擎"""

    def calculate_price(self, req: PricingRequest) -> PricingResult:
        if req.strategy == PricingStrategy.COST_PLUS:
            return self._cost_plus(req)
        elif req.strategy == PricingStrategy.COMPETITIVE:
            return self._competitive(req)
        elif req.strategy == PricingStrategy.ELASTICITY:
            return self._elasticity(req)
        elif req.strategy == PricingStrategy.DYNAMIC:
            return self._dynamic(req)
        raise ValueError(f"不支持的策略: {req.strategy}")

    def _cost_plus(self, req: PricingRequest) -> PricingResult:
        suggested = req.cost / (1 - req.target_margin)
        suggested = self._clamp(suggested, req)
        margin = (suggested - req.cost) / suggested
        return PricingResult(
            sku_id=req.sku_id,
            suggested_price=round(suggested, 2),
            current_price=req.current_price,
            price_change=round(suggested - req.current_price, 2),
            price_change_pct=round((suggested - req.current_price) / req.current_price * 100, 2),
            expected_margin=round(margin, 4),
            expected_demand_change=0.0,
            strategy_used="cost_plus",
            confidence=0.6,
            reasoning=f"基于成本{req.cost:.2f}和目标毛利率{req.target_margin:.0%}计算",
        )

    def _competitive(self, req: PricingRequest) -> PricingResult:
        if not req.competitor_prices:
            return self._cost_plus(req)
        avg = np.mean(req.competitor_prices)
        suggested = avg * 0.98
        suggested = self._clamp(suggested, req)
        margin = (suggested - req.cost) / suggested
        return PricingResult(
            sku_id=req.sku_id,
            suggested_price=round(suggested, 2),
            current_price=req.current_price,
            price_change=round(suggested - req.current_price, 2),
            price_change_pct=round((suggested - req.current_price) / req.current_price * 100, 2),
            expected_margin=round(margin, 4),
            expected_demand_change=0.0,
            strategy_used="competitive",
            confidence=0.7,
            reasoning=f"基于{len(req.competitor_prices)}个竞品均价{avg:.2f}",
        )

    def _elasticity(self, req: PricingRequest) -> PricingResult:
        e = req.elasticity if req.elasticity else -1.5
        if e < -1:
            optimal = req.cost * e / (e + 1)
        else:
            optimal = req.current_price
        suggested = self._clamp(optimal, req)
        margin = (suggested - req.cost) / suggested
        demand_change = e * (suggested / req.current_price - 1)
        return PricingResult(
            sku_id=req.sku_id,
            suggested_price=round(suggested, 2),
            current_price=req.current_price,
            price_change=round(suggested - req.current_price, 2),
            price_change_pct=round((suggested - req.current_price) / req.current_price * 100, 2),
            expected_margin=round(margin, 4),
            expected_demand_change=round(demand_change, 4),
            strategy_used="elasticity",
            confidence=0.75,
            reasoning=f"基于价格弹性{e:.2f}的利润最大化定价",
        )

    def _dynamic(self, req: PricingRequest) -> PricingResult:
        results = [self._cost_plus(req)]
        if req.competitor_prices:
            results.append(self._competitive(req))
        if req.elasticity:
            results.append(self._elasticity(req))
        weights = [0.2, 0.3, 0.5][:len(results)]
        total_w = sum(weights)
        suggested = sum(r.suggested_price * w for r, w in zip(results, weights)) / total_w
        suggested = self._clamp(suggested, req)
        margin = (suggested - req.cost) / suggested
        return PricingResult(
            sku_id=req.sku_id,
            suggested_price=round(suggested, 2),
            current_price=req.current_price,
            price_change=round(suggested - req.current_price, 2),
            price_change_pct=round((suggested - req.current_price) / req.current_price * 100, 2),
            expected_margin=round(margin, 4),
            expected_demand_change=0.0,
            strategy_used="dynamic",
            confidence=0.8,
            reasoning=f"综合{len(results)}种策略加权",
        )

    def _clamp(self, price: float, req: PricingRequest) -> float:
        lo = req.current_price * (1 - req.max_price_change)
        hi = req.current_price * (1 + req.max_price_change)
        lo = max(lo, req.cost * 1.02)
        return max(min(price, hi), lo)
```

```python
@router.get("/alerts", response_model=InventoryAlertsResponse, summary="库存预警")
async def get_inventory_alerts(
    warehouse_id: int = Query(default=1, description="仓库ID"),
    alert_type: str = Query(default="all", description="预警类型: all/stockout/surplus/slow"),
    db: AsyncSession = Depends(get_db),
):
    """查询库存预警列表"""
    stmt = select(Inventory).where(Inventory.warehouse_id == warehouse_id)
    result = await db.execute(stmt)
    inventories = result.scalars().all()

    alerts = []
    # 优化：批量查询替代N+1查询
    sku_ids = [inv.sku_id for inv in inventories]
    avg_demands = await _batch_get_avg_demand(db, sku_ids, warehouse_id)
    sku_map = await _batch_get_skus(db, sku_ids)

    for inv in inventories:
        sku = sku_map.get(inv.sku_id)
        if not sku:
            continue

        inv_position = inv.available_qty + inv.in_transit_qty

        # 缺货预警
        if inv_position <= inv.safety_stock or inv_position <= inv.reorder_point:
            avg_demand = avg_demands.get(inv.sku_id, 1.0)
            days_left = inv.available_qty / max(avg_demand, 1)
            alerts.append(InventoryAlertItem(
                sku_id=inv.sku_id,
                sku_code=sku.sku_code,
                warehouse_id=inv.warehouse_id,
                available_qty=inv.available_qty,
                safety_stock=inv.safety_stock,
                alert_type="stockout",
                days_until_stockout=round(days_left, 1),
            ))
        # 积压预警
        elif inv.available_qty > inv.safety_stock * 3:
            alerts.append(InventoryAlertItem(
                sku_id=inv.sku_id,
                sku_code=sku.sku_code,
                warehouse_id=inv.warehouse_id,
                available_qty=inv.available_qty,
                safety_stock=inv.safety_stock,
                alert_type="surplus",
                days_until_stockout=None,
            ))

    # 过滤
    if alert_type != "all":
        alerts = [a for a in alerts if a.alert_type == alert_type]

    return InventoryAlertsResponse(total=len(alerts), alerts=alerts)

            continue

        avg_demand = await _get_avg_demand(db, inv.sku_id, inv.warehouse_id)
        turnover_days = inv.available_qty / max(avg_demand, 0.1) if avg_demand > 0 else 999
        abc, xyz = await _get_abc_xyz_class(db, inv.sku_id)

        stock_value = inv.available_qty * float(sku.cost_price)
        total_value += stock_value
        total_turnover += turnover_days

        if turnover_days < 30:
            health = "healthy"
        elif turnover_days < 60:
            health = "warning"
        elif turnover_days < 90:
            health = "slow"
        else:
            health = "overstock"

        items.append(InventoryAnalysisItem(
            sku_id=inv.sku_id,
            sku_code=sku.sku_code,
            available_qty=inv.available_qty,

            turnover_days=round(turnover_days, 1),
            abc_class=abc,
            xyz_class=xyz,
            health_status=health,
        ))

    avg_turnover = total_turnover / max(len(items), 1)

    return InventoryAnalysisResponse(
        total_skus=len(items),
        total_value=round(total_value, 2),
        avg_turnover_days=round(avg_turnover, 1),
        items=items,
```

## 安全库存计算引擎

```python
import numpy as np
from typing import Dict


class SafetyStockCalculator:
    """安全库存计算器"""

    Z_SCORES = {
        0.85: 1.04, 0.90: 1.28, 0.95: 1.65,
        0.97: 1.88, 0.98: 2.05, 0.99: 2.33,
    }

    @classmethod
    def get_z(cls, service_level: float) -> float:
        if service_level in cls.Z_SCORES:
            return cls.Z_SCORES[service_level]
        levels = sorted(cls.Z_SCORES.keys())
        for i in range(len(levels) - 1):
            if levels[i] <= service_level <= levels[i + 1]:
                ratio = (service_level - levels[i]) / (levels[i + 1] - levels[i])
                return cls.Z_SCORES[levels[i]] + ratio * (cls.Z_SCORES[levels[i + 1]] - cls.Z_SCORES[levels[i]])
        return 2.33

    @classmethod
    def calculate(
        cls,
        avg_daily_demand: float,
        demand_std: float,
        avg_lead_time: float,
        lead_time_std: float,
        service_level: float = 0.95,
    ) -> Dict:
        z = cls.get_z(service_level)
        sigma_l = np.sqrt(avg_lead_time * demand_std ** 2 + avg_daily_demand ** 2 * lead_time_std ** 2)
        ss = z * sigma_l
        lt_demand = avg_daily_demand * avg_lead_time
        rop = lt_demand + ss

        return {
            "safety_stock": round(ss, 1),
            "reorder_point": round(rop, 1),
            "lead_time_demand": round(lt_demand, 1),
            "z_score": round(z, 2),
            "service_level": service_level,
        }
```

## ABC-XYZ分类与策略匹配

```python
import numpy as np
from typing import Tuple


class ABCXYZClassifier:
    """ABC-XYZ分类器"""

    ABC_THRESHOLDS = (0.80, 0.95)
    XYZ_THRESHOLDS = (0.5, 1.0)

    @classmethod
    def classify(
        cls,
        sales_value: float,
        total_value: float,
        cumulative_ratio: float,
        sales_series: list,
    ) -> Tuple[str, str]:
        """
        对商品进行ABC-XYZ分类

        Args:
            sales_value: 该商品销售价值
            total_value: 总销售价值
            cumulative_ratio: 累计价值占比
            sales_series: 销量序列

        Returns:
            (ABC分类, XYZ分类)
        """
        # ABC分类
        if cumulative_ratio <= cls.ABC_THRESHOLDS[0]:
            abc = "A"
        elif cumulative_ratio <= cls.ABC_THRESHOLDS[1]:
            abc = "B"
        else:
            abc = "C"

        # XYZ分类(基于变异系数)
        if len(sales_series) < 2 or np.mean(sales_series) == 0:
            xyz = "Z"
        else:
            cv = float(np.std(sales_series) / np.mean(sales_series))
            if cv < cls.XYZ_THRESHOLDS[0]:
                xyz = "X"
            elif cv < cls.XYZ_THRESHOLDS[1]:
                xyz = "Y"
            else:
                xyz = "Z"

        return abc, xyz

    @classmethod
    def get_strategy(cls, abc: str, xyz: str) -> dict:
        """获取分类对应的库存策略"""
        strategies = {
            "AX": {"method": "continuous_ss", "service_level": 0.98, "review": "daily", "forecast": "sku_level"},
            "AY": {"method": "continuous_ss", "service_level": 0.95, "review": "daily", "forecast": "sku_level"},
            "AZ": {"method": "make_to_order", "service_level": 0.90, "review": "daily", "forecast": "safety_min"},
            "BX": {"method": "periodic_rs", "service_level": 0.95, "review": "weekly", "forecast": "category_level"},
            "BY": {"method": "periodic_rs", "service_level": 0.90, "review": "weekly", "forecast": "category_level"},
            "BZ": {"method": "periodic_rs", "service_level": 0.85, "review": "weekly", "forecast": "minimal"},
            "CX": {"method": "periodic_rs", "service_level": 0.90, "review": "biweekly", "forecast": "aggregate"},
            "CY": {"method": "periodic_rs", "service_level": 0.85, "review": "biweekly", "forecast": "aggregate"},
            "CZ": {"method": "make_to_order", "service_level": 0.80, "review": "monthly", "forecast": "none"},
        }
        return strategies.get(f"{abc}{xyz}", strategies["BY"])
```

```python
class ReplenishPlanGenerator:
    """补货计划自动生成器"""

        self.db = db
        self.engine = InventoryOptimizerEngine()

        result = await self.db.execute(stmt)
        inventories = result.scalars().all()

        plans = []
        for inv in inventories:
            plan = await self._generate_single(inv)
            if plan:
                plans.append(plan)

        return plans

    async def _generate_single(self, inv) -> ReplenishPlan:
        """为单个库存记录生成补货计划"""
        from app.services.inventory_optimizer import (
            ReplenishRequest, ReplenishStrategy, SafetyStockCalculator,
        )

        # 获取需求统计(简化)
        avg_demand = 10.0  # 实际应从销售数据计算
        demand_std = 5.0
        lead_time = 7.0
        lead_time_std = 1.5

        # 计算安全库存
        ss_result = SafetyStockCalculator.calculate(
            avg_daily_demand=avg_demand,
            demand_std=demand_std,
            avg_lead_time=lead_time,
            lead_time_std=lead_time_std,
        )

        inv_position = inv.available_qty + inv.in_transit_qty
```
