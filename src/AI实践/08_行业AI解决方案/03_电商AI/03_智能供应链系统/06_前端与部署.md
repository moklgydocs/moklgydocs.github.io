# 前端与部署

## React前端架构

前端采用React + Ant Design Pro + ECharts构建，提供供应链智能系统的可视化操作界面。

### 技术栈

| 技术 | 版本 | 用途 |
|------|------|------|
| React | 18 | UI框架 |
| TypeScript | 5 | 类型安全 |
| Ant Design | 5 | UI组件库 |
| Ant Design Pro | 6 | 企业级脚手架 |
| ECharts | 5 | 数据可视化 |
| Axios | 1 | HTTP客户端 |
| React Router | 6 | 路由管理 |
| Zustand | 4 | 状态管理 |

```typescript
// src/services/forecast.ts
import request from '../utils/request';

export interface ForecastDataPoint {
  date: string;
  predicted: number;
  predicted_lower: number;
  predicted_upper: number;
  actual?: number;
}

export interface ForecastResponse {
  sku_id: number;
  warehouse_id: number;
  model_type: string;
  forecast_days: number;
  data: ForecastDataPoint[];
}

export interface AccuracyMetrics {
  sku_id: number;
  model_type: string;
  mape: number | null;
  rmse: number | null;
  wmape: number | null;
  mae: number | null;
  sample_count: number;
}

export const forecastService = {
  predict: (skuId: number, days: number = 30, modelType: string = 'prophet') =>
    request.post<any, ForecastResponse>('/api/v1/forecast/predict', {
      sku_id: skuId,
      days,
      model_type: modelType,
    }),

  getResults: (skuId: number, page: number = 1) =>
    request.get('/api/v1/forecast/results', {
      params: { sku_id: skuId, page, page_size: 30 },
    }),

  getAccuracy: (skuId: number, modelType?: string) =>
    request.get('/api/v1/forecast/accuracy', {
      params: { sku_id: skuId, model_type: modelType },
    }),

  train: (skuId: number, modelType: string = 'prophet') =>
    request.post('/api/v1/forecast/train', {
      sku_id: skuId,
      model_type: modelType,
    }),
};
```

### ForecastDashboard（预测看板）

```tsx
// src/pages/ForecastDashboard/index.tsx
import React, { useState, useEffect } from 'react';
import { Card, Select, Row, Col, Statistic, Table, Tag, Space, Spin } from 'antd';
import {
  LineChartOutlined,
  RiseOutlined,
  FallOutlined,
} from '@ant-design/icons';
import ReactECharts from 'echarts-for-react';
import { forecastService, ForecastDataPoint, AccuracyMetrics } from '../../services/forecast';

const ForecastDashboard: React.FC = () => {
  const [skuId, setSkuId] = useState<number>(1);
  const [forecastData, setForecastData] = useState<ForecastDataPoint[]>([]);
  const [accuracy, setAccuracy] = useState<AccuracyMetrics | null>(null);
  const [loading, setLoading] = useState(false);

  useEffect(() => {
    loadForecastData();
  }, [skuId]);

  const loadForecastData = async () => {
    setLoading(true);
    try {
      const [forecastRes, accuracyRes] = await Promise.all([
        forecastService.predict(skuId, 30, 'prophet'),
        forecastService.getAccuracy(skuId),
      ]);
      setForecastData(forecastRes.data || []);
      setAccuracy(accuracyRes);
    } catch (error) {
      console.error('加载预测数据失败:', error);
    } finally {
      setLoading(false);
    }
  };

  const getChartOption = () => {
    const dates = forecastData.map((d) => d.date);
    const predicted = forecastData.map((d) => d.predicted);
    const lower = forecastData.map((d) => d.predicted_lower);
    const upper = forecastData.map((d) => d.predicted_upper);
    const actual = forecastData.map((d) => d.actual);

    return {
      tooltip: { trigger: 'axis' },
      legend: {
        data: ['预测值', '置信下限', '置信上限', '实际值'],
      },
      grid: { left: '3%', right: '4%', bottom: '3%', containLabel: true },
      xAxis: { type: 'category', data: dates },
      yAxis: { type: 'value', name: '销量' },
      series: [
        {
          name: '预测值',
          type: 'line',
          data: predicted,
          lineStyle: { width: 3 },
          itemStyle: { color: '#1890ff' },
        },
        {
          name: '置信下限',
          type: 'line',
          data: lower,
          lineStyle: { type: 'dashed', width: 1 },
          itemStyle: { color: '#91d5ff' },
          areaStyle: { opacity: 0.1 },
        },
        {
          name: '置信上限',
          type: 'line',
          data: upper,
          lineStyle: { type: 'dashed', width: 1 },
          itemStyle: { color: '#91d5ff' },
          areaStyle: { opacity: 0.1 },
        },
        {
          name: '实际值',
          type: 'line',
          data: actual,
          lineStyle: { width: 2 },
          itemStyle: { color: '#52c41a' },
        },
      ],
    };
  };

  return (
    <div>
      <Card title="需求预测看板" extra={
        <Space>
          <span>SKU:</span>
          <Select
            value={skuId}
            onChange={setSkuId}
            style={{ width: 200 }}
            options={[
              { value: 1, label: 'SKU001-01 蓝牙耳机(黑)' },
              { value: 2, label: 'SKU001-02 蓝牙耳机(白)' },
              { value: 3, label: 'SKU002-01 智能手表(黑)' },
              { value: 4, label: 'SKU003-01 有机牛奶' },
            ]}
          />
        </Space>
      }>
        <Row gutter={16} style={{ marginBottom: 24 }}>
          <Col span={6}>
            <Statistic
              title="MAPE"
              value={accuracy?.mape || '-'}
              suffix="%"
              prefix={accuracy && accuracy.mape && accuracy.mape < 20 ? <RiseOutlined /> : <FallOutlined />}
              valueStyle={{ color: accuracy && accuracy.mape && accuracy.mape < 20 ? '#3f8600' : '#cf1322' }}
            />
          </Col>
          <Col span={6}>
            <Statistic title="RMSE" value={accuracy?.rmse || '-'} />
          </Col>
          <Col span={6}>
            <Statistic title="WMAPE" value={accuracy?.wmape || '-'} suffix="%" />
          </Col>
          <Col span={6}>
            <Statistic title="样本数" value={accuracy?.sample_count || '-'} />
          </Col>
        </Row>

        <Spin spinning={loading}>
          <ReactECharts option={getChartOption()} style={{ height: 400 }} />
        </Spin>
      </Card>
    </div>
  );
};

export default ForecastDashboard;
```

### InventoryView（库存视图）

```tsx
// src/pages/InventoryView/index.tsx
import React, { useState, useEffect } from 'react';
import { Card, Table, Tag, Row, Col, Statistic, Badge, Tooltip } from 'antd';
import { AlertOutlined, StockOutlined } from '@ant-design/icons';
import ReactECharts from 'echarts-for-react';
import { inventoryService, InventoryAlert } from '../../services/inventory';

const InventoryView: React.FC = () => {
  const [alerts, setAlerts] = useState<InventoryAlert[]>([]);
  const [loading, setLoading] = useState(false);

  useEffect(() => {
    loadAlerts();
  }, []);

  const loadAlerts = async () => {
    setLoading(true);
    try {
      const res = await inventoryService.getAlerts();
      setAlerts(res.alerts || []);
    } catch (error) {
      console.error('加载预警数据失败:', error);
    } finally {
      setLoading(false);
    }
  };

  const getGaugeOption = (available: number, safetyStock: number) => {
    const ratio = available / Math.max(safetyStock, 1);
    return {
      series: [{
        type: 'gauge',
        startAngle: 200,
        endAngle: -20,
        min: 0,
        max: 3,
        detail: { formatter: '{value}x' },
        data: [{ value: Math.round(ratio * 10) / 10, name: '库存/安全库存' }],
        axisLine: {
          lineStyle: {
            width: 20,
            color: [
              [0.5, '#ff4d4f'],
              [1, '#faad14'],
              [3, '#52c41a'],
            ],
          },
        },
        pointer: { width: 5 },
      }],
    };
  };

  const alertColumns = [
    { title: 'SKU编码', dataIndex: 'sku_code', key: 'sku_code' },
    { title: '可用库存', dataIndex: 'available_qty', key: 'available_qty' },
    { title: '安全库存', dataIndex: 'safety_stock', key: 'safety_stock' },
    {
      title: '预警类型',
      dataIndex: 'alert_type',
      key: 'alert_type',
      render: (type: string) => (
        <Tag color={type === 'stockout' ? 'red' : type === 'surplus' ? 'orange' : 'blue'}>
          {type === 'stockout' ? '缺货' : type === 'surplus' ? '积压' : '预警'}
        </Tag>
      ),
    },
    {
      title: '预计断货天数',
      dataIndex: 'days_until_stockout',
      key: 'days_until_stockout',
      render: (days: number | null) => days !== null ? (
        <span style={{ color: days < 3 ? '#ff4d4f' : '#faad14' }}>{days.toFixed(1)}天</span>
      ) : '-',
    },
    {
      title: '库存水位',
      key: 'gauge',
      render: (_: any, record: InventoryAlert) => (
        <ReactECharts
          option={getGaugeOption(record.available_qty, record.safety_stock)}
          style={{ height: 80, width: 120 }}
        />
      ),
    },
  ];

  const stockoutCount = alerts.filter((a) => a.alert_type === 'stockout').length;
  const surplusCount = alerts.filter((a) => a.alert_type === 'surplus').length;

  return (
    <div>
      <Card title="库存预警视图">
        <Row gutter={16} style={{ marginBottom: 24 }}>
          <Col span={8}>
            <Statistic
              title="缺货预警"
              value={stockoutCount}
              prefix={<AlertOutlined />}
              valueStyle={{ color: stockoutCount > 0 ? '#cf1322' : '#3f8600' }}
            />
          </Col>
          <Col span={8}>
            <Statistic
              title="积压预警"
              value={surplusCount}
              prefix={<StockOutlined />}
              valueStyle={{ color: surplusCount > 0 ? '#faad14' : '#3f8600' }}
            />
          </Col>
          <Col span={8}>
            <Statistic title="总预警数" value={alerts.length} />
          </Col>
        </Row>

        <Table
          columns={alertColumns}
          dataSource={alerts}
          rowKey="sku_id"
          loading={loading}
          pagination={{ pageSize: 10 }}
        />
      </Card>
    </div>
  );
};

export default InventoryView;
```

### PricingPanel（定价面板）

```tsx
// src/pages/PricingPanel/index.tsx
import React, { useState } from 'react';
import { Card, Select, Button, Row, Col, Statistic, Descriptions, Tag, Space, message } from 'antd';
import { DollarOutlined, SwapOutlined } from '@ant-design/icons';
import ReactECharts from 'echarts-for-react';
import request from '../../utils/request';

interface PriceSuggestion {
  suggested_price: number;
  current_price: number;
  price_change: number;
  price_change_pct: number;
  expected_margin: number;
  strategy_used: string;
  confidence: number;
  reasoning: string;
}

const PricingPanel: React.FC = () => {
  const [skuId, setSkuId] = useState<number>(1);
  const [strategy, setStrategy] = useState<string>('elasticity');
  const [suggestion, setSuggestion] = useState<PriceSuggestion | null>(null);
  const [loading, setLoading] = useState(false);

  const calculatePrice = async () => {
    setLoading(true);
    try {
      const res = await request.post('/api/v1/pricing/calculate', {
        sku_id: skuId,
        strategy,
      });
      setSuggestion(res.suggestions?.[0] || null);
      message.success('定价计算完成');
    } catch (error) {
      message.error('计算失败');
    } finally {
      setLoading(false);
    }
  };

  const getPriceTrendOption = () => {
    // 模拟价格趋势数据
    const dates = Array.from({ length: 30 }, (_, i) => `2024-12-${i + 1}`);
    const currentPrices = Array.from({ length: 30 }, () => suggestion?.current_price || 199);
    const suggestedPrices = Array.from({ length: 30 }, (_, i) =>
      i >= 25 ? (suggestion?.suggested_price || 199) : (suggestion?.current_price || 199)
    );

    return {
      tooltip: { trigger: 'axis' },
      legend: { data: ['当前价格', '建议价格'] },
      xAxis: { type: 'category', data: dates },
      yAxis: { type: 'value', name: '价格(元)', scale: true },
      series: [
        {
          name: '当前价格',
          type: 'line',
          data: currentPrices,
          lineStyle: { width: 2 },
          itemStyle: { color: '#8c8c8c' },
        },
        {
          name: '建议价格',
          type: 'line',
          data: suggestedPrices,
          lineStyle: { width: 3, type: 'dashed' },
          itemStyle: { color: '#1890ff' },
          markLine: {
            data: [{ yAxis: suggestion?.suggested_price || 0, name: '建议价' }],
          },
        },
      ],
    };
  };

  return (
    <div>
      <Card title="智能定价面板" extra={
        <Space>
          <span>SKU:</span>
          <Select value={skuId} onChange={setSkuId} style={{ width: 200 }}
            options={[
              { value: 1, label: 'SKU001-01 蓝牙耳机(黑)' },
              { value: 3, label: 'SKU002-01 智能手表(黑)' },
              { value: 4, label: 'SKU003-01 有机牛奶' },
            ]}
          />
          <Select value={strategy} onChange={setStrategy} style={{ width: 150 }}
            options={[
              { value: 'cost_plus', label: '成本加成' },
              { value: 'competitive', label: '竞争定价' },
              { value: 'elasticity', label: '弹性定价' },
              { value: 'dynamic', label: '动态定价' },
            ]}
          />
          <Button type="primary" icon={<SwapOutlined />} onClick={calculatePrice} loading={loading}>
            计算建议价格
          </Button>
        </Space>
      }>
        {suggestion && (
          <>
            <Row gutter={16} style={{ marginBottom: 24 }}>
              <Col span={6}>
                <Statistic title="当前价格" value={suggestion.current_price} prefix="¥" />
              </Col>
              <Col span={6}>
                <Statistic
                  title="建议价格"
                  value={suggestion.suggested_price}
                  prefix="¥"
                  valueStyle={{
                    color: suggestion.price_change > 0 ? '#cf1322' : '#3f8600',
                  }}
                />
              </Col>
              <Col span={6}>
                <Statistic
                  title="价格变动"
                  value={suggestion.price_change_pct}
                  suffix="%"
                  prefix={suggestion.price_change > 0 ? '+' : ''}
                  valueStyle={{
                    color: suggestion.price_change > 0 ? '#cf1322' : '#3f8600',
                  }}
                />
              </Col>
              <Col span={6}>
                <Statistic
                  title="预期毛利率"
                  value={suggestion.expected_margin * 100}
                  suffix="%"
                  precision={1}
                />
              </Col>
            </Row>

            <Descriptions bordered column={2} style={{ marginBottom: 24 }}>
              <Descriptions.Item label="使用策略">
                <Tag color="blue">{suggestion.strategy_used}</Tag>
              </Descriptions.Item>
              <Descriptions.Item label="置信度">
                <Tag color={suggestion.confidence > 0.7 ? 'green' : 'orange'}>
                  {(suggestion.confidence * 100).toFixed(0)}%
                </Tag>
              </Descriptions.Item>
              <Descriptions.Item label="分析说明" span={2}>
                {suggestion.reasoning}
              </Descriptions.Item>
            </Descriptions>

            <ReactECharts option={getPriceTrendOption()} style={{ height: 300 }} />
          </>
        )}
      </Card>
    </div>
  );
};

export default PricingPanel;
```

### SupplyChainMap（供应链地图）

```tsx
// src/pages/SupplyChainMap/index.tsx
import React, { useState, useEffect } from 'react';
import { Card, Row, Col, Statistic, Table, Tag } from 'antd';
import { EnvironmentOutlined, CarOutlined } from '@ant-design/icons';
import ReactECharts from 'echarts-for-react';
import request from '../../utils/request';

interface SupplierInfo {
  id: number;
  name: string;
  city: string;
  grade: string;
  total_score: number;
  lead_time_days: number;
}

const SupplyChainMap: React.FC = () => {
  const [suppliers, setSuppliers] = useState<SupplierInfo[]>([]);
  const [dashboardData, setDashboardData] = useState<any>(null);

  useEffect(() => {
    loadData();
  }, []);

  const loadData = async () => {
    try {
      const [supRes, dashRes] = await Promise.all([
        request.get('/api/v1/suppliers', { params: { is_active: true } }),
        request.get('/api/v1/supply-chain/dashboard'),
      ]);
      setSuppliers(supRes || []);
      setDashboardData(dashRes);
    } catch (error) {
      console.error('加载数据失败:', error);
    }
  };

  const getMapOption = () => {
    // 模拟仓储和供应商分布数据
    const warehouses = [
      { name: '华东总仓(上海)', value: [121.47, 31.23, 50000] },
      { name: '华南区域仓(广州)', value: [113.26, 23.13, 20000] },
      { name: '华北区域仓(北京)', value: [116.40, 39.90, 20000] },
    ];

    const supplierPoints = suppliers.map((s) => {
      const cityCoords: Record<string, [number, number]> = {
        '深圳': [114.07, 22.62],
        '东莞': [113.75, 23.05],
        '呼和浩特': [111.75, 40.84],
      };
      const coords = cityCoords[s.city] || [116.40, 39.90];
      return { name: s.name, value: [...coords, s.total_score] };
    });

    return {
      tooltip: { trigger: 'item' },
      legend: { data: ['仓库', '供应商'] },
      geo: {
        map: 'china',
        roam: true,
        label: { show: false },
        itemStyle: { areaColor: '#f3f3f3', borderColor: '#999' },
        emphasis: { itemStyle: { areaColor: '#e0e0e0' } },
      },
      series: [
        {
          name: '仓库',
          type: 'effectScatter',
          coordinateSystem: 'geo',
          data: warehouses,
          symbolSize: (val: number[]) => Math.sqrt(val[2]) / 5,
          itemStyle: { color: '#1890ff' },
          label: { show: true, formatter: '{b}', position: 'right' },
        },
        {
          name: '供应商',
          type: 'scatter',
          coordinateSystem: 'geo',
          data: supplierPoints,
          symbolSize: 15,
          itemStyle: { color: '#52c41a' },
          label: { show: true, formatter: '{b}', position: 'right', fontSize: 10 },
        },
      ],
    };
  };

  const supplierColumns = [
    { title: '供应商', dataIndex: 'name', key: 'name' },
    { title: '城市', dataIndex: 'city', key: 'city' },
    { title: '交期(天)', dataIndex: 'lead_time_days', key: 'lead_time_days' },
    {
      title: '等级',
      dataIndex: 'grade',
      key: 'grade',
      render: (grade: string) => (
        <Tag color={grade === 'A' ? 'green' : grade === 'B' ? 'blue' : 'orange'}>{grade}</Tag>
      ),
    },
    { title: '评分', dataIndex: 'total_score', key: 'total_score' },
  ];

  return (
    <div>
      <Row gutter={16} style={{ marginBottom: 16 }}>
        <Col span={6}>
          <Card>
            <Statistic title="活跃供应商" value={dashboardData?.supplier_stats?.total || 0} prefix={<EnvironmentOutlined />} />
          </Card>
        </Col>
        <Col span={6}>
          <Card>
            <Statistic title="本月采购额" value={dashboardData?.purchase_stats?.month_amount || 0} prefix="¥" />
          </Card>
        </Col>
        <Col span={6}>
          <Card>
            <Statistic title="待处理订单" value={dashboardData?.purchase_stats?.pending_orders || 0} />
          </Card>
        </Col>
        <Col span={6}>
          <Card>
            <Statistic title="库存健康率" value={(dashboardData?.inventory_stats?.health_rate || 0) * 100} suffix="%" precision={1} />

        <ReactECharts option={getMapOption()} style={{ height: 500 }} />
```
