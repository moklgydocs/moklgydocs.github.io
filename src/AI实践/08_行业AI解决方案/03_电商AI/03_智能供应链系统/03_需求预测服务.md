## FastAPI路由完整代码

### 请求/响应模型

```python
from pydantic import BaseModel, Field
from typing import Optional, List
from enum import Enum
from datetime import datetime


class ModelType(str, Enum):
    PROPHET = "prophet"
    XGBOOST = "xgboost"
    ENSEMBLE = "ensemble"


class TrainRequest(BaseModel):
    """训练请求"""
    sku_id: int = Field(..., description="SKU ID")
    warehouse_id: int = Field(default=1, description="仓库ID")
    model_type: ModelType = Field(default=ModelType.PROPHET, description="模型类型")
    forecast_days: int = Field(default=30, ge=7, le=365, description="预测天数")
    retrain: bool = Field(default=False, description="是否重新训练")


class TrainResponse(BaseModel):
    """训练响应"""
    task_id: str = Field(..., description="Celery任务ID")
    sku_id: int
    model_type: str
    status: str = "submitted"
    message: str


class PredictRequest(BaseModel):
    """预测请求"""
    sku_id: int = Field(..., description="SKU ID")
    warehouse_id: int = Field(default=1, description="仓库ID")
    days: int = Field(default=30, ge=1, le=90, description="预测天数")
    model_type: ModelType = Field(default=ModelType.PROPHET, description="模型类型")


class ForecastDataPoint(BaseModel):
    """单个预测数据点"""
    date: str
    predicted: float
    predicted_lower: float
    predicted_upper: float
    actual: Optional[float] = None


class PredictResponse(BaseModel):
    """预测响应"""
    sku_id: int
    warehouse_id: int
    model_type: str
    forecast_days: int
    data: List[ForecastDataPoint]


class BatchPredictRequest(BaseModel):
    """批量预测请求"""
    sku_ids: List[int] = Field(..., min_length=1, max_length=100, description="SKU ID列表")
    warehouse_id: int = Field(default=1, description="仓库ID")
    days: int = Field(default=30, ge=1, le=90, description="预测天数")
    model_type: ModelType = Field(default=ModelType.PROPHET)


class ForecastResultItem(BaseModel):
    """预测结果项"""
    id: int
    sku_id: int
    warehouse_id: int
    forecast_date: str
    predicted_qty: float
    predicted_lower: Optional[float]
    predicted_upper: Optional[float]
    actual_qty: Optional[float]
    model_type: str


class ForecastResultResponse(BaseModel):
    """预测结果查询响应"""
    total: int
    items: List[ForecastResultItem]


class AccuracyMetrics(BaseModel):
    """预测准确率指标"""
    sku_id: int
    model_type: str
    mape: Optional[float]
    rmse: Optional[float]
    wmape: Optional[float]
    mae: Optional[float]
    sample_count: int
    last_evaluated: Optional[str]
```

### 路由定义

```python
from fastapi import APIRouter, Depends, HTTPException, Query
from sqlalchemy.ext.asyncio import AsyncSession
from sqlalchemy import select, and_, desc
from typing import Optional

from app.database import get_db
from app.models.forecast import ForecastResult
from app.schemas.forecast import (
    TrainRequest, TrainResponse, PredictRequest, PredictResponse,
    BatchPredictRequest, ForecastResultResponse, ForecastResultItem,
    AccuracyMetrics, ForecastDataPoint,
)
from app.tasks.forecast import train_forecast_model, batch_predict_task
from app.services.forecast_engine import ProphetEngine, XGBoostEngine

router = APIRouter(prefix="/api/v1/forecast", tags=["预测服务"])


@router.post("/train", response_model=TrainResponse, summary="训练预测模型")
async def train_model(
    req: TrainRequest,
    db: AsyncSession = Depends(get_db),
):
    """
    提交预测模型训练任务，异步执行。
    返回Celery任务ID用于查询训练状态。
    """
    task = train_forecast_model.delay(
        sku_id=req.sku_id,
        warehouse_id=req.warehouse_id,
        model_type=req.model_type.value,
        forecast_days=req.forecast_days,
        retrain=req.retrain,
    )
    return TrainResponse(
        task_id=task.id,
        sku_id=req.sku_id,
        model_type=req.model_type.value,
        status="submitted",
        message=f"训练任务已提交，任务ID: {task.id}",
    )


@router.post("/predict", response_model=PredictResponse, summary="生成预测")
async def predict(
    req: PredictRequest,
    db: AsyncSession = Depends(get_db),
):
    """
    对指定SKU生成销量预测，同步返回预测结果。
    如果模型未训练则自动使用默认参数。
    """
    engine = _get_engine(req.model_type)

    # 从数据库获取历史销量
    sales_data = await _get_sales_history(db, req.sku_id, req.warehouse_id, days=365)

    if len(sales_data) < 30:
        raise HTTPException(status_code=400, detail="历史数据不足30天，无法生成预测")

    # 执行预测
    forecast_df = engine.predict(sales_data, periods=req.days)

    # 构建响应
    data_points = []
    for _, row in forecast_df.iterrows():
        data_points.append(ForecastDataPoint(
            date=str(row["date"]),
            predicted=round(row["predicted"], 1),
            predicted_lower=round(row["predicted_lower"], 1),
            predicted_upper=round(row["predicted_upper"], 1),
        ))

    # 保存预测结果到数据库
    await _save_forecast_results(db, req.sku_id, req.warehouse_id, req.model_type.value, forecast_df)

    return PredictResponse(
        sku_id=req.sku_id,
        warehouse_id=req.warehouse_id,
        model_type=req.model_type.value,
        forecast_days=req.days,
        data=data_points,
    )


@router.get("/results", response_model=ForecastResultResponse, summary="查询预测结果")
async def get_results(
    sku_id: int = Query(..., description="SKU ID"),
    warehouse_id: int = Query(default=1, description="仓库ID"),
    start_date: Optional[str] = Query(default=None, description="开始日期"),
    end_date: Optional[str] = Query(default=None, description="结束日期"),
    page: int = Query(default=1, ge=1),
    page_size: int = Query(default=30, ge=1, le=100),
    db: AsyncSession = Depends(get_db),
):
    """查询已保存的预测结果"""
    conditions = [
        ForecastResult.sku_id == sku_id,
        ForecastResult.warehouse_id == warehouse_id,
    ]
    if start_date:
        conditions.append(ForecastResult.forecast_date >= start_date)
    if end_date:
        conditions.append(ForecastResult.forecast_date <= end_date)

    # 查询总数
    count_stmt = select(ForecastResult).where(and_(*conditions))
    result = await db.execute(count_stmt)
    total = len(result.all())

    # 分页查询
    stmt = (
        select(ForecastResult)
        .where(and_(*conditions))
        .order_by(desc(ForecastResult.forecast_date))
        .offset((page - 1) * page_size)
        .limit(page_size)
    )
    result = await db.execute(stmt)
    items = result.scalars().all()

    return ForecastResultResponse(
        total=total,
        items=[
            ForecastResultItem(
                id=item.id,
                sku_id=item.sku_id,
                warehouse_id=item.warehouse_id,
                forecast_date=str(item.forecast_date),
                predicted_qty=float(item.predicted_qty),
                predicted_lower=float(item.predicted_lower) if item.predicted_lower else None,
                predicted_upper=float(item.predicted_upper) if item.predicted_upper else None,
                actual_qty=float(item.actual_qty) if item.actual_qty else None,
                model_type=item.model_type or "",
            )
            for item in items
        ],
    )

    )


@router.get("/accuracy", response_model=AccuracyMetrics, summary="预测准确率")
async def get_accuracy(
    sku_id: int = Query(..., description="SKU ID"),
    model_type: Optional[str] = Query(default=None, description="模型类型"),
    db: AsyncSession = Depends(get_db),
):
    """
    查询预测准确率指标，基于已有预测结果与实际值的对比。
    """
    conditions = [
        ForecastResult.sku_id == sku_id,
        ForecastResult.actual_qty.isnot(None),
    ]
    if model_type:
        conditions.append(ForecastResult.model_type == model_type)

    stmt = select(ForecastResult).where(and_(*conditions))
    result = await db.execute(stmt)
    records = result.scalars().all()

    if not records:
        raise HTTPException(status_code=404, detail="没有可评估的预测数据")

    # 计算准确率指标
    import numpy as np
    y_true = np.array([float(r.actual_qty) for r in records])
    y_pred = np.array([float(r.predicted_qty) for r in records])

    mask = y_true != 0
    mape = float(np.mean(np.abs((y_true[mask] - y_pred[mask]) / y_true[mask])) * 100) if mask.any() else None
    rmse = float(np.sqrt(np.mean((y_true - y_pred) ** 2)))
    mae = float(np.mean(np.abs(y_true - y_pred)))
    wmape = float(np.sum(np.abs(y_true - y_pred)) / np.sum(np.abs(y_true)) * 100) if np.sum(np.abs(y_true)) > 0 else None

    return AccuracyMetrics(
        sku_id=sku_id,
        model_type=model_type or "all",
        mape=round(mape, 2) if mape else None,
        rmse=round(rmse, 2),
        wmape=round(wmape, 2) if wmape else None,
        mae=round(mae, 2),
        sample_count=len(records),
        last_evaluated=str(records[-1].created_at) if records else None,
    )


def _get_engine(model_type: ModelType):
    """获取预测引擎实例"""
    if model_type == ModelType.PROPHET:
        return ProphetEngine()
    elif model_type == ModelType.XGBOOST:
        return XGBoostEngine()
    else:
        return ProphetEngine()


async def _get_sales_history(db: AsyncSession, sku_id: int, warehouse_id: int, days: int = 365):
    """从数据库获取历史销量数据"""
    from app.models.order import SalesItem, SalesOrder
    from datetime import timedelta

    cutoff = datetime.now() - timedelta(days=days)
    stmt = (
        select(SalesItem.sku_id, SalesOrder.created_at, SalesItem.quantity)
        .join(SalesOrder, SalesItem.order_id == SalesOrder.id)
        .where(
            and_(
                SalesItem.sku_id == sku_id,
                SalesOrder.warehouse_id == warehouse_id,
                SalesOrder.created_at >= cutoff,
                SalesOrder.status == "delivered",
            )
        )
        .order_by(SalesOrder.created_at)
    )
    result = await db.execute(stmt)
    rows = result.all()

    import pandas as pd
    if not rows:
        return pd.DataFrame(columns=["ds", "y"])

    df = pd.DataFrame(rows, columns=["sku_id", "ds", "y"])
    df["ds"] = pd.to_datetime(df["ds"]).dt.date
    daily_sales = df.groupby(["sku_id", "ds"])["y"].sum().reset_index()
    daily_sales.columns = ["sku_id", "ds", "y"]
    return daily_sales[["ds", "y"]]


async def _save_forecast_results(
    db: AsyncSession,
    sku_id: int,
    warehouse_id: int,
    model_type: str,
    forecast_df,
):
    """保存预测结果到数据库"""
    from datetime import datetime

    for _, row in forecast_df.iterrows():
        result = ForecastResult(
            sku_id=sku_id,
            warehouse_id=warehouse_id,
            forecast_date=row["date"] if isinstance(row["date"], datetime) else datetime.strptime(str(row["date"]), "%Y-%m-%d"),
            predicted_qty=row["predicted"],
            predicted_lower=row.get("predicted_lower"),
            predicted_upper=row.get("predicted_upper"),
            model_type=model_type,
            model_version="v1",
        )
        db.add(result)
    await db.flush()
```

## Prophet预测引擎

```python
import pandas as pd
import numpy as np
from prophet import Prophet
from typing import Optional
import joblib
import os
from app.config import settings


class ProphetEngine:
    """Prophet预测引擎"""

    MODEL_DIR = os.path.join(settings.BASE_DIR, "models", "prophet")

    def __init__(
        self,
        growth: str = "linear",
        seasonality_mode: str = "multiplicative",
        yearly_seasonality: int = 10,
        weekly_seasonality: int = 3,
        changepoint_prior_scale: float = 0.05,
        seasonality_prior_scale: float = 10.0,
    ):
        self.growth = growth
        self.seasonality_mode = seasonality_mode
        self.yearly_seasonality = yearly_seasonality
        self.weekly_seasonality = weekly_seasonality
        self.changepoint_prior_scale = changepoint_prior_scale
        self.seasonality_prior_scale = seasonality_prior_scale
        self.model: Optional[Prophet] = None

    def train(self, df: pd.DataFrame) -> dict:
        """
        训练Prophet模型

        Args:
            df: 包含ds(日期)和y(销量)列的DataFrame

        Returns:
            训练指标
        """
        train_df = df[["ds", "y"]].copy()
        train_df["ds"] = pd.to_datetime(train_df["ds"])

        # 构建节假日
        holidays = self._build_holidays(train_df["ds"].dt.year.unique().tolist())

        self.model = Prophet(
            growth=self.growth,
            seasonality_mode=self.seasonality_mode,
            yearly_seasonality=self.yearly_seasonality,
            weekly_seasonality=self.weekly_seasonality,
            changepoint_prior_scale=self.changepoint_prior_scale,
            seasonality_prior_scale=self.seasonality_prior_scale,
            holidays=holidays if not holidays.empty else None,
        )
        self.model.add_seasonality(name="monthly", period=30.5, fourier_order=5)
        self.model.fit(train_df)

        # 评估
        forecast_train = self.model.predict(train_df)
        y_true = train_df["y"].values
        y_pred = forecast_train["yhat"].values

        return {
            "train_samples": len(train_df),
            "mape": self._mape(y_true, y_pred),
            "rmse": self._rmse(y_true, y_pred),
        }

    def predict(self, df: pd.DataFrame, periods: int = 30) -> pd.DataFrame:
        """
        生成预测

        Args:
            df: 历史数据
            periods: 预测天数

        Returns:
            预测结果DataFrame
        """
        # 自动训练
        if self.model is None:
            self.train(df)

        future = self.model.make_future_dataframe(periods=periods, freq="D", include_history=False)
        forecast = self.model.predict(future)

        result = pd.DataFrame({
            "date": forecast["ds"],
            "predicted": forecast["yhat"].clip(lower=0),
            "predicted_lower": forecast["yhat_lower"].clip(lower=0),
            "predicted_upper": forecast["yhat_upper"].clip(lower=0),
        })
        return result

    def save_model(self, sku_id: int, warehouse_id: int) -> str:
        """保存模型到磁盘"""
        os.makedirs(self.MODEL_DIR, exist_ok=True)
        path = os.path.join(self.MODEL_DIR, f"prophet_{sku_id}_{warehouse_id}.joblib")
        joblib.dump(self.model, path)
        return path

    def load_model(self, sku_id: int, warehouse_id: int) -> bool:
        """从磁盘加载模型"""
        path = os.path.join(self.MODEL_DIR, f"prophet_{sku_id}_{warehouse_id}.joblib")
        if os.path.exists(path):
            self.model = joblib.load(path)
            return True
        return False

    @staticmethod
    def _build_holidays(years: list) -> pd.DataFrame:
        """构建中国节假日"""
        holidays = []
        for year in years:
            holidays.extend([
                {"holiday": "promo_618", "ds": f"{year}-06-18", "lower_window": -17, "upper_window": 0},
                {"holiday": "promo_1111", "ds": f"{year}-11-11", "lower_window": -10, "upper_window": 0},
                {"holiday": "national_day", "ds": f"{year}-10-01", "lower_window": 0, "upper_window": 6},
            ])
        return pd.DataFrame(holidays) if holidays else pd.DataFrame()

    @staticmethod
    def _mape(y_true: np.ndarray, y_pred: np.ndarray) -> float:
        mask = y_true != 0
        if not mask.any():
            return float("inf")
        return float(np.mean(np.abs((y_true[mask] - y_pred[mask]) / y_true[mask])) * 100)

    @staticmethod
    def _rmse(y_true: np.ndarray, y_pred: np.ndarray) -> float:
        return float(np.sqrt(np.mean((y_true - y_pred) ** 2)))
```

## XGBoost预测引擎

```python
import pandas as pd
import numpy as np
import xgboost as xgb
from typing import Optional
import joblib
import os
from app.config import settings


class XGBoostEngine:
    """XGBoost预测引擎"""

    MODEL_DIR = os.path.join(settings.BASE_DIR, "models", "xgboost")

    def __init__(
        self,
        n_estimators: int = 500,
        max_depth: int = 6,
        learning_rate: float = 0.05,
        subsample: float = 0.8,
        colsample_bytree: float = 0.8,
    ):
        self.n_estimators = n_estimators
        self.max_depth = max_depth
        self.learning_rate = learning_rate
        self.subsample = subsample
        self.colsample_bytree = colsample_bytree
        self.model: Optional[xgb.XGBRegressor] = None
        self.feature_names: list = []

    def _create_features(self, df: pd.DataFrame) -> pd.DataFrame:
        """创建特征"""
        df = df.copy()
        df["ds"] = pd.to_datetime(df["ds"])

        # 日历特征
        df["year"] = df["ds"].dt.year
        df["month"] = df["ds"].dt.month
        df["day"] = df["ds"].dt.day
        df["dayofweek"] = df["ds"].dt.dayofweek
        df["dayofyear"] = df["ds"].dt.dayofyear
        df["weekofyear"] = df["ds"].dt.isocalendar().week.astype(int)
        df["is_weekend"] = (df["ds"].dt.dayofweek >= 5).astype(int)
        df["is_month_start"] = df["ds"].dt.is_month_start.astype(int)
        df["is_month_end"] = df["ds"].dt.is_month_end.astype(int)
        df["quarter"] = df["ds"].dt.quarter

        # 促销标记
        df["is_promo"] = 0
        df.loc[
            (df["ds"].dt.month == 6) & (df["ds"].dt.day <= 18) |
            (df["ds"].dt.month == 11) & (df["ds"].dt.day <= 11),
            "is_promo"
        ] = 1

        # 滞后特征
        for lag in [1, 7, 14, 28]:
            df[f"lag_{lag}"] = df["y"].shift(lag)

        # 滚动统计
        for window in [7, 14, 28]:
            df[f"rolling_mean_{window}"] = df["y"].rolling(window).mean()
            df[f"rolling_std_{window}"] = df["y"].rolling(window).std()
            df[f"rolling_min_{window}"] = df["y"].rolling(window).min()
            df[f"rolling_max_{window}"] = df["y"].rolling(window).max()

        # 同比环比
        df["yoy_ratio"] = df["y"] / df["y"].shift(364) - 1
        df["wow_ratio"] = df["y"] / df["y"].shift(7) - 1

        return df

    def train(self, df: pd.DataFrame) -> dict:
        """训练XGBoost模型"""
        df_feat = self._create_features(df)
        df_feat = df_feat.dropna()

        feature_cols = [c for c in df_feat.columns if c not in ["ds", "y"]]
        self.feature_names = feature_cols

        X = df_feat[feature_cols].values
        y = df_feat["y"].values

        # 时间序列切分(后20%作为验证集)
        split_idx = int(len(X) * 0.8)
        X_train, X_val = X[:split_idx], X[split_idx:]
        y_train, y_val = y[:split_idx], y[split_idx:]

        self.model = xgb.XGBRegressor(
            n_estimators=self.n_estimators,
            max_depth=self.max_depth,
            learning_rate=self.learning_rate,
            subsample=self.subsample,
            colsample_bytree=self.colsample_bytree,
            objective="reg:squarederror",
            random_state=42,
        )
        self.model.fit(
            X_train, y_train,
            eval_set=[(X_val, y_val)],
            verbose=False,
        )

        # 评估
        y_pred_val = self.model.predict(X_val)
        return {
            "train_samples": len(X_train),
            "val_samples": len(X_val),
            "val_mape": self._mape(y_val, y_pred_val),
            "val_rmse": self._rmse(y_val, y_pred_val),
        }

    def predict(self, df: pd.DataFrame, periods: int = 30) -> pd.DataFrame:
        """生成预测"""
        if self.model is None:
            self.train(df)

        # 用历史数据逐步预测
        df_extended = df.copy()
        last_date = pd.to_datetime(df["ds"].max())

        predictions = []
        for i in range(periods):
            next_date = last_date + pd.Timedelta(days=i + 1)

            # 构建特征(使用已有数据)
            temp_row = pd.DataFrame({"ds": [next_date], "y": [0]})
            df_temp = pd.concat([df_extended, temp_row], ignore_index=True)
            df_feat = self._create_features(df_temp)

            if df_feat.iloc[-1].isna().any():
                # 用最近可用值填充
                last_valid = df_feat.dropna().iloc[-1]
                for col in self.feature_names:
                    if pd.isna(df_feat.iloc[-1][col]):
                        df_feat.iloc[-1, df_feat.columns.get_loc(col)] = last_valid[col]

            X_pred = df_feat[self.feature_names].iloc[[-1]].values
            pred = max(float(self.model.predict(X_pred)[0]), 0)

            predictions.append({
                "date": str(next_date.date()),
                "predicted": round(pred, 1),
                "predicted_lower": round(pred * 0.85, 1),
                "predicted_upper": round(pred * 1.15, 1),
            })

            # 将预测值加入历史以支持下一步预测
            df_extended = pd.concat([
                df_extended,
                pd.DataFrame({"ds": [next_date], "y": [pred]}),
            ], ignore_index=True)

        return pd.DataFrame(predictions)

    def save_model(self, sku_id: int, warehouse_id: int) -> str:
        os.makedirs(self.MODEL_DIR, exist_ok=True)
        path = os.path.join(self.MODEL_DIR, f"xgb_{sku_id}_{warehouse_id}.joblib")
        joblib.dump({"model": self.model, "features": self.feature_names}, path)
        return path

    def load_model(self, sku_id: int, warehouse_id: int) -> bool:
        path = os.path.join(self.MODEL_DIR, f"xgb_{sku_id}_{warehouse_id}.joblib")
        if os.path.exists(path):
            data = joblib.load(path)
            self.model = data["model"]
            self.feature_names = data["features"]
            return True
        return False

    @staticmethod
    def _mape(y_true: np.ndarray, y_pred: np.ndarray) -> float:
        mask = y_true != 0
        if not mask.any():
            return float("inf")
        return float(np.mean(np.abs((y_true[mask] - y_pred[mask]) / y_true[mask])) * 100)

    @staticmethod
    def _rmse(y_true: np.ndarray, y_pred: np.ndarray) -> float:
        return float(np.sqrt(np.mean((y_true - y_pred) ** 2)))
```

```python
            ((dates.dt.month == 11) & (dates.dt.day <= 11)) |
            ((dates.dt.month == 12) & (dates.dt.day <= 12))

                pd.Timestamp(year=d.year, month=11, day=11),
                pd.Timestamp(year=d.year, month=12, day=12),
            ]
            min_dist = min(abs((d - pd).days) for pd in promo_dates)
            days_to_promo.append(min_dist)

        return pd.DataFrame({
            "is_promo": is_promo,
            "days_to_promo": days_to_promo,
        })

    @staticmethod
    def compute_historical_features(
        series: pd.Series,
        lags: List[int] = None,
        windows: List[int] = None,
    ) -> pd.DataFrame:
        """计算历史统计特征"""
        if lags is None:
            lags = [1, 7, 14, 28]
        if windows is None:
            windows = [7, 14, 28]

        features = {}
        for lag in lags:
```

```python
from app.celery_app import celery_app

@celery_app.task(bind=True, queue="ml", time_limit=1800)
def batch_predict_task(
    self,
    sku_ids: list,
    warehouse_id: int,
    days: int = 30,
    model_type: str = "prophet",
):
    """批量预测任务"""
    results = []
    total = len(sku_ids)

    for i, sku_id in enumerate(sku_ids):
        self.update_state(
            state="PROGRESS",
            meta={"current": i + 1, "total": total, "sku_id": sku_id},
        )
        try:
            import asyncio
            df = asyncio.run(_get_sales_data(sku_id, warehouse_id))

            if len(df) < 30:
                results.append({"sku_id": sku_id, "status": "skipped", "reason": "数据不足"})
                continue

            if model_type == "prophet":
                engine = ProphetEngine()
            else:
                engine = XGBoostEngine()

            # 尝试加载已有模型
            if not engine.load_model(sku_id, warehouse_id):
                engine.train(df)

            forecast_df = engine.predict(df, periods=days)
            asyncio.run(_save_results(sku_id, warehouse_id, model_type, forecast_df))

            results.append({"sku_id": sku_id, "status": "completed"})

        except Exception as e:
            results.append({"sku_id": sku_id, "status": "failed", "error": str(e)})

    return {"total": total, "results": results}


async def _get_sales_data(sku_id: int, warehouse_id: int) -> pd.DataFrame:
    """获取销量历史数据"""
    from app.models.order import SalesItem, SalesOrder
    from sqlalchemy import select, and_
    from datetime import datetime, timedelta

    async with async_session() as session:
        cutoff = datetime.now() - timedelta(days=365)
        stmt = (
            select(SalesItem.sku_id, SalesOrder.created_at, SalesItem.quantity)
            .join(SalesOrder, SalesItem.order_id == SalesOrder.id)
            .where(
                and_(
                    SalesItem.sku_id == sku_id,
                    SalesOrder.warehouse_id == warehouse_id,
                    SalesOrder.created_at >= cutoff,
                    SalesOrder.status == "delivered",
                )
            )
            .order_by(SalesOrder.created_at)
        )
        result = await session.execute(stmt)
        rows = result.all()

        if not rows:
            return pd.DataFrame(columns=["ds", "y"])

        df = pd.DataFrame(rows, columns=["sku_id", "ds", "y"])
        df["ds"] = pd.to_datetime(df["ds"]).dt.date
        daily_sales = df.groupby(["sku_id", "ds"])["y"].sum().reset_index()
        daily_sales.columns = ["sku_id", "ds", "y"]
        return daily_sales[["ds", "y"]]

    async with async_session() as session:
        for _, row in forecast_df.iterrows():
            date_val = row["date"]
            if isinstance(date_val, str):
                date_val = datetime.strptime(date_val, "%Y-%m-%d")

                predicted_upper=row.get("predicted_upper"),
                model_type=model_type,
                model_version="v1",
            )
```

```python
        model = self._get(model_id, version)
```

### 层级预测调和

当SKU级和品类级预测独立生成时，各层级预测值不一致（SKU级预测之和不等于品类级预测）。需通过调和算法保证层级一致性。

```python
import numpy as np

class ForecastReconciler:
    """层级预测调和器"""

    def __init__(self, hierarchy: dict):
        """
        Args:
            hierarchy: 层级关系 {"category_A": ["sku_1", "sku_2"], ...}
        """
        self.hierarchy = hierarchy

    def bottom_up(self, sku_forecasts: dict[str, float]) -> dict[str, float]:
        """自底向上调和：SKU级预测加总到品类级"""
        result = dict(sku_forecasts)
        for category, skus in self.hierarchy.items():
            result[category] = sum(sku_forecasts.get(sku, 0) for sku in skus)
        return result

    def top_down(self, category_forecasts: dict[str, float],
                 sku_proportions: dict[str, float]) -> dict[str, float]:
        """自顶向下调和：品类级预测按比例分配到SKU级"""
        result = dict(category_forecasts)
        for category, skus in self.hierarchy.items():
            cat_forecast = category_forecasts.get(category, 0)
            total_prop = sum(sku_proportions.get(sku, 0) for sku in skus)
            for sku in skus:
                prop = sku_proportions.get(sku, 0)
                result[sku] = cat_forecast * (prop / total_prop) if total_prop > 0 else 0
        return result

    def mint_reconciliation(
        self,
        base_forecasts: np.ndarray,
        summing_matrix: np.ndarray,
    ) -> np.ndarray:
        """MinT (Minimum Trace) 调和：最优无偏调和"""
        # S: 求和矩阵 (n_agg x n_bottom)
        # W: 基础预测误差协方差矩阵
        # 调和公式: y_hat_reconciled = S * (S'W^{-1}S)^{-1} * S' * W^{-1} * y_hat_base
        n_bottom = summing_matrix.shape[1]
        W = np.eye(len(base_forecasts))  # 简化：假设独立同方差
        S = summing_matrix
        SW_inv = S.T @ np.linalg.inv(W)
        reconciled = S @ np.linalg.inv(SW_inv @ S) @ SW_inv @ base_forecasts
        return reconciled
```
