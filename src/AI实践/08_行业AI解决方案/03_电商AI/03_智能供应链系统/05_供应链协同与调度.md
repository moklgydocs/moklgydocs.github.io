```python
@router.post("/suppliers", response_model=SupplierResponse, summary="创建/更新供应商")
async def manage_supplier(
    req: SupplierCreateRequest,
    db: AsyncSession = Depends(get_db),
):
    """创建新供应商"""
    # 检查编码唯一性
    stmt = select(Supplier).where(Supplier.code == req.code)
    result = await db.execute(stmt)
    if result.scalar_one_or_none():
        raise HTTPException(status_code=400, detail=f"供应商编码 {req.code} 已存在")

    supplier = Supplier(
        code=req.code,
        name=req.name,
        contact_person=req.contact_person,
        contact_phone=req.contact_phone,
        email=req.email,
        address=req.address,
        city=req.city,
        lead_time_days=req.lead_time_days,
        min_order_qty=req.min_order_qty,
    )
    db.add(supplier)
    await db.flush()
    await db.refresh(supplier)

    return _supplier_to_response(supplier)


@router.put("/suppliers/{supplier_id}", response_model=SupplierResponse, summary="更新供应商")
async def update_supplier(
    supplier_id: int,
    req: SupplierUpdateRequest,
    db: AsyncSession = Depends(get_db),
):
    """更新供应商信息"""
    supplier = await db.get(Supplier, supplier_id)
    if not supplier:
        raise HTTPException(status_code=404, detail="供应商不存在")

    update_data = req.model_dump(exclude_unset=True)
    for field, value in update_data.items():
        setattr(supplier, field, value)

    await db.flush()
    await db.refresh(supplier)
    return _supplier_to_response(supplier)


@router.post("/suppliers/{supplier_id}/evaluate", response_model=SupplierResponse, summary="评估供应商")
async def evaluate_supplier(
    supplier_id: int,
    req: SupplierEvaluationRequest,
    db: AsyncSession = Depends(get_db),
):
    """评估供应商并更新评分"""
    supplier = await db.get(Supplier, supplier_id)
    if not supplier:
        raise HTTPException(status_code=404, detail="供应商不存在")

    evaluator = SupplierEvaluator()
    eval_result = evaluator.evaluate(
        SupplierEvaluationRequest(
            supplier_id=str(supplier.id),
            supplier_name=supplier.name,
            scores=SupplierScore(
                quality_score=req.quality_score,
                price_score=req.price_score,
                delivery_score=req.delivery_score,
                service_score=req.service_score,
                risk_score=req.risk_score,
            ),
        )
    )

    # 更新评分
    supplier.quality_score = req.quality_score
    supplier.price_score = req.price_score
    supplier.delivery_score = req.delivery_score
    supplier.service_score = req.service_score
    supplier.risk_score = req.risk_score
    supplier.total_score = eval_result.total_score
    supplier.grade = SupplierGrade(eval_result.grade)

    await db.flush()
    await db.refresh(supplier)
    return _supplier_to_response(supplier)

@router.post("/purchase/orders", response_model=PurchaseOrderResponse, summary="创建采购订单")
async def create_purchase_order(
    req: PurchaseOrderCreateRequest,
    db: AsyncSession = Depends(get_db),
):
    """创建采购订单"""
    supplier = await db.get(Supplier, req.supplier_id)
    if not supplier:
        raise HTTPException(status_code=404, detail="供应商不存在")

    # 生成订单号
    import time
    order_no = f"PO-{int(time.time())}-{req.supplier_id:03d}"

    # 计算总金额
    total_amount = 0.0
    order_items = []
    for item_data in req.items:
        sku = await db.get(SKU, item_data.get("sku_id"))
        if not sku:
            raise HTTPException(status_code=400, detail=f"SKU {item_data.get('sku_id')} 不存在")

        qty = item_data["quantity"]
        price = item_data.get("unit_price", float(sku.cost_price))
        amount = qty * price
        total_amount += amount

        order_items.append(PurchaseItem(
            sku_id=sku.id,
            quantity=qty,
            unit_price=price,
            amount=amount,
        ))

    order = PurchaseOrder(
        order_no=order_no,
        supplier_id=req.supplier_id,
        warehouse_id=req.warehouse_id,
        status=PurchaseOrderStatus.DRAFT,
        total_amount=total_amount,
        expected_date=req.expected_date,
        remark=req.remark,
    )
    db.add(order)
    await db.flush()

    for item in order_items:
        item.order_id = order.id
        db.add(item)

    await db.flush()
    await db.refresh(order)

    # 查询明细
    stmt = select(PurchaseItem).where(PurchaseItem.order_id == order.id)
    result = await db.execute(stmt)
    items = result.scalars().all()

    return PurchaseOrderResponse(
        id=order.id,
        order_no=order.order_no,
        supplier_id=order.supplier_id,
        warehouse_id=order.warehouse_id,
        status=order.status.value,
        total_amount=float(order.total_amount),
        expected_date=str(order.expected_date) if order.expected_date else None,
        items=[{
            "sku_id": i.sku_id,
            "quantity": i.quantity,
            "unit_price": float(i.unit_price),
            "amount": float(i.amount),
        } for i in items],
        created_at=str(order.created_at),
    )
```

## 供应商评估模型

```python
from pydantic import BaseModel, Field
from typing import Optional, List, Dict
import numpy as np


class SupplierScore(BaseModel):
    quality_score: float = Field(..., ge=0, le=100)
    price_score: float = Field(..., ge=0, le=100)
    delivery_score: float = Field(..., ge=0, le=100)
    service_score: float = Field(..., ge=0, le=100)
    risk_score: float = Field(..., ge=0, le=100)


class SupplierEvaluationRequest(BaseModel):
    supplier_id: str
    supplier_name: str
    scores: SupplierScore
    weights: Optional[Dict[str, float]] = None


class SupplierEvaluationResponse(BaseModel):
    supplier_id: str
    supplier_name: str
    total_score: float
    grade: str
    quality_score: float
    price_score: float
    delivery_score: float
    service_score: float
    risk_score: float
    recommendations: List[str]


class SupplierEvaluator:
    """供应商评估引擎"""

    DEFAULT_WEIGHTS = {
        "quality": 0.30,
        "price": 0.25,
        "delivery": 0.25,
        "service": 0.10,
        "risk": 0.10,
    }

    GRADE_THRESHOLDS = {"A": 90, "B": 80, "C": 70, "D": 60}

    def evaluate(self, req: SupplierEvaluationRequest) -> SupplierEvaluationResponse:
        weights = req.weights or self.DEFAULT_WEIGHTS
        scores = req.scores

        total = (
            scores.quality_score * weights.get("quality", 0.30)
            + scores.price_score * weights.get("price", 0.25)
            + scores.delivery_score * weights.get("delivery", 0.25)
            + scores.service_score * weights.get("service", 0.10)
            + scores.risk_score * weights.get("risk", 0.10)
        )

        grade = self._grade(total)
        recs = self._recommendations(scores, total)

        return SupplierEvaluationResponse(
            supplier_id=req.supplier_id,
            supplier_name=req.supplier_name,
            total_score=round(total, 1),
            grade=grade,
            quality_score=scores.quality_score,
            price_score=scores.price_score,
            delivery_score=scores.delivery_score,
            service_score=scores.service_score,
            risk_score=scores.risk_score,
            recommendations=recs,
        )

    def _grade(self, score: float) -> str:
        for grade, threshold in self.GRADE_THRESHOLDS.items():
            if score >= threshold:
                return grade
        return "E"

    def _recommendations(self, scores: SupplierScore, total: float) -> List[str]:
        recs = []
        if scores.quality_score < 70:
            recs.append("质量评分偏低，建议加强来料检验并要求供应商提供改进计划")
        if scores.price_score < 70:
            recs.append("价格竞争力不足，建议引入替代供应商竞价")
        if scores.delivery_score < 70:
            recs.append("交期表现不稳定，建议设定交付SLA并建立预警机制")
        if scores.service_score < 60:
            recs.append("服务水平低于预期，建议评估是否需要更换供应商")
        if scores.risk_score < 60:
            recs.append("风险评分偏低，建议制定备选供应商计划")
        if total >= 90:
            recs.append("综合表现优秀，建议纳入战略合作供应商")
        if not recs:
            recs.append("表现良好，建议持续监控")
        return recs

    def compare(self, evaluations: List[SupplierEvaluationRequest]) -> List[Dict]:
        """对比多个供应商"""
        results = []
        for req in evaluations:
            result = self.evaluate(req)
            results.append({
                "supplier_id": result.supplier_id,
                "supplier_name": result.supplier_name,
                "total_score": result.total_score,
                "grade": result.grade,
            })
        return sorted(results, key=lambda x: x["total_score"], reverse=True)
```

## 采购优化引擎

```python
import numpy as np
from pydantic import BaseModel, Field
from typing import Optional, List, Dict
from enum import Enum


class PurchaseOptimizationRequest(BaseModel):
    sku_id: int
    annual_demand: float = Field(..., gt=0)
    unit_cost: float = Field(..., gt=0)
    ordering_cost: float = Field(default=200, gt=0)
    holding_cost_rate: float = Field(default=0.25, gt=0)
    lead_time_days: float = Field(default=7, gt=0)
    suppliers: List[Dict] = Field(..., min_length=1)


class SupplierRecommendation(BaseModel):
    supplier_id: str
    supplier_name: str
    total_cost: float
    unit_price: float
    score: float
    rank: int


class PurchaseOptimizationResult(BaseModel):
    sku_id: int
    optimal_order_quantity: float
    optimal_order_frequency: float
    recommended_supplier: Optional[SupplierRecommendation]
    all_suppliers: List[SupplierRecommendation]
    total_annual_cost: float


class PurchaseOptimizer:
    """采购优化引擎"""

    def optimize(self, req: PurchaseOptimizationRequest) -> PurchaseOptimizationResult:
        supplier_results = []

        for idx, sup in enumerate(req.suppliers):
            price = sup.get("unit_price", req.unit_cost)
            quality_score = sup.get("quality_score", 80)
            delivery_score = sup.get("delivery_score", 80)

            annual_purchase = req.annual_demand * price
            H = price * req.holding_cost_rate
            eoq = np.sqrt(2 * req.annual_demand * req.ordering_cost / H) if H > 0 else req.annual_demand

            annual_ordering = (req.annual_demand / eoq) * req.ordering_cost
            annual_holding = (eoq / 2) * H

            total_cost = annual_purchase + annual_ordering + annual_holding

            cost_score = max(0, 100 - (total_cost / (req.annual_demand * req.unit_cost) - 1) * 200)
            overall_score = cost_score * 0.5 + quality_score * 0.3 + delivery_score * 0.2

            supplier_results.append({
                "supplier_id": str(sup.get("id", idx)),
                "supplier_name": sup.get("name", f"供应商{idx+1}"),
                "total_cost": round(total_cost, 2),
                "unit_price": price,
                "score": round(overall_score, 1),
            })

        supplier_results.sort(key=lambda x: x["score"], reverse=True)
        for i, s in enumerate(supplier_results):
            s["rank"] = i + 1

        best = supplier_results[0] if supplier_results else None

        if best:
            H = best["unit_price"] * req.holding_cost_rate
            eoq = np.sqrt(2 * req.annual_demand * req.ordering_cost / H) if H > 0 else req.annual_demand
        else:
            eoq = np.sqrt(2 * req.annual_demand * req.ordering_cost / (req.unit_cost * req.holding_cost_rate))

        return PurchaseOptimizationResult(
            sku_id=req.sku_id,
            optimal_order_quantity=round(eoq, 0),
            optimal_order_frequency=round(req.annual_demand / eoq, 1),
            recommended_supplier=SupplierRecommendation(**best) if best else None,
            all_suppliers=[SupplierRecommendation(**s) for s in supplier_results],
            total_annual_cost=best["total_cost"] if best else 0,
        )
```

## 调度服务：配送路径规划简化版

```python
import numpy as np
from pydantic import BaseModel, Field
from typing import List, Tuple, Optional


class DeliveryPoint(BaseModel):
    """配送点"""
    id: str
    name: str
    longitude: float
    latitude: float
    demand: float = 0


class RoutePlan(BaseModel):
    """路线规划结果"""
    vehicle_id: str
    stops: List[str]
    total_distance: float
    total_demand: float
    estimated_time_minutes: float


class RoutingRequest(BaseModel):
    """路径规划请求"""
    depot: DeliveryPoint = Field(..., description="仓库/起点")
    delivery_points: List[DeliveryPoint] = Field(..., min_length=1)
    vehicle_capacity: float = Field(default=1000, gt=0)
    num_vehicles: int = Field(default=1, ge=1)
    avg_speed_kmh: float = Field(default=30, gt=0)


class RoutingResponse(BaseModel):
    """路径规划响应"""
    routes: List[RoutePlan]
    total_distance: float
    total_vehicles_used: int
    utilization_rate: float


class DeliveryRouter:
    """配送路径规划服务(简化版)"""

    def plan(self, req: RoutingRequest) -> RoutingResponse:
        """规划配送路线"""
        if not req.delivery_points:
            return RoutingResponse(routes=[], total_distance=0, total_vehicles_used=0, utilization_rate=0)

        # 构建距离矩阵
        all_points = [req.depot] + req.delivery_points
        n = len(all_points)
        dist_matrix = np.zeros((n, n))
        for i in range(n):
            for j in range(n):
                dist_matrix[i][j] = self._haversine_distance(
                    (all_points[i].latitude, all_points[i].longitude),
                    (all_points[j].latitude, all_points[j].longitude),
                )

        # 使用节约算法求解
        routes = self._savings_algorithm(
            dist_matrix=dist_matrix,
            demands=[0] + [p.demand for p in req.delivery_points],
            capacity=req.vehicle_capacity,
            num_vehicles=req.num_vehicles,
            point_ids=[p.id for p in all_points],
        )

        # 计算路线详情
        route_plans = []
        total_dist = 0
        total_demand = 0

        for idx, route in enumerate(routes):
            dist = self._route_distance(dist_matrix, route)
            demand = sum(req.delivery_points[i - 1].demand for i in route if i > 0)
            time_min = dist / (req.avg_speed_kmh / 60)

            route_plans.append(RoutePlan(
                vehicle_id=f"V{idx + 1}",
                stops=[all_points[i].id for i in route],
                total_distance=round(dist, 1),
                total_demand=demand,
                estimated_time_minutes=round(time_min, 0),
            ))
            total_dist += dist
            total_demand += demand

        total_capacity = req.vehicle_capacity * req.num_vehicles
        utilization = total_demand / total_capacity if total_capacity > 0 else 0

        return RoutingResponse(
            routes=route_plans,
            total_distance=round(total_dist, 1),
            total_vehicles_used=len(route_plans),
            utilization_rate=round(utilization, 3),
        )

    def _savings_algorithm(
        self,
        dist_matrix: np.ndarray,
        demands: list,
        capacity: float,
        num_vehicles: int,
        point_ids: list,
    ) -> List[List[int]]:
        """Clarke-Wright节约算法"""
        n = len(demands) - 1  # 客户数

        # 初始: 每个客户一条路线
        routes = [[i] for i in range(1, n + 1)]
        route_loads = [demands[i] for i in range(1, n + 1)]
        route_of = {i: i - 1 for i in range(1, n + 1)}

        # 计算节约值
        savings = []
        for i in range(1, n + 1):
            for j in range(i + 1, n + 1):
                save = dist_matrix[0][i] + dist_matrix[0][j] - dist_matrix[i][j]
                savings.append((save, i, j))
        savings.sort(reverse=True)

        # 合并路线
        for save, i, j in savings:
            ri = route_of[i]
            rj = route_of[j]
            if ri == rj:
                continue
            if route_loads[ri] + route_loads[rj] > capacity:
                continue

            route_i = routes[ri]
            route_j = routes[rj]
            if not route_i or not route_j:
                continue

            can_merge = False
            if route_i[-1] == i and route_j[0] == j:
                new_route = route_i + route_j
                can_merge = True
            elif route_i[0] == i and route_j[-1] == j:
                new_route = route_j + route_i
                can_merge = True

            if not can_merge:
                continue

            routes[ri] = new_route
            routes[rj] = []
            route_loads[ri] += route_loads[rj]
            route_loads[rj] = 0
            for c in new_route:
                route_of[c] = ri

        return [r for r in routes if r]

    @staticmethod
    def _route_distance(dist_matrix: np.ndarray, route: list) -> float:
        if not route:
            return 0
        total = dist_matrix[0][route[0]]
        for k in range(len(route) - 1):
            total += dist_matrix[route[k]][route[k + 1]]
        total += dist_matrix[route[-1]][0]
        return float(total)

    @staticmethod
    def _haversine_distance(p1: Tuple[float, float], p2: Tuple[float, float]) -> float:
        """计算两点间的球面距离(km)"""
        R = 6371
        lat1, lon1 = np.radians(p1[0]), np.radians(p1[1])
        lat2, lon2 = np.radians(p2[0]), np.radians(p2[1])
        dlat = lat2 - lat1
        dlon = lon2 - lon1
        a = np.sin(dlat / 2) ** 2 + np.cos(lat1) * np.cos(lat2) * np.sin(dlon / 2) ** 2
        return R * 2 * np.arcsin(np.sqrt(a))
```

```python
from fastapi import APIRouter, Depends, Query

@dashboard_router.get("/dashboard", summary="供应链看板数据")
async def get_dashboard(
    warehouse_id: int = Query(default=1),
    db: AsyncSession = Depends(get_db),
):
    """获取供应链看板核心数据"""
    now = datetime.now()
    month_start = now.replace(day=1, hour=0, minute=0, second=0, microsecond=0)

    # 供应商统计
    stmt = select(Supplier).where(Supplier.is_active == True)
    result = await db.execute(stmt)

    supplier_stats = {
        "total": len(active_suppliers),
        "grade_a": len([s for s in active_suppliers if s.grade.value == "A"]),
        "grade_b": len([s for s in active_suppliers if s.grade.value == "B"]),
        "grade_c": len([s for s in active_suppliers if s.grade.value == "C"]),
    }

    # 采购统计
    stmt = select(PurchaseOrder).where(PurchaseOrder.created_at >= month_start)
    result = await db.execute(stmt)
    month_orders = result.scalars().all()
    purchase_stats = {
        "month_orders": len(month_orders),
        "month_amount": sum(float(o.total_amount) for o in month_orders),
        "pending_orders": len([o for o in month_orders if o.status == PurchaseOrderStatus.SUBMITTED]),
        "completed_orders": len([o for o in month_orders if o.status == PurchaseOrderStatus.RECEIVED]),
    }

    # 库存健康度
    stmt = select(Inventory).where(Inventory.warehouse_id == warehouse_id)
    result = await db.execute(stmt)
    inventories = result.scalars().all()

    stockout_count = 0
    surplus_count = 0
    total_value = 0.0

    for inv in inventories:
        sku = await db.get(SKU, inv.sku_id)
        if sku:
            total_value += inv.available_qty * float(sku.cost_price)
        if inv.available_qty <= inv.safety_stock:
            stockout_count += 1
        elif inv.available_qty > inv.safety_stock * 3:
            surplus_count += 1

    inventory_stats = {
        "total_skus": len(inventories),
        "total_value": round(total_value, 2),
        "stockout_count": stockout_count,
        "surplus_count": surplus_count,
        "health_rate": round(1 - stockout_count / max(len(inventories), 1), 3),
    }

    return {
        "supplier_stats": supplier_stats,
        "purchase_stats": purchase_stats,
        "inventory_stats": inventory_stats,
        "timestamp": str(now),
    }
```

```python
    PURCHASE_ORDER_CREATED = "purchase_order_created"

    FORECAST_UPDATED = "forecast_updated"


class SupplyChainEvent:
    """供应链事件"""

    def __init__(
        self,
        event_type: EventType,
        entity_id: str,
        entity_type: str,
        data: dict,
        severity: str = "info",
    ):
        self.event_type = event_type
        self.entity_id = entity_id
        self.entity_type = entity_type
        self.data = data
        self.severity = severity
        self.timestamp = datetime.now()

    def to_dict(self) -> dict:
        return {
            "event_type": self.event_type.value,
            "entity_id": self.entity_id,
            "entity_type": self.entity_type,
            "data": self.data,
            "severity": self.severity,
            "timestamp": self.timestamp.isoformat(),
        }


class EventNotifier:
    """事件通知服务"""

    def __init__(self, redis_client):
        self.redis = redis_client

    async def notify(self, event: SupplyChainEvent) -> None:
        """发送事件通知"""
        channel = f"events:{event.entity_type}"
        await self.redis.publish(channel, json.dumps(event.to_dict()))

        # 同时存储到事件列表
        key = f"events:recent:{event.entity_type}"
        await self.redis.lpush(key, json.dumps(event.to_dict()))
        await self.redis.ltrim(key, 0, 99)  # 保留最近100条
```

### 采购审批工作流

AI生成的补货建议和采购订单必须经过人工审核确认后方可执行。

```python
from enum import Enum
from datetime import datetime
from uuid import uuid4

class ApprovalStatus(str, Enum):
    PENDING = "pending"
    APPROVED = "approved"
    REJECTED = "rejected"
    MODIFIED = "modified"

# 库存平衡公式
# 期末库存 = 期初库存 + 入库量 - 出库量
# 期末库存 = 期初库存 + 采购入库 + 生产入库 - 销售出库 - 调拨出库 - 损耗

def check_inventory_balance(beginning: float, inbound: float, outbound: float,
                             actual_ending: float, tolerance: float = 0.01) -> bool:
    """校验库存平衡：期末 = 期初 + 入库 - 出库"""
    expected_ending = beginning + inbound - outbound
    return abs(expected_ending - actual_ending) <= tolerance * beginning

class PurchaseApprovalWorkflow:
    """采购审批工作流"""

    def __init__(self, db_session):
        self.db = db_session

    # 需要审批的阈值
    AUTO_APPROVE_THRESHOLD = 50_000   # 5万以下自动审批
    MANAGER_APPROVE_THRESHOLD = 200_000  # 20万以下经理审批
    DIRECTOR_APPROVE_THRESHOLD = 1_000_000  # 100万以下总监审批

    async def submit_for_approval(
        self,
        purchase_order: dict,
        submitted_by: str,
    ) -> dict:
        """提交采购订单审批"""
        total_amount = purchase_order["total_amount"]

        approval = {
            "approval_id": f"APR-{uuid4().hex[:8]}",
            "po_id": purchase_order["po_id"],
            "total_amount": total_amount,
            "status": ApprovalStatus.PENDING,
            "submitted_by": submitted_by,
            "submitted_at": datetime.now(),
        }

        # 自动审批（低金额+低风险）
        if (total_amount <= self.AUTO_APPROVE_THRESHOLD
            and purchase_order.get("risk_level") != "high"):
            approval["status"] = ApprovalStatus.APPROVED
            approval["auto_approved"] = True
        else:
            # 确定审批级别
            if total_amount <= self.MANAGER_APPROVE_THRESHOLD:
                approval["required_approver"] = "manager"
            elif total_amount <= self.DIRECTOR_APPROVE_THRESHOLD:
                approval["required_approver"] = "director"
            else:
                approval["required_approver"] = "vp"

            # 高风险订单需要额外说明
            if purchase_order.get("risk_level") == "high":
                approval["requires_justification"] = True

        await self.db.save(approval)
        return approval

    async def approve(
        self,
        approval_id: str,
        approver: str,
        decision: str,  # approved / rejected / modified
        modified_order: dict | None = None,
        notes: str | None = None,
    ) -> dict:
        """审批处理"""
        approval = await self.db.get(approval_id)

        approval["status"] = decision
        approval["approved_by"] = approver
        approval["approved_at"] = datetime.now()
        approval["approval_notes"] = notes

        if decision == "modified" and modified_order:
            approval["modified_order"] = modified_order

        await self.db.update(approval)
        return approval
```
