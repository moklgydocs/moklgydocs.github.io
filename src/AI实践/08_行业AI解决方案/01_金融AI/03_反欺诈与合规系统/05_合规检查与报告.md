# 合规检查与报告

## 概述

合规检查与报告服务是反欺诈与合规系统的重要组成部分，负责KYC客户识别、AML反洗钱检查、大额交易报告、SAR可疑活动报告自动生成以及审计日志记录。本章实现完整的合规业务逻辑和API接口。

### KYC服务实现

```python
# app/services/compliance_service.py
import uuid
from datetime import datetime
from typing import Optional
from sqlalchemy.orm import Session

from app.models.account import Account
from app.models.compliance import ComplianceCheck, CheckType, CheckStatus, SARReport, SARStatus
from app.models.fraud import FraudAlert, AlertStatus
from app.schemas.compliance import KYCCheckRequest, AMLCheckRequest
from app.utils.redis_client import redis_client


class KYCService:
    """KYC客户识别服务"""

    # 高风险国家/地区代码
    HIGH_RISK_COUNTRIES = {"KP", "IR", "SY", "CU", "VE", "MM", "AF"}

    # 高风险行业
    HIGH_RISK_INDUSTRIES = {
        "虚拟货币", "赌博", "武器", "贵金属交易",
        "地下钱庄", "网络博彩", "虚拟资产",
    }

    def __init__(self, db: Session):
        self.db = db

    def perform_kyc_check(self, request: KYCCheckRequest) -> dict:
        """执行KYC检查"""
        account = self.db.query(Account).filter(
            Account.id == request.account_id
        ).first()
        if not account:
            raise ValueError(f"账户 {request.account_id} 不存在")

        check_id = f"KYC-{uuid.uuid4().hex[:12].upper()}"
        checks = []
        overall_passed = True
        risk_level = "low"
        findings = []

        # 1. 证件校验
        id_check = self._validate_id(request.id_type, request.id_number)
        checks.append(id_check)
        if not id_check["passed"]:
            overall_passed = False
            findings.append(id_check["detail"])

        # 2. 黑名单检查
        blacklist_check = self._check_blacklist(request.full_name, request.id_number)
        checks.append(blacklist_check)
        if not blacklist_check["passed"]:
            overall_passed = False
            risk_level = "critical"
            findings.append(blacklist_check["detail"])

        # 3. 高风险国家检查
        country_check = self._check_country(request.nationality)
        checks.append(country_check)
        if not country_check["passed"]:
            risk_level = max(risk_level, "medium", key=self._risk_order)
            findings.append(country_check["detail"])

        # 4. PEP检查
        pep_check = self._check_pep(request.is_pep)
        checks.append(pep_check)
        if not pep_check["passed"]:
            risk_level = max(risk_level, "medium", key=self._risk_order)
            findings.append(pep_check["detail"])

        # 5. 高风险行业检查
        industry_check = self._check_industry(request.industry)
        checks.append(industry_check)
        if not industry_check["passed"]:
            risk_level = max(risk_level, "medium", key=self._risk_order)
            findings.append(industry_check["detail"])

        # 6. 尽职调查级别
        dd_level = self._determine_dd_level(risk_level)

        # 保存检查记录
        compliance_check = ComplianceCheck(
            check_id=check_id,
            account_id=request.account_id,
            check_type=CheckType.KYC,
            status=CheckStatus.PASSED if overall_passed else CheckStatus.FAILED,
            risk_level=risk_level,
            details={"checks": checks, "dd_level": dd_level},
            findings="; ".join(findings) if findings else "KYC检查通过",
            checked_by="system",
            checked_at=datetime.utcnow(),
        )
        self.db.add(compliance_check)

        # 更新账户风险等级
        account.risk_level = risk_level
        account.kyc_level = dd_level
        self.db.commit()
        self.db.refresh(compliance_check)

        return {
            "check_id": compliance_check.check_id,
            "status": compliance_check.status.value,
            "risk_level": risk_level,
            "dd_level": dd_level,
            "checks": checks,
            "findings": findings,
            "overall_passed": overall_passed,
        }

    def _validate_id(self, id_type: str, id_number: str) -> dict:
        """证件校验"""
        check = {
            "check_type": "id_validation",
            "passed": True,
            "detail": "",
        }

        if id_type == "身份证" and len(id_number) == 18:
            # 校验码验证
            weights = [7, 9, 10, 5, 8, 4, 2, 1, 6, 3, 7, 9, 10, 5, 8, 4, 2]
            check_chars = "10X98765432"
            try:
                total = sum(int(id_number[i]) * weights[i] for i in range(17))
                if id_number[17].upper() != check_chars[total % 11]:
                    check["passed"] = False
                    check["detail"] = "身份证校验码不匹配"
            except (ValueError, IndexError):
                check["passed"] = False
                check["detail"] = "身份证号格式无效"
        elif id_type == "护照":
            if len(id_number) < 5:
                check["passed"] = False
                check["detail"] = "护照号格式无效"

        return check

    def _check_blacklist(self, name: str, id_number: str) -> dict:
        """黑名单检查"""
        check = {
            "check_type": "blacklist",
            "passed": True,
            "detail": "",
        }

        # 模拟黑名单检查
        # 实际应用中应调用黑名单数据库或API
        sanction_lists = {
            "un_sanction": set(),
            "ofac": set(),
            "pep": set(),
        }

        for list_name, blacklist in sanction_lists.items():
            if name in blacklist or id_number in blacklist:
                check["passed"] = False
                check["detail"] = f"命中{name}制裁名单"
                break

        return check

    def _check_country(self, nationality: Optional[str]) -> dict:
        """高风险国家检查"""
        check = {
            "check_type": "country_risk",
            "passed": True,
            "detail": "",
        }

        if nationality and nationality in self.HIGH_RISK_COUNTRIES:
            check["passed"] = False
            check["detail"] = f"高风险国家/地区: {nationality}"

        return check

    def _check_pep(self, is_pep: bool) -> dict:
        """PEP检查"""
        return {
            "check_type": "pep",
            "passed": not is_pep,
            "detail": "政治公众人物(PEP)，需加强尽职调查" if is_pep else "",
        }

    def _check_industry(self, industry: Optional[str]) -> dict:
        """高风险行业检查"""
        check = {
            "check_type": "industry_risk",
            "passed": True,
            "detail": "",
        }

        if industry and industry in self.HIGH_RISK_INDUSTRIES:
            check["passed"] = False
            check["detail"] = f"高风险行业: {industry}"

        return check

    @staticmethod
    def _determine_dd_level(risk_level: str) -> str:
        """确定尽职调查级别"""
        level_map = {
            "low": "SDD",       # 简化尽职调查
            "medium": "CDD",    # 标准尽职调查
            "high": "EDD",      # 加强尽职调查
            "critical": "EDD",  # 加强尽职调查
        }
        return level_map.get(risk_level, "CDD")

    @staticmethod
    def _risk_order(level: str) -> int:
        """风险等级排序"""
        order = {"low": 0, "medium": 1, "high": 2, "critical": 3}
        return order.get(level, 0)
```

## 反洗钱合规检查流程

```python
class AMLService:
    """AML反洗钱检查服务"""

    # 大额交易报告阈值
    LARGE_AMOUNT_THRESHOLDS = {
        "cash": 50000,
        "corporate_transfer": 200000,
        "personal_transfer": 50000,
        "cross_border": 200000,
    }

    def __init__(self, db: Session):
        self.db = db

    def perform_aml_check(self, request: AMLCheckRequest) -> dict:
        """执行AML检查"""
        account = self.db.query(Account).filter(
            Account.id == request.account_id
        ).first()
        if not account:
            raise ValueError(f"账户 {request.account_id} 不存在")

        check_id = f"AML-{uuid.uuid4().hex[:12].upper()}"
        checks = []
        risk_level = "low"
        findings = []
        should_report = False

        # 1. 大额交易检查
        large_amount_check = self._check_large_amounts(request.account_id)
        checks.append(large_amount_check)
        if large_amount_check["has_large_amount"]:
            should_report = True
            risk_level = max(risk_level, "high", key=self._risk_order)
            findings.append(large_amount_check["detail"])

        # 2. 结构性拆分检查
        structuring_check = self._check_structuring(request.account_id)
        checks.append(structuring_check)
        if structuring_check["detected"]:
            should_report = True
            risk_level = max(risk_level, "high", key=self._risk_order)
            findings.append(structuring_check["detail"])

        # 3. 频繁交易检查
        frequency_check = self._check_frequency(request.account_id)
        checks.append(frequency_check)
        if frequency_check["suspicious"]:
            risk_level = max(risk_level, "medium", key=self._risk_order)
            findings.append(frequency_check["detail"])

        # 4. 高风险对手方检查
        counterparty_check = self._check_counterparties(request.account_id)
        checks.append(counterparty_check)
        if counterparty_check["high_risk"]:
            risk_level = max(risk_level, "medium", key=self._risk_order)
            findings.append(counterparty_check["detail"])

        # 5. 客户风险等级评估
        customer_risk = self._assess_customer_risk(account)
        risk_level = max(risk_level, customer_risk, key=self._risk_order)

        # 保存检查记录
        compliance_check = ComplianceCheck(
            check_id=check_id,
            account_id=request.account_id,
            transaction_id=request.transaction_id,
            check_type=CheckType.AML,
            status=CheckStatus.REVIEWING if should_report else CheckStatus.PASSED,
            risk_level=risk_level,
            details={"checks": checks, "should_report": should_report},
            findings="; ".join(findings) if findings else "AML检查通过",
            checked_by="system",
            checked_at=datetime.utcnow(),
        )
        self.db.add(compliance_check)

        # 更新账户风险等级
        account.risk_level = risk_level
        self.db.commit()
        self.db.refresh(compliance_check)

        return {
            "check_id": compliance_check.check_id,
            "status": compliance_check.status.value,
            "risk_level": risk_level,
            "should_report": should_report,
            "checks": checks,
            "findings": findings,
        }

    def _check_large_amounts(self, account_id: int) -> dict:
        """检查大额交易"""
        from app.models.transaction import Transaction

        # 查询近24小时交易
        from datetime import timedelta
        cutoff = datetime.utcnow() - timedelta(hours=24)
        transactions = self.db.query(Transaction).filter(
            Transaction.account_id == account_id,
            Transaction.timestamp >= cutoff,
        ).all()

        large_amounts = []
        for tx in transactions:
            threshold = self.LARGE_AMOUNT_THRESHOLDS.get(
                tx.transaction_type.value
                if hasattr(tx.transaction_type, 'value') else tx.transaction_type,
                50000,
            )
            if tx.amount >= threshold:
                large_amounts.append({
                    "transaction_id": tx.transaction_id,
                    "amount": tx.amount,
                    "type": tx.transaction_type.value
                    if hasattr(tx.transaction_type, 'value') else str(tx.transaction_type),
                    "threshold": threshold,
                })

        return {
            "check_type": "large_amount",
            "has_large_amount": len(large_amounts) > 0,
            "large_transactions": large_amounts,
            "detail": f"发现{len(large_amounts)}笔大额交易需报告" if large_amounts else "",
        }

    def _check_structuring(self, account_id: int) -> dict:
        """检查结构性拆分"""
        from app.models.transaction import Transaction
        from datetime import timedelta

        cutoff = datetime.utcnow() - timedelta(hours=24)
        transactions = self.db.query(Transaction).filter(
            Transaction.account_id == account_id,
            Transaction.timestamp >= cutoff,
        ).all()

        total = sum(tx.amount for tx in transactions)
        threshold = 50000
        all_below_threshold = all(tx.amount < threshold for tx in transactions)

        detected = (
            total >= threshold * 0.9
            and all_below_threshold
            and len(transactions) >= 3
        )

        return {
            "check_type": "structuring",
            "detected": detected,
            "total_amount": total,
            "transaction_count": len(transactions),
            "detail": (
                f"疑似结构性拆分: {len(transactions)}笔交易合计{total:.0f}元，"
                f"均低于{threshold}元报告阈值"
            ) if detected else "",
        }

    def _check_frequency(self, account_id: int) -> dict:
        """检查交易频率"""
        from app.models.transaction import Transaction
        from datetime import timedelta

        # 24小时交易频率
        cutoff_24h = datetime.utcnow() - timedelta(hours=24)
        count_24h = self.db.query(Transaction).filter(
            Transaction.account_id == account_id,
            Transaction.timestamp >= cutoff_24h,
        ).count()

        # 1小时交易频率
        cutoff_1h = datetime.utcnow() - timedelta(hours=1)
        count_1h = self.db.query(Transaction).filter(
            Transaction.account_id == account_id,
            Transaction.timestamp >= cutoff_1h,
        ).count()

        suspicious = count_1h > 20 or count_24h > 50

        return {
            "check_type": "frequency",
            "suspicious": suspicious,
            "count_1h": count_1h,
            "count_24h": count_24h,
            "detail": (
                f"交易频率异常: 1小时{count_1h}笔, 24小时{count_24h}笔"
            ) if suspicious else "",
        }

    def _check_counterparties(self, account_id: int) -> dict:
        """检查对手方风险"""
        from app.models.transaction import Transaction
        from datetime import timedelta

        cutoff = datetime.utcnow() - timedelta(hours=72)
        counterparties = self.db.query(Transaction).filter(
            Transaction.account_id == account_id,
            Transaction.timestamp >= cutoff,
        ).with_entities(Transaction.counterparty_account_id).distinct().all()

        # 简化的高风险对手方检查
        high_risk = False
        detail = ""

        return {
            "check_type": "counterparty",
            "high_risk": high_risk,
            "unique_counterparties": len(counterparties),
            "detail": detail,
        }

    def _assess_customer_risk(self, account: Account) -> str:
        """评估客户综合风险"""
        risk_score = 0.0

        if account.risk_level == "high":
            risk_score += 0.3
        elif account.risk_level == "medium":
            risk_score += 0.1

        if account.kyc_level == "SDD":
            risk_score += 0.1

        if account.balance > 10000000:
            risk_score += 0.1

        if risk_score >= 0.5:
            return "high"
        elif risk_score >= 0.2:
            return "medium"
        return "low"

    @staticmethod
    def _risk_order(level: str) -> int:
        order = {"low": 0, "medium": 1, "high": 2, "critical": 3}
        return order.get(level, 0)
```

class LargeAmountReportService:
    """大额交易自动报告服务"""

    def __init__(self, db: Session):
        self.db = db

    def generate_report(self, account_id: int, transaction_id: int = None) -> dict:
        """生成大额交易报告"""
        from app.models.transaction import Transaction
        from app.models.account import Account

        account = self.db.query(Account).filter(Account.id == account_id).first()
        if not account:
            raise ValueError("账户不存在")

        # 获取相关交易
        query = self.db.query(Transaction).filter(Transaction.account_id == account_id)
        if transaction_id:
            query = query.filter(Transaction.id == transaction_id)
        transactions = query.order_by(Transaction.timestamp.desc()).limit(100).all()

        report = {
            "report_type": "大额交易报告",
            "report_date": datetime.utcnow().strftime("%Y-%m-%d"),
            "reporting_institution": "",
            "subject": {
                "account_id": account.id,
                "account_number": account.account_number,
                "account_type": account.account_type.value
                if hasattr(account.account_type, 'value') else str(account.account_type),
                "risk_level": account.risk_level,
            },
            "large_transactions": [
                {
                    "transaction_id": tx.transaction_id,
                    "amount": tx.amount,
                    "currency": tx.currency,
                    "type": tx.transaction_type.value
                    if hasattr(tx.transaction_type, 'value') else str(tx.transaction_type),
                    "timestamp": tx.timestamp.isoformat(),
                    "channel": tx.channel,
                }
                for tx in transactions
                if tx.amount >= 50000
            ],
            "total_large_amount": sum(tx.amount for tx in transactions if tx.amount >= 50000),
        }

        # 保存合规检查记录
        compliance_check = ComplianceCheck(
            check_id=f"LAR-{uuid.uuid4().hex[:12].upper()}",
            account_id=account_id,
            transaction_id=transaction_id,
            check_type=CheckType.LARGE_AMOUNT,
            status=CheckStatus.PASSED,
            risk_level=account.risk_level,
            details=report,
            findings=f"大额交易报告: {len(report['large_transactions'])}笔",
            checked_by="system",
            checked_at=datetime.utcnow(),
        )
        self.db.add(compliance_check)
        self.db.commit()

        report["check_id"] = compliance_check.check_id
        return report

## SAR可疑活动报告自动生成

```python
class SARReportService:
    """SAR报告自动生成服务"""

    def __init__(self, db: Session, llm_client=None):
        self.db = db
        self.llm_client = llm_client

    def generate_sar(self, compliance_check_id: int, use_llm: bool = True) -> dict:
        """生成SAR报告"""
        compliance_check = self.db.query(ComplianceCheck).filter(
            ComplianceCheck.id == compliance_check_id
        ).first()
        if not compliance_check:
            raise ValueError("合规检查记录不存在")

        account = self.db.query(Account).filter(
            Account.id == compliance_check.account_id
        ).first()

        report_id = f"SAR-{uuid.uuid4().hex[:12].upper()}"

        # 获取相关交易
        from app.models.transaction import Transaction
        from datetime import timedelta
        cutoff = datetime.utcnow() - timedelta(days=30)
        transactions = self.db.query(Transaction).filter(
            Transaction.account_id == account.id,
            Transaction.timestamp >= cutoff,
        ).order_by(Transaction.timestamp.desc()).limit(100).all()

        # 构建报告内容
        activity_description = self._generate_activity_description(
            compliance_check, transactions
        )
        suspicious_indicators = self._extract_suspicious_indicators(compliance_check)

        # LLM辅助生成
        llm_description = None
        if use_llm and self.llm_client:
            try:
                llm_description = self.llm_client.generate_sar_description({
                    "account_id": account.id,
                    "amount": sum(tx.amount for tx in transactions),
                    "transaction_type": "",
                    "triggered_rules": suspicious_indicators,
                    "ml_score": 0,
                    "recent_tx_count": len(transactions),
                })
            except Exception:
                llm_description = None

        # 生成建议
        recommendation = self._generate_recommendation(compliance_check.risk_level)

        # 保存SAR报告
        sar_report = SARReport(
            report_id=report_id,
            compliance_check_id=compliance_check_id,
            account_id=account.id,
            status=SARStatus.DRAFT,
            subject_info={
                "account_number": account.account_number,
                "account_type": account.account_type.value
                if hasattr(account.account_type, 'value') else str(account.account_type),
                "risk_level": account.risk_level,
            },
            activity_description=llm_description or activity_description,
            suspicious_indicators=suspicious_indicators,
            transaction_details={
                "total_amount": sum(tx.amount for tx in transactions),
                "transaction_count": len(transactions),
                "high_risk_count": sum(
                    1 for tx in transactions
                    if tx.risk_level in ["high", "critical"]
                ),
            },
            recommendation=recommendation,
            generated_by_llm=llm_description is not None,
        )
        self.db.add(sar_report)
        self.db.commit()
        self.db.refresh(sar_report)

        return {
            "report_id": sar_report.report_id,
            "status": sar_report.status.value,
            "activity_description": sar_report.activity_description,
            "suspicious_indicators": sar_report.suspicious_indicators,
            "recommendation": sar_report.recommendation,
            "generated_by_llm": sar_report.generated_by_llm,
            "created_at": sar_report.created_at.isoformat(),
        }

    def _generate_activity_description(self, check: ComplianceCheck, transactions: list) -> str:
        """生成活动描述"""
        parts = [
            f"账户{check.account_id}在近期交易中触发了{check.check_type.value}检查。",
        ]

        if check.findings:
            parts.append(f"检查发现: {check.findings}")

        if transactions:
            total_amount = sum(tx.amount for tx in transactions)
            parts.append(
                f"近30天共{len(transactions)}笔交易，合计金额{total_amount:.2f}元。"
            )

            high_risk = [tx for tx in transactions if tx.risk_level in ["high", "critical"]]
            if high_risk:
                parts.append(f"其中{len(high_risk)}笔交易被标记为高风险。")

        return " ".join(parts)

    def _extract_suspicious_indicators(self, check: ComplianceCheck) -> dict:
        """提取可疑指标"""
        indicators = {}

        if check.details and isinstance(check.details, dict):
            checks = check.details.get("checks", [])
            for c in checks:
                if c.get("check_type") and not c.get("passed", True):
                    indicators[c["check_type"]] = c.get("detail", "")

        return indicators

    @staticmethod
    def _generate_recommendation(risk_level: str) -> str:
        """生成处理建议"""
        recommendations = {
            "critical": "建议立即冻结账户，向监管机构提交SAR报告，配合执法调查",
            "high": "建议强化监控，限制交易额度，提交SAR报告",
            "medium": "建议加强关注，提升客户风险等级，持续监控",
            "low": "建议继续常规监控，记录本次分析结果",
        }
        return recommendations.get(risk_level, "建议继续常规监控")
```
