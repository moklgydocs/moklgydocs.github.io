# 交易监控服务

## 概述

交易监控服务是反欺诈与合规系统的核心入口，负责接收交易数据、执行实时风控规则、记录交易流水，并提供交易查询和监控仪表盘数据接口。

```python
@router.post("", response_model=APIResponse, summary="交易上报")
def create_transaction(
    tx_data: TransactionCreate,
    db: Session = Depends(get_db),
):
    """上报单笔交易，执行实时风控检查"""
    # 验证账户存在且有效
    account = db.query(Account).filter(Account.id == tx_data.account_id).first()
    if not account:
        raise HTTPException(status_code=404, detail="账户不存在")
    if account.status != AccountStatus.ACTIVE:
        raise HTTPException(status_code=400, detail="账户状态异常")

    # 创建交易记录
    tx_service = TransactionService(db)
    transaction = tx_service.create_transaction(tx_data)

    # 执行风控规则检查
    risk_result = tx_service.evaluate_risk(transaction)

    # 更新交易状态和风险信息
    if risk_result["should_block"]:
        transaction.status = TransactionStatus.BLOCKED
        transaction.risk_score = risk_result["risk_score"]
        transaction.risk_level = risk_result["risk_level"]
    elif risk_result["risk_level"] in ["high", "critical"]:
        transaction.status = TransactionStatus.REVIEWING
        transaction.risk_score = risk_result["risk_score"]
        transaction.risk_level = risk_result["risk_level"]
    else:
        transaction.status = TransactionStatus.COMPLETED
        transaction.risk_score = risk_result["risk_score"]
        transaction.risk_level = risk_result["risk_level"]

    transaction.processed_at = datetime.utcnow()
    db.commit()
    db.refresh(transaction)

    return APIResponse(
        data=TransactionResponse.model_validate(transaction).model_dump()
    )


@router.get("", response_model=APIResponse, summary="交易查询")
def list_transactions(
    account_id: Optional[int] = Query(None),
    transaction_type: Optional[str] = Query(None),
    status: Optional[str] = Query(None),
    min_amount: Optional[float] = Query(None),
    max_amount: Optional[float] = Query(None),
    risk_level: Optional[str] = Query(None),
    start_time: Optional[datetime] = Query(None),
    end_time: Optional[datetime] = Query(None),
    page: int = Query(1, ge=1),
    page_size: int = Query(20, ge=1, le=100),
    db: Session = Depends(get_db),
):
    """分页查询交易记录，支持多条件筛选"""
    query = db.query(Transaction)

    if account_id:
        query = query.filter(Transaction.account_id == account_id)
    if transaction_type:
        query = query.filter(Transaction.transaction_type == transaction_type)
    if status:
        query = query.filter(Transaction.status == status)
    if min_amount is not None:
        query = query.filter(Transaction.amount >= min_amount)
    if max_amount is not None:
        query = query.filter(Transaction.amount <= max_amount)
    if risk_level:
        query = query.filter(Transaction.risk_level == risk_level)
    if start_time:
        query = query.filter(Transaction.timestamp >= start_time)
    if end_time:
        query = query.filter(Transaction.timestamp <= end_time)

    total = query.count()
    total_pages = math.ceil(total / page_size)
    items = (
        query.order_by(Transaction.timestamp.desc())
        .offset((page - 1) * page_size)
        .limit(page_size)
        .all()
    )

    return APIResponse(data=TransactionListResponse(
        items=[TransactionResponse.model_validate(tx).model_dump() for tx in items],
        total=total,
        page=page,
        page_size=page_size,
        total_pages=total_pages,
    ).model_dump())

@router.post("/batch", response_model=APIResponse, summary="批量交易导入")
def batch_create_transactions(
    batch_data: TransactionBatchCreate,
    db: Session = Depends(get_db),
):
    """批量导入交易记录"""
    tx_service = TransactionService(db)
    results = []

    for tx_data in batch_data.transactions:
        try:
            transaction = tx_service.create_transaction(tx_data)
            risk_result = tx_service.evaluate_risk(transaction)

            if risk_result["should_block"]:
                transaction.status = TransactionStatus.BLOCKED
            else:
                transaction.status = TransactionStatus.COMPLETED
            transaction.risk_score = risk_result["risk_score"]
            transaction.risk_level = risk_result["risk_level"]
            transaction.processed_at = datetime.utcnow()

            results.append({
                "transaction_id": transaction.transaction_id,
                "status": transaction.status.value,
                "risk_level": transaction.risk_level,
            })
        except Exception as e:
            results.append({
                "account_id": tx_data.account_id,
                "error": str(e),
            })

    db.commit()

    return APIResponse(data={
        "total": len(batch_data.transactions),
        "results": results,
    })


@router.get("/dashboard/stats", response_model=APIResponse, summary="监控仪表盘数据")
def get_dashboard_stats(db: Session = Depends(get_db)):
    """获取交易监控仪表盘统计数据"""
    now = datetime.utcnow()
    last_24h = now - timedelta(hours=24)

    # 总交易统计
    total_transactions = db.query(func.count(Transaction.id)).filter(
        Transaction.timestamp >= last_24h
    ).scalar() or 0

    total_amount = db.query(func.sum(Transaction.amount)).filter(
        Transaction.timestamp >= last_24h
    ).scalar() or 0.0

    # 高风险交易
    high_risk_count = db.query(func.count(Transaction.id)).filter(
        and_(
            Transaction.timestamp >= last_24h,
            Transaction.risk_level.in_(["high", "critical"]),
        )
    ).scalar() or 0

    # 被阻断交易
    blocked_count = db.query(func.count(Transaction.id)).filter(
        and_(
            Transaction.timestamp >= last_24h,
            Transaction.status == TransactionStatus.BLOCKED,
        )
    ).scalar() or 0

    # 待处理告警
    pending_alerts = db.query(func.count(FraudAlert.id)).filter(
        FraudAlert.status == AlertStatus.PENDING
    ).scalar() or 0

    # 交易类型分布
    type_dist = dict(
        db.query(Transaction.transaction_type, func.count(Transaction.id))
        .filter(Transaction.timestamp >= last_24h)
        .group_by(Transaction.transaction_type)
        .all()
    )

    # 风险等级分布
    risk_dist = dict(
        db.query(Transaction.risk_level, func.count(Transaction.id))
        .filter(and_(Transaction.timestamp >= last_24h, Transaction.risk_level.isnot(None)))
        .group_by(Transaction.risk_level)
        .all()
    )

    # 小时趋势
    hourly_trend = []
    for hour in range(24):
        hour_start = now.replace(minute=0, second=0, microsecond=0) - timedelta(hours=23 - hour)
        hour_end = hour_start + timedelta(hours=1)
        count = db.query(func.count(Transaction.id)).filter(
            and_(Transaction.timestamp >= hour_start, Transaction.timestamp < hour_end)
        ).scalar() or 0
        hourly_trend.append({
            "hour": hour_start.strftime("%H:00"),
            "count": count,
        })

    stats = DashboardStats(
        total_transactions=total_transactions,
        total_amount=float(total_amount),
        high_risk_count=high_risk_count,
        blocked_count=blocked_count,
        pending_alerts=pending_alerts,
        transaction_type_distribution=type_dist,
        risk_level_distribution=risk_dist,
        hourly_trend=hourly_trend,
    )

    return APIResponse(data=stats.model_dump())
```

```python
# app/services/transaction_service.py
import uuid
from datetime import datetime, timedelta
from sqlalchemy.orm import Session
from sqlalchemy import and_

from app.models.transaction import Transaction, TransactionType, TransactionStatus
from app.models.account import Account, AccountStatus
from app.schemas.transaction import TransactionCreate
from app.services.fraud_engine import RuleEngine
from app.utils.redis_client import redis_client


class TransactionService:
    """交易服务"""

    def __init__(self, db: Session):
        self.db = db
        self.rule_engine = RuleEngine()

    def create_transaction(self, tx_data: TransactionCreate) -> Transaction:
        """创建交易记录"""
        # 验证账户
        account = self.db.query(Account).filter(
            Account.id == tx_data.account_id
        ).first()
        if not account:
            raise ValueError(f"账户 {tx_data.account_id} 不存在")
        if account.status != AccountStatus.ACTIVE:
            raise ValueError(f"账户 {tx_data.account_id} 状态异常: {account.status}")

        # 检查日累计限额
        today_start = datetime.utcnow().replace(
            hour=0, minute=0, second=0, microsecond=0
        )
        daily_total = self.db.query(Transaction).filter(
            and_(
                Transaction.account_id == tx_data.account_id,
                Transaction.timestamp >= today_start,
                Transaction.status != TransactionStatus.BLOCKED,
            )
        ).with_entities(
            self.db.func.sum(Transaction.amount)
        ).scalar() or 0.0

        if daily_total + tx_data.amount > account.daily_limit:
            raise ValueError(
                f"超出日累计限额: 当前累计{daily_total}, "
                f"本笔{tx_data.amount}, 限额{account.daily_limit}"
            )

        # 生成交易ID
        transaction_id = f"TX{uuid.uuid4().hex[:16].upper()}"

        # 创建交易记录
        transaction = Transaction(
            transaction_id=transaction_id,
            account_id=tx_data.account_id,
            counterparty_account_id=tx_data.counterparty_account_id,
            transaction_type=tx_data.transaction_type,
            amount=tx_data.amount,
            currency=tx_data.currency,
            description=tx_data.description,
            channel=tx_data.channel,
            location=tx_data.location,
            latitude=tx_data.latitude,
            longitude=tx_data.longitude,
            ip_address=tx_data.ip_address,
            device_id=tx_data.device_id,
            status=TransactionStatus.PENDING,
            timestamp=tx_data.timestamp,
        )

        self.db.add(transaction)
        self.db.flush()

        # 更新Redis计数器
        self._update_redis_counters(tx_data)

        return transaction

    def evaluate_risk(self, transaction: Transaction) -> dict:
        """评估交易风险"""
        # 构建规则引擎上下文
        context = self._build_rule_context(transaction)

        # 执行规则引擎
        rule_results = self.rule_engine.evaluate(context)

        # 计算综合风险评分
        risk_score = self._calculate_risk_score(rule_results)
        risk_level = self._determine_risk_level(risk_score)

        should_block = risk_score >= 0.8 or any(
            r.get("risk_level") == "critical" for r in rule_results
        )

        return {
            "rule_results": rule_results,
            "risk_score": risk_score,
            "risk_level": risk_level,
            "should_block": should_block,
        }

    def _build_rule_context(self, transaction: Transaction) -> dict:
        """构建规则引擎上下文"""
        # 从Redis获取实时统计
        account_id = transaction.account_id
        tx_count_5min = redis_client.get_counter(f"tx:count:{account_id}:300")
        tx_count_1h = redis_client.get_counter(f"tx:count:{account_id}:3600")
        tx_count_24h = redis_client.get_counter(f"tx:count:{account_id}:86400")
        tx_amount_1h = redis_client.get_float(f"tx:amount:{account_id}:3600")

        context = {
            "transaction_id": transaction.transaction_id,
            "account_id": transaction.account_id,
            "amount": transaction.amount,
            "transaction_type": transaction.transaction_type.value
            if isinstance(transaction.transaction_type, TransactionType)
            else transaction.transaction_type,
            "currency": transaction.currency,
            "channel": transaction.channel or "",
            "location": transaction.location or "",
            "hour": transaction.timestamp.hour,
            "day_of_week": transaction.timestamp.weekday(),
            "is_weekend": transaction.timestamp.weekday() >= 5,
            "is_night": transaction.timestamp.hour < 6 or transaction.timestamp.hour >= 22,
            "tx_count_300s": tx_count_5min,
            "tx_count_3600s": tx_count_1h,
            "tx_count_86400s": tx_count_24h,
            "tx_amount_3600s": tx_amount_1h,
        }

        return context

    def _calculate_risk_score(self, rule_results: list) -> float:
        """计算综合风险评分"""
        if not rule_results:
            return 0.0

        level_weights = {
            "critical": 1.0,
            "high": 0.7,
            "medium": 0.4,
            "low": 0.2,
        }

        max_score = max(
            level_weights.get(r.get("risk_level", "low"), 0.2)
            for r in rule_results
        )

        # 多规则触发加权
        count_factor = min(len(rule_results) * 0.1, 0.3)
        score = min(max_score + count_factor, 1.0)

        return round(score, 4)

    def _determine_risk_level(self, risk_score: float) -> str:
        """确定风险等级"""
        if risk_score >= 0.8:
            return "critical"
        elif risk_score >= 0.6:
            return "high"
        elif risk_score >= 0.3:
            return "medium"
        else:
            return "low"

    def _update_redis_counters(self, tx_data: TransactionCreate):
        """更新Redis实时计数器"""
        account_id = tx_data.account_id
        windows = {300: "300", 3600: "3600", 86400: "86400"}

        for seconds, label in windows.items():
            redis_client.increment_counter(f"tx:count:{account_id}:{label}", seconds)
            redis_client.increment_float(
                f"tx:amount:{account_id}:{label}", tx_data.amount, seconds
            )
```

## 实时风控规则引擎

```python
# app/services/fraud_engine.py
from dataclasses import dataclass
from typing import Any
from datetime import datetime


@dataclass
class RuleCondition:
    """规则条件"""
    field: str
    operator: str  # gt, lt, eq, gte, lte, in, contains
    threshold: Any


class Rule:
    """风控规则"""

    def __init__(
        self,
        rule_id: str,
        name: str,
        rule_type: str,
        conditions: list[RuleCondition],
        logic: str = "AND",
        risk_level: str = "medium",
        description: str = "",
    ):
        self.rule_id = rule_id
        self.name = name
        self.rule_type = rule_type
        self.conditions = conditions
        self.logic = logic
        self.risk_level = risk_level
        self.description = description

    def evaluate(self, context: dict) -> dict | None:
        """评估规则"""
        results = [self._eval_condition(c, context) for c in self.conditions]

        triggered = (
            all(results) if self.logic == "AND"
            else any(results) if self.logic == "OR"
            else False
        )

        if triggered:
            return {
                "rule_id": self.rule_id,
                "rule_name": self.name,
                "rule_type": self.rule_type,
                "risk_level": self.risk_level,
                "description": self.description,
                "triggered_conditions": [
                    {"field": c.field, "threshold": c.threshold}
                    for c, r in zip(self.conditions, results) if r
                ],
            }
        return None

    def _eval_condition(self, condition: RuleCondition, context: dict) -> bool:
        """评估单个条件"""
        value = context.get(condition.field)
        if value is None:
            return False

        ops = {
            "gt": lambda v, t: v > t,
            "lt": lambda v, t: v < t,
            "eq": lambda v, t: v == t,
            "gte": lambda v, t: v >= t,
            "lte": lambda v, t: v <= t,
            "in": lambda v, t: v in t,
            "contains": lambda v, t: t in str(v),
        }
        op_fn = ops.get(condition.operator)
        if op_fn is None:
            return False
        try:
            return op_fn(value, condition.threshold)
        except (TypeError, ValueError):
            return False


class RuleEngine:

            ),
        ]

    def evaluate(self, context: dict) -> list[dict]:
        """评估所有规则"""
        results = []
        for rule in self.rules:
            result = rule.evaluate(context)
            if result:
                results.append(result)
        return results

    def add_rule(self, rule: Rule):
        """添加规则"""
        self.rules.append(rule)

    def remove_rule(self, rule_id: str):
        """移除规则"""
        self.rules = [r for r in self.rules if r.rule_id != rule_id]

    def get_rules(self) -> list[dict]:
        """获取所有规则"""
        return [
            {
                "rule_id": r.rule_id,
                "name": r.name,
                "type": r.rule_type,
                "risk_level": r.risk_level,
                "description": r.description,
                "conditions": [
                    {"field": c.field, "operator": c.operator, "threshold": c.threshold}
                    for c in r.conditions
                ],
                "logic": r.logic,
            }
            for r in self.rules
        ]
```

```python
import redis
import json
from typing import Optional, Any
import os


class RedisClient:
    """Redis客户端"""

    def __init__(self):
        self.client = redis.from_url(
            os.getenv("REDIS_URL", "redis://localhost:6379/0"),
            decode_responses=True,
        )

    def increment_counter(self, key: str, window_seconds: int = 3600) -> int:
        """递增计数器"""
        pipe = self.client.pipeline()
        pipe.incr(key)
        pipe.expire(key, window_seconds)
        result = pipe.execute()
        return result[0]

    def get_counter(self, key: str) -> int:
        """获取计数器值"""
        value = self.client.get(key)
        return int(value) if value else 0

    def increment_float(self, key: str, amount: float, window_seconds: int = 3600) -> float:
        """递增浮点数"""
        pipe = self.client.pipeline()
        pipe.incrbyfloat(key, amount)
        pipe.expire(key, window_seconds)
        result = pipe.execute()
        return float(result[0])

    def get_float(self, key: str) -> float:
        """获取浮点数值"""
        value = self.client.get(key)
        return float(value) if value else 0.0

    def set_cache(self, key: str, value: Any, ttl: int = 300) -> bool:
        """设置缓存"""
        return self.client.set(key, json.dumps(value, default=str), ex=ttl)

    def get_cache(self, key: str) -> Optional[Any]:
        """获取缓存"""
        value = self.client.get(key)
        return json.loads(value) if value else None

    def set_hash(self, key: str, mapping: dict) -> bool:
        """设置哈希"""
        return self.client.hset(key, mapping=mapping)

    def get_hash(self, key: str) -> dict:
        """获取哈希"""
        return self.client.hgetall(key)

    def publish(self, channel: str, message: dict) -> int:
        """发布消息"""
```

```python
from fastapi import WebSocket, WebSocketDisconnect
from typing import Set
import asyncio
import json


class ConnectionManager:
    """WebSocket连接管理器"""

    def __init__(self):
        self.active_connections: Set[WebSocket] = set()

    async def connect(self, websocket: WebSocket):
        await websocket.accept()
        self.active_connections.add(websocket)

    def disconnect(self, websocket: WebSocket):
        self.active_connections.discard(websocket)

    async def broadcast(self, message: dict):
        """广播消息给所有连接"""
        disconnected = set()
        for connection in self.active_connections:
            try:
                await connection.send_json(message)
            except Exception:
                disconnected.add(connection)
        self.active_connections -= disconnected
```

```python
from enum import Enum
from datetime import datetime

class AlertStatus(str, Enum):
    NEW = "new"                    # 新告警，待分配
    ASSIGNED = "assigned"          # 已分配分析师
    INVESTIGATING = "investigating" # 调查中
    ESCALATED = "escalated"         # 升级处理
    CONFIRMED_FRAUD = "confirmed"  # 确认欺诈
    FALSE_POSITIVE = "false_positive" # 误报
    CLOSED = "closed"               # 已关闭

def _compute_priority(fraud_score: float, risk_level: str) -> str:
    """根据欺诈分数和风险等级计算优先级"""
    if fraud_score >= 0.8 or risk_level == "HIGH":
        return "CRITICAL"
    elif fraud_score >= 0.5 or risk_level == "MEDIUM":
        return "HIGH"
    else:
        return "MEDIUM"

class FraudAlertCase:
    """欺诈告警工单"""

    @staticmethod
    async def create_case(

        transaction_id: str,
        fraud_score: float,
        risk_level: str,
        rule_triggers: list[str],
        db,
    ) -> dict:
        """从告警创建调查工单"""
        case = {
            "case_id": f"CASE-{uuid4().hex[:8]}",
            "alert_id": alert_id,
            "transaction_id": transaction_id,
            "fraud_score": fraud_score,
            "risk_level": risk_level,
            "rule_triggers": rule_triggers,
            "status": AlertStatus.NEW,
            "assigned_to": None,
            "priority": _compute_priority(fraud_score, risk_level),
            "created_at": datetime.now(),
            "sla_deadline": datetime.now() + timedelta(
                hours=2 if risk_level == "critical" else 24
            ),
            "actions": [],
        }
        query = insert(FraudAlertCase).values(**case)
        await db.execute(query)
        return case

    @staticmethod
    async def assign_case(case_id: str, analyst_id: str, db) -> None:
        """分配分析师"""
        await db.execute(
```
