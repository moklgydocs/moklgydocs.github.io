# 前端与部署

## React前端架构

前端基于React + Ant Design Pro构建，ECharts提供数据可视化能力。

### 技术选型

| 技术 | 版本 | 用途 |
|------|------|------|
| React | 18.x | UI框架 |
| Ant Design Pro | 6.x | 企业级UI组件 |
| UmiJS | 4.x | 应用框架 |
| ECharts | 5.x | 数据可视化 |
| Axios | 1.x | HTTP客户端 |
| Day.js | 1.x | 日期处理 |

### TransactionMonitor（交易监控仪表盘）

```tsx
// src/pages/TransactionMonitor/index.tsx
import React, { useEffect, useState } from 'react';
import { Row, Col, Card, Statistic, Table, Tag, Space } from 'antd';
import {
  ArrowUpOutlined,
  WarningOutlined,
  StopOutlined,
  ClockCircleOutlined,
} from '@ant-design/icons';
import ReactECharts from 'echarts-for-react';
import { getDashboardStats } from '@/services/api';
import dayjs from 'dayjs';


const TransactionMonitor: React.FC = () => {
  const [stats, setStats] = useState<any>(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    fetchStats();
    const interval = setInterval(fetchStats, 30000);
    return () => clearInterval(interval);
  }, []);

  const fetchStats = async () => {
    try {
      const res = await getDashboardStats();
      if (res.code === 0) {
        setStats(res.data);
      }
    } finally {
      setLoading(false);
    }
  };

  const riskLevelColor: Record<string, string> = {
    low: 'green',
    medium: 'orange',
    high: 'red',
    critical: 'magenta',
  };

  // 24小时交易趋势图配置
  const hourlyTrendOption = {
    title: { text: '24小时交易趋势', left: 'center' },
    tooltip: { trigger: 'axis' },
    xAxis: {
      type: 'category',
      data: stats?.hourly_trend?.map((h: any) => h.hour) || [],
    },
    yAxis: { type: 'value', name: '交易笔数' },
    series: [{
      data: stats?.hourly_trend?.map((h: any) => h.count) || [],
      type: 'line',
      smooth: true,
      areaStyle: { opacity: 0.3 },
      itemStyle: { color: '#1890ff' },
    }],
    grid: { left: '10%', right: '5%', bottom: '15%', top: '15%' },
  };

  // 交易类型分布图配置
  const typeDistOption = {
    title: { text: '交易类型分布', left: 'center' },
    tooltip: { trigger: 'item' },
    series: [{
      type: 'pie',
      radius: ['40%', '70%'],
      data: Object.entries(stats?.transaction_type_distribution || {}).map(
        ([name, value]) => ({ name, value })
      ),
      label: { show: true, formatter: '{b}: {d}%' },
    }],
  };

  // 风险等级分布图配置
  const riskDistOption = {
    title: { text: '风险等级分布', left: 'center' },
    tooltip: { trigger: 'item' },
    series: [{
      type: 'pie',
      radius: ['40%', '70%'],
      data: Object.entries(stats?.risk_level_distribution || {}).map(
        ([name, value]) => ({
          name: name || '未评估',
          value,
          itemStyle: { color: riskLevelColor[name] || '#999' },
        })
      ),
      label: { show: true, formatter: '{b}: {d}%' },
    }],
  };

  return (
    <div>
      <Row gutter={[16, 16]}>
        <Col span={6}>
          <Card loading={loading}>
            <Statistic
              title="24小时交易总量"
              value={stats?.total_transactions || 0}
              prefix={<ArrowUpOutlined />}
            />
          </Card>
        </Col>
        <Col span={6}>
          <Card loading={loading}>
            <Statistic
              title="24小时交易总额"
              value={stats?.total_amount || 0}
              precision={2}
              prefix="¥"
            />
          </Card>
        </Col>
        <Col span={6}>
          <Card loading={loading}>
            <Statistic
              title="高风险交易"
              value={stats?.high_risk_count || 0}
              prefix={<WarningOutlined />}
              valueStyle={{ color: '#cf1322' }}
            />
          </Card>
        </Col>
        <Col span={6}>
          <Card loading={loading}>
            <Statistic
              title="被阻断交易"
              value={stats?.blocked_count || 0}
              prefix={<StopOutlined />}
              valueStyle={{ color: '#fa541c' }}
            />
          </Card>
        </Col>
      </Row>

      <Row gutter={[16, 16]} style={{ marginTop: 16 }}>
        <Col span={6}>
          <Card loading={loading}>
            <Statistic
              title="待处理告警"
              value={stats?.pending_alerts || 0}
              prefix={<ClockCircleOutlined />}
              valueStyle={{ color: '#d48806' }}
            />
          </Card>
        </Col>
        <Col span={18}>
          <Card>
            <ReactECharts option={hourlyTrendOption} style={{ height: 300 }} />
          </Card>
        </Col>
      </Row>

      <Row gutter={[16, 16]} style={{ marginTop: 16 }}>
        <Col span={12}>
          <Card>
            <ReactECharts option={typeDistOption} style={{ height: 300 }} />
          </Card>
        </Col>
        <Col span={12}>
          <Card>
            <ReactECharts option={riskDistOption} style={{ height: 300 }} />
          </Card>
        </Col>
      </Row>
    </div>
  );
};

export default TransactionMonitor;
```

### FraudAlertList（欺诈告警列表）

```tsx
// src/pages/FraudAlertList/index.tsx
import React, { useEffect, useState } from 'react';
import { Table, Tag, Button, Modal, Descriptions, Space, message } from 'antd';
import { EyeOutlined, CheckOutlined, CloseOutlined } from '@ant-design/icons';
import { getFraudAlerts, updateAlertStatus } from '@/services/api';
import dayjs from 'dayjs';


const FraudAlertList: React.FC = () => {
  const [alerts, setAlerts] = useState<any[]>([]);
  const [total, setTotal] = useState(0);
  const [page, setPage] = useState(1);
  const [loading, setLoading] = useState(false);
  const [detailVisible, setDetailVisible] = useState(false);
  const [currentAlert, setCurrentAlert] = useState<any>(null);

  useEffect(() => {
    fetchAlerts();
  }, [page]);

  const fetchAlerts = async () => {
    setLoading(true);
    try {
      const res = await getFraudAlerts({ page, page_size: 20 });
      if (res.code === 0) {
        setAlerts(res.data.items || []);
        setTotal(res.data.total || 0);
      }
    } finally {
      setLoading(false);
    }
  };

  const handleStatusUpdate = async (alertId: string, status: string) => {
    try {
      await updateAlertStatus(alertId, status);
      message.success('状态更新成功');
      fetchAlerts();
    } catch {
      message.error('状态更新失败');
    }
  };

  const riskLevelColor: Record<string, string> = {
    low: 'green',
    medium: 'orange',
    high: 'red',
    critical: 'magenta',
  };

  const statusColor: Record<string, string> = {
    pending: 'gold',
    confirmed: 'red',
    false_positive: 'green',
    escalated: 'volcano',
    resolved: 'blue',
  };

  const statusLabel: Record<string, string> = {
    pending: '待处理',
    confirmed: '已确认',
    false_positive: '误报',
    escalated: '已升级',
    resolved: '已解决',
  };

  const columns = [
    {
      title: '告警ID',
      dataIndex: 'alert_id',
      key: 'alert_id',
      width: 150,
    },
    {
      title: '交易ID',
      dataIndex: 'transaction_id',
      key: 'transaction_id',
      width: 150,
    },
    {
      title: '风险评分',
      dataIndex: 'risk_score',
      key: 'risk_score',
      width: 100,
      render: (score: number) => (
        <span style={{ color: score >= 0.8 ? '#cf1322' : score >= 0.5 ? '#fa8c16' : '#52c41a' }}>
          {(score * 100).toFixed(1)}%
        </span>
      ),
    },
    {
      title: '风险等级',
      dataIndex: 'risk_level',
      key: 'risk_level',
      width: 100,
      render: (level: string) => (
        <Tag color={riskLevelColor[level]}>{level?.toUpperCase()}</Tag>
      ),
    },
    {
      title: 'ML评分',
      dataIndex: 'ml_score',
      key: 'ml_score',
      width: 100,
      render: (score: number) => score ? `${(score * 100).toFixed(1)}%` : '-',
    },
    {
      title: '状态',
      dataIndex: 'status',
      key: 'status',
      width: 100,
      render: (status: string) => (
        <Tag color={statusColor[status]}>{statusLabel[status] || status}</Tag>
      ),
    },
    {
      title: '触发规则',
      dataIndex: 'triggered_rules',
      key: 'triggered_rules',
      width: 200,
      render: (rules: any) => {
        if (!rules) return '-';
        const ruleList = Array.isArray(rules) ? rules : [];
        return ruleList.map((r: any, i: number) => (
          <Tag key={i}>{typeof r === 'string' ? r : r.rule_name || r}</Tag>
        ));
      },
    },
    {
      title: '时间',
      dataIndex: 'created_at',
      key: 'created_at',
      width: 180,
      render: (t: string) => t ? dayjs(t).format('YYYY-MM-DD HH:mm:ss') : '-',
    },
    {
      title: '操作',
      key: 'action',
      width: 200,
      render: (_: any, record: any) => (
        <Space>
          <Button
            size="small"
            icon={<EyeOutlined />}
            onClick={() => {
              setCurrentAlert(record);
              setDetailVisible(true);
            }}
          >
            详情
          </Button>
          {record.status === 'pending' && (
            <>
              <Button
                size="small"
                type="primary"
                danger
                icon={<CheckOutlined />}
                onClick={() => handleStatusUpdate(record.alert_id, 'confirmed')}
              >
                确认
              </Button>
              <Button
                size="small"
                icon={<CloseOutlined />}
                onClick={() => handleStatusUpdate(record.alert_id, 'false_positive')}
              >
                误报
              </Button>
            </>
          )}
        </Space>
      ),
    },
  ];

  return (
    <div>
      <Table
        columns={columns}
        dataSource={alerts}
        rowKey="id"
        loading={loading}
        pagination={{
          current: page,
          total,
          pageSize: 20,
          onChange: setPage,
          showTotal: (t) => `共 ${t} 条`,
        }}
        scroll={{ x: 1400 }}
      />

      <Modal
        title="告警详情"
        open={detailVisible}
        onCancel={() => setDetailVisible(false)}
        width={700}
        footer={null}
      >
        {currentAlert && (
          <Descriptions bordered column={2}>
            <Descriptions.Item label="告警ID">{currentAlert.alert_id}</Descriptions.Item>
            <Descriptions.Item label="交易ID">{currentAlert.transaction_id}</Descriptions.Item>
            <Descriptions.Item label="风险评分">
              <span style={{ color: currentAlert.risk_score >= 0.8 ? '#cf1322' : '#fa8c16' }}>
                {(currentAlert.risk_score * 100).toFixed(2)}%
              </span>
            </Descriptions.Item>
            <Descriptions.Item label="风险等级">
              <Tag color={riskLevelColor[currentAlert.risk_level]}>
                {currentAlert.risk_level?.toUpperCase()}
              </Tag>
            </Descriptions.Item>
            <Descriptions.Item label="ML评分">
              {currentAlert.ml_score ? `${(currentAlert.ml_score * 100).toFixed(2)}%` : '-'}
            </Descriptions.Item>
            <Descriptions.Item label="状态">
              <Tag color={statusColor[currentAlert.status]}>
                {statusLabel[currentAlert.status]}
              </Tag>
            </Descriptions.Item>
            <Descriptions.Item label="解释" span={2}>
              {currentAlert.explanation || '-'}
            </Descriptions.Item>
            <Descriptions.Item label="处理人" span={2}>
              {currentAlert.assigned_to || '待分配'}
            </Descriptions.Item>
          </Descriptions>
        )}
      </Modal>
    </div>
  );
};

export default FraudAlertList;
```

### RuleManagement（规则管理）

```tsx
// src/pages/RuleManagement/index.tsx
import React, { useEffect, useState } from 'react';
import {
  Table, Button, Modal, Form, Input, Select, Switch,
  InputNumber, Tag, Space, message, Popconfirm,
} from 'antd';
import { PlusOutlined, EditOutlined, DeleteOutlined } from '@ant-design/icons';
import { getFraudRules, updateFraudRule } from '@/services/api';


const RuleManagement: React.FC = () => {
  const [rules, setRules] = useState<any[]>([]);
  const [loading, setLoading] = useState(false);
  const [editVisible, setEditVisible] = useState(false);
  const [currentRule, setCurrentRule] = useState<any>(null);
  const [form] = Form.useForm();

  useEffect(() => {
    fetchRules();
  }, []);

  const fetchRules = async () => {
    setLoading(true);
    try {
      const res = await getFraudRules();
      if (res.code === 0) {
        setRules(res.data || []);
      }
    } finally {
      setLoading(false);
    }
  };

  const handleEdit = (rule: any) => {
    setCurrentRule(rule);
    form.setFieldsValue(rule);
    setEditVisible(true);
  };

  const handleSave = async () => {
    try {
      const values = await form.validateFields();
      if (currentRule) {
        await updateFraudRule(currentRule.id, values);
        message.success('规则更新成功');
      }
      setEditVisible(false);
      fetchRules();
    } catch {
      message.error('保存失败');
    }
  };

  const handleToggle = async (ruleId: number, enabled: boolean) => {
    try {
      await updateFraudRule(ruleId, { enabled });
      message.success(enabled ? '规则已启用' : '规则已禁用');
      fetchRules();
    } catch {
      message.error('操作失败');
    }
  };

  const riskLevelColor: Record<string, string> = {
    low: 'green', medium: 'orange', high: 'red', critical: 'magenta',
  };

  const columns = [
    {
      title: '规则ID',
      dataIndex: 'rule_id',
      key: 'rule_id',
      width: 100,
    },
    {
      title: '规则名称',
      dataIndex: 'name',
      key: 'name',
      width: 150,

    {
      title: '类型',
      dataIndex: 'rule_type',
      key: 'rule_type',

          threshold: '阈值', composite: '组合', temporal: '时序', pattern: '模式',
        };
        return <Tag>{typeMap[type] || type}</Tag>;
      },
    },
    {
      title: '风险等级',
      dataIndex: 'risk_level',
      key: 'risk_level',
      width: 100,
      render: (level: string) => (
        <Tag color={riskLevelColor[level]}>{level?.toUpperCase()}</Tag>
      ),
    },
    {
      title: '描述',
      dataIndex: 'description',
      key: 'description',
      ellipsis: true,
    },
    {
      title: '命中次数',
      dataIndex: 'hit_count',
      key: 'hit_count',
      width: 100,
      sorter: (a: any, b: any) => a.hit_count - b.hit_count,
    },
    {
      title: '误报次数',
      dataIndex: 'false_positive_count',
      key: 'false_positive_count',
      width: 100,
    },
    {
      title: '启用',
      dataIndex: 'enabled',
      key: 'enabled',
      width: 80,
      render: (enabled: boolean, record: any) => (
        <Switch checked={enabled} onChange={(v) => handleToggle(record.id, v)} />
      ),
    },
    {
      title: '优先级',
      dataIndex: 'priority',
      key: 'priority',
```
