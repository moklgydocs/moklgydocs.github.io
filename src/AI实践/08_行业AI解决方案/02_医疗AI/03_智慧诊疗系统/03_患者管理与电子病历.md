```python
from pydantic import BaseModel, Field, field_validator
from typing import Optional
from datetime import date

class PatientCreate(BaseModel):
    """患者注册请求"""
    name: str = Field(..., min_length=2, max_length=50, description="姓名")
    gender: str = Field(..., pattern="^(男|女)$", description="性别")
    birth_date: date = Field(..., description="出生日期")
    id_card: str = Field(..., min_length=18, max_length=18, description="身份证号")
    phone: Optional[str] = Field(None, pattern=r"^1[3-9]\d{9}$", description="手机号")
    address: Optional[str] = Field(None, max_length=200, description="住址")
    emergency_contact: Optional[str] = Field(None, max_length=50, description="紧急联系人")
    emergency_phone: Optional[str] = Field(None, description="紧急联系电话")
    blood_type: Optional[str] = Field(None, pattern="^(A|B|AB|O|Rh-|Rh\+)$", description="血型")
    allergies: Optional[str] = Field("", description="过敏史，逗号分隔")
    medical_history: Optional[str] = Field("", description="既往史")
    family_history: Optional[str] = Field("", description="家族史")
    insurance_type: Optional[str] = Field(None, description="医保类型")
    insurance_no: Optional[str] = Field(None, description="医保号")

class PatientUpdate(BaseModel):
    """患者信息更新"""
    phone: Optional[str] = None
    address: Optional[str] = None
    emergency_contact: Optional[str] = None
    emergency_phone: Optional[str] = None
    allergies: Optional[str] = None
    medical_history: Optional[str] = None
    family_history: Optional[str] = None

class PatientResponse(BaseModel):
    """患者信息响应"""
    id: str
    name: str
    gender: str
    birth_date: date
    id_card: str  # 实际返回时脱敏
    phone: Optional[str] = None
    address: Optional[str] = None
    emergency_contact: Optional[str] = None
    blood_type: Optional[str] = None
    allergies: str = ""
    medical_history: str = ""
    family_history: str = ""
    insurance_type: Optional[str] = None
    is_active: str = "active"
    created_at: str

    model_config = {"from_attributes": True}

class PatientListResponse(BaseModel):
    """患者列表响应"""
    total: int
    items: list[PatientResponse]
    page: int
    page_size: int
```

```python
    present_illness: Optional[str] = None

    physical_exam: Optional[str] = None
```

```python
# app/routers/patients.py
from fastapi import APIRouter, Depends, HTTPException, Query
from sqlalchemy.ext.asyncio import AsyncSession
from sqlalchemy import select, func, or_
from typing import Optional

from app.database import get_db
from app.models.patient import Patient
from app.schemas.patient import (
    PatientCreate, PatientUpdate, PatientResponse, PatientListResponse,
)
from app.core.security import get_current_user, require_role, UserRole, User
from app.utils.masking import mask_id_card, mask_phone

router = APIRouter(prefix="/api/v1/patients", tags=["患者管理"])

@router.post("", response_model=PatientResponse, status_code=201)
async def create_patient(
    data: PatientCreate,
    db: AsyncSession = Depends(get_db),
    current_user: User = Depends(require_role(UserRole.DOCTOR, UserRole.ADMIN)),
):
    """患者注册"""
    # 检查身份证号是否已注册
    existing = await db.execute(
        select(Patient).where(Patient.id_card == data.id_card)
    )
    if existing.scalar_one_or_none():
        raise HTTPException(status_code=409, detail="该身份证号已注册")

    now = datetime.now().isoformat()
    patient = Patient(
        **data.model_dump(),
        is_active="active",
        created_at=now,
        updated_at=now,
    )
    db.add(patient)
    await db.flush()
    await db.refresh(patient)

    # 返回时脱敏
    response = PatientResponse.model_validate(patient)
    response.id_card = mask_id_card(response.id_card)
    if response.phone:
        response.phone = mask_phone(response.phone)
    return response

@router.get("/{patient_id}", response_model=PatientResponse)
async def get_patient(
    patient_id: str,
    db: AsyncSession = Depends(get_db),
    current_user: User = Depends(get_current_user),
):
    """获取患者信息"""
    result = await db.execute(
        select(Patient).where(Patient.id == patient_id, Patient.is_active == "active")
    )
    patient = result.scalar_one_or_none()
    if not patient:
        raise HTTPException(status_code=404, detail="患者不存在")

    response = PatientResponse.model_validate(patient)
    response.id_card = mask_id_card(response.id_card)
    return response

@router.get("", response_model=PatientListResponse)
async def list_patients(
    page: int = Query(1, ge=1),
    page_size: int = Query(20, ge=1, le=100),
    search: Optional[str] = Query(None, description="搜索关键词(姓名/身份证号)"),
    db: AsyncSession = Depends(get_db),
    current_user: User = Depends(get_current_user),
):
    """患者列表(分页+搜索)"""
    query = select(Patient).where(Patient.is_active == "active")

    if search:
        query = query.where(
            or_(
                Patient.name.contains(search),
                Patient.id_card.contains(search),
            )
        )

    # 计算总数
    count_query = select(func.count()).select_from(query.subquery())
    total = (await db.execute(count_query)).scalar()

    # 分页
    offset = (page - 1) * page_size
    query = query.order_by(Patient.created_at.desc()).offset(offset).limit(page_size)
    result = await db.execute(query)
    patients = result.scalars().all()

    items = []
    for p in patients:
        resp = PatientResponse.model_validate(p)
        resp.id_card = mask_id_card(resp.id_card)
        items.append(resp)

    return PatientListResponse(total=total, items=items, page=page, page_size=page_size)

@router.put("/{patient_id}", response_model=PatientResponse)
async def update_patient(
    patient_id: str,
    data: PatientUpdate,
    db: AsyncSession = Depends(get_db),
    current_user: User = Depends(require_role(UserRole.DOCTOR, UserRole.ADMIN)),
):
    """更新患者信息"""
    result = await db.execute(
        select(Patient).where(Patient.id == patient_id, Patient.is_active == "active")
    )
    patient = result.scalar_one_or_none()
    if not patient:
        raise HTTPException(status_code=404, detail="患者不存在")

    update_data = data.model_dump(exclude_unset=True)
    for key, value in update_data.items():
        setattr(patient, key, value)
    patient.updated_at = datetime.now().isoformat()

    await db.flush()
    await db.refresh(patient)

    response = PatientResponse.model_validate(patient)
    response.id_card = mask_id_card(response.id_card)
    return response
```

```python
# app/routers/records.py
from fastapi import APIRouter, Depends, HTTPException, Query
from sqlalchemy.ext.asyncio import AsyncSession
from sqlalchemy import select, func, or_
from typing import Optional
from datetime import datetime

from app.database import get_db
from app.models.patient import Patient
from app.models.medical_record import MedicalRecord, Diagnosis
from app.models.examination import LabTest, ImagingExam

# --- 病历CRUD ---

@router.post("/patients/{patient_id}/records", response_model=RecordResponse, status_code=201)
async def create_record(
    patient_id: str,
    data: RecordCreate,
    db: AsyncSession = Depends(get_db),
    current_user: User = Depends(require_role(UserRole.DOCTOR)),
):
    """创建病历"""

        raise HTTPException(status_code=404, detail="患者不存在")

        patient_id=patient_id,
        doctor_id=current_user.id,
        encounter_type=data.encounter_type,
        chief_complaint=data.chief_complaint,
        present_illness=data.present_illness,
        past_history=data.past_history,

        family_history=data.family_history,
        allergy_history=data.allergy_history,
        physical_exam=data.physical_exam,
        admission_date=data.admission_date,
        status="draft",
        created_at=now,
        updated_at=now,
    )
    db.add(record)
    await db.flush()
    await db.refresh(record)

    return RecordResponse.model_validate(record)

@router.get("/patients/{patient_id}/records", response_model=RecordListResponse)
async def list_records(
    patient_id: str,
    page: int = Query(1, ge=1),
    page_size: int = Query(20, ge=1, le=100),
    encounter_type: Optional[str] = Query(None),
    db: AsyncSession = Depends(get_db),
    current_user: User = Depends(get_current_user),
):
    """获取患者病历列表"""
    query = select(MedicalRecord).where(MedicalRecord.patient_id == patient_id)

    if encounter_type:
        query = query.where(MedicalRecord.encounter_type == encounter_type)

    count_query = select(func.count()).select_from(query.subquery())
    total = (await db.execute(count_query)).scalar()

    offset = (page - 1) * page_size
    query = query.order_by(MedicalRecord.created_at.desc()).offset(offset).limit(page_size)
    result = await db.execute(query)
    records = result.scalars().all()

    items = [RecordResponse.model_validate(r) for r in records]
    return RecordListResponse(total=total, items=items, page=page, page_size=page_size)

@router.get("/records/{record_id}", response_model=RecordResponse)
async def get_record(
    record_id: str,
    db: AsyncSession = Depends(get_db),
    current_user: User = Depends(get_current_user),
):
    """获取病历详情"""
    record = await db.get(MedicalRecord, record_id)
    if not record:
        raise HTTPException(status_code=404, detail="病历不存在")
    return RecordResponse.model_validate(record)

@router.put("/records/{record_id}", response_model=RecordResponse)
async def update_record(
    record_id: str,
    data: RecordUpdate,
    db: AsyncSession = Depends(get_db),
    current_user: User = Depends(require_role(UserRole.DOCTOR)),
):
    """更新病历"""
    record = await db.get(MedicalRecord, record_id)
    if not record:
        raise HTTPException(status_code=404, detail="病历不存在")

    if record.status == "signed":
        raise HTTPException(status_code=400, detail="已签名的病历不可修改")

    update_data = data.model_dump(exclude_unset=True)
    for key, value in update_data.items():
        setattr(record, key, value)
    record.updated_at = datetime.now().isoformat()

    await db.flush()
    await db.refresh(record)
    return RecordResponse.model_validate(record)

@router.post("/records/{record_id}/diagnoses", response_model=DiagnosisResponse, status_code=201)
async def add_diagnosis(
    record_id: str,
    data: DiagnosisCreate,
    db: AsyncSession = Depends(get_db),
    current_user: User = Depends(require_role(UserRole.DOCTOR)),
):
    """添加诊断"""
    record = await db.get(MedicalRecord, record_id)
    if not record:
        raise HTTPException(status_code=404, detail="病历不存在")

    # 自动填充ICD-10编码
    if not data.icd10_code and data.disease_name in DISEASE_ICD10_MAP:
        data.icd10_code = DISEASE_ICD10_MAP[data.disease_name]

    # 如果设置为主诊断，取消其他主诊断
    if data.is_primary:
        existing_primary = await db.execute(
            select(Diagnosis).where(
                Diagnosis.record_id == record_id,
                Diagnosis.is_primary == True,  # noqa: E712
            )
        )
        for existing in existing_primary.scalars().all():
            existing.is_primary = False

    diagnosis = Diagnosis(
        record_id=record_id,
        disease_name=data.disease_name,
        icd10_code=data.icd10_code,
        is_primary=data.is_primary,
        diagnosis_type=data.diagnosis_type,
        doctor_note=data.doctor_note,
        diagnosed_at=datetime.now().isoformat(),
    )
    db.add(diagnosis)
    await db.flush()
    await db.refresh(diagnosis)

    return DiagnosisResponse.model_validate(diagnosis)
```

## EMR服务业务逻辑

```python
# app/services/emr_service.py
from sqlalchemy.ext.asyncio import AsyncSession
from sqlalchemy import select
from typing import Optional
from datetime import datetime

from app.models.medical_record import MedicalRecord, Diagnosis
from app.models.examination import LabTest, ImagingExam
from app.models.patient import Patient
from app.schemas.medical_record import (
    RecordCreate, RecordResponse,
    DiagnosisCreate, DiagnosisResponse,
    LabTestCreate, LabTestResponse,
)
from app.utils.icd10_mapping import DISEASE_ICD10_MAP

class EMRService:
    """电子病历服务"""

    def __init__(self, db: AsyncSession):
        self.db = db

    async def create_record(
        self,
        patient_id: str,
        doctor_id: str,
        data: RecordCreate,
    ) -> MedicalRecord:
        """创建病历"""
        # 验证患者
        patient = await self.db.get(Patient, patient_id)
        if not patient:
            raise ValueError(f"患者不存在: {patient_id}")

        now = datetime.now().isoformat()

        # 自动填充过敏史和既往史
        allergy_history = data.allergy_history or patient.allergies
        past_history = data.past_history or patient.medical_history

        record = MedicalRecord(
            patient_id=patient_id,
            doctor_id=doctor_id,
            encounter_type=data.encounter_type,
            chief_complaint=data.chief_complaint,
            present_illness=data.present_illness,
            past_history=past_history,
            personal_history=data.personal_history,
            family_history=data.family_history,
            allergy_history=allergy_history,
            physical_exam=data.physical_exam,
            admission_date=data.admission_date,
            status="draft",
            created_at=now,
            updated_at=now,
        )
        self.db.add(record)
        await self.db.flush()
        await self.db.refresh(record)
        return record

    async def add_diagnosis(
        self,
        record_id: str,
        data: DiagnosisCreate,
    ) -> Diagnosis:
        """添加诊断"""
        record = await self.db.get(MedicalRecord, record_id)
        if not record:
            raise ValueError(f"病历不存在: {record_id}")

        # 自动ICD-10编码
        icd10_code = data.icd10_code
        if not icd10_code:
            icd10_code = DISEASE_ICD10_MAP.get(data.disease_name)

        # 主诊断处理
        if data.is_primary:
            result = await self.db.execute(
                select(Diagnosis).where(
                    Diagnosis.record_id == record_id,
                    Diagnosis.is_primary == True,  # noqa: E712
                )
            )
            for existing in result.scalars().all():
                existing.is_primary = False

        diagnosis = Diagnosis(
            record_id=record_id,
            disease_name=data.disease_name,
            icd10_code=icd10_code,
            is_primary=data.is_primary,
            diagnosis_type=data.diagnosis_type,
            doctor_note=data.doctor_note,
            diagnosed_at=datetime.now().isoformat(),
        )
        self.db.add(diagnosis)
        await self.db.flush()
        return diagnosis

    async def add_lab_test(
        self,
        record_id: str,
        data: LabTestCreate,
    ) -> LabTest:
        """添加检验结果"""
        record = await self.db.get(MedicalRecord, record_id)
        if not record:
            raise ValueError(f"病历不存在: {record_id}")

        # 危急值检查
        is_critical = self._check_critical_value(data.test_code or "", data.result_value or "")

        lab_test = LabTest(
            patient_id=record.patient_id,
            record_id=record_id,
            test_name=data.test_name,
            test_code=data.test_code,
            category=data.category,
            specimen_type=data.specimen_type,
            result_value=data.result_value,
            result_text=data.result_text,
            unit=data.unit,
            reference_range=data.reference_range,
            is_abnormal=data.is_abnormal,
            abnormal_flag=data.abnormal_flag,
            ordering_doctor="",
            ordered_at=datetime.now().isoformat(),
            resulted_at=datetime.now().isoformat() if data.result_value else None,
        )
        self.db.add(lab_test)
        await self.db.flush()
        return lab_test

    def _check_critical_value(self, test_code: str, result_value: str) -> bool:
        """检查危急值"""
        CRITICAL_RULES = {
            "K": (2.8, 6.2),
            "GLU": (2.2, 22.2),
            "WBC": (2.0, 30.0),
            "PLT": (30.0, 1000.0),
            "TNI": (None, 2.0),
        }
        if test_code not in CRITICAL_RULES:
            return False
        try:
            value = float(result_value)
            low, high = CRITICAL_RULES[test_code]
            if low is not None and value <= low:
                return True
            if high is not None and value >= high:
                return True
        except (ValueError, TypeError):
            pass
        return False

    async def get_patient_record_history(
        self,
        patient_id: str,
        limit: int = 20,
    ) -> list[dict]:
        """获取患者历史病历摘要"""
        result = await self.db.execute(
            select(MedicalRecord)
            .where(MedicalRecord.patient_id == patient_id)
            .order_by(MedicalRecord.created_at.desc())
            .limit(limit)
        )
        records = result.scalars().all()

        history = []
        for record in records:
            # 获取诊断
            diag_result = await self.db.execute(
                select(Diagnosis).where(Diagnosis.record_id == record.id)
            )
            diagnoses = diag_result.scalars().all()

            history.append({
                "record_id": record.id,
                "encounter_type": record.encounter_type,
                "chief_complaint": record.chief_complaint,
                "diagnoses": [
                    {"disease": d.disease_name, "icd10": d.icd10_code, "is_primary": d.is_primary}
                    for d in diagnoses
                ],
                "created_at": record.created_at,
            })

        return history
```

## 数据权限控制

```python
# app/core/access_control.py
from fastapi import Depends, HTTPException
from sqlalchemy.ext.asyncio import AsyncSession
from sqlalchemy import select

from app.database import get_db
from app.models.medical_record import MedicalRecord
from app.models.patient import Patient
from app.core.security import User, UserRole, get_current_user

async def check_record_access(
    record_id: str,
    current_user: User = Depends(get_current_user),
    db: AsyncSession = Depends(get_db),
) -> MedicalRecord:
    """检查用户是否有权访问该病历"""
    result = await db.execute(
        select(MedicalRecord).where(MedicalRecord.id == record_id)
    )
    record = result.scalar_one_or_none()
    if not record:
        raise HTTPException(status_code=404, detail="病历不存在")

    # 管理员可访问所有
    if current_user.role == UserRole.ADMIN:
        return record

    # 医生只能访问自己科室的患者
    if current_user.role == UserRole.DOCTOR:
        # 获取患者所属科室(简化：通过最近就诊判断)
        if record.doctor_id == current_user.id:
            return record

        # 检查是否同科室
        result = await db.execute(
            select(MedicalRecord.doctor_id).where(MedicalRecord.patient_id == record.patient_id)
        )
        # 如果当前用户曾接诊过该患者，允许访问
        return record

    raise HTTPException(status_code=403, detail="无权访问该病历")

async def check_patient_access(
    patient_id: str,
    current_user: User = Depends(get_current_user),
    db: AsyncSession = Depends(get_db),
) -> Patient:
    """检查用户是否有权访问该患者信息"""
    patient = await db.get(Patient, patient_id)
    if not patient:
        raise HTTPException(status_code=404, detail="患者不存在")

    if current_user.role == UserRole.ADMIN:
        return patient

    # 医生：允许访问(实际中应根据科室过滤)
    if current_user.role == UserRole.DOCTOR:
        return patient

    # 患者：只能访问自己
    if current_user.role == UserRole.PATIENT:
        if current_user.id != patient_id:
            raise HTTPException(status_code=403, detail="只能访问本人信息")

    return patient
```

## 脱敏工具

```python
# app/utils/masking.py
import re

def mask_id_card(id_card: str) -> str:
    """身份证号脱敏"""
    if len(id_card) < 8:
        return "***"
    return id_card[:3] + "*" * (len(id_card) - 7) + id_card[-4:]

def mask_phone(phone: str) -> str:
    """手机号脱敏"""
    if len(phone) < 7:
        return "***"
    return phone[:3] + "****" + phone[-4:]

def mask_name(name: str) -> str:
    """姓名脱敏"""
    if len(name) <= 1:
        return name
    if len(name) == 2:
        return name[0] + "*"
    return name[0] + "*" * (len(name) - 2) + name[-1]

def mask_address(address: str) -> str:
    """地址脱敏：保留到区"""
    pattern = r"^(.*?[省市].*?[区县])"
    match = re.match(pattern, address)
    if match:
        return match.group(1) + "***"
    return "***"
```
