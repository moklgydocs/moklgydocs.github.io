# 医学影像AI

## 影像AI核心任务

医学影像AI的核心任务可以分为四类：检测、分割、分类和配准。

| 任务 | 定义 | 输入 | 输出 | 典型应用 |
|------|------|------|------|----------|
| 检测(Detection) | 定位影像中感兴趣区域的位置 | 影像+候选区域 | 边界框+置信度 | 肺结节检测、骨折检测 |
| 分割(Segmentation) | 像素级别的区域划分 | 影像 | 像素级掩码 | 器官分割、肿瘤分割 |
| 分类(Classification) | 对影像或区域进行类别判定 | 影像/ROI | 类别+概率 | 良恶性分类、疾病分级 |
| 配准(Registration) | 将不同时间/模态影像对齐 | 两幅影像 | 空间变换矩阵 | 治疗前后对比、多模态融合 |

## CNN架构在医学影像的应用

### U-Net：医学影像分割的经典架构

U-Net是医学影像分割领域最具影响力的网络架构，采用编码器-解码器结构配合跳跃连接。

```python
import torch
import torch.nn as nn
import torch.nn.functional as F

class DoubleConv(nn.Module):
    """双卷积块: Conv3x3 -> BN -> ReLU -> Conv3x3 -> BN -> ReLU"""
    def __init__(self, in_channels: int, out_channels: int):
        super().__init__()
        self.double_conv = nn.Sequential(
            nn.Conv2d(in_channels, out_channels, kernel_size=3, padding=1, bias=False),
            nn.BatchNorm2d(out_channels),
            nn.ReLU(inplace=True),
            nn.Conv2d(out_channels, out_channels, kernel_size=3, padding=1, bias=False),
            nn.BatchNorm2d(out_channels),
            nn.ReLU(inplace=True),
        )

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        return self.double_conv(x)

class UNet(nn.Module):
    """U-Net医学影像分割网络"""
    def __init__(self, in_channels: int = 1, num_classes: int = 2, features: list[int] = None):
        super().__init__()
        if features is None:
            features = [64, 128, 256, 512]

        # 编码器(下采样)
        self.downs = nn.ModuleList()
        self.pool = nn.MaxPool2d(kernel_size=2, stride=2)
        for feature in features:
            self.downs.append(DoubleConv(in_channels, feature))
            in_channels = feature

        # 瓶颈层
        self.bottleneck = DoubleConv(features[-1], features[-1] * 2)

        # 解码器(上采样)
        self.ups = nn.ModuleList()
        for feature in reversed(features):
            self.ups.append(nn.ConvTranspose2d(feature * 2, feature, kernel_size=2, stride=2))
            self.ups.append(DoubleConv(feature * 2, feature))

        # 输出层
        self.final_conv = nn.Conv2d(features[0], num_classes, kernel_size=1)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        skip_connections = []

        # 编码路径
        for down in self.downs:
            x = down(x)
            skip_connections.append(x)
            x = self.pool(x)

        x = self.bottleneck(x)
        skip_connections = skip_connections[::-1]  # 反转

        # 解码路径
        for idx in range(0, len(self.ups), 2):
            x = self.ups[idx](x)  # 上采样
            skip = skip_connections[idx // 2]

            # 处理尺寸不匹配
            if x.shape != skip.shape:
                x = F.interpolate(x, size=skip.shape[2:], mode="bilinear", align_corners=True)

            concat_skip = torch.cat((skip, x), dim=1)
            x = self.ups[idx + 1](concat_skip)

        return self.final_conv(x)

# 模型实例化
model = UNet(in_channels=1, num_classes=2)  # 单通道输入，二分类分割
```

import torch
import torch.nn as nn
from torchvision import models

class LesionClassifier(nn.Module):
    """基于ResNet的病变分类器"""
    def __init__(self, num_classes: int = 4, pretrained: bool = True):
        super().__init__()
        self.backbone = models.resnet50(weights=models.ResNet50_Weights.DEFAULT if pretrained else None)

        # 替换第一层以适配灰度/单通道医学影像
        original_conv = self.backbone.conv1
        self.backbone.conv1 = nn.Conv2d(
            1, 64, kernel_size=7, stride=2, padding=3, bias=False,
        )
        if pretrained:
            # 将RGB预训练权重平均到单通道
            with torch.no_grad():
                self.backbone.conv1.weight[:, 0, :, :] = original_conv.weight.mean(dim=1)

        # 替换分类头
        in_features = self.backbone.fc.in_features
        self.backbone.fc = nn.Sequential(
            nn.Dropout(0.3),
            nn.Linear(in_features, 256),
            nn.ReLU(),
            nn.Dropout(0.2),
            nn.Linear(256, num_classes),
        )

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        return self.backbone(x)

### RetinaNet检测：肺结节检测

```python
import torch
import torch.nn as nn
import torch.nn.functional as F
from torchvision.models import resnet50

class FeaturePyramidNetwork(nn.Module):
    """特征金字塔网络"""
    def __init__(self, in_channels_list: list[int], out_channels: int = 256):
        super().__init__()
        self.lateral_convs = nn.ModuleList()
        self.smooth_convs = nn.ModuleList()

        for in_channels in in_channels_list:
            self.lateral_convs.append(nn.Conv2d(in_channels, out_channels, 1))
            self.smooth_convs.append(nn.Conv2d(out_channels, out_channels, 3, padding=1))

    def forward(self, features: list[torch.Tensor]) -> list[torch.Tensor]:
        laterals = [conv(f) for conv, f in zip(self.lateral_convs, features)]

        # 自顶向下融合
        for i in range(len(laterals) - 1, 0, -1):
            upsampled = F.interpolate(laterals[i], size=laterals[i - 1].shape[2:], mode="nearest")
            laterals[i - 1] = laterals[i - 1] + upsampled

        return [smooth(lat) for smooth, lat in zip(self.smooth_convs, laterals)]

class FocalLoss(nn.Module):
    """Focal Loss: 解决正负样本不均衡问题"""
    def __init__(self, alpha: float = 0.25, gamma: float = 2.0):
        super().__init__()
        self.alpha = alpha
        self.gamma = gamma

    def forward(self, predictions: torch.Tensor, targets: torch.Tensor) -> torch.Tensor:
        bce = F.binary_cross_entropy_with_logits(predictions, targets, reduction="none")
        pt = torch.exp(-bce)
        focal_loss = self.alpha * (1 - pt) ** self.gamma * bce
        return focal_loss.mean()

class NoduleDetector(nn.Module):
    """肺结节检测器(简化版RetinaNet)"""
    def __init__(self, num_anchors: int = 9, num_classes: int = 2):
        super().__init__()
        backbone = resnet50(weights=models.ResNet50_Weights.DEFAULT)
        self.layer0 = nn.Sequential(backbone.conv1, backbone.bn1, backbone.relu, backbone.maxpool)
        self.layer1 = backbone.layer1  # 256
        self.layer2 = backbone.layer2  # 512
        self.layer3 = backbone.layer3  # 1024
        self.layer4 = backbone.layer4  # 2048

        self.fpn = FeaturePyramidNetwork([256, 512, 1024, 2048], out_channels=256)

        # 检测头
        self.cls_head = nn.Conv2d(256, num_anchors * num_classes, 3, padding=1)
        self.reg_head = nn.Conv2d(256, num_anchors * 4, 3, padding=1)

        self.num_anchors = num_anchors
        self.num_classes = num_classes

    def forward(self, x: torch.Tensor) -> tuple[list[torch.Tensor], list[torch.Tensor]]:
        c1 = self.layer1(self.layer0(x))
        c2 = self.layer2(c1)
        c3 = self.layer3(c2)
        c4 = self.layer4(c3)

        features = self.fpn([c1, c2, c3, c4])

        cls_outputs = []
        reg_outputs = []
        for feat in features:
            cls_out = self.cls_head(feat)
            reg_out = self.reg_head(feat)
            B, _, H, W = cls_out.shape
            cls_out = cls_out.view(B, self.num_anchors, self.num_classes, H, W)
            reg_out = reg_out.view(B, self.num_anchors, 4, H, W)
            cls_outputs.append(cls_out)
            reg_outputs.append(reg_out)

        return cls_outputs, reg_outputs
```

import numpy as np
from typing import Optional
from pydantic import BaseModel, Field

class DICOMMetadata(BaseModel):
    """DICOM元数据"""
    patient_id: str
    study_uid: str
    series_uid: str
    modality: str
    study_date: str
    rows: int
    columns: int
    slice_thickness: Optional[float] = None
    pixel_spacing: Optional[list[float]] = None
    window_center: Optional[float] = None
    window_width: Optional[float] = None
    rescale_intercept: Optional[float] = None
    rescale_slope: Optional[float] = None

class DICOMPreprocessor:
    """DICOM影像预处理"""

    @staticmethod
    def apply_window(image: np.ndarray, center: float, width: float) -> np.ndarray:
        """窗宽窗位调整"""
        min_val = center - width / 2
        max_val = center + width / 2
        windowed = np.clip(image, min_val, max_val)
        windowed = (windowed - min_val) / (width) * 255.0
        return windowed.astype(np.uint8)

    @staticmethod
    def get_lung_window(image: np.ndarray) -> np.ndarray:
        """肺窗: W=1500, C=-600"""
        return DICOMPreprocessor.apply_window(image, center=-600, width=1500)

    @staticmethod
    def get_mediastinal_window(image: np.ndarray) -> np.ndarray:
        """纵隔窗: W=350, C=40"""
        return DICOMPreprocessor.apply_window(image, center=40, width=350)

    @staticmethod
    def get_bone_window(image: np.ndarray) -> np.ndarray:
        """骨窗: W=2500, C=480"""
        return DICOMPreprocessor.apply_window(image, center=480, width=2500)

    @staticmethod
    def normalize_to_01(image: np.ndarray) -> np.ndarray:
        """归一化到[0, 1]"""
        img = image.astype(np.float32)
        min_val, max_val = img.min(), img.max()
        if max_val - min_val < 1e-6:
            return np.zeros_like(img)
        return (img - min_val) / (max_val - min_val)

    @staticmethod
    def resample_slice(
        image: np.ndarray,
        original_spacing: tuple[float, float, float],
        target_spacing: tuple[float, float, float] = (1.0, 1.0, 1.0),
    ) -> np.ndarray:
        """重采样到统一间距(简化版)"""
        from scipy.ndimage import zoom
        resize_factor = [
            original_spacing[i] / target_spacing[i]
            for i in range(3)
        ]
        return zoom(image, resize_factor, order=1)

import torch

import random

            ),

            T.RandomErasing(p=0.1, scale=(0.02, 0.1)),
            T.ToTensor(),
            T.Normalize(mean=[0.5], std=[0.5]),
        ])

    @staticmethod
    def get_val_transforms(image_size: int = 512) -> T.Compose:
        """验证集数据增强(仅resize和归一化)"""
        return T.Compose([
            T.Resize((image_size, image_size)),
            T.ToTensor(),
            T.Normalize(mean=[0.5], std=[0.5]),
        ])

    @staticmethod
    def add_gaussian_noise(image: torch.Tensor, mean: float = 0.0, std: float = 0.02) -> torch.Tensor:
        """添加高斯噪声(模拟影像噪声)"""
        noise = torch.randn_like(image) * std + mean
        return torch.clamp(image + noise, 0.0, 1.0)

    @staticmethod
    def simulate_motion_blur(image: torch.Tensor, kernel_size: int = 5) -> torch.Tensor:
        """模拟运动伪影(简化版)"""
        import torch.nn.functional as F
        kernel = torch.zeros(1, 1, kernel_size, kernel_size)
        kernel[0, 0, kernel_size // 2, :] = 1.0 / kernel_size
        padded = F.pad(image.unsqueeze(0), (kernel_size // 2,) * 4, mode="reflect")
        blurred = F.conv2d(padded, kernel)
        return blurred.squeeze(0)

### 评估指标

```python
import numpy as np
from typing import Optional

def dice_coefficient(pred: np.ndarray, target: np.ndarray, smooth: float = 1e-6) -> float:
    """Dice系数: 分割质量的核心指标"""
    pred_flat = pred.flatten().astype(np.float64)
    target_flat = target.flatten().astype(np.float64)
    intersection = np.sum(pred_flat * target_flat)
    return (2.0 * intersection + smooth) / (np.sum(pred_flat) + np.sum(target_flat) + smooth)

def iou_score(pred: np.ndarray, target: np.ndarray, smooth: float = 1e-6) -> float:
    """交并比(IoU/Jaccard)"""
    pred_flat = pred.flatten().astype(np.float64)
    target_flat = target.flatten().astype(np.float64)
    intersection = np.sum(pred_flat * target_flat)
    union = np.sum(pred_flat) + np.sum(target_flat) - intersection
    return (intersection + smooth) / (union + smooth)

def hausdorff_distance(pred: np.ndarray, target: np.ndarray, percentile: float = 95.0) -> float:
    """Hausdorff距离(95th percentile)"""
    from scipy.spatial.distance import cdist
    pred_points = np.argwhere(pred > 0).astype(np.float64)
    target_points = np.argwhere(target > 0).astype(np.float64)

    if len(pred_points) == 0 or len(target_points) == 0:
        return float("inf")

    dist_matrix = cdist(pred_points, target_points)
    h1 = np.percentile(np.min(dist_matrix, axis=1), percentile)
    h2 = np.percentile(np.min(dist_matrix, axis=0), percentile)
    return max(h1, h2)

def compute_auc_roc(y_true: np.ndarray, y_scores: np.ndarray) -> float:
    """AUC-ROC计算(简化版)"""
    sorted_indices = np.argsort(-y_scores)
    y_sorted = y_true[sorted_indices]
    tpr_list, fpr_list = [0.0], [0.0]
    tp, fp = 0, 0
    total_pos = np.sum(y_true == 1)
    total_neg = np.sum(y_true == 0)

    for i in range(len(y_sorted)):
        if y_sorted[i] == 1:
            tp += 1
        else:
            fp += 1
        tpr_list.append(tp / total_pos if total_pos > 0 else 0)
        fpr_list.append(fp / total_neg if total_neg > 0 else 0)

    # 梯形法计算面积
    auc = 0.0
    for i in range(1, len(fpr_list)):
        auc += (fpr_list[i] - fpr_list[i - 1]) * (tpr_list[i] + tpr_list[i - 1]) / 2
    return auc

def compute_mean_ap(
    predictions: list[dict],
    iou_threshold: float = 0.5,
) -> float:
    """mAP计算(简化版)"""
    aps = []
    for cls in set(p["class"] for p in predictions):
        cls_preds = [p for p in predictions if p["class"] == cls]
        cls_preds.sort(key=lambda x: x["confidence"], reverse=True)

        tp_count, fp_count = 0, 0
        precisions, recalls = [], []
        total_gt = sum(1 for p in cls_preds if p["is_true_positive"] or p["is_false_negative"])

        for pred in cls_preds:
            if pred.get("is_true_positive"):
                tp_count += 1
            else:
                fp_count += 1
            precisions.append(tp_count / (tp_count + fp_count))
            recalls.append(tp_count / total_gt if total_gt > 0 else 0)

        # 11点插值
        ap = 0.0
        for t in np.arange(0, 1.1, 0.1):
            mask = [p for p, r in zip(precisions, recalls) if r >= t]
            p = max(mask) if mask else 0
            ap += p / 11
        aps.append(ap)

    return np.mean(aps) if aps else 0.0
```

### 训练流程

```python
import torch
from torch.utils.data import DataLoader, Dataset
from torch.optim import AdamW
from torch.optim.lr_scheduler import CosineAnnealingLR
from typing import Optional
import numpy as np

class MedicalImageDataset(Dataset):
    """医学影像数据集基类"""
    def __init__(
        self,
        image_paths: list[str],
        labels: Optional[list] = None,
        transform: Optional[object] = None,
    ):
        self.image_paths = image_paths
        self.labels = labels
        self.transform = transform

    def __len__(self) -> int:
        return len(self.image_paths)

    def __getitem__(self, idx: int) -> tuple:
        # 实际中此处加载DICOM并进行预处理
        image = self._load_image(self.image_paths[idx])
        if self.transform:
            image = self.transform(image)
        if self.labels is not None:
            return image, self.labels[idx]
        return image

    def _load_image(self, path: str) -> np.ndarray:
        """加载影像(需根据实际格式实现)"""
        raise NotImplementedError

def train_one_epoch(
    model: torch.nn.Module,
    dataloader: DataLoader,
    criterion: torch.nn.Module,
    optimizer: torch.optim.Optimizer,
    device: torch.device,
) -> dict:
    """训练一个epoch"""
    model.train()
    total_loss = 0.0
    correct = 0
    total = 0

    for images, labels in dataloader:
        images = images.to(device)
        labels = labels.to(device)

        optimizer.zero_grad()
        outputs = model(images)
        loss = criterion(outputs, labels)
        loss.backward()

        # 梯度裁剪(防止梯度爆炸)
        torch.nn.utils.clip_grad_norm_(model.parameters(), max_norm=1.0)
        optimizer.step()

        total_loss += loss.item() * images.size(0)
        _, predicted = outputs.max(1)
        total += labels.size(0)
        correct += predicted.eq(labels).sum().item()

    return {
        "loss": total_loss / total,
        "accuracy": correct / total,
    }

def train_model(
    model: torch.nn.Module,
    train_loader: DataLoader,
    val_loader: DataLoader,
    num_epochs: int = 50,
    learning_rate: float = 1e-4,
    device: Optional[torch.device] = None,
) -> dict:
    """完整训练流程"""
    if device is None:
        device = torch.device("cuda" if torch.cuda.is_available() else "cpu")

    model = model.to(device)
    optimizer = AdamW(model.parameters(), lr=learning_rate, weight_decay=1e-4)
    scheduler = CosineAnnealingLR(optimizer, T_max=num_epochs)
    criterion = torch.nn.CrossEntropyLoss()

    best_val_acc = 0.0
    history = {"train_loss": [], "val_loss": [], "train_acc": [], "val_acc": []}

    for epoch in range(num_epochs):
        train_metrics = train_one_epoch(model, train_loader, criterion, optimizer, device)
        val_metrics = train_one_epoch(model, val_loader, criterion, optimizer, device)

        scheduler.step()

        history["train_loss"].append(train_metrics["loss"])
        history["val_loss"].append(val_metrics["loss"])
        history["train_acc"].append(train_metrics["accuracy"])
        history["val_acc"].append(val_metrics["accuracy"])

        if val_metrics["accuracy"] > best_val_acc:
            best_val_acc = val_metrics["accuracy"]
            torch.save(model.state_dict(), "best_model.pth")

    return history
```

### 医学影像预训练策略

```python
import torch
import torch.nn as nn
from torchvision import models

class TransferLearningClassifier(nn.Module):
    """迁移学习分类器"""
    def __init__(
        self,
        num_classes: int,
        backbone: str = "resnet50",
        freeze_backbone: bool = True,
        pretrained: bool = True,
    ):
        super().__init__()

        # 加载预训练backbone
        if backbone == "resnet50":
            self.backbone = models.resnet50(
                weights=models.ResNet50_Weights.DEFAULT if pretrained else None,
            )
            in_features = self.backbone.fc.in_features

            # 冻结backbone(初期训练)
            if freeze_backbone:
                for param in self.backbone.parameters():
                    param.requires_grad = False

            # 替换分类头
            self.backbone.fc = nn.Sequential(
                nn.Dropout(0.3),
                nn.Linear(in_features, 512),
                nn.ReLU(),
                nn.BatchNorm1d(512),
                nn.Dropout(0.2),
                nn.Linear(512, num_classes),
            )

    def unfreeze_backbone(self):
        """解冻backbone进行微调"""
        for param in self.backbone.parameters():
            param.requires_grad = True

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        return self.backbone(x)

# 两阶段训练策略
def two_stage_training(
    model: TransferLearningClassifier,
    train_loader: torch.utils.data.DataLoader,
    val_loader: torch.utils.data.DataLoader,
    device: torch.device,
    stage1_epochs: int = 10,
    stage2_epochs: int = 20,
):
    """两阶段训练：先训分类头，再微调整体"""
    criterion = nn.CrossEntropyLoss()

    # Stage 1: 冻结backbone，只训练分类头
    optimizer = torch.optim.Adam(
        filter(lambda p: p.requires_grad, model.parameters()),
        lr=1e-3,
    )
    for epoch in range(stage1_epochs):
        train_one_epoch(model, train_loader, criterion, optimizer, device)

    # Stage 2: 解冻backbone，用更小学习率微调
    model.unfreeze_backbone()
    optimizer = torch.optim.AdamW(model.parameters(), lr=1e-5, weight_decay=1e-4)
    scheduler = CosineAnnealingLR(optimizer, T_max=stage2_epochs)

    for epoch in range(stage2_epochs):
        train_one_epoch(model, train_loader, criterion, optimizer, device)
        scheduler.step()
```

### 可解释性

```python
import torch
import torch.nn.functional as F
import numpy as np

class GradCAM:
    """Grad-CAM: 梯度加权类激活映射"""
    def __init__(self, model: torch.nn.Module, target_layer: torch.nn.Module):
        self.model = model
        self.target_layer = target_layer
        self.gradients = None
        self.activations = None
        self._register_hooks()

    def _register_hooks(self):
        def forward_hook(module, input, output):
            self.activations = output.detach()

        def backward_hook(module, grad_input, grad_output):
            self.gradients = grad_output[0].detach()

        self.target_layer.register_forward_hook(forward_hook)
        self.target_layer.register_full_backward_hook(backward_hook)

    def generate_cam(
        self,
        input_image: torch.Tensor,
        target_class: Optional[int] = None,
    ) -> np.ndarray:
        """生成类激活映射"""
        self.model.eval()
        output = self.model(input_image)

        if target_class is None:
            target_class = output.argmax(dim=1).item()

        self.model.zero_grad()
        one_hot = torch.zeros_like(output)
        one_hot[0, target_class] = 1
        output.backward(gradient=one_hot)

        # 全局平均池化得到权重
        weights = self.gradients.mean(dim=(2, 3), keepdim=True)
        cam = (weights * self.activations).sum(dim=1, keepdim=True)
        cam = F.relu(cam)

        # 归一化到[0, 1]
        cam = cam.squeeze().cpu().numpy()
        cam = (cam - cam.min()) / (cam.max() - cam.min() + 1e-8)

        return cam
```

### 影像AI产品化关键注意事项

1. **AI辅助定位**：所有AI输出必须明确标注"辅助"角色，不能替代医师判断
2. **性能监控**：上线后需持续监控模型性能，检测数据漂移
3. **版本管理**：模型更新需经过验证和审批，不可随意更新
4. **审计追踪**：所有AI辅助决策需记录日志，可追溯
5. **失败模式**：需明确AI失败的边界条件，设置兜底机制

### NMPA算法变更管理要求

根据NMPA《医疗器械软件注册审查指导原则》，AI医疗器械的算法管理有严格要求：

- **锁定算法 (Locked Algorithm)**：生产环境中的算法版本必须与注册审批版本一致，不允许在线持续学习
- **版本控制**：每次算法更新（含训练数据变化、模型结构变化、超参数调整）均视为重大变更，需重新评估是否需要变更注册
- **变更分级**：
  - 重大变更：算法架构变化 → 需重新注册
  - 中等变更：训练数据显著变化 → 需变更备案
  - 轻微变更：后处理逻辑调整 → 内部验证后可执行
- **临床评价**：每次算法变更后需在临床数据集上重新验证sensitivity/specificity/AUROC
- **追溯性**：保留完整的训练数据血统、模型参数和验证结果
