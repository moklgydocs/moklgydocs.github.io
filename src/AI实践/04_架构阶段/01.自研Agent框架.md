# 自研 Agent 框架设计

当 LangChain/LangGraph 不满足需求时，了解如何设计自己的 Agent 框架。更重要的是：通过从零设计，彻底理解框架背后的抽象。

## 1. 为什么考虑自研

| 场景 | 说明 |
|------|------|
| 深度定制 | 需要特殊的记忆机制、工具调度策略 |
| 性能要求 | 框架开销无法接受 |
| 依赖管控 | 企业禁止引入大型开源框架 |
| 学习目的 | 理解框架原理，而非依赖黑盒 |

> 大多数情况下用 LangGraph 就够了。自研框架的成本很高。

## 2. Agent 核心抽象

```python
# 最小化 Agent 框架的核心接口

from abc import ABC, abstractmethod
from typing import Any, AsyncIterator
from dataclasses import dataclass, field

@dataclass
class Message:
    role: str  # "system" | "user" | "assistant" | "tool"
    content: str
    tool_call_id: str | None = None
    tool_calls: list[dict] | None = None

@dataclass  
class ToolResult:
    tool_call_id: str
    name: str
    output: str
    error: bool = False

class Tool(ABC):
    @property
    @abstractmethod
    def name(self) -> str: ...
    
    @property
    @abstractmethod
    def description(self) -> str: ...
    
    @property
    @abstractmethod
    def parameters(self) -> dict: ...
    
    @abstractmethod
    async def execute(self, **kwargs) -> str: ...
    
    def to_openai_schema(self) -> dict:
        return {
            "type": "function",
            "function": {
                "name": self.name,
                "description": self.description,
                "parameters": self.parameters,
            }
        }

class Memory(ABC):
    @abstractmethod
    async def add(self, message: Message) -> None: ...
    
    @abstractmethod
    async def get(self, max_tokens: int | None = None) -> list[Message]: ...
    
    @abstractmethod
    async def clear(self) -> None: ...

class AgentRuntime(ABC):
    """Agent 执行引擎"""
    
    @abstractmethod
    async def run(self, user_input: str) -> str: ...
    
    @abstractmethod
    async def stream(self, user_input: str) -> AsyncIterator[str]: ...
```

## 3. 简单 ReAct Agent 实现

```python
import json
import asyncio
from openai import AsyncOpenAI

class SimpleReActAgent:
    def __init__(
        self,
        llm_client: AsyncOpenAI,
        model: str,
        system_prompt: str,
        tools: list[Tool],
        memory: Memory,
        max_iterations: int = 10,
    ):
        self.client = llm_client
        self.model = model
        self.system_prompt = system_prompt
        self.tools = {t.name: t for t in tools}
        self.memory = memory
        self.max_iterations = max_iterations
    
    async def run(self, user_input: str) -> str:
        await self.memory.add(Message(role="user", content=user_input))
        
        for iteration in range(self.max_iterations):
            history = await self.memory.get()
            
            response = await self.client.chat.completions.create(
                model=self.model,
                messages=[
                    {"role": "system", "content": self.system_prompt},
                    *[{"role": m.role, "content": m.content, **({"tool_call_id": m.tool_call_id} if m.tool_call_id else {}), **({"tool_calls": m.tool_calls} if m.tool_calls else {})} for m in history],
                ],
                tools=[t.to_openai_schema() for t in self.tools.values()],
            )
            
            msg = response.choices[0].message
            
            # 没有工具调用 → 最终答案
            if not msg.tool_calls:
                await self.memory.add(Message(role="assistant", content=msg.content))
                return msg.content
            
            # 有工具调用 → 执行工具
            await self.memory.add(Message(
                role="assistant",
                content=msg.content or "",
                tool_calls=[tc.model_dump() for tc in msg.tool_calls],
            ))
            
            # 并行执行所有工具
            tool_results = await asyncio.gather(*[
                self._execute_tool(tc) for tc in msg.tool_calls
            ])
            
            for result in tool_results:
                await self.memory.add(Message(
                    role="tool",
                    content=result.output,
                    tool_call_id=result.tool_call_id,
                ))
        
        return "达到最大迭代次数，任务未完成"
    
    async def _execute_tool(self, tool_call) -> ToolResult:
        name = tool_call.function.name
        args = json.loads(tool_call.function.arguments)
        
        if name not in self.tools:
            return ToolResult(
                tool_call_id=tool_call.id,
                name=name,
                output=f"工具 '{name}' 不存在",
                error=True,
            )
        
        try:
            output = await self.tools[name].execute(**args)
            return ToolResult(tool_call_id=tool_call.id, name=name, output=str(output))
        except Exception as e:
            return ToolResult(
                tool_call_id=tool_call.id,
                name=name,
                output=f"执行错误: {e}",
                error=True,
            )
```

## 4. 记忆系统设计

```python
from collections import deque

class SlidingWindowMemory(Memory):
    """滑动窗口记忆：保留最近 N 条消息"""
    
    def __init__(self, max_messages: int = 20):
        self.messages: deque[Message] = deque(maxlen=max_messages)
        self._system: Message | None = None
    
    async def add(self, message: Message) -> None:
        self.messages.append(message)
    
    async def get(self, max_tokens: int | None = None) -> list[Message]:
        return list(self.messages)
    
    async def clear(self) -> None:
        self.messages.clear()

class SummaryMemory(Memory):
    """摘要记忆：历史消息超限后压缩为摘要"""
    
    def __init__(self, llm_client, max_messages: int = 20):
        self.client = llm_client
        self.max_messages = max_messages
        self.messages: list[Message] = []
        self.summary: str = ""
    
    async def add(self, message: Message) -> None:
        self.messages.append(message)
        
        if len(self.messages) > self.max_messages:
            await self._compress()
    
    async def _compress(self):
        """将较旧的消息压缩为摘要"""
        to_compress = self.messages[:-self.max_messages // 2]
        self.messages = self.messages[-self.max_messages // 2:]
        
        msgs_text = "\n".join(
            f"{m.role}: {m.content}" for m in to_compress
        )
        
        response = await self.client.chat.completions.create(
            model="gpt-4o-mini",
            messages=[{
                "role": "user",
                "content": f"请用300字内摘要以下对话：\n\n{msgs_text}"
            }]
        )
        
        new_summary = response.choices[0].message.content
        if self.summary:
            self.summary += "\n\n" + new_summary
        else:
            self.summary = new_summary
    
    async def get(self, max_tokens: int | None = None) -> list[Message]:
        result = []
        if self.summary:
            result.append(Message(
                role="system",
                content=f"[历史对话摘要]\n{self.summary}"
            ))
        result.extend(self.messages)
        return result
    
    async def clear(self) -> None:
        self.messages.clear()
        self.summary = ""
```

class CodeInterpreterTool(Tool):
    name = "run_python"
    description = "执行 Python 代码，返回运行结果"
    parameters = {
        "type": "object",
        "properties": {
            "code": {"type": "string", "description": "Python 代码"},
        },
        "required": ["code"]
    }
    
    async def execute(self, code: str) -> str:
        import subprocess
        result = subprocess.run(
            ["python", "-c", code],
            capture_output=True, text=True, timeout=10,
            cwd="/tmp/sandbox",  # 沙箱目录
        )
        if result.returncode == 0:
            return result.stdout or "(无输出)"
        return f"错误:\n{result.stderr}"

## 6. 主流框架对比

| 框架 | 优势 | 劣势 | 适用场景 |
|------|------|------|---------|
| **LangGraph** | 状态机清晰、调试工具好 | 学习曲线陡 | 复杂 Agent 流程 |
| **LangChain** | 组件丰富、生态大 | 抽象层多、版本变化快 | 快速原型 |
| **AutoGen** | 多 Agent 对话灵活 | 代码执行安全风险 | 代码生成 Agent |
| **CrewAI** | 角色扮演直观 | 灵活性不足 | 固定角色分工 |
| **自研** | 完全可控 | 维护成本高 | 特殊需求/学习 |
