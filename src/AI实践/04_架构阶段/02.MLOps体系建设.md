# MLOps 体系建设

MLOps 是把 AI 系统当作软件工程来管理：版本化、自动化测试、持续部署、监控告警。本文聚焦 LLM 应用的 MLOps，不涉及传统 ML 训练流水线。

## 1. LLM 应用的 MLOps 特殊性

与传统软件的区别：

| 方面 | 传统软件 | LLM 应用 |
|------|---------|---------|
| 测试 | 单元测试确定性 | 输出随机，需语义评估 |
| 版本化 | 代码版本 | 代码 + Prompt + 模型版本 |
| 部署 | 发布新代码 | 发布新 Prompt / 切换新模型 |
| 监控 | 错误率/延迟 | 质量分/幻觉率/用户满意度 |
| 回滚 | 回滚代码 | 回滚 Prompt/模型，清缓存 |

## 2. Prompt 版本管理

```
prompts/
├── v1/
│   ├── rag_answer.txt
│   └── code_review.txt  
├── v2/
│   ├── rag_answer.txt   ← 改进版
│   └── code_review.txt
└── current -> v2/       ← 软链接指向当前版本
```

```python
# prompt_registry.py
import hashlib
import json
from pathlib import Path
from datetime import datetime

class PromptRegistry:
    def __init__(self, store_path: str = "./prompts"):
        self.store = Path(store_path)
        self.store.mkdir(exist_ok=True)
        self.index_file = self.store / "index.json"
        self._load_index()
    
    def _load_index(self):
        if self.index_file.exists():
            self.index = json.loads(self.index_file.read_text())
        else:
            self.index = {}
    
    def save(self, name: str, content: str, metadata: dict = {}) -> str:
        """保存 Prompt 版本，返回版本 ID"""
        version_id = hashlib.sha256(content.encode()).hexdigest()[:12]
        
        prompt_path = self.store / name / f"{version_id}.txt"
        prompt_path.parent.mkdir(exist_ok=True)
        prompt_path.write_text(content, encoding="utf-8")
        
        if name not in self.index:
            self.index[name] = []
        
        self.index[name].append({
            "version_id": version_id,
            "created_at": datetime.now().isoformat(),
            "metadata": metadata,
        })
        
        self.index_file.write_text(json.dumps(self.index, ensure_ascii=False, indent=2))
        return version_id
    
    def get(self, name: str, version_id: str = "latest") -> str:
        """获取 Prompt 内容"""
        if version_id == "latest":
            if name not in self.index or not self.index[name]:
                raise KeyError(f"Prompt '{name}' 不存在")
            version_id = self.index[name][-1]["version_id"]
        
        prompt_path = self.store / name / f"{version_id}.txt"
        return prompt_path.read_text(encoding="utf-8")
    
    def list_versions(self, name: str) -> list[dict]:
        return self.index.get(name, [])

registry = PromptRegistry()
```

## 3. A/B 测试框架

```python
import random
import asyncio
from dataclasses import dataclass
from typing import Callable

@dataclass
class Variant:
    name: str
    weight: float  # 流量比例，如 0.5 = 50%
    prompt_version: str
    model: str | None = None

class ABTestFramework:
    def __init__(self, variants: list[Variant]):
        assert abs(sum(v.weight for v in variants) - 1.0) < 0.001
        self.variants = variants
        self.results: dict[str, list[float]] = {v.name: [] for v in variants}
    
    def assign_variant(self, user_id: str) -> Variant:
        """稳定分配：同一用户总是分到同一 variant"""
        seed = int(hashlib.md5(user_id.encode()).hexdigest(), 16) % 10000 / 10000
        
        cumulative = 0
        for variant in self.variants:
            cumulative += variant.weight
            if seed < cumulative:
                return variant
        
        return self.variants[-1]
    
    def record_result(self, variant_name: str, score: float):
        """记录评估分数（如用户评分、质量分）"""
        self.results[variant_name].append(score)
    
    def get_statistics(self) -> dict:
        stats = {}
        for name, scores in self.results.items():
            if scores:
                stats[name] = {
                    "count": len(scores),
                    "mean": sum(scores) / len(scores),
                    "min": min(scores),
                    "max": max(scores),
                }
        return stats
    
    def winner(self) -> str | None:
        """返回统计显著的优胜者（简化版，生产用 t-test）"""
        stats = self.get_statistics()
        if not stats:
            return None
        
        # 简单比较均值
        best = max(stats.items(), key=lambda x: x[1]["mean"])
        return best[0] if best[1]["count"] >= 100 else None  # 至少 100 样本

# 使用示例
ab_test = ABTestFramework([
    Variant("control", 0.5, "v1", model="gpt-4o-mini"),
    Variant("treatment", 0.5, "v2", model="gpt-4o-mini"),
])

async def handle_request(user_id: str, question: str) -> str:
    variant = ab_test.assign_variant(user_id)
    prompt = registry.get("rag_answer", version_id=variant.prompt_version)
    answer = await generate_answer(question, prompt)
    return answer, variant.name
```

name: AI Quality Gates

on:
  pull_request:
    paths:
      - "prompts/**"
      - "src/rag/**"

    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      
      - name: Setup Python
        uses: actions/setup-python@v4
        with:
          python-version: "3.11"
      
      - name: Install dependencies
        run: pip install -r requirements.txt
      
      - name: Run quality tests
        env:
          OPENAI_API_KEY: ${{ secrets.OPENAI_API_KEY }}
        run: pytest tests/test_rag_quality.py -v --tb=short
      
      - name: Run security tests
        run: pytest tests/test_security.py -v
      
      - name: Post results as PR comment
        uses: actions/github-script@v6
        with:
          script: |
            const fs = require('fs');
            const results = JSON.parse(fs.readFileSync('test-results.json'));
            github.rest.issues.createComment({
              issue_number: context.issue.number,
              owner: context.repo.owner,
              repo: context.repo.repo,
              body: `## AI Quality Results\n${JSON.stringify(results, null, 2)}`
            });
