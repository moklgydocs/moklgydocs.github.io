# 模型网关设计

模型网关是 AI 系统的统一入口：屏蔽多厂商差异、统一限流鉴权、实现成本控制和智能路由。

## 1. 为什么需要模型网关

```
没有网关（分散调用）：
Service A → OpenAI API
Service B → DeepSeek API        密钥分散 / 成本难统计 / 无法统一限流
Service C → 通义千问 API

有网关（统一入口）：
Service A ↘
Service B → LLM Gateway → OpenAI / DeepSeek / 通义
Service C ↗              统一鉴权/限流/成本/路由
```

## 2. 网关核心功能设计

```python
from fastapi import FastAPI, HTTPException, Depends, Request
from fastapi.responses import StreamingResponse
from pydantic import BaseModel
import httpx
import time
import asyncio
from collections import defaultdict

app = FastAPI(title="LLM Gateway")

# ===== 数据模型 =====

class GatewayRequest(BaseModel):
    model: str  # 逻辑模型名，如 "fast" / "smart" / "cheap"
    messages: list[dict]
    temperature: float = 0.7
    max_tokens: int | None = None
    stream: bool = False

class ModelConfig:
    def __init__(self, provider: str, model_id: str, api_key: str, base_url: str):
        self.provider = provider
        self.model_id = model_id
        self.api_key = api_key
        self.base_url = base_url

# ===== 模型路由配置 =====

MODEL_ROUTES = {
    # 逻辑名 → 物理模型（支持 fallback 列表）
    "fast": [
        ModelConfig("deepseek", "deepseek-chat", "sk-ds-...", "https://api.deepseek.com"),
    ],
    "smart": [
        ModelConfig("openai", "gpt-4o", "sk-...", "https://api.openai.com/v1"),
        ModelConfig("deepseek", "deepseek-chat", "sk-ds-...", "https://api.deepseek.com"),  # fallback
    ],
    "cheap": [
        ModelConfig("openai", "gpt-4o-mini", "sk-...", "https://api.openai.com/v1"),
    ],
}
```

## 3. 智能路由策略

```python
import random

class Router:
    def __init__(self):
        self.health: dict[str, bool] = {}  # 模型健康状态
        self.error_counts: dict[str, int] = defaultdict(int)
        self.last_error_time: dict[str, float] = {}
        self.circuit_open_until: dict[str, float] = {}
    
    def select_model(self, logical_name: str) -> ModelConfig | None:
        """选择可用的模型（熔断 + fallback）"""
        candidates = MODEL_ROUTES.get(logical_name, [])
        
        for model in candidates:
            key = f"{model.provider}:{model.model_id}"
            
            # 熔断检查
            if key in self.circuit_open_until:
                if time.time() < self.circuit_open_until[key]:
                    continue  # 熔断期内，跳过
                else:
                    del self.circuit_open_until[key]  # 半开放，尝试恢复
            
            return model
        
        return None
    
    def record_failure(self, model: ModelConfig):
        key = f"{model.provider}:{model.model_id}"
        self.error_counts[key] += 1
        self.last_error_time[key] = time.time()
        
        # 5 次失败内开启熔断
        if self.error_counts[key] >= 5:
            self.circuit_open_until[key] = time.time() + 60  # 熔断 60 秒
            self.error_counts[key] = 0
            print(f"熔断触发: {key}，60秒后尝试恢复")
    
    def record_success(self, model: ModelConfig):
        key = f"{model.provider}:{model.model_id}"
        self.error_counts[key] = 0

router = Router()
```

## 4. 限流实现

```python
import asyncio
from collections import defaultdict

class RateLimiter:
    """令牌桶限流"""
    
    def __init__(self, rate: int, per: int = 60):
        """rate 次/per 秒"""
        self.rate = rate
        self.per = per
        self.buckets: dict[str, list[float]] = defaultdict(list)
        self._lock = asyncio.Lock()
    
    async def acquire(self, key: str) -> bool:
        async with self._lock:
            now = time.time()
            window_start = now - self.per
            
            # 清理过期记录
            self.buckets[key] = [t for t in self.buckets[key] if t > window_start]
            
            if len(self.buckets[key]) >= self.rate:
                return False
            
            self.buckets[key].append(now)
            return True

# 多级限流
rate_limiters = {
    "per_user_per_minute": RateLimiter(rate=20, per=60),
    "per_user_per_day": RateLimiter(rate=1000, per=86400),
    "global_per_second": RateLimiter(rate=100, per=1),
}

async def check_rate_limits(api_key: str) -> bool:
    for name, limiter in rate_limiters.items():
        key = f"{name}:{api_key}" if "user" in name else name
        if not await limiter.acquire(key):
            raise HTTPException(429, f"超过限流限制: {name}")
    return True
```

## 5. 完整网关接口

```python
@app.post("/v1/chat/completions")
async def chat_completions(req: GatewayRequest, request: Request):
    # 1. API Key 校验
    api_key = request.headers.get("Authorization", "").replace("Bearer ", "")
    if not validate_api_key(api_key):
        raise HTTPException(401, "无效的 API Key")
    
    # 2. 限流
    await check_rate_limits(api_key)
    
    # 3. 路由选择
    model_config = router.select_model(req.model)
    if not model_config:
        raise HTTPException(503, f"模型 '{req.model}' 暂时不可用")
    
    # 4. 转发请求
    start_time = time.time()
    
    try:
        if req.stream:
            return await stream_forward(req, model_config, api_key)
        else:
            return await forward_request(req, model_config, api_key, start_time)
    except httpx.HTTPError as e:
        router.record_failure(model_config)
        raise HTTPException(502, f"上游模型错误: {str(e)}")

async def forward_request(req: GatewayRequest, config: ModelConfig, api_key: str, start_time: float):
    async with httpx.AsyncClient(timeout=60) as client:
        resp = await client.post(
            f"{config.base_url}/chat/completions",
            headers={"Authorization": f"Bearer {config.api_key}"},
            json={
                "model": config.model_id,
                "messages": req.messages,
                "temperature": req.temperature,
                "max_tokens": req.max_tokens,
            }
        )
        resp.raise_for_status()
        data = resp.json()
    
    router.record_success(config)
    latency = (time.time() - start_time) * 1000
    
    # 成本追踪
    usage = data.get("usage", {})
    cost = calculate_cost(config.model_id, usage.get("prompt_tokens", 0), usage.get("completion_tokens", 0))
    track_usage(api_key, config.model_id, latency, cost)
    
    return data

async def stream_forward(req: GatewayRequest, config: ModelConfig, api_key: str):
    async def gen():
        async with httpx.AsyncClient(timeout=None) as client:
            async with client.stream(
                "POST",
                f"{config.base_url}/chat/completions",
                headers={"Authorization": f"Bearer {config.api_key}"},
                json={
                    "model": config.model_id,
                    "messages": req.messages,
                    "temperature": req.temperature,
                    "stream": True,
                }
            ) as resp:
                async for line in resp.aiter_lines():
                    yield line + "\n\n"
    
    return StreamingResponse(gen(), media_type="text/event-stream")
```

## 6. 成本预算控制

```python
class BudgetController:
    def __init__(self, daily_limit_usd: float):
        self.daily_limit = daily_limit_usd
        self.spent_today: float = 0.0
        self.last_reset = time.strftime("%Y-%m-%d")
    
    def check_budget(self) -> bool:
        today = time.strftime("%Y-%m-%d")
        if today != self.last_reset:
            self.spent_today = 0.0
            self.last_reset = today
        
        return self.spent_today < self.daily_limit
    
    def record_spend(self, amount_usd: float):
        self.spent_today += amount_usd
        
        remaining_pct = (1 - self.spent_today / self.daily_limit) * 100
        
        if self.spent_today >= self.daily_limit * 0.9:
            # 90% 预算 → 自动降级到便宜模型
            print(f"预警：已使用 {100 - remaining_pct:.0f}% 预算，切换到低成本模型")

budget = BudgetController(daily_limit_usd=50.0)
```
