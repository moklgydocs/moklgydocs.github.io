# MCP 协议开发实战

MCP（Model Context Protocol）是 Anthropic 推出的开放协议，让 AI 助手能安全、标准化地连接外部数据源和工具。本文从零实现一个 MCP Server，并接入 Claude Desktop。

## 1. MCP 架构概览

```mermaid
graph LR
    Client["AI Client\n(Claude Desktop)"]
    Server["MCP Server\n(你写的)"]
    DB["数据源\n(数据库/API/文件)"]
    
    Client <-->|MCP 协议| Server
    Server <--> DB
```

**三大原语：**

| 原语 | 用途 | 类比 |
|------|------|------|
| **Resources** | 暴露数据给模型读取 | 上下文/文件 |
| **Tools** | 让模型调用执行操作 | Function Calling |
| **Prompts** | 预制可复用的提示模板 | 提示词库 |

## 3. 最简 MCP Server

```python
# server.py
from mcp.server.fastmcp import FastMCP

# 创建 MCP Server
mcp = FastMCP("我的工具服务")

# 注册工具
@mcp.tool()
def add(a: int, b: int) -> int:
    """将两个数字相加"""
    return a + b

@mcp.tool()
def get_time() -> str:
    """获取当前时间"""
    from datetime import datetime
    return datetime.now().strftime("%Y-%m-%d %H:%M:%S")

# 运行（stdio 模式，供 Claude Desktop 使用）
if __name__ == "__main__":
    mcp.run()
```

## 4. 完整实战：文件管理 MCP Server

```python
# file_manager_server.py
import os
import json
from pathlib import Path
from mcp.server.fastmcp import FastMCP
from mcp import types

mcp = FastMCP("文件管理器")

# 安全根目录（限制访问范围）
SAFE_ROOT = Path(os.getenv("MCP_FILE_ROOT", "./workspace")).resolve()

def safe_path(relative_path: str) -> Path:
    """确保路径在安全根目录内，防止路径遍历攻击"""
    full_path = (SAFE_ROOT / relative_path).resolve()
    if not str(full_path).startswith(str(SAFE_ROOT)):
        raise ValueError(f"禁止访问安全目录之外的路径: {relative_path}")
    return full_path

# ============ Resources：暴露文件列表 ============

@mcp.resource("file://workspace/list")
def list_workspace() -> str:
    """列出工作区的所有文件"""
    files = []
    for p in SAFE_ROOT.rglob("*"):
        if p.is_file():
            files.append({
                "path": str(p.relative_to(SAFE_ROOT)),
                "size": p.stat().st_size,
                "modified": p.stat().st_mtime,
            })
    return json.dumps(files, ensure_ascii=False, indent=2)

@mcp.resource("file://workspace/{filename}")
def read_file(filename: str) -> str:
    """读取指定文件内容"""
    path = safe_path(filename)
    if not path.exists():
        raise FileNotFoundError(f"文件不存在: {filename}")
    return path.read_text(encoding="utf-8")

# ============ Tools：文件操作 ============

@mcp.tool()
def create_file(filename: str, content: str) -> str:
    """
    在工作区创建新文件
    
    Args:
        filename: 文件名（相对路径）
        content: 文件内容
    """
    path = safe_path(filename)
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(content, encoding="utf-8")
    return f"文件已创建: {filename}"

@mcp.tool()
def append_to_file(filename: str, content: str) -> str:
    """向文件末尾追加内容"""
    path = safe_path(filename)
    if not path.exists():
        return f"文件不存在: {filename}"
    with path.open("a", encoding="utf-8") as f:
        f.write(content)
    return f"内容已追加到: {filename}"

@mcp.tool()
def search_in_files(keyword: str, file_pattern: str = "*.txt") -> str:
    """
    在工作区文件中搜索关键词
    
    Args:
        keyword: 搜索关键词
        file_pattern: 文件匹配模式，默认 *.txt
    """
    results = []
    for path in SAFE_ROOT.glob(file_pattern):
        if path.is_file():
            try:
                content = path.read_text(encoding="utf-8")
                if keyword in content:
                    # 找到匹配行
                    for i, line in enumerate(content.splitlines(), 1):
                        if keyword in line:
                            results.append(f"{path.name}:{i}: {line.strip()}")
            except Exception:
                pass
    
    if results:
        return "\n".join(results)
    return f"未找到包含「{keyword}」的内容"

# ============ Prompts：预制提示模板 ============

@mcp.prompt()
def code_review_prompt(filename: str) -> list[types.PromptMessage]:
    """生成代码审查提示"""
    return [
        types.PromptMessage(
            role="user",
            content=types.TextContent(
                type="text",
                text=f"请审查文件 {filename} 的代码，重点关注：\n1. 潜在 Bug\n2. 安全漏洞\n3. 性能问题\n4. 代码规范"
            )
        )
    ]

if __name__ == "__main__":
    mcp.run()
```

## 5. 接入 Claude Desktop

### 配置文件位置

- **MacOS**: `~/Library/Application Support/Claude/claude_desktop_config.json`
- **Windows**: `%APPDATA%\Claude\claude_desktop_config.json`

### 配置内容

```json
{
  "mcpServers": {
    "file-manager": {
      "command": "python",
      "args": ["/absolute/path/to/file_manager_server.py"],
      "env": {
        "MCP_FILE_ROOT": "/Users/you/workspace"
      }
    }
  }
}
```

重启 Claude Desktop，在对话框右下角选择"文件管理器"工具即可使用。

## 6. HTTP 模式（多客户端共享）

```python
# http_server.py - 通过 HTTP 暴露 MCP 服务（SSE 传输）
from mcp.server.fastmcp import FastMCP

mcp = FastMCP("HTTP MCP Server")

@mcp.tool()
def query_database(sql: str) -> str:
    """执行只读 SQL 查询"""
    # ...
    pass

# 启动 HTTP 服务
if __name__ == "__main__":
    mcp.run(transport="streamable-http", host="0.0.0.0", port=8080)
```

```python
# client.py - Python 客户端调用
import asyncio
from mcp import ClientSession
from mcp.client.streamable_http import streamablehttp_client

async def main():
    async with streamablehttp_client("http://localhost:8080/mcp") as (read, write, _):
        async with ClientSession(read, write) as session:
            await session.initialize()
            
            # 列出可用工具
            tools = await session.list_tools()
            print("可用工具:", [t.name for t in tools.tools])
            
            # 调用工具
            result = await session.call_tool(
                "query_database",
                {"sql": "SELECT COUNT(*) FROM users"}
            )
            print("查询结果:", result.content[0].text)

asyncio.run(main())
```

## 7. 实战：数据库查询 MCP Server

```python
# db_mcp_server.py
import asyncio
import json
import asyncpg
from mcp.server.fastmcp import FastMCP
from contextlib import asynccontextmanager

mcp = FastMCP("数据库助手")

# 数据库连接池（使用 lifespan 管理）
pool: asyncpg.Pool = None

@asynccontextmanager
async def lifespan():
    global pool
    pool = await asyncpg.create_pool(
        dsn="postgresql://user:pass@localhost/mydb",
        min_size=2,
        max_size=10,
    )
    yield
    await pool.close()

@mcp.tool()
async def query(sql: str) -> str:
    """
    执行只读 SQL 查询（SELECT 语句）
    
    Args:
        sql: SQL 查询语句，只允许 SELECT
    """
    # 安全检查：只允许 SELECT
    sql_clean = sql.strip().upper()
    if not sql_clean.startswith("SELECT"):
        return "错误：只允许 SELECT 查询"
    
    async with pool.acquire() as conn:
        rows = await conn.fetch(sql)
        result = [dict(row) for row in rows[:100]]  # 最多返回 100 行
        return json.dumps(result, ensure_ascii=False, default=str)

@mcp.tool()
async def describe_table(table_name: str) -> str:
    """获取表结构"""
    async with pool.acquire() as conn:
        rows = await conn.fetch("""
            SELECT column_name, data_type, is_nullable
            FROM information_schema.columns
            WHERE table_name = $1
            ORDER BY ordinal_position
        """, table_name)
        return json.dumps([dict(r) for r in rows], ensure_ascii=False)

@mcp.resource("db://tables")
async def list_tables() -> str:
    """列出所有可查询的表"""
    async with pool.acquire() as conn:
        rows = await conn.fetch("""
            SELECT table_name FROM information_schema.tables
            WHERE table_schema = 'public'
        """)
        tables = [r["table_name"] for r in rows]
        return json.dumps(tables, ensure_ascii=False)

if __name__ == "__main__":
    mcp.run()
```
