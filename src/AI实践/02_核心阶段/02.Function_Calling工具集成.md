# Function Calling 与工具集成

Function Calling 是让 LLM 与真实世界交互的桥梁。模型通过分析用户意图，输出结构化的工具调用参数，由应用层执行后将结果反馈给模型。

## 1. 工作原理

```
用户："查一下苹果和微软的股价，算一下谁涨幅更大"
  ↓
LLM 分析意图
  ↓
输出工具调用: [get_stock("AAPL"), get_stock("MSFT")]
  ↓
应用层并行执行工具
  ↓
返回结果 → LLM 生成最终回答
```

### Pydantic 定义（推荐）

```python
from pydantic import BaseModel, Field
from openai import AsyncOpenAI
import json

client = AsyncOpenAI()

class GetWeatherArgs(BaseModel):
    city: str = Field(description="城市名称")
    unit: str = Field(default="celsius", description="温度单位")

class CreateEventArgs(BaseModel):
    title: str = Field(description="事件标题")
    start_time: str = Field(description="开始时间 ISO 8601")
    end_time: str = Field(description="结束时间 ISO 8601")
    location: str | None = Field(default=None, description="地点")

def pydantic_to_tool(name: str, description: str, model: type[BaseModel]) -> dict:
    """把 Pydantic 模型转换为工具定义"""
    return {
        "type": "function",
        "function": {
            "name": name,
            "description": description,
            "parameters": model.model_json_schema(),
        }
    }

tools = [
    pydantic_to_tool("get_weather", "获取天气", GetWeatherArgs),
    pydantic_to_tool("create_calendar_event", "创建日历事件", CreateEventArgs),
]
```

## 3. 完整工具调用流程

```python
import asyncio
import json
from openai.types.chat import ChatCompletionMessageToolCall

# 工具实现
async def get_weather(city: str, unit: str = "celsius") -> dict:
    # 实际中调用天气 API
    return {"city": city, "temperature": 25, "condition": "晴", "unit": unit}

async def create_calendar_event(title: str, start_time: str, end_time: str, location: str | None = None) -> dict:
    # 实际中调用日历 API
    return {"event_id": "evt_001", "title": title, "status": "created"}

# 工具注册表
TOOL_REGISTRY = {
    "get_weather": get_weather,
    "create_calendar_event": create_calendar_event,
}

async def execute_tool(tool_call: ChatCompletionMessageToolCall) -> str:
    """执行单个工具调用"""
    name = tool_call.function.name
    args = json.loads(tool_call.function.arguments)
    
    if name not in TOOL_REGISTRY:
        return f"错误：工具 '{name}' 不存在"
    
    try:
        result = await TOOL_REGISTRY[name](**args)
        return json.dumps(result, ensure_ascii=False)
    except Exception as e:
        return f"工具执行错误：{str(e)}"

async def agent_loop(user_message: str) -> str:
    """完整的 Agent 循环"""
    messages = [{"role": "user", "content": user_message}]
    
    while True:
        response = await client.chat.completions.create(
            model="gpt-4o",
            messages=messages,
            tools=tools,
            tool_choice="auto",  # 让模型自己决定是否调用工具
        )
        
        message = response.choices[0].message
        messages.append(message.model_dump())
        
        # 没有工具调用，返回最终答案
        if not message.tool_calls:
            return message.content
        
        # 并行执行所有工具调用
        tool_results = await asyncio.gather(*[
            execute_tool(tc) for tc in message.tool_calls
        ])
        
        # 将工具结果追加到消息列表
        for tool_call, result in zip(message.tool_calls, tool_results):
            messages.append({
                "role": "tool",
                "tool_call_id": tool_call.id,
                "content": result,
            })
    
# 测试
result = asyncio.run(agent_loop(
    "查一下北京和上海的天气，帮我安排明天下午 3 点到 4 点的会议，主题是「项目评审」，地点北京办公室"
))
print(result)
```

## 4. 并行工具调用

GPT-4o 和 Claude 都支持一次请求中调用多个工具（Parallel Tool Calling）。

```python
# 模型可能返回多个工具调用：
# tool_calls = [
#   ToolCall(id="call_1", name="get_weather", args='{"city":"北京"}'),
#   ToolCall(id="call_2", name="get_weather", args='{"city":"上海"}'),
# ]

# 关键：并行执行，不要串行
async def execute_parallel_tools(tool_calls: list) -> list[dict]:
    tasks = [execute_tool(tc) for tc in tool_calls]
    results = await asyncio.gather(*tasks, return_exceptions=True)
    
    tool_messages = []
    for tc, result in zip(tool_calls, results):
        if isinstance(result, Exception):
            content = f"工具执行失败：{str(result)}"
        else:
            content = result
        
        tool_messages.append({
            "role": "tool",
            "tool_call_id": tc.id,
            "content": content,
        })
    
    return tool_messages
```

## 5. 错误处理与重试

```python
from tenacity import retry, stop_after_attempt, wait_exponential

class ToolExecutor:
    def __init__(self, max_retries: int = 3):
        self.max_retries = max_retries
    
    async def execute_with_retry(self, tool_call: ChatCompletionMessageToolCall) -> str:
        name = tool_call.function.name
        args = json.loads(tool_call.function.arguments)
        
        for attempt in range(self.max_retries):
            try:
                result = await TOOL_REGISTRY[name](**args)
                return json.dumps(result, ensure_ascii=False)
            except Exception as e:
                if attempt == self.max_retries - 1:
                    # 最后一次也失败，返回错误信息让模型处理
                    return json.dumps({
                        "error": str(e),
                        "suggestion": "工具执行失败，请尝试其他方法"
                    })
                await asyncio.sleep(2 ** attempt)  # 指数退避
        
        return '{"error": "超过最大重试次数"}'
```

## 6. 工具设计最佳实践

### Description 写作技巧

```python
# 差：太模糊
{"name": "search", "description": "搜索"}

# 好：明确何时用、用什么参数
{
    "name": "search_knowledge_base",
    "description": """在内部知识库中搜索相关文档。
    
    适用场景：
    - 用户询问产品功能、使用方法
    - 需要查找内部政策或流程
    
    不适用：
    - 实时天气、股价等外部数据（用 get_weather / get_stock）
    - 个人用户账户信息（用 get_user_info）
    """,
}
```

### 参数粒度设计

```python
# 太细粒度：多次调用才能完成任务
send_email_to("boss@company.com")
set_email_subject("季度报告")
add_email_body("请查收...")
send_email()  # 4 次调用

# 合适粒度：一次调用完成
send_email(
    to="boss@company.com",
    subject="季度报告",
    body="请查收...",
)
```

### 工具权限控制

```python
from enum import Enum

class ToolPermission(Enum):
    READ = "read"      # 只读操作
    WRITE = "write"    # 写操作
    ADMIN = "admin"    # 管理操作

TOOL_PERMISSIONS = {
    "search_web": ToolPermission.READ,
    "create_file": ToolPermission.WRITE,
    "delete_all": ToolPermission.ADMIN,
}

def get_allowed_tools(user_role: str) -> list[dict]:
    """根据用户角色过滤可用工具"""
    allowed_perms = {
        "viewer": [ToolPermission.READ],
        "editor": [ToolPermission.READ, ToolPermission.WRITE],
        "admin": list(ToolPermission),
    }.get(user_role, [])

        t for name, t in ALL_TOOLS.items()
        if TOOL_PERMISSIONS.get(name) in allowed_perms
    ]
```
