# HTTPS 与证书配置

2025 年了，没有 HTTPS 的网站浏览器直接标红。Let's Encrypt 免费证书 + 自动续期，没理由不上。

---

## Let's Encrypt 免费证书

### 安装 Certbot

```bash
# CentOS/RHEL
sudo yum install -y certbot python3-certbot-nginx

# Ubuntu
sudo apt install -y certbot python3-certbot-nginx
```

### 申请证书

```bash
# 自动配置（Certbot 自动修改 Nginx 配置）
sudo certbot --nginx -d erp.company.com -d api.company.com

# 仅申请证书，不改 Nginx 配置
sudo certbot certonly --nginx -d erp.company.com

# 证书位置
# /etc/letsencrypt/live/erp.company.com/fullchain.pem   ← 证书链
# /etc/letsencrypt/live/erp.company.com/privkey.pem     ← 私钥
```

### 自动续期

```bash
# 测试续期
sudo certbot renew --dry-run

# 自动续期（certbot 安装时已配置 systemd timer 或 cron）
# 查看定时任务
sudo systemctl list-timers | grep certbot

# 手动添加 cron（如果没有）
# 每天凌晨 3 点检查续期
0 3 * * * certbot renew --quiet --post-hook "nginx -s reload"
```

## 手动配置 SSL

```nginx
server {
    listen 80;
    server_name erp.company.com;
    # HTTP 强制跳转 HTTPS
    return 301 https://$server_name$request_uri;
}

server {
    listen 443 ssl http2;
    server_name erp.company.com;

    # 证书
    ssl_certificate /etc/letsencrypt/live/erp.company.com/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/erp.company.com/privkey.pem;

    # SSL 协议（只开 1.2 和 1.3）
    ssl_protocols TLSv1.2 TLSv1.3;

    # 加密套件
    ssl_ciphers ECDHE-ECDSA-AES128-GCM-SHA256:ECDHE-RSA-AES128-GCM-SHA256:ECDHE-ECDSA-AES256-GCM-SHA384:ECDHE-RSA-AES256-GCM-SHA384;
    ssl_prefer_server_ciphers on;

    # SSL Session 缓存
    ssl_session_cache shared:SSL:10m;
    ssl_session_timeout 10m;

    # OCSP Stapling
    ssl_stapling on;
    ssl_stapling_verify on;
    resolver 8.8.8.8 valid=300s;

    # HSTS（强制浏览器使用 HTTPS）
    add_header Strict-Transport-Security "max-age=31536000; includeSubDomains" always;

    # 反向代理
    location / {
        proxy_pass http://127.0.0.1:5000;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }
}
```

## SSL 通用配置（抽取复用）

```nginx
# /etc/nginx/conf.d/ssl-common.conf（不要放 server 块，被 include）
ssl_protocols TLSv1.2 TLSv1.3;
ssl_ciphers ECDHE-ECDSA-AES128-GCM-SHA256:ECDHE-RSA-AES128-GCM-SHA256:ECDHE-ECDSA-AES256-GCM-SHA384:ECDHE-RSA-AES256-GCM-SHA384;
ssl_prefer_server_ciphers on;
ssl_session_cache shared:SSL:10m;
ssl_session_timeout 10m;
ssl_stapling on;
ssl_stapling_verify on;
resolver 8.8.8.8 valid=300s;
add_header Strict-Transport-Security "max-age=31536000; includeSubDomains" always;
```

```nginx
# 站点配置里引用
server {
    listen 443 ssl http2;
    server_name erp.company.com;

    ssl_certificate /etc/nginx/ssl/erp.company.com.pem;
    ssl_certificate_key /etc/nginx/ssl/erp.company.com.key;
    include /etc/nginx/conf.d/ssl-common.conf;

    location / { ... }
}
```

## 自签名证书（开发/内网）

```bash
# 生成自签名证书（有效期 365 天）
openssl req -x509 -nodes -days 365 \
  -newkey rsa:2048 \
  -keyout /etc/nginx/ssl/self-signed.key \
  -out /etc/nginx/ssl/self-signed.pem \
  -subj "/CN=dev.company.local"
```

::: warning
自签名证书浏览器会报不安全。内网/开发环境可以用，生产环境用 Let's Encrypt。
:::

## Docker 环境 + Certbot

```yaml
services:
  nginx:
    image: nginx:alpine
    ports:
      - "80:80"
      - "443:443"
    volumes:
      - ./conf.d:/etc/nginx/conf.d:ro
      - ./ssl:/etc/nginx/ssl:ro
      - certbot-webroot:/var/www/certbot:ro
      - certbot-certs:/etc/letsencrypt:ro

  certbot:
    image: certbot/certbot
    volumes:
      - certbot-webroot:/var/www/certbot
      - certbot-certs:/etc/letsencrypt
    entrypoint: "/bin/sh -c 'trap exit TERM; while :; do certbot renew; sleep 12h & wait $${!}; done'"

volumes:
  certbot-webroot:
  certbot-certs:
```

```nginx
# Nginx 配置中允许 Certbot 验证
server {
    listen 80;
    server_name erp.company.com;

    location /.well-known/acme-challenge/ {
        root /var/www/certbot;
    }

    location / {
        return 301 https://$server_name$request_uri;
    }
}
```

```bash
# 首次申请
docker compose run --rm certbot certonly \
  --webroot -w /var/www/certbot \
  -d erp.company.com --email admin@company.com --agree-tos
```
