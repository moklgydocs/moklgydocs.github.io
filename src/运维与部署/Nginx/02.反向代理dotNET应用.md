# 反向代理 .NET 应用

Kestrel 不直接面对公网。Nginx 挡在前面做反向代理，处理 HTTPS、静态文件、负载均衡。

---

## 架构

```
客户端 → Nginx(:80/:443) → Kestrel(:5000) → ASP.NET Core 应用

具体来说：
  浏览器 → HTTPS → Nginx → HTTP → Kestrel
  Nginx 负责：SSL终止、静态文件、请求转发、日志
  Kestrel 负责：运行 .NET 应用
```

---

## 基础反向代理

```nginx
# /etc/nginx/conf.d/erp-api.conf
server {
    listen 80;
    server_name api.company.com;

    location / {
        proxy_pass http://127.0.0.1:5000;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }
}
```

### 关键 Header 说明

```nginx
proxy_set_header Host $host;
# → 后端拿到的 Host 是原始域名，不是 127.0.0.1

proxy_set_header X-Real-IP $remote_addr;
# → 后端拿到真实客户端 IP

proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
# → 代理链上所有 IP

proxy_set_header X-Forwarded-Proto $scheme;
# → 后端知道原始请求是 HTTP 还是 HTTPS
```

### ASP.NET Core 配合配置

```csharp
// Program.cs
builder.Services.Configure<ForwardedHeadersOptions>(options =>
{
    options.ForwardedHeaders = ForwardedHeaders.XForwardedFor | ForwardedHeaders.XForwardedProto;
    options.KnownProxies.Add(IPAddress.Parse("172.17.0.1")); // Nginx 的 IP
});

app.UseForwardedHeaders();
```

## WebSocket 代理

```nginx
# SignalR / WebSocket
location /hubs/ {
    proxy_pass http://127.0.0.1:5000;
    proxy_http_version 1.1;
    proxy_set_header Upgrade $http_upgrade;
    proxy_set_header Connection "upgrade";
    proxy_set_header Host $host;
    proxy_set_header X-Real-IP $remote_addr;
    proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
    proxy_set_header X-Forwarded-Proto $scheme;

    # WebSocket 长连接超时
    proxy_read_timeout 3600s;
    proxy_send_timeout 3600s;
}
```

## gRPC 代理

```nginx
# gRPC 需要 HTTP/2
server {
    listen 443 ssl http2;
    server_name grpc.company.com;

    ssl_certificate /etc/nginx/ssl/grpc.company.com.pem;
    ssl_certificate_key /etc/nginx/ssl/grpc.company.com.key;

    location / {
        grpc_pass grpc://127.0.0.1:5001;
        grpc_set_header Host $host;
        grpc_set_header X-Real-IP $remote_addr;

        # 超时
        grpc_read_timeout 300s;
        grpc_send_timeout 300s;
    }
}
```

## 负载均衡

```nginx
# 定义后端集群
upstream erp_backend {
    # 轮询（默认）
    server 127.0.0.1:5001;
    server 127.0.0.1:5002;
    server 127.0.0.1:5003;
}

# 加权轮询
upstream erp_backend_weighted {
    server 127.0.0.1:5001 weight=3;   # 处理 3/6 请求
    server 127.0.0.1:5002 weight=2;   # 处理 2/6 请求
    server 127.0.0.1:5003 weight=1;   # 处理 1/6 请求
}

# IP Hash（同一客户端总是访问同一后端）
upstream erp_backend_sticky {
    ip_hash;
    server 127.0.0.1:5001;
    server 127.0.0.1:5002;
}

# 最少连接
upstream erp_backend_least {
    least_conn;
    server 127.0.0.1:5001;
    server 127.0.0.1:5002;
}

server {
    listen 80;
    server_name erp.company.com;

    location / {
        proxy_pass http://erp_backend;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }
}
```

### 健康检查（被动）

```nginx
upstream erp_backend {
    server 127.0.0.1:5001 max_fails=3 fail_timeout=30s;
    server 127.0.0.1:5002 max_fails=3 fail_timeout=30s;
    server 127.0.0.1:5003 backup;   # 备用，前面全挂了才用
}
# 3 次失败后，30 秒内不再转发到该节点
```

## Docker 环境反向代理

```nginx
# Docker Compose 网络中，用服务名做反向代理
upstream erp_api {
    server erp-api:5000;     # Docker 服务名
}

upstream sso_api {
    server sso-api:5000;
}

server {
    listen 80;
    server_name erp.company.com;
    location / { proxy_pass http://erp_api; }
}

server {
    listen 80;
    server_name sso.company.com;
    location / { proxy_pass http://sso_api; }
}
```

## 前后端分离部署

```nginx
server {
    listen 80;
    server_name erp.company.com;

    # 前端（Vue/React 构建产物）
    location / {
        root /usr/share/nginx/html/erp;
        index index.html;
        try_files $uri $uri/ /index.html;   # SPA 路由
    }

    # API 代理
    location /api/ {
        proxy_pass http://127.0.0.1:5000/api/;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }

    # SignalR
    location /hubs/ {
        proxy_pass http://127.0.0.1:5000/hubs/;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
        proxy_set_header Host $host;
        proxy_read_timeout 3600s;
    }
}
```

## 速查

| 场景 | 关键配置 |
|------|---------|
| 基础代理 | `proxy_pass http://127.0.0.1:5000` |
| 传真实 IP | `proxy_set_header X-Real-IP $remote_addr` |
| WebSocket | `proxy_set_header Upgrade $http_upgrade` |
| gRPC | `grpc_pass grpc://127.0.0.1:5001` |
| 负载均衡 | `upstream { server ...; server ...; }` |
| 前后端分离 | `try_files $uri /index.html` + `location /api/` |
