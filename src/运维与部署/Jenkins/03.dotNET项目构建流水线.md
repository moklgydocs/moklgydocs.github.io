# .NET 项目构建流水线

从 GitLab 拉代码 → 编译 → 测试 → Docker 打包 → 推送镜像 → 部署到服务器。完整的 .NET 项目 CI/CD 流水线。

---

## 完整的 Jenkinsfile

```groovy
pipeline {
    agent any

    environment {
        APP_NAME = 'erp-api'
        DOCKER_REGISTRY = 'registry.company.com'
        DEPLOY_SERVER = 'deploy@192.168.1.100'
        DEPLOY_PATH = '/opt/apps/erp'
    }

    options {
        timeout(time: 20, unit: 'MINUTES')
        timestamps()
        disableConcurrentBuilds()
        buildDiscarder(logRotator(numToKeepStr: '20'))
    }

    triggers {
        gitlab(triggerOnPush: true, triggerOnMergeRequest: true, branchFilterType: 'All')
    }

    stages {
        // ===== 1. 拉取代码 =====
        stage('Checkout') {
            steps {
                checkout scm
                script {
                    env.GIT_TAG = sh(script: 'git describe --tags --always', returnStdout: true).trim()
                    env.GIT_MSG = sh(script: 'git log -1 --pretty=%B', returnStdout: true).trim()
                    echo "版本: ${env.GIT_TAG}"
                    echo "提交: ${env.GIT_MSG}"
                }
            }
        }

        // ===== 2. 编译 =====
        stage('Build') {
            steps {
                sh '''
                    docker run --rm \
                      -v $(pwd):/src \
                      -w /src \
                      mcr.microsoft.com/dotnet/sdk:8.0 \
                      dotnet publish ERP.Host/ERP.Host.csproj \
                        -c Release \
                        -o /src/publish \
                        --no-restore
                '''
                // 或者如果 Jenkins 服务器装了 .NET SDK
                // sh 'dotnet publish ERP.Host/ERP.Host.csproj -c Release -o ./publish'
            }
        }

        // ===== 3. 测试 =====
        stage('Test') {
            steps {
                sh '''
                    docker run --rm \
                      -v $(pwd):/src \
                      -w /src \
                      mcr.microsoft.com/dotnet/sdk:8.0 \
                      dotnet test --no-build --verbosity normal
                '''
            }
            post {
                always {
                    // 收集测试报告（如果有的话）
                    junit allowEmptyResults: true, testResults: '**/TestResults/*.xml'
                }
            }
        }

        // ===== 4. Docker 打包 =====
        stage('Docker Build') {
            steps {
                sh """
                    docker build \
                      -t ${DOCKER_REGISTRY}/${APP_NAME}:${env.GIT_TAG} \
                      -t ${DOCKER_REGISTRY}/${APP_NAME}:latest \
                      -f ERP.Host/Dockerfile .
                """
            }
        }

        // ===== 5. 推送镜像 =====
        stage('Push Image') {
            steps {
                withCredentials([usernamePassword(
                    credentialsId: 'docker-registry-cred',
                    usernameVariable: 'REG_USER',
                    passwordVariable: 'REG_PASS')]) {
                    sh """
                        echo ${REG_PASS} | docker login ${DOCKER_REGISTRY} -u ${REG_USER} --password-stdin
                        docker push ${DOCKER_REGISTRY}/${APP_NAME}:${env.GIT_TAG}
                        docker push ${DOCKER_REGISTRY}/${APP_NAME}:latest
                        docker logout ${DOCKER_REGISTRY}
                    """
                }
            }
        }

        // ===== 6. 部署到开发环境 =====
        stage('Deploy Dev') {
            when { branch 'develop' }
            steps {
                withCredentials([sshUserPrivateKey(
                    credentialsId: 'deploy-ssh-key',
                    keyFileVariable: 'SSH_KEY')]) {
                    sh """
                        ssh -i ${SSH_KEY} -o StrictHostKeyChecking=no ${DEPLOY_SERVER} '
                            cd ${DEPLOY_PATH}
                            sed -i "s/APP_VERSION=.*/APP_VERSION=${env.GIT_TAG}/" .env
                            docker compose pull erp-api
                            docker compose up -d erp-api
                            sleep 10
                            curl -sf http://localhost:5000/health || exit 1
                        '
                    """
                }
                echo "✅ 开发环境部署成功: ${env.GIT_TAG}"
            }
        }

        // ===== 7. 部署到生产环境（需要手动确认） =====
        stage('Deploy Prod') {
            when { branch 'main' }
            steps {
                input message: "确认部署 ${env.GIT_TAG} 到生产环境？", ok: '确认部署'
                withCredentials([sshUserPrivateKey(
                    credentialsId: 'prod-ssh-key',
                    keyFileVariable: 'SSH_KEY')]) {
                    sh """
                        ssh -i ${SSH_KEY} -o StrictHostKeyChecking=no prod-deploy@prod-server '
                            cd /opt/apps/erp
                            sed -i "s/APP_VERSION=.*/APP_VERSION=${env.GIT_TAG}/" .env
                            docker compose pull erp-api
                            docker compose up -d erp-api
                            sleep 15
                            curl -sf http://localhost:5000/health || exit 1
                        '
                    """
                }
                echo "✅ 生产环境部署成功: ${env.GIT_TAG}"
            }
        }
    }

    post {
        success {
            updateGitlabCommitStatus name: 'build', state: 'success'
        }
        failure {
            updateGitlabCommitStatus name: 'build', state: 'failed'
            mail to: 'dev-team@company.com',
                 subject: "❌ ${APP_NAME} 构建失败 #${BUILD_NUMBER}",
                 body: """
                    项目: ${APP_NAME}
                    分支: ${env.GIT_BRANCH}
                    提交: ${env.GIT_MSG}
                    查看: ${BUILD_URL}
                 """
        }
        always {
            // 清理本地镜像
            sh "docker rmi ${DOCKER_REGISTRY}/${APP_NAME}:${env.GIT_TAG} || true"
            cleanWs()
        }
    }
}
```

## 流水线图解

```
develop 分支推送：
  Checkout → Build → Test → Docker Build → Push Image → Deploy Dev
       ↓        ↓       ↓         ↓            ↓           ↓
    拉代码    编译    跑测试    打Docker镜像   推到仓库   SSH部署到开发机

main 分支推送：
  Checkout → Build → Test → Docker Build → Push Image → [手动确认] → Deploy Prod
```

## 两种构建策略

### 策略一：Jenkins 服务器上装 .NET SDK

```groovy
stage('Build') {
    steps {
        sh 'dotnet restore'
        sh 'dotnet publish -c Release -o ./publish'
    }
}
```

优点：快，不用每次拉 Docker 镜像
缺点：要维护 Jenkins 上的 SDK 版本

### 策略二：用 Docker 构建（推荐）

```groovy
stage('Build') {
    steps {
        sh '''
            docker run --rm \
              -v $(pwd):/src -w /src \
              mcr.microsoft.com/dotnet/sdk:8.0 \
              dotnet publish -c Release -o /src/publish
        '''
    }
}
```

优点：环境隔离，版本随意切换
缺点：首次拉镜像慢

### 策略三：直接在 Jenkins 里 docker build（最推荐）

```groovy
stage('Docker Build') {
    steps {
        // Dockerfile 里已经有多阶段构建，直接 build 即可
        sh 'docker build -t erp-api:${BUILD_NUMBER} .'
    }
}
```

优点：最简单，Dockerfile 负责一切
缺点：无

## 多分支 Pipeline

```
Jenkins → 新建 Item → Multibranch Pipeline

Branch Sources:
  → Git
  → Repository URL: http://gitlab.company.com/group/erp-api.git
  → Credentials: gitlab-cred

Build Configuration:
  → by Jenkinsfile
  → Script Path: Jenkinsfile

Scan Multibranch Pipeline Triggers:
  → 间隔: 1 minute

效果：
  自动发现所有分支
  有 Jenkinsfile 的分支自动创建 Job
  分支删除后 Job 也删除
```
