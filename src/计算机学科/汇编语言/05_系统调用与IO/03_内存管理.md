# 内存管理

## 从程序视角看内存

在汇编层面，你直接面对的是**虚拟地址空间**——操作系统给每个进程营造的"独占内存"幻觉。理解进程的内存布局，是理解变量存储、动态分配、栈溢出等问题的前提。

```mermaid
flowchart TB
    subgraph 进程虚拟地址空间 32位
        direction BT
        K["0xC0000000 内核空间<br/>（用户不可访问）"]
        S2["栈区 ← 向低地址增长"]
        M2["内存映射区 mmap"]
        H["堆区 → 向高地址增长"]
        B["BSS段 未初始化数据"]
        D["数据段 已初始化数据"]
        T["代码段 只读可执行"]
        R["0x08048000 程序起始"]
    end
```

## 进程内存布局详解

### 代码段（.text）

存放机器指令，权限为**可读可执行**（r-x）。尝试写入会触发段错误。

```nasm
section .text
    global _start
_start:
    ; 这里的指令存储在代码段
    ; 地址通常从 0x08048000 开始（32位）
```

### 数据段（.data）

存放已初始化的全局变量和静态变量，权限为**可读可写**（rw-）。

```nasm
section .data
    counter dd 0              ; 已初始化全局变量
    message db 'Hello', 0     ; 已初始化字符串
```

### BSS 段（.bss）

存放未初始化的全局变量，运行时由内核清零。**不占可执行文件空间**。

```nasm
section .bss
    buffer resb 1024          ; 未初始化缓冲区
    array  resd 100           ; 未初始化数组
```

### 堆（Heap）

动态分配的内存区域，通过 `brk`/`sbrk` 或 `mmap` 系统调用扩展：

```mermaid
flowchart LR
    subgraph 堆的增长
        direction LR
        A["brk 当前位置"] --> |"sbrk(+N)"| B["新 brk"]
        B -.-> |"新分配N字节"| C["可用堆空间"]
    end
```

### 栈（Stack）

函数调用、局部变量、返回地址的存储区域，向低地址增长：

```nasm
; 栈的增长方向
push eax    ; ESP -= 4, [ESP] = EAX  （向低地址）
pop eax     ; EAX = [ESP], ESP += 4  （向高地址）
```

::: important 栈与堆的碰撞
如果栈和堆向对方增长到重叠，就会导致内存耗尽。Linux 会在栈溢出时发送 SIGSEGV 信号，杀死进程。可用 `ulimit -s` 查看栈大小限制。
:::

## brk / sbrk - 堆内存分配

`brk` 系统调用改变进程数据段的末尾地址（即"program break"），是最基础的堆内存管理方式。

```nasm
; int brk(void *addr)
; 设置新的 program break
mov eax, 45             ; sys_brk (32位)
mov ebx, new_addr       ; 新的断点地址
int 0x80

; 获取当前 program break
mov eax, 45
xor ebx, ebx            ; 0 = 查询当前值
int 0x80
; EAX = 当前 program break 地址
```

### 实现 malloc 的核心逻辑

```nasm
; simple_malloc.asm - 简易 malloc/free 实现
; 编译: nasm -f elf32 simple_malloc.asm -o simple_malloc.o
; 链接: gcc -m32 simple_malloc.o -o simple_malloc -nostdlib

section .bss
    current_brk resd 1        ; 当前 program break

section .data
    test_ptr dd 0

section .text
    global _start

_start:
    ; 初始化：获取当前 brk
    call init_brk

    ; 分配 1024 字节
    push dword 1024
    call my_malloc
    add esp, 4
    mov [test_ptr], eax       ; 保存分配的指针

    ; 向分配的内存写入数据
    mov ecx, 256
    mov edi, [test_ptr]
    mov al, 0xAA
    cld
    rep stosb                 ; 填充 0xAA

    ; 使用完毕，退出
    mov ebx, 0
    mov eax, 1
    int 0x80

; ============================================
; init_brk - 初始化堆
; ============================================
init_brk:
    push ebp
    mov ebp, esp
    mov eax, 45              ; sys_brk
    xor ebx, ebx             ; 查询当前值
    int 0x80
    mov [current_brk], eax
    mov esp, ebp
    pop ebp
    ret

; ============================================
; my_malloc - 简易内存分配
; 输入: [ebp+8] = 字节数
; 输出: EAX = 分配的内存地址（或 0 失败）
; ============================================
my_malloc:
    push ebp
    mov ebp, esp

    ; 对齐到 4 字节
    mov eax, [ebp+8]
    add eax, 3
    and eax, ~3              ; 向上对齐

    ; 保存大小（简单实现：在分配的内存前4字节存储大小）
    add eax, 4               ; 额外 4 字节存 header

    ; 扩展 brk
    mov ecx, [current_brk]   ; 当前 brk
    add ecx, eax             ; 新 brk
    push ecx
    mov eax, 45              ; sys_brk
    mov ebx, ecx             ; 新 brk 地址
    int 0x80
    pop ecx

    cmp eax, 0
    jl .malloc_fail

    ; 返回分配的内存地址（跳过 header）
    mov eax, [current_brk]
    mov edx, [ebp+8]         ; 原始请求大小
    add edx, 3
    and edx, ~3
    mov [eax], edx            ; 存储大小到 header
    add eax, 4               ; 跳过 header

    ; 更新 current_brk
    mov [current_brk], ecx

    jmp .malloc_done

.malloc_fail:
    xor eax, eax              ; 返回 NULL

.malloc_done:
    mov esp, ebp
    pop ebp
    ret
```

## mmap - 内存映射

`mmap` 比 `brk` 更灵活，可以映射文件到内存、分配匿名内存：

```nasm
; void *mmap(void *addr, size_t length, int prot,
;             int flags, int fd, off_t offset)
mov eax, 90             ; sys_mmap (32位)

; 参数压栈（从右到左）
push dword 0            ; offset = 0
push dword -1           ; fd = -1 (匿名映射)
push dword 0x22         ; flags = MAP_PRIVATE | MAP_ANONYMOUS
push dword 3            ; prot = PROT_READ | PROT_WRITE
push dword 4096         ; length = 4096 字节
push dword 0            ; addr = NULL (内核选择地址)
call mmap
add esp, 24
; EAX = 映射的内存地址
```

### mmap 保护标志

| 标志 | 值 | 含义 |
|------|-----|------|
| PROT_READ | 1 | 可读 |
| PROT_WRITE | 2 | 可写 |
| PROT_EXEC | 4 | 可执行 |

### mmap 映射标志

| 标志 | 值 | 含义 |
|------|-----|------|
| MAP_SHARED | 1 | 共享映射 |
| MAP_PRIVATE | 2 | 私有映射 |
| MAP_ANONYMOUS | 0x20 | 匿名映射（不关联文件） |

```mermaid
flowchart TB
    subgraph mmap 用途
        A["匿名映射 → malloc"]
        B["文件映射 → 文件IO"]
        C["共享内存 → IPC"]
    end
```

## munmap - 释放映射

```nasm
; int munmap(void *addr, size_t length)
mov eax, 91             ; sys_munmap
mov ebx, mapped_addr
mov ecx, mapped_length
int 0x80
```

## 实战：用 mmap 分配可执行内存

```nasm
; exec_mem.asm - 分配可执行内存并运行动态生成的代码
; 编译: nasm -f elf32 exec_mem.asm -o exec_mem.o
; 链接: ld -m elf_i386 exec_mem.o -o exec_mem

section .data
    success_msg db 'Dynamic code executed! Result: ', 0
    success_len equ $ - success_msg
    newline db 0xA

section .bss
    result resd 1

section .text
    global _start

_start:
    ; mmap 分配可读可写可执行的内存
    mov eax, 90           ; sys_mmap
    xor ebx, ebx          ; addr = NULL
    mov ecx, 4096         ; length
    mov edx, 7            ; prot = PROT_READ|PROT_WRITE|PROT_EXEC
    mov esi, 0x22         ; flags = MAP_PRIVATE|MAP_ANONYMOUS
    mov edi, -1           ; fd = -1
    xor ebp, ebp          ; offset = 0
    int 0x80

    ; EAX = 映射的内存地址
    cmp eax, 0
    jl .mmap_fail

    ; 在映射内存中生成代码：mov eax, 42; ret
    mov ecx, eax          ; 保存基地址
    mov byte [eax], 0xB8  ; mov eax, imm32
    mov dword [eax+1], 42
    mov byte [eax+5], 0xC3 ; ret

    ; 执行动态生成的代码
    call ecx
    ; EAX = 42

    mov [result], eax

    ; 释放映射
    mov eax, 91           ; sys_munmap
    mov ebx, ecx          ; 地址
    mov ecx, 4096         ; 大小
    int 0x80

    ; 用退出码返回结果
    mov ebx, [result]
    mov eax, 1
    int 0x80

.mmap_fail:
    mov eax, 1
    mov ebx, 1
    int 0x80
```

::: warning 可执行内存与安全
分配 RWX（可读可写可执行）内存在安全上是危险的——攻击者可以写入恶意代码并执行。现代操作系统通过 NX bit（No-eXecute）和 W^X（Write XOR Execute）策略限制此类操作。上面的例子仅作教学，生产环境应避免 RWX 映射。
:::

## 内存保护：mprotect

可以在运行时修改内存区域的权限：

```nasm
; int mprotect(void *addr, size_t len, int prot)
mov eax, 125            ; sys_mprotect (32位)
mov ebx, page_addr      ; 必须页对齐
mov ecx, page_size      ; 必须是页大小的倍数
mov edx, prot           ; 新权限
int 0x80
```

常见模式：先 RW 写入代码，再改为 RX 执行：

```nasm
; 1. 分配 RW 内存
; mmap(..., PROT_READ|PROT_WRITE, ...)

; 2. 写入代码到该内存
; mov [code_page], ...

; 3. 改为 RX 执行
; mprotect(code_page, 4096, PROT_READ|PROT_EXEC)

; 4. 调用代码
; call code_page
```

## 虚拟内存与物理内存

```mermaid
flowchart LR
    subgraph 虚拟地址
        VA["0x08048000<br/>进程看到"]
    end
    subgraph 页表
        PT["页目录 → 页表 → 页框"]
    end
    subgraph 物理内存
        PA["0x12345000<br/>实际位置"]
    end
    VA --> PT --> PA
```

- 每个进程有独立的页表，虚拟地址映射到不同的物理地址
- 未访问的页不占物理内存（按需分配）
- 访问未映射的页触发**缺页中断**（Page Fault），内核负责分配物理页

::: tip 页大小
x86 默认页大小 4KB，x86-64 支持 2MB/1GB 大页（Huge Pages）。内存相关系统调用的参数必须是页对齐的。
:::

## 小结

```mermaid
mindmap
  root((内存管理))
    内存布局
      代码段 .text r-x
      数据段 .data rw-
      BSS段 .bss rw-
      堆 动态分配
      栈 函数调用
    系统调用
      brk/sbrk 堆扩展
      mmap 内存映射
      munmap 释放映射
      mprotect 修改权限
    关键概念
      虚拟地址 vs 物理地址
      页表与缺页中断
      页对齐 4KB
      NX bit 安全
```

::: tip 面试要点
- 进程内存布局：代码段→数据段→BSS→堆→ mmap→栈→内核空间
- `brk` 扩展堆，`mmap` 分配映射内存——两者是 malloc 的底层实现
- BSS 段不占可执行文件空间，运行时由内核清零
- 栈向低地址增长，堆向高地址增长，碰头则内存耗尽
- mmap 分配匿名内存是最灵活的分配方式，支持自定义权限
- mprotect 可修改内存权限，实现"先写后执行"（W^X）安全策略
- 内存操作必须页对齐（4KB），否则系统调用会失败
:::
