# .NET 规则引擎集成

## 规则引擎选型

| 引擎 | 优点 | 缺点 | 适用场景 |
|------|------|------|---------|
| **Microsoft RulesEngine** | 微软官方，JSON 配置，活跃维护 | 表达式灵活性有限 | 业务规则较固定的场景 |
| **NCalc2** | 表达式语言强大，轻量 | 无内置策略管理 | 需要复杂数学/逻辑表达式 |
| **NRules** | 完整 Rete 算法，高性能 | 学习曲线陡 | 超复杂规则、高频匹配场景 |

## 方案一：Microsoft RulesEngine

```bash
dotnet add package RulesEngine
```

### PDP 实现

```csharp
public class RulesEnginePdp : IPolicyDecisionEngine
{
    private readonly RulesEngine.RulesEngine _engine;

    public RulesEnginePdp(IConfiguration config)
    {
        var workflows = LoadWorkflowsFromConfig(config);
        _engine = new RulesEngine.RulesEngine(workflows, null);
    }

    public async Task<Decision> DecideAsync(AttributeContext context)
    {
        // 构建规则引擎输入（使用强类型 DTO）
        var inputs = new[]
        {
            new RuleParameter("subject",     MapSubject(context.Subject)),
            new RuleParameter("resource",    MapResource(context.Resource)),
            new RuleParameter("action",      MapAction(context.Action)),
            new RuleParameter("environment", MapEnvironment(context.Environment))
        };

        var results = await _engine.ExecuteAllRulesAsync("ContractReadPolicy", inputs);

        // 所有规则通过才放行
        var allPassed = results.All(r => r.IsSuccess);

        if (allPassed)
            return Decision.Allow();

        var failedRule = results.First(r => !r.IsSuccess);
        return Decision.Deny(failedRule.Rule.RuleName,
            failedRule.ActionResult.Output?.ToString());
    }

    private SubjectDto MapSubject(Dictionary<string, object> attrs) => new()
    {
        UserId     = attrs.GetValueOrDefault("userId")?.ToString(),
        Department = attrs.GetValueOrDefault("department")?.ToString(),
        Roles      = attrs.GetValueOrDefault("roles") as List<string> ?? new(),
        Level      = Convert.ToInt32(attrs.GetValueOrDefault("level", 0))
    };
}
```

## 方案二：NCalc2 动态表达式

```bash
dotnet add package NCalc2
```

```csharp
public class NCalcPdp : IPolicyDecisionEngine
{
    public Decision Evaluate(AttributeContext ctx, IEnumerable<AbacPolicy> policies)
    {
        foreach (var policy in policies.OrderByDescending(p => p.Priority))
        {
            if (!EvaluatePolicy(ctx, policy, out var reason))
            {
                if (policy.Effect == EffectType.Deny)
                    return Decision.Deny(policy.PolicyId, reason);
                // Permit 规则不匹配时继续评估下一条
                continue;
            }

            if (policy.Effect == EffectType.Deny)
                return Decision.Deny(policy.PolicyId, "Deny rule matched");
            if (policy.Effect == EffectType.Allow)
                return Decision.Allow(policy.PolicyId);
        }

        return Decision.Deny("default", "No applicable policy found");
    }

    private bool EvaluatePolicy(AttributeContext ctx, AbacPolicy policy, out string failReason)
    {
        failReason = null;

        // 将属性上下文展平为 NCalc 变量
        var expression = new NCalc.Expression(policy.ExpressionText);

        expression.EvaluateParameter += (name, args) =>
        {
            args.Result = name switch
            {
                var n when n.StartsWith("subject_")  => ctx.Subject.GetValueOrDefault(n[8..]),
                var n when n.StartsWith("resource_") => ctx.Resource.GetValueOrDefault(n[9..]),
                var n when n.StartsWith("env_")      => ctx.Environment.GetValueOrDefault(n[4..]),
                _ => null
            };
        };

        // 注册自定义函数
        expression.EvaluateFunction += (name, args) =>
        {
            if (name == "Contains" && args.Parameters.Length == 2)
            {
                var list = args.Parameters[0].Evaluate() as IEnumerable<string>;
                var value = args.Parameters[1].Evaluate()?.ToString();
                args.Result = list?.Contains(value) ?? false;
            }
        };

        try
        {
            var result = expression.Evaluate();
            if (result is bool b) return b;
            failReason = "Expression did not return boolean";
            return false;
        }
        catch (Exception ex)
        {
            failReason = $"Eval error: {ex.Message}";
            return false;
        }
    }
}
```

## 策略存储与热更新

```csharp
public class DatabasePolicyStore : IPolicyStore
{
    private readonly IMemoryCache _cache;
    private readonly AppDbContext _db;

    public async Task<List<AbacPolicy>> GetPoliciesAsync(string resourceType, string action)
    {
        var cacheKey = $"policies:{resourceType}:{action}";

        return await _cache.GetOrCreateAsync(cacheKey, async entry =>
        {
            entry.AbsoluteExpirationRelativeToNow = TimeSpan.FromMinutes(10);

            return await _db.Policies
                .Where(p => p.ResourceType == resourceType
                         && p.Actions.Contains(action)
                         && p.IsActive)
                .OrderByDescending(p => p.Priority)
                .ToListAsync();
        });
    }

    // 策略更新时清除缓存（支持热更新）
    public async Task UpdatePolicyAsync(AbacPolicy policy)
    {
        await _db.SaveChangesAsync();

        // 清除所有受影响的缓存
        _cache.Remove($"policies:{policy.ResourceType}:read");
        _cache.Remove($"policies:{policy.ResourceType}:write");
        _cache.Remove($"policies:{policy.ResourceType}:delete");
    }
}
```
