```rust
// src/error.rs
// AHT20 传感器操作的错误类型。
// 注意：no_std 环境不能使用 thiserror（依赖 std），直接手写 Display。

/// AHT20 传感器操作可能产生的错误。
///
/// # C# 类比
/// 相当于 `enum SensorException`，但错误是值（不是异常），调用方通过 `match` 处理。
#[derive(Debug, Clone, Copy, PartialEq, Eq, defmt::Format)]
#[non_exhaustive]
pub enum SensorError {
    /// I²C 总线通信错误（NACK、总线仲裁失败、超时等）
    I2c,
    /// 传感器响应格式不符合预期（数据字节数或状态标志异常）
    InvalidResponse,
    /// 传感器处于测量忙状态（status byte bit7 = 1），调用方应等待后重试
    Busy,
    /// 传感器未完成出厂校准（status byte bit3 = 0），通常表示硬件损坏
    NotCalibrated,
}
```

## 三、`sensor/aht20.rs`（AHT20 I²C 驱动）

```rust
// src/sensor/aht20.rs
// AHT20 温湿度传感器驱动。
//
// 协议参考：AHT20 Product Manual v1.1
// I²C 地址：0x38（固定，不可配置）
// 测量分辨率：温度 0.01°C，湿度 0.024%RH

use embassy_rp::{
    i2c::{Async, I2c},
    peripherals::I2C0,
};
use embassy_time::Timer;

use crate::error::SensorError;

/// AHT20 I²C 地址（7-bit）
const AHT20_ADDR: u8 = 0x38;

/// 初始化命令：CMD_INIT = 0xBE，参数 0x08 0x00
/// 上电后发送，使传感器进入校准模式
const CMD_INIT: [u8; 3] = [0xBE, 0x08, 0x00];

/// 触发测量命令：CMD_TRIGGER = 0xAC，参数 0x33 0x00
/// 每次采集前发送，传感器在 80ms 内完成测量
const CMD_MEASURE: [u8; 3] = [0xAC, 0x33, 0x00];

/// 测量完成等待时间（毫秒），datasheet 要求 ≥ 75ms，取 80ms 留有余量
const MEASURE_WAIT_MS: u64 = 80;

/// 传感器一次采集的结果。
///
/// 使用整数 × 10 编码（decidegrees / decipercent）避免浮点运算：
/// - `temperature_decidegrees = 253` 表示 25.3°C
/// - `humidity_decipercent = 601` 表示 60.1%RH
///
/// Cortex-M0+ 没有硬件 FPU，浮点运算依赖 soft-float 库（~10KB Flash），
/// 整数编码完全等价且零额外开销。
#[derive(Debug, Clone, Copy, defmt::Format)]
pub struct SensorReading {
    /// 温度（°C × 10），有符号，范围 -500 ~ +850
    pub temperature_decidegrees: i16,
    /// 相对湿度（%RH × 10），无符号，范围 0 ~ 1000
    pub humidity_decipercent: u16,
}

/// AHT20 驱动状态。
///
/// 驱动不持有 I²C 总线（每次操作时借用），允许同一任务内共享 I²C 总线给多个设备。
pub struct Aht20 {
    /// 是否已完成初始化序列
    initialized: bool,
}

impl Aht20 {
    /// 创建新的 AHT20 驱动实例（未初始化状态）。
    #[must_use]
    pub const fn new() -> Self {
        Self { initialized: false }
    }

    /// 执行 AHT20 初始化序列。
    ///
    /// 必须在第一次 `read()` 之前调用。
    ///
    /// # 时序
    /// 1. 等待 40ms（AHT20 上电内部初始化）
    /// 2. 发送初始化命令 `[0xBE, 0x08, 0x00]`
    /// 3. 等待 10ms（校准完成）
    /// 4. 检查 status byte：bit3 = 1 表示已校准
    ///
    /// # Errors
    /// - `SensorError::I2c`：I²C 通信失败
    /// - `SensorError::NotCalibrated`：校准标志位未置位（硬件异常）
    pub async fn init(
        &mut self,
        i2c: &mut I2c<'_, I2C0, Async>,
    ) -> Result<(), SensorError> {
        // AHT20 上电后需要至少 40ms 内部初始化
        Timer::after_millis(40).await;

        // 发送初始化/校准命令
        i2c.write(AHT20_ADDR, &CMD_INIT)
            .await
            .map_err(|_| SensorError::I2c)?;

        // 等待校准完成
        Timer::after_millis(10).await;

        // 读取 1 字节 status，验证校准标志
        let mut status = [0u8; 1];
        i2c.read(AHT20_ADDR, &mut status)
            .await
            .map_err(|_| SensorError::I2c)?;

        // bit3 = Calibrated bit（1 = 已校准）
        if status[0] & 0x08 == 0 {
            defmt::error!(
                "AHT20 init: calibration bit not set, status=0x{:02X}",
                status[0]
            );
            return Err(SensorError::NotCalibrated);
        }

        self.initialized = true;
        defmt::info!("AHT20 init OK, status=0x{:02X}", status[0]);
        Ok(())
    }

    /// 触发一次温湿度采集并返回结果。
    ///
    /// # 时序
    /// 1. 发送测量命令 `[0xAC, 0x33, 0x00]`
    /// 2. 等待 80ms（测量时间）
    /// 3. 读取 6 字节响应
    /// 4. 检查 status byte 的 Busy 位（bit7）
    /// 5. 从 40-bit 原始数据中提取温度/湿度
    ///
    /// # Errors
    /// - `SensorError::I2c`：I²C 通信失败
    /// - `SensorError::Busy`：测量未完成（80ms 后仍 Busy，属于异常）
    /// - `SensorError::InvalidResponse`：响应数据格式异常
    pub async fn read(
        &mut self,
        i2c: &mut I2c<'_, I2C0, Async>,
    ) -> Result<SensorReading, SensorError> {
        if !self.initialized {
            // 未初始化时尝试自动初始化（容错）
            self.init(i2c).await?;
        }

        // 1. 触发测量
        i2c.write(AHT20_ADDR, &CMD_MEASURE)
            .await
            .map_err(|_| SensorError::I2c)?;

        // 2. 等待 80ms（测量时间）
        // embassy async：等待期间 executor 可运行其他 task，CPU 不自旋
        Timer::after_millis(MEASURE_WAIT_MS).await;

        // 3. 读取 6 字节响应
        // 响应格式：[status, hum_hi, hum_mid, hum+temp, temp_mid, temp_lo]
        let mut buf = [0u8; 6];
        i2c.read(AHT20_ADDR, &mut buf)
            .await
            .map_err(|_| SensorError::I2c)?;

        defmt::trace!(
            "AHT20 raw: {:02X} {:02X} {:02X} {:02X} {:02X} {:02X}",
            buf[0], buf[1], buf[2], buf[3], buf[4], buf[5]
        );

        // 4. 检查 Busy 位（bit7 = 1 表示测量仍在进行）
        if buf[0] & 0x80 != 0 {
            defmt::warn!("AHT20 still busy after {}ms wait", MEASURE_WAIT_MS);
            return Err(SensorError::Busy);
        }

        // 5. 解析 AHT20 20-bit 原始数据
        //
        // 字节布局（5 个数据字节，40 bits）：
        //  ┌─── 湿度 (20 bits) ───┐┌─── 温度 (20 bits) ───┐
        //  buf[1][7:0] buf[2][7:0] buf[3][7:4] | buf[3][3:0] buf[4][7:0] buf[5][7:0]
        //
        // C# 等价：
        //   uint rawHum  = ((uint)buf[1] << 12) | ((uint)buf[2] << 4) | ((uint)buf[3] >> 4);
        //   uint rawTemp = (((uint)buf[3] & 0x0F) << 16) | ((uint)buf[4] << 8) | buf[5];

        let raw_hum: u32 = ((buf[1] as u32) << 12)
            | ((buf[2] as u32) << 4)
            | ((buf[3] as u32) >> 4);

        let raw_temp: u32 = (((buf[3] & 0x0F) as u32) << 16)
            | ((buf[4] as u32) << 8)
            | (buf[5] as u32);

        // 6. 换算（整数 × 10，避免浮点）
        //
        // 公式（datasheet）：
        //   RH(%) = raw_hum  / 2^20 × 100
        //   T(°C) = raw_temp / 2^20 × 200 - 50
        //
        // 整数化（× 10 保留 1 位小数）：
        //   humidity_decipercent   = (raw_hum  × 1000) >> 20   [单位: 0.1%RH]
        //   temperature_decidegrees = (raw_temp × 2000) >> 20 - 500 [单位: 0.1°C]
        //
        // 注意：raw_hum × 1000 可能超 u32（raw_hum 最大约 10^6，× 1000 ≈ 10^9，
        //       刚好在 u32 边缘），用 u64 中间运算再截断。

        let humidity_decipercent = ((raw_hum as u64 * 1_000) >> 20) as u16;
        let temperature_decidegrees =
            (((raw_temp as u64 * 2_000) >> 20) as i32 - 500) as i16;

        defmt::debug!(
            "AHT20: raw_hum={} raw_temp={} → hum={}.{}%RH temp={}.{}°C",
            raw_hum,
            raw_temp,
            humidity_decipercent / 10,
            humidity_decipercent % 10,
            temperature_decidegrees / 10,
            temperature_decidegrees % 10,
        );

        Ok(SensorReading {
            temperature_decidegrees,
            humidity_decipercent,
        })
    }
}
```

## 五、`reporter/uart.rs`（UART JSON 上报）

```rust
// src/reporter/uart.rs
// 将传感器读数序列化为 JSON 并通过 UART 发送。
//
// 输出格式（每 10 秒一条）：
//   {"ts":12345,"temp":25.3,"hum":60.1}\r\n
//
// 字段说明：
//   ts:   上电后毫秒数（embassy_time::Instant::now().as_millis()）
//   temp: 温度，°C，1位小数（可为负值，如 -2.3）
//   hum:  湿度，%RH，1位小数，范围 0.0~100.0

use core::fmt::Write as _;

use embassy_rp::{
    peripherals::UART0,
    uart::{Blocking, UartTx},
};
use heapless::String;

use crate::sensor::aht20::SensorReading;

/// 将一条传感器读数格式化为 JSON 并通过 UART TX 发送。
///
/// # 格式
/// `{"ts":{ts},"temp":{t_int}.{t_frac},"hum":{h_int}.{h_frac}}\r\n`
///
/// # 参数
/// - `uart_tx`: UART TX 句柄（blocking 模式，发送完整 FIFO flush 后返回）
/// - `ts_ms`:   上电后毫秒时间戳
/// - `reading`: AHT20 采集结果
pub fn report(
    uart_tx: &mut UartTx<'static, UART0, Blocking>,
    ts_ms: u64,
    reading: &SensorReading,
) {
    // 使用 heapless::String<128> 进行 no_std JSON 格式化
    // 典型输出长度约 40 字节，128 字节容量绰绰有余
    let mut buf: String<128> = String::new();

    // 温度有符号处理：
    //   temperature_decidegrees = 253  →  25.3°C   (正)
    //   temperature_decidegrees = -23  →  -2.3°C   (负)
    let t = reading.temperature_decidegrees;
    let sign = if t < 0 { "-" } else { "" };
    let t_abs = t.unsigned_abs(); // i16::unsigned_abs() → u16，安全的绝对值

    let h = reading.humidity_decipercent;

    // write! 在 no_std 中可用（来自 core::fmt，heapless::String 实现 core::fmt::Write）
    // 使用转义大括号 {{ }} 输出字面量 { }
    let _ = write!(
        buf,
        "{{\"ts\":{ts_ms},\"temp\":{sign}{t_int}.{t_frac},\"hum\":{h_int}.{h_frac}}}\r\n",
        ts_ms  = ts_ms,
        sign   = sign,
        t_int  = t_abs / 10,
        t_frac = t_abs % 10,
        h_int  = h / 10,
        h_frac = h % 10,
    );

    // UART blocking_write：阻塞直到所有字节送入 TX FIFO
    // 在 80ms 测量间隔 + 10s 定时器间隔内，40字节 @ 9600bps ≈ 42ms，不影响时序
    if let Err(_) = uart_tx.blocking_write(buf.as_bytes()) {
        defmt::error!("UART TX failed, dropping report");
    }
}

// ─── 单元测试（宿主机上运行，无需硬件）────────────────────────────────
#[cfg(test)]
mod tests {
    use super::*;

    // 辅助函数：将 SensorReading 格式化为 JSON 字符串（不依赖 UART 硬件）
    fn format_json(ts_ms: u64, reading: &SensorReading) -> String<128> {
        let mut buf: String<128> = String::new();
        let t = reading.temperature_decidegrees;
        let sign = if t < 0 { "-" } else { "" };
        let t_abs = t.unsigned_abs();
        let h = reading.humidity_decipercent;
        let _ = write!(
            buf,
            "{{\"ts\":{ts_ms},\"temp\":{sign}{t_int}.{t_frac},\"hum\":{h_int}.{h_frac}}}\r\n",
            ts_ms  = ts_ms,
            sign   = sign,
            t_int  = t_abs / 10,
            t_frac = t_abs % 10,
            h_int  = h / 10,
            h_frac = h % 10,
        );
        buf
    }

    #[test]
    fn test_format_positive_temp() {
        let reading = SensorReading {
            temperature_decidegrees: 253,  // 25.3°C
            humidity_decipercent: 601,     // 60.1%RH
        };
        let json = format_json(12345, &reading);
        assert_eq!(json.as_str(), "{\"ts\":12345,\"temp\":25.3,\"hum\":60.1}\r\n");
    }

    #[test]
    fn test_format_negative_temp() {
        let reading = SensorReading {
            temperature_decidegrees: -23,  // -2.3°C
            humidity_decipercent: 450,     // 45.0%RH
        };
        let json = format_json(0, &reading);
        assert_eq!(json.as_str(), "{\"ts\":0,\"temp\":-2.3,\"hum\":45.0}\r\n");
    }

    #[test]
    fn test_format_zero_temp() {
        let reading = SensorReading {
            temperature_decidegrees: 0,
            humidity_decipercent: 1000,   // 100.0%RH（饱和）
        };
        let json = format_json(99999, &reading);
        assert_eq!(json.as_str(), "{\"ts\":99999,\"temp\":0.0,\"hum\":100.0}\r\n");
    }

    #[test]
    fn test_format_boundary_temp() {
        // 最低：-50.0°C = -500 decidegrees
        let reading = SensorReading {
            temperature_decidegrees: -500,
            humidity_decipercent: 0,
        };
        let json = format_json(1, &reading);
        assert_eq!(json.as_str(), "{\"ts\":1,\"temp\":-50.0,\"hum\":0.0}\r\n");
    }
}
```

## 六、`main.rs`（Embassy 入口）

```rust
// src/main.rs
// MoNode：Embassy on RP2040 温湿度采集节点
//
// 硬件连接：
//   GPIO0  → UART0 TX（3.3V TTL，接上位机 USB-UART 模块的 RX）
//   GPIO4  → I2C0 SDA（4.7kΩ 上拉至 3.3V）
//   GPIO5  → I2C0 SCL（4.7kΩ 上拉至 3.3V）
//   3V3    → AHT20 VDD
//   GND    → AHT20 GND

// no_std：禁用标准库，使用 'core' + 'embassy' 替代
#![no_std]
// no_main：禁用默认 main 入口，由 cortex-m-rt 提供 reset handler
#![no_main]

use embassy_executor::Spawner;
use embassy_rp::{
    bind_interrupts,
    i2c::{self, Config as I2cConfig, I2c},
    peripherals::{I2C0, UART0},
    uart::{self, Config as UartConfig, UartTx},
};
use embassy_time::{Instant, Timer};
use {defmt_rtt as _, panic_probe as _};  // 链接 RTT 输出 + panic 处理

mod error;
mod reporter;
mod sensor;

use reporter::uart::report;
use sensor::aht20::Aht20;

// bind_interrupts! 宏：将 I2C0_IRQ 中断绑定到 embassy-rp 的 I2C 异步处理器。
// 必须在 crate 根（main.rs）中调用，生成 'static 的 Irqs 类型。
// C# 类比：相当于注册 I2C 控制器的中断服务例程 (ISR)。
bind_interrupts!(struct Irqs {
    I2C0_IRQ => i2c::InterruptHandler<I2C0>;
});

// 主采样间隔：10 秒
const SAMPLE_INTERVAL_MS: u64 = 10_000;

/// Embassy 入口：初始化外设，然后 spawner task。
///
/// `Spawner` 是 Embassy 的任务调度句柄，`spawn()` 类似 C# 的 `Task.Run()`。
/// 本函数返回后 executor 继续运行已注册的 task，main 本身成为 idle。
#[embassy_executor::main]
async fn main(spawner: Spawner) {
    // 初始化 RP2040 外设，消耗（移动）芯片上所有外设句柄
    // 相当于 C# `new IotHub()` —— 之后外设只能通过具体句柄访问
    let p = embassy_rp::init(Default::default());

    // ── UART0 TX 初始化：9600 bps，GPIO0 ─────────────────────────────
    // blocking 模式：`blocking_write` 直接推 FIFO，不需要 DMA / 中断
    // 对 40 字节/10s 的吞吐量完全够用（9600 bps → 960 B/s，发 40B 需 42ms）
    let mut uart_config = UartConfig::default();
    uart_config.baudrate = 9600;
    let uart_tx = UartTx::new(p.UART0, p.PIN_0, uart::NoDma, uart_config);

    // ── I2C0 异步初始化：100kHz，GPIO4 SDA / GPIO5 SCL ───────────────
    // new_async：使用中断驱动模式（Irqs 绑定 I2C0_IRQ），
    // async 等待期间 CPU 不轮询，可被 executor 调度给其他 task
    let mut i2c_config = I2cConfig::default();
    i2c_config.frequency = 100_000; // Standard Mode: 100 kHz
    let i2c = I2c::new_async(p.I2C0, p.PIN_5, p.PIN_4, Irqs, i2c_config);
    //                              ^^^^^^  ^^^^^^
    //                              SCL     SDA（embassy-rp API 顺序：SCL 在前）

    defmt::info!("MoNode starting, sample interval: {}s", SAMPLE_INTERVAL_MS / 1000);

    // spawn sensor_loop task：将外设所有权移入 task
    // unwrap()：spawn 只在 TaskPool 满时失败（只有 1 个任务，不可能满）
    spawner.spawn(sensor_loop(i2c, uart_tx)).unwrap();
}

/// 传感器采集主循环（永远不返回）。
///
/// 此 task 持有 I2C 和 UART TX 的所有权，在 10 秒间隔内循环采集并上报。
/// `-> !`：never 类型，Embassy task 的惯用签名（无限循环）。
///
/// # 编译期约束
/// `#[embassy_executor::task]` 要求无泛型参数，使用 embassy-rp 具体类型。
#[embassy_executor::task]
async fn sensor_loop(
    mut i2c: I2c<'static, I2C0, i2c::Async>,
    mut uart_tx: UartTx<'static, UART0, uart::Blocking>,
) -> ! {
    let mut sensor = Aht20::new();

    // AHT20 初始化（带重试，避免初始化失败后整机 hang）
    loop {
        match sensor.init(&mut i2c).await {
            Ok(()) => {
                defmt::info!("AHT20 initialized successfully");
                break;
            }
            Err(e) => {
                defmt::error!("AHT20 init failed: {:?}, retrying in 1s...", e);
                Timer::after_millis(1_000).await;
            }
        }
    }

    // 主采集循环（永不退出）
    loop {
        let ts_ms = Instant::now().as_millis();

        match sensor.read(&mut i2c).await {
            Ok(reading) => {
                defmt::info!(
                    "AHT20 reading: temp={}.{}°C hum={}.{}%RH",
                    reading.temperature_decidegrees / 10,
                    reading.temperature_decidegrees % 10,
                    reading.humidity_decipercent / 10,
                    reading.humidity_decipercent % 10,
                );
                // 通过 UART 发送 JSON 数据给上位机
                report(&mut uart_tx, ts_ms, &reading);
            }
            Err(crate::error::SensorError::Busy) => {
                // Busy 不是致命错误：等待 20ms 后重试（不等待完整 10s）
                defmt::warn!("AHT20 busy, retrying in 20ms");
                Timer::after_millis(20).await;
                continue; // 不等 10s，立即重试
            }
            Err(e) => {
                // 其他错误（I²C 失败等）：记录日志，等待下一个采集周期
                defmt::error!("AHT20 read error: {:?}", e);
            }
        }

        // 等待到下一个采集周期
        // embassy async：Timer future 挂起，CPU 进入 WFI（Wait For Interrupt）低功耗模式
        Timer::after_millis(SAMPLE_INTERVAL_MS).await;
    }
}
```

## 八、代码注释风格

### defmt 日志级别使用规范

| 级别 | 场景 | 示例 |
|------|------|------|
| `defmt::error!` | 硬件错误、I²C NACK、不可恢复状态 | `"AHT20 init failed: {:?}"` |
| `defmt::warn!` | 可恢复异常（Busy、重试） | `"AHT20 busy, retrying"` |
| `defmt::info!` | 正常状态汇报 | `"MoNode starting"`, `"AHT20 reading: ..."` |
| `defmt::debug!` | 中间计算结果（开发期） | `"raw_hum={} raw_temp={}"` |
| `defmt::trace!` | 最细粒度（I²C 原始字节） | `"AHT20 raw: {:02X} ..."` |

defmt 日志字符串在编译期被替换为 4 字节索引，**不占用 SRAM**，对 Flash 预算的影响极小（约 200 字节/条格式字符串，存在 `.rodata` 段）。

### C# vs Rust 错误处理对比

```csharp
// C# 风格（异常）
try {
    var reading = await sensor.ReadAsync();
    await uart.WriteAsync(Encoding.UTF8.GetBytes(JsonSerializer.Serialize(reading)));
} catch (I2CException ex) {
    logger.LogError(ex, "I2C communication failed");
}
```

```rust
// Rust 风格（Result）
match sensor.read(&mut i2c).await {
    Ok(reading)  => report(&mut uart_tx, ts_ms, &reading),
    Err(SensorError::Busy) => { /* 等待重试 */ }
    Err(e)       => defmt::error!("read error: {:?}", e),
}
// 没有异常传播，错误处理在调用点显式完成
// 编译器强制处理所有 Err 分支（穷举 match）
```

## 九、`no_std` 特化代码说明

### `core::fmt::Write` 在 `no_std` 中

标准库的 `std::fmt::Write` 和 `core::fmt::Write` 是同一个 trait，`use core::fmt::Write as _` 将 `write!` 宏可用的目标类型引入作用域，不会与其他 `Write` trait 冲突。

```rust
use core::fmt::Write as _;  // 只引入 trait 方法，不引入名称（_ 别名）
let mut s: heapless::String<64> = heapless::String::new();
write!(s, "Hello {}", 42).ok();
// s 现在包含 "Hello 42"
```

### `unsigned_abs()` 处理负温度

```rust
// i16::unsigned_abs() 返回 u16，安全地取绝对值（不溢出）
// i16::MIN.abs() 会 overflow panic，但 i16::MIN.unsigned_abs() 返回 32768u16
let t: i16 = -500;       // -50.0°C（AHT20 最低）
let abs: u16 = t.unsigned_abs();  // 500u16
let int = abs / 10;                // 50
let frac = abs % 10;               // 0
// 输出："-50.0"
```
