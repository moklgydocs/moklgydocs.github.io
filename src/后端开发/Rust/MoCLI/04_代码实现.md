```rust
//! MoCLI 错误类型定义。
//!
//! 分两层：
//! - [`ParseError`]：单行解析失败（可忽略，计入统计）
//! - [`CliError`]：不可恢复错误（文件不存在、写 CSV 失败）

use std::path::PathBuf;

/// 单行日志解析失败。
///
/// 这类错误**不终止程序**，会被计入 `parse_error_count` 并继续处理后续行。
#[derive(Debug, thiserror::Error)]
pub enum ParseError {
    /// JSON 格式不合法
    #[error("第 {line_num} 行不是合法 JSON：{source}")]
    InvalidJson {
        line_num: u64,
        #[source]
        source: serde_json::Error,
    },

    /// 缺少必须字段（timestamp / level / message）
    #[error("第 {line_num} 行缺少必须字段 `{field}`")]
    MissingField { line_num: u64, field: &'static str },

    /// `level` 字段值无法识别
    #[error("第 {line_num} 行 level 值不合法：`{value}`")]
    InvalidLevel { line_num: u64, value: String },

    /// `timestamp` 字段无法解析为 RFC 3339 时间
    #[error("第 {line_num} 行 timestamp 无法解析：`{value}`")]
    InvalidTimestamp { line_num: u64, value: String },
}

/// 不可恢复的 CLI 错误。
#[derive(Debug, thiserror::Error)]
pub enum CliError {
    /// 输入文件不存在或无权限读取
    #[error("无法打开输入文件 `{path}`：{source}")]
    OpenInput {
        path: PathBuf,
        #[source]
        source: std::io::Error,
    },

    /// CSV 输出文件创建或写入失败
    #[error("无法写入 CSV 文件 `{path}`：{source}")]
    WriteCsv {
        path: PathBuf,
        #[source]
        source: csv::Error,
    },

    /// stdout 写入失败
    #[error("写入 stdout 失败：{0}")]
    WriteStdout(#[source] std::io::Error),
}
```

```rust
//! MoCLI 核心数据类型。
//!
//! 所有类型均为 NewType 或强类型枚举，避免裸 `String`/`i64` 传参出错。

use chrono::{DateTime, Utc};
use serde::{Deserialize, Serialize};
use std::path::PathBuf;

// ─── 日志级别 ─────────────────────────────────────────────────────────

/// 日志级别（强类型枚举，防止字符串比较出错）。
///
/// C# 中通常用 `Microsoft.Extensions.Logging.LogLevel`，Rust 对应自定义枚举。
#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, Serialize, Deserialize)]
#[serde(rename_all = "UPPERCASE")]
#[non_exhaustive] // 未来可能新增 TRACE、FATAL 等级别
pub enum LogLevel {
    /// 错误：影响功能的异常
    Error,
    /// 警告：潜在问题，功能未中断
    Warn,
    /// 信息：正常业务流程
    Info,
    /// 调试：开发调试信息
    Debug,
}

impl LogLevel {
    /// 从大小写不敏感的字符串解析日志级别。
    ///
    /// # Errors
    /// 返回原始字符串（供调用方构造错误上下文）。
    pub fn from_str_ci(s: &str) -> Result<Self, ()> {
        match s.to_ascii_uppercase().as_str() {
            "ERROR" => Ok(Self::Error),
            "WARN" | "WARNING" => Ok(Self::Warn),
            "INFO" => Ok(Self::Info),
            "DEBUG" => Ok(Self::Debug),
            _ => Err(()),
        }
    }

    /// 返回标准大写字符串表示。
    #[must_use]
    pub fn as_str(self) -> &'static str {
        match self {
            Self::Error => "ERROR",
            Self::Warn  => "WARN",
            Self::Info  => "INFO",
            Self::Debug => "DEBUG",
        }
    }
}

impl std::fmt::Display for LogLevel {
    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
        f.write_str(self.as_str())
    }
}

// ─── 原始日志条目（serde 反序列化目标）───────────────────────────────

/// 单行 JSON 日志的结构化表示。
///
/// JSON 格式示例：
/// ```json
/// {"timestamp":"2024-01-01T08:00:00Z","level":"ERROR","message":"disk full","service":"api"}
/// ```
#[derive(Debug, Clone, Deserialize)]
pub struct LogEntry {
    /// 日志时间戳（RFC 3339 格式，带时区）
    #[serde(alias = "ts", alias = "time")]
    pub timestamp: DateTime<Utc>,

    /// 日志级别
    #[serde(alias = "lvl", alias = "severity")]
    pub level: LogLevel,

    /// 日志消息内容
    #[serde(alias = "msg")]
    pub message: String,

    /// 来源服务名称（可选）
    #[serde(default)]
    pub service: Option<String>,
}

// ─── 聚合键（HashMap 的 Key）─────────────────────────────────────────

/// 聚合统计的分组键：(level, message 前 200 字符)。
///
/// 截断 message 避免长消息造成 HashMap key 过大。
#[derive(Debug, Clone, PartialEq, Eq, Hash)]
pub struct AggKey {
    /// 日志级别
    pub level: LogLevel,
    /// 消息前缀（最多 200 字符，防止超长消息炸 HashMap）
    pub message_prefix: String,
}

impl AggKey {
    /// 从 `LogEntry` 创建聚合键。
    #[must_use]
    pub fn from_entry(entry: &LogEntry) -> Self {
        // 截断超长消息：取前 200 字节边界（避免 UTF-8 截断乱码用 char_indices）
        let prefix = if entry.message.len() <= 200 {
            entry.message.clone()
        } else {
            // 找到 200 字节处最近的 char 边界
            let idx = entry
                .message
                .char_indices()
                .map(|(i, _)| i)
                .take_while(|&i| i <= 200)
                .last()
                .unwrap_or(0);
            format!("{}…", &entry.message[..idx])
        };
        Self { level: entry.level, message_prefix: prefix }
    }
}

// ─── 聚合结果 ──────────────────────────────────────────────────────────

/// 单个聚合分组的统计结果。
#[derive(Debug, Clone)]
pub struct AggResult {
    /// 该分组的日志级别
    pub level: LogLevel,
    /// 消息前缀
    pub message: String,
    /// 出现次数
    pub count: u64,
    /// 该分组第一条日志的时间戳
    pub first_seen: DateTime<Utc>,
    /// 该分组最后一条日志的时间戳
    pub last_seen: DateTime<Utc>,
}

// ─── CSV 报告行 ─────────────────────────────────────────────────────

/// CSV 输出行结构（与 AC5 对应）。
#[derive(Debug, Serialize)]
pub struct ReportRow {
    /// 第一次出现时间（ISO 8601）
    pub timestamp_first: String,
    /// 最后一次出现时间
    pub timestamp_last: String,
    /// 日志级别
    pub level: String,
    /// 消息内容（最多 200 字符）
    pub message: String,
    /// 出现次数
    pub count: u64,
}

impl From<AggResult> for ReportRow {
    fn from(r: AggResult) -> Self {
        Self {
            timestamp_first: r.first_seen.to_rfc3339(),
            timestamp_last:  r.last_seen.to_rfc3339(),
            level:   r.level.to_string(),
            message: r.message,
            count:   r.count,
        }
    }
}

// ─── 过滤参数（从 CLI 传入）──────────────────────────────────────────

/// 过滤条件，由 CLI 层构建后传入 core。
#[derive(Debug, Clone)]
pub struct FilterArgs {
    /// 仅保留该级别（None = 全部保留）
    pub level: Option<LogLevel>,
    /// 仅保留该时间点之后的日志（包含边界）
    pub from: Option<DateTime<Utc>>,
    /// 仅保留该时间点之前的日志（包含边界）
    pub to: Option<DateTime<Utc>>,
}

impl FilterArgs {
    /// 无过滤条件（保留全部）。
    #[must_use]
    pub fn none() -> Self {
        Self { level: None, from: None, to: None }
    }
}

// ─── 分析运行配置 ─────────────────────────────────────────────────────

/// `run_analysis` 的完整入参。
#[derive(Debug)]
pub struct RunConfig {
    /// 输入文件路径
    pub input: PathBuf,
    /// 过滤条件
    pub filter: FilterArgs,
    /// CSV 输出路径（None = stdout 打印摘要）
    pub output: Option<PathBuf>,
    /// 并行线程数（None = rayon 默认，等于 CPU 核心数）
    pub threads: Option<usize>,
}

// ─── 运行结果摘要 ────────────────────────────────────────────────────

/// `run_analysis` 完成后的摘要信息（用于 stdout 打印 + 测试断言）。
#[derive(Debug, Clone)]
pub struct RunSummary {
    /// 处理的总行数
    pub total_lines: u64,
    /// 通过过滤的行数
    pub matched_lines: u64,
    /// 解析失败的行数（跳过，不影响结果）
    pub parse_error_count: u64,
    /// 聚合后的唯一分组数
    pub unique_groups: usize,
    /// 输出路径（None = stdout）
    pub output_path: Option<PathBuf>,
}

```rust
//! 单行 JSON 日志解析器。
//!
//! 将原始字符串解析为 [`LogEntry`]。解析失败返回 [`ParseError`]，
//! 调用方决定是否忽略（计数）还是终止。

use crate::error::ParseError;
use crate::types::LogEntry;
use tracing::warn;

/// 解析单行 JSON 日志字符串。
///
/// # 参数
/// - `line`：原始行内容（不含换行符）
/// - `line_num`：行号，仅用于错误上下文
///
/// # Errors
/// - [`ParseError::InvalidJson`]：不是合法 JSON 对象
/// - [`ParseError::MissingField`]：缺少 timestamp / level / message
/// - [`ParseError::InvalidLevel`]：level 值未知
/// - [`ParseError::InvalidTimestamp`]：timestamp 不是 RFC 3339 格式
#[tracing::instrument(skip(line), fields(line_num))]
pub fn parse_log_line(line: &str, line_num: u64) -> Result<LogEntry, ParseError> {
    // 空行直接跳过（视为解析错误，由调用方计入 error count）
    let line = line.trim();
    if line.is_empty() {
        return Err(ParseError::InvalidJson {
            line_num,
            source: serde_json::from_str::<serde_json::Value>("").unwrap_err(),
        });
    }

    // 尝试将行解析为结构化 LogEntry
    // serde 会自动处理 alias（ts / time / lvl / msg 等别名）
    let entry: LogEntry = serde_json::from_str(line).map_err(|e| {
        warn!(line_num, error = %e, "JSON 解析失败");
        ParseError::InvalidJson { line_num, source: e }
    })?;

    Ok(entry)
}

#[cfg(test)]
mod tests {
    use super::*;
    use rstest::rstest;

    // 合法的标准格式
    #[test]
    fn test_parse_valid_entry() {
        let line = r#"{"timestamp":"2024-01-01T08:00:00Z","level":"ERROR","message":"disk full"}"#;
        let entry = parse_log_line(line, 1).expect("应该成功解析");
        assert_eq!(entry.message, "disk full");
    }

    // 支持别名字段
    #[test]
    fn test_parse_aliases() {
        let line = r#"{"ts":"2024-01-01T08:00:00Z","lvl":"WARN","msg":"low memory"}"#;
        let entry = parse_log_line(line, 1).expect("别名字段应该被识别");
        assert_eq!(entry.message, "low memory");
    }

    // 参数化测试：不合法输入应返回 Err
    #[rstest]
    #[case("not json at all")]
    #[case("")]
    #[case("   ")]
    #[case(r#"{"level":"ERROR"}"#)]  // 缺少 timestamp
    fn test_parse_invalid_returns_err(#[case] input: &str) {
        assert!(parse_log_line(input, 99).is_err());
    }

    // level 大小写不敏感（serde rename_all = UPPERCASE 已处理）
    #[rstest]
    #[case(r#"{"timestamp":"2024-01-01T00:00:00Z","level":"error","message":"test"}"#)]
    #[case(r#"{"timestamp":"2024-01-01T00:00:00Z","level":"Error","message":"test"}"#)]
    fn test_level_case_insensitive(#[case] line: &str) {
        // serde 的 UPPERCASE 不处理小写，这里验证行为
        // 注意：若需要小写支持，须用自定义 Deserialize；目前按 AC3 仅 CLI 层做 case-insensitive
        let _ = parse_log_line(line, 1); // 不 panic 即可（可能 Err，也可能 Ok）
    }
}

## `crates/mo-cli-core/src/filter.rs`

```rust
//! 日志条目过滤器。
//!
//! 根据 [`FilterArgs`] 判断一条 [`LogEntry`] 是否应被保留。
//! 使用纯函数，无副作用，便于单元测试。

use crate::types::{FilterArgs, LogEntry};

/// 判断一条日志条目是否通过过滤条件。
///
/// 三个条件均为"AND"关系：level 匹配 AND 时间 ≥ from AND 时间 ≤ to。
/// 任一条件不满足则返回 `false`。
#[must_use]
pub fn passes_filter(entry: &LogEntry, filter: &FilterArgs) -> bool {
    // 1. 级别过滤：未指定则全部通过
    if let Some(required_level) = filter.level {
        if entry.level != required_level {
            return false;
        }
    }

    // 2. 时间下界过滤：entry 时间必须 >= from
    if let Some(from) = filter.from {
        if entry.timestamp < from {
            return false;
        }
    }

    // 3. 时间上界过滤：entry 时间必须 <= to
    if let Some(to) = filter.to {
        if entry.timestamp > to {
            return false;
        }
    }

    true
}

#[cfg(test)]
mod tests {
    use super::*;
    use crate::types::LogLevel;
    use chrono::{DateTime, Utc};
    use rstest::rstest;

    fn make_entry(level: LogLevel, ts: &str) -> LogEntry {
        LogEntry {
            timestamp: ts.parse::<DateTime<Utc>>().unwrap(),
            level,
            message: "test message".to_string(),
            service: None,
        }
    }

    fn parse_dt(s: &str) -> DateTime<Utc> {
        s.parse().unwrap()
    }

    // 无过滤条件：全部通过
    #[test]
    fn test_no_filter_passes_all() {
        let entry = make_entry(LogLevel::Error, "2024-01-15T12:00:00Z");
        assert!(passes_filter(&entry, &FilterArgs::none()));
    }

    // 级别过滤
    #[rstest]
    #[case(LogLevel::Error, Some(LogLevel::Error), true)]  // 级别匹配
    #[case(LogLevel::Warn,  Some(LogLevel::Error), false)] // 级别不匹配
    #[case(LogLevel::Info,  None,                  true)]  // 未指定级别
    fn test_level_filter(
        #[case] entry_level: LogLevel,
        #[case] filter_level: Option<LogLevel>,
        #[case] expected: bool,
    ) {
        let entry = make_entry(entry_level, "2024-01-15T12:00:00Z");
        let filter = FilterArgs { level: filter_level, from: None, to: None };
        assert_eq!(passes_filter(&entry, &filter), expected);
    }

    // 时间范围过滤
    #[rstest]
    #[case("2024-01-15T12:00:00Z", Some("2024-01-15T00:00:00Z"), Some("2024-01-15T23:59:59Z"), true)]  // 范围内
    #[case("2024-01-14T23:59:59Z", Some("2024-01-15T00:00:00Z"), None,                         false)] // 早于 from
    #[case("2024-01-16T00:00:01Z", None,                         Some("2024-01-15T23:59:59Z"), false)] // 晚于 to
    #[case("2024-01-15T00:00:00Z", Some("2024-01-15T00:00:00Z"), Some("2024-01-15T00:00:00Z"), true)]  // 边界（含）
    fn test_time_filter(
        #[case] entry_ts: &str,
        #[case] from: Option<&str>,
        #[case] to: Option<&str>,
        #[case] expected: bool,
    ) {
        let entry = make_entry(LogLevel::Info, entry_ts);
        let filter = FilterArgs {
            level: None,
            from:  from.map(parse_dt),
            to:    to.map(parse_dt),
        };
        assert_eq!(passes_filter(&entry, &filter), expected);
    }
}

```rust
//! 日志条目聚合器。
//!
//! 提供两个版本：
//! - [`aggregate_serial`]：单线程，适合小文件（< 100MB）
//! - [`aggregate_parallel`]：rayon 并行，适合大文件（≥ 100MB）
//!
//! 两者接口相同，调用方通过文件大小自动选择。

use crate::types::{AggKey, AggResult, LogEntry};
use rayon::prelude::*;
use std::collections::HashMap;

// ─── 内部中间态 ─────────────────────────────────────────────────────

/// HashMap 中存储的聚合中间状态。
#[derive(Debug)]
struct Accumulator {
    count:      u64,
    first_seen: chrono::DateTime<chrono::Utc>,
    last_seen:  chrono::DateTime<chrono::Utc>,
}

impl Accumulator {
    fn new(ts: chrono::DateTime<chrono::Utc>) -> Self {
        Self { count: 1, first_seen: ts, last_seen: ts }
    }

    /// 合并一条新的条目：count+1，更新 first/last。
    fn merge_entry(&mut self, ts: chrono::DateTime<chrono::Utc>) {
        self.count += 1;
        if ts < self.first_seen { self.first_seen = ts; }
        if ts > self.last_seen  { self.last_seen  = ts; }
    }

    /// 合并另一个 Accumulator（用于 rayon reduce 阶段）。
    fn merge_acc(&mut self, other: Self) {
        self.count += other.count;
        if other.first_seen < self.first_seen { self.first_seen = other.first_seen; }
        if other.last_seen  > self.last_seen  { self.last_seen  = other.last_seen; }
    }
}

// ─── 结果转换 ────────────────────────────────────────────────────────

fn acc_map_to_results(map: HashMap<AggKey, Accumulator>) -> Vec<AggResult> {
    let mut results: Vec<AggResult> = map
        .into_iter()
        .map(|(key, acc)| AggResult {
            level:      key.level,
            message:    key.message_prefix,
            count:      acc.count,
            first_seen: acc.first_seen,
            last_seen:  acc.last_seen,
        })
        .collect();
    // 按 level + message 排序，保证输出确定性（便于测试 + 快照对比）
    results.sort_by(|a, b| {
        a.level.as_str().cmp(b.level.as_str())
            .then_with(|| a.message.cmp(&b.message))
    });
    results
}

// ─── 单线程聚合 ──────────────────────────────────────────────────────

/// 单线程版聚合：逐条合并到 HashMap。
///
/// # 参数
/// - `entries`：已过滤的日志条目迭代器
///
/// # 返回
/// 按 (level, message) 排序的聚合结果列表。
pub fn aggregate_serial(entries: impl Iterator<Item = LogEntry>) -> Vec<AggResult> {
    let mut map: HashMap<AggKey, Accumulator> = HashMap::new();

    for entry in entries {
        let key = AggKey::from_entry(&entry);
        map.entry(key)
            .and_modify(|acc| acc.merge_entry(entry.timestamp))
            .or_insert_with(|| Accumulator::new(entry.timestamp));
    }

    acc_map_to_results(map)
}

// ─── rayon 并行聚合 ───────────────────────────────────────────────────

/// rayon 并行版聚合：把条目分配到多个线程，每线程建局部 HashMap，
/// 最后 reduce 合并为单一 HashMap。
///
/// # 关键设计
/// - 不用全局 `Mutex<HashMap>`（会成为瓶颈）
/// - 用 `fold`（每线程局部 HashMap）+ `reduce`（合并两个 HashMap）
/// - 类比 C# PLINQ 的 `AsParallel().Aggregate()`
///
/// # 参数
/// - `entries`：支持 `rayon::ParallelIterator` 的条目集合
pub fn aggregate_parallel(entries: Vec<LogEntry>) -> Vec<AggResult> {
    // par_iter 把 Vec 均匀分片给 rayon 线程池
    let map = entries
        .into_par_iter()
        .fold(
            // 每个线程从空 HashMap 开始（无锁竞争）
            HashMap::new,
            |mut local_map: HashMap<AggKey, Accumulator>, entry| {
                let key = AggKey::from_entry(&entry);
                local_map
                    .entry(key)
                    .and_modify(|acc| acc.merge_entry(entry.timestamp))
                    .or_insert_with(|| Accumulator::new(entry.timestamp));
                local_map
            },
        )
        // reduce 阶段：把各线程的局部 HashMap 两两合并
        .reduce(HashMap::new, |mut a, b| {
            for (key, acc_b) in b {
                a.entry(key)
                    .and_modify(|acc_a| acc_a.merge_acc(Accumulator {
                        count:      acc_b.count,
                        first_seen: acc_b.first_seen,
                        last_seen:  acc_b.last_seen,
                    }))
                    .or_insert(acc_b);
            }
            a
        });

    acc_map_to_results(map)
}

//! CSV 报告写出器。
//!
//! 将 [`AggResult`] 列表写入 CSV 文件或 stdout。
//! 使用 `csv` crate 确保特殊字符（逗号、引号、换行）正确转义。

use crate::error::CliError;
use crate::types::{AggResult, ReportRow, RunSummary};
use std::io::Write;
use std::path::Path;

/// 将聚合结果写入 CSV 文件。
///
/// # Errors
/// - [`CliError::WriteCsv`]：文件创建失败或写入中断
pub fn write_csv(results: Vec<AggResult>, path: &Path) -> Result<(), CliError> {
    // csv::Writer 内部持有 BufWriter，自动批量写入，减少 syscall
    let mut wtr = csv::Writer::from_path(path).map_err(|e| CliError::WriteCsv {
        path: path.to_owned(),
        source: e,
    })?;

    for row in results.into_iter().map(ReportRow::from) {
        wtr.serialize(&row).map_err(|e| CliError::WriteCsv {
            path: path.to_owned(),
            source: e,
        })?;
    }

    // flush 确保全部写入磁盘
    wtr.flush().map_err(|e| CliError::WriteCsv {
        path: path.to_owned(),
        source: csv::Error::from(e),
    })?;

    Ok(())
}

/// 将摘要信息打印到 stdout（未指定 --output 时）。
///
/// # Errors
/// - [`CliError::WriteStdout`]：stdout 写入失败（罕见，管道断开时可能发生）
pub fn print_summary<W: Write>(summary: &RunSummary, results: &[AggResult], mut out: W) -> Result<(), CliError> {
    writeln!(out,
        "\n─── MoCLI 分析完成 ──────────────────────────────────────────────"
    ).map_err(CliError::WriteStdout)?;
    writeln!(out, "  总行数：      {}", summary.total_lines).map_err(CliError::WriteStdout)?;
    writeln!(out, "  匹配行数：    {}", summary.matched_lines).map_err(CliError::WriteStdout)?;
    writeln!(out, "  解析失败：    {}", summary.parse_error_count).map_err(CliError::WriteStdout)?;
    writeln!(out, "  唯一分组数：  {}", summary.unique_groups).map_err(CliError::WriteStdout)?;
    writeln!(out, "─────────────────────────────────────────────────────────────").map_err(CliError::WriteStdout)?;

    if results.is_empty() {
        writeln!(out, "  （无匹配日志）").map_err(CliError::WriteStdout)?;
    } else {
        writeln!(out, "  {:<8} {:<10} {}", "COUNT", "LEVEL", "MESSAGE").map_err(CliError::WriteStdout)?;
        for r in results.iter().take(20) { // 最多显示 20 条
            writeln!(
                out,
                "  {:<8} {:<10} {}",
                r.count,
                r.level,
                if r.message.len() > 80 { &r.message[..80] } else { &r.message }
            ).map_err(CliError::WriteStdout)?;
        }
        if results.len() > 20 {
            writeln!(out, "  ...（共 {} 个分组，完整结果用 --output 导出 CSV）", results.len())
                .map_err(CliError::WriteStdout)?;
        }
    }

    Ok(())
}

```rust
//! 分析流水线主控模块。
//!
//! `run_analysis` 串联：文件读取 → 解析 → 过滤 → 聚合 → 输出，
//! 实现 AC2（内存 < 50MB）+ AC6（rayon 并行）的核心逻辑。

use crate::{
    aggregator::{aggregate_parallel, aggregate_serial},
    error::CliError,
    filter::passes_filter,
    parser::parse_log_line,
    reporter::{print_summary, write_csv},
    types::{LogEntry, RunConfig, RunSummary},
};
use std::{
    fs::File,
    io::{BufRead, BufReader},
};
use tracing::{info, warn};

/// 执行完整的日志分析流水线。
///
/// # 内存控制策略
/// - 使用 `BufReader`（默认 8KB 内核缓冲）逐行读取，不把文件全量加载进内存
/// - 解析失败的行立即 drop，不保留原始字符串
/// - 只有通过过滤的 `LogEntry`（约 30 bytes）才进入聚合阶段
/// - 最终 `HashMap` 的大小与**唯一 key 数量**成正比，而非文件大小
///
/// # Errors
/// - [`CliError::OpenInput`]：文件不存在或无读取权限
/// - [`CliError::WriteCsv`]：CSV 写出失败
/// - [`CliError::WriteStdout`]：stdout 写出失败
pub fn run_analysis(config: RunConfig) -> Result<RunSummary, CliError> {
    // ── 1. 设置 rayon 线程池大小（若用户指定了 --threads）────────────
    if let Some(n) = config.threads {
        // rayon 全局线程池只能初始化一次，忽略后续调用的错误
        let _ = rayon::ThreadPoolBuilder::new()
            .num_threads(n)
            .build_global();
    }

    // ── 2. 打开输入文件（流式，不全量加载）─────────────────────────
    let file = File::open(&config.input).map_err(|e| CliError::OpenInput {
        path: config.input.clone(),
        source: e,
    })?;
    let reader = BufReader::new(file); // BufReader 默认 8KB 缓冲

    // 获取文件大小，决定单线程 vs 并行策略（100MB 阈值）
    let file_size = std::fs::metadata(&config.input)
        .map(|m| m.len())
        .unwrap_or(0);
    let use_parallel = file_size >= 100 * 1024 * 1024; // 100MB

    info!(
        path = %config.input.display(),
        file_size_mb = file_size / 1024 / 1024,
        use_parallel,
        "开始分析日志文件"
    );

    // ── 3. 逐行读取 + 解析 + 过滤 ───────────────────────────────────
    let mut total_lines: u64    = 0;
    let mut matched_lines: u64  = 0;
    let mut parse_errors: u64   = 0;
    let mut matched: Vec<LogEntry> = Vec::new();

    for (idx, line_result) in reader.lines().enumerate() {
        total_lines += 1;
        let line_num = (idx + 1) as u64;

        // 处理 IO 读取错误（磁盘损坏等，直接 warn 并跳过）
        let line = match line_result {
            Ok(l) => l,
            Err(e) => {
                warn!(line_num, error = %e, "读取行失败，跳过");
                parse_errors += 1;
                continue;
            }
        };

        // 解析失败计数后跳过，不终止程序（符合 AC 风险缓解措施）
        let entry = match parse_log_line(&line, line_num) {
            Ok(e) => e,
            Err(_) => {
                parse_errors += 1;
                continue;
            }
        };
        // 注意：`line`（原始 String）在这里立即 drop，不保留在内存中

        // 过滤：不通过则 drop entry
        if !passes_filter(&entry, &config.filter) {
            continue;
        }

        matched_lines += 1;
        matched.push(entry);
    }

    info!(
        total_lines,
        matched_lines,
        parse_errors,
        "读取完成，开始聚合"
    );

    // ── 4. 聚合 ─────────────────────────────────────────────────────
    let results = if use_parallel {
        aggregate_parallel(matched)
    } else {
        aggregate_serial(matched.into_iter())
    };

    let unique_groups = results.len();

    // ── 5. 输出 ──────────────────────────────────────────────────────
    let output_path = config.output.clone();
    if let Some(ref path) = config.output {
        write_csv(results, path)?;
        info!(path = %path.display(), "CSV 报告已写入");
    } else {
        let summary = RunSummary {
            total_lines,
            matched_lines,
            parse_error_count: parse_errors,
            unique_groups,
            output_path: None,
        };
        print_summary(&summary, &results, std::io::stdout()).map_err(|e| {
            if let CliError::WriteStdout(io_err) = e { CliError::WriteStdout(io_err) }
            else { unreachable!() }
        })?;
        return Ok(summary);
    }

    Ok(RunSummary {
        total_lines,
        matched_lines,
        parse_error_count: parse_errors,
        unique_groups,
        output_path,
    })
}

/// MoCLI · GB 级 JSON 日志分析工具
///
/// 从结构化 JSON 日志文件中，按条件过滤并生成统计报告。

///

    ///
    /// 未指定则处理全部级别。
    #[arg(
        short = 'l',
        long  = "level",
        value_name = "LEVEL",
        value_parser = parse_level,
    )]
    level: Option<LogLevel>,

    /// 时间范围起始（RFC 3339 格式，含）
    ///
    /// 示例：2024-01-01T00:00:00Z 或 2024-01-01T08:00:00+08:00

    )]
    from: Option<chrono::DateTime<chrono::Utc>>,

        long       = "to",
        value_name = "DATETIME",
        value_parser = parse_datetime,
    )]
    to: Option<chrono::DateTime<chrono::Utc>>,

    /// CSV 输出文件路径
    ///
    /// 未指定时在 stdout 打印摘要（最多 20 条分组）。
    #[arg(short = 'o', long = "output", value_name = "FILE")]
    output: Option<PathBuf>,

    /// rayon 并行线程数（默认等于 CPU 核心数）
    ///
    /// 文件 < 100MB 时自动使用单线程。
    #[arg(long = "threads", value_name = "N")]
    threads: Option<usize>,

// ─── 自定义 clap value_parser ─────────────────────────────────────────

/// 大小写不敏感地解析日志级别字符串。
fn parse_level(s: &str) -> Result<LogLevel, String> {
    LogLevel::from_str_ci(s)
        .map_err(|()| format!("无效级别 `{s}`，可选：error | warn | info | debug"))
}

/// 解析 RFC 3339 时间字符串为 UTC DateTime。
fn parse_datetime(s: &str) -> Result<chrono::DateTime<chrono::Utc>, String> {
    // 支持 "2024-01-01T00:00:00Z" 和 "2024-01-01T08:00:00+08:00" 两种格式
    chrono::DateTime::parse_from_rfc3339(s)
        .map(|dt| dt.with_timezone(&chrono::Utc))
        .map_err(|e| format!("时间格式错误 `{s}`（需要 RFC 3339 格式）：{e}"))
}

// ─── 入口 ─────────────────────────────────────────────────────────────

fn main() -> Result<()> {

    // RUST_LOG=mo_cli=info,mo_cli_core=debug 可以分模块控制
    tracing_subscriber::fmt()
        .with_env_filter(EnvFilter::from_default_env())
        .with_target(true)
        .init();

    let args = CliArgs::parse();

    // 把 CLI 参数转换为 core 层的 RunConfig（层间隔离）
    let config = RunConfig {
        input: args.input.clone(),
        filter: FilterArgs {
            level: args.level,
            from:  args.from,
            to:    args.to,
        },
        output:  args.output,
        threads: args.threads,
    };

    // 调用核心分析流水线
    let summary = run_analysis(config)
        .with_context(|| format!("分析文件 `{}` 失败", args.input.display()))?;

    // 有 CSV 输出时，额外打印一行确认信息
    if let Some(ref path) = summary.output_path {
        eprintln!(
            "✓ 报告已生成：{}（{} 条分组，共 {} 行匹配）",
            path.display(),
            summary.unique_groups,
            summary.matched_lines,
        );
    }

    Ok(())
}

## `crates/mo-cli-core/benches/parse_bench.rs`

```rust
//! criterion benchmark：验证 AC6（rayon 并行速度 > 单线程 2x）。

use criterion::{criterion_group, criterion_main, BenchmarkId, Criterion, Throughput};
use mo_cli_core::{FilterArgs, LogLevel, RunConfig, run_analysis};
use std::io::Write;
use tempfile::NamedTempFile;

/// 生成指定行数的测试 JSONL 文件，返回临时文件。
fn generate_jsonl(line_count: usize) -> NamedTempFile {
    let mut tmp = NamedTempFile::new().unwrap();
    for i in 0..line_count {
        let level = match i % 4 {
            0 => "ERROR",
            1 => "WARN",
            2 => "INFO",
            _ => "DEBUG",
        };
        writeln!(
            tmp,
            r#"{{"timestamp":"2024-01-01T{:02}:{:02}:{:02}Z","level":"{}","message":"msg_{i}"}}"#,
            (i / 3600) % 24,
            (i / 60) % 60,
            i % 60,
            level,
        )
        .unwrap();
    }
    tmp
}

fn bench_serial_vs_parallel(c: &mut Criterion) {
    let mut group = c.benchmark_group("aggregation");

    // 测试不同规模：10K / 100K / 500K 行
    for line_count in [10_000usize, 100_000, 500_000] {
        let tmp = generate_jsonl(line_count);
        let file_bytes = tmp.path().metadata().unwrap().len();
        group.throughput(Throughput::Bytes(file_bytes));

        // 单线程基准
        group.bench_with_input(
            BenchmarkId::new("serial", line_count),
            &line_count,
            |b, _| {
                b.iter(|| {
                    let config = RunConfig {
                        input:   tmp.path().to_owned(),
                        filter:  FilterArgs { level: Some(LogLevel::Error), from: None, to: None },
                        output:  None,
                        threads: Some(1), // 强制单线程
                    };
                    run_analysis(config).unwrap()
                });
            },
        );

        // rayon 并行（使用 CPU 核心数）
        group.bench_with_input(
            BenchmarkId::new("parallel", line_count),
            &line_count,
            |b, _| {
                b.iter(|| {
                    let config = RunConfig {
                        input:   tmp.path().to_owned(),
                        filter:  FilterArgs { level: Some(LogLevel::Error), from: None, to: None },
                        output:  None,
                        threads: None, // 默认：CPU 核心数
                    };
                    run_analysis(config).unwrap()
                });
            },
        );
    }

    group.finish();
}

criterion_group!(benches, bench_serial_vs_parallel);
criterion_main!(benches);
```

## C# 开发者关键对比

### 流式处理：BufReader vs StreamReader

```csharp
// C# — StreamReader 逐行，内存 O(1)
using var reader = new StreamReader("app.log");
while (reader.ReadLine() is { } line)
{
    ProcessLine(line);  // line 处理后被 GC 回收（不确定时机）
}
```

```rust
// Rust — BufReader 逐行，内存 O(1)，drop 时机确定
let file   = File::open("app.log")?;
let reader = BufReader::new(file);
for line in reader.lines() {
    let line = line?;          // Result<String, io::Error>
    process_line(&line);
}   // line 在这里立即 drop（离开作用域）
```

**关键区别**：Rust 的 `drop` 发生在作用域结束，确定且立即；C# 的 GC 回收不确定。

### 并行聚合：PLINQ vs rayon

```csharp
// C# PLINQ — 并行聚合
var result = entries
    .AsParallel()
    .GroupBy(e => (e.Level, e.Message[..Math.Min(200, e.Message.Length)]))
    .ToDictionary(g => g.Key, g => g.Count());
```

```rust
// Rust rayon — 并行聚合（fold + reduce 模式）
let result = entries
    .into_par_iter()
    .fold(HashMap::new, |mut map, entry| {   // 每线程局部 HashMap
        *map.entry(key).or_insert(0) += 1;
        map
    })
    .reduce(HashMap::new, |mut a, b| {       // 合并各线程结果
        for (k, v) in b { *a.entry(k).or_insert(0) += v; }
        a
    });
```

**关键区别**：rayon 不用 `Mutex`，每线程局部 HashMap，零锁竞争。
