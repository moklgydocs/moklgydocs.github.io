# 数据类型与表设计

::: tip 核心观点
PostgreSQL 拥有所有关系型数据库中最丰富的类型系统——从基础类型到数组、范围、JSONB、几何类型，几乎不需要任何 ORM 映射就能直接建模业务数据。善用 PG 类型，能大幅减少应用层代码。
:::

## 数值类型

### 整数与自增

```sql
-- 整数类型选择
CREATE TABLE products (
    id          integer GENERATED ALWAYS AS IDENTITY PRIMARY KEY,  -- PG 10+ 推荐
    category_id smallint NOT NULL,   -- 小范围分类，-32768 ~ 32767
    view_count  integer DEFAULT 0,   -- 通用整数
    ean_code    bigint               -- 条码等大数字
);

-- 插入时自增列自动赋值
INSERT INTO products (category_id, ean_code) VALUES (1, 6901234567890);
-- id 自动生成
```

### SERIAL vs IDENTITY

```sql
-- ❌ 旧方式：SERIAL（PG 10 前唯一选择）
CREATE TABLE old_style (
    id SERIAL PRIMARY KEY,       -- 内部是 integer + sequence
    name text
);
-- 问题：可以手动插入任意 id，容易产生序列冲突
INSERT INTO old_style (id, name) VALUES (1, 'hack');  -- 允许！

-- ✅ 新方式：IDENTITY（PG 10+ 推荐）
CREATE TABLE new_style (
    id integer GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    name text
);
-- ALWAYS 模式禁止手动指定 id
INSERT INTO new_style (id, name) VALUES (1, 'hack');
-- ERROR: cannot insert a non-DEFAULT value into identity column

-- 但可以 OVERRIDING SYSTEM VALUE 应急
INSERT INTO new_style (id, name)
OVERRIDING SYSTEM VALUE VALUES (1, 'migration_data');
```

::: important 始终使用 IDENTITY 替代 SERIAL
`GENERATED ALWAYS AS IDENTITY` 是 SQL 标准语法，比 SERIAL 更安全：
- **ALWAYS**：默认禁止手动插入 id，防止序列冲突
- **BY DEFAULT**：允许手动指定，但优先使用序列
- 迁移数据时用 `OVERRIDING SYSTEM VALUE` 显式声明
:::

### 精确小数与货币

```sql
-- numeric(precision, scale) - 精确计算，适合金融
CREATE TABLE accounts (
    id        integer GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    balance   numeric(15, 2) NOT NULL DEFAULT 0,  -- 15位总精度，2位小数
    currency  char(3) DEFAULT 'CNY'
);

-- 精确计算不会丢失精度
SELECT 0.1::numeric + 0.2::numeric;   -- 0.3 ✓
SELECT 0.1::float8 + 0.2::float8;     -- 0.30000000000000004 ✗

-- money 类型（不推荐，绑定 locale）
SELECT '¥1,234.56'::money;
```

## 字符与文本

```sql
-- PG 中 text 和 varchar 几乎没有性能差异
CREATE TABLE articles (
    id       integer GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    title    varchar(200) NOT NULL,    -- 有长度限制时用 varchar
    content  text,                      -- 无限制用 text，性能一样
    slug     char(10)                   -- 定长，仅用于固定格式编码
);

-- text 的操作函数
SELECT length('你好世界');                    -- 4（字符数，不是字节数）
SELECT left('Hello World', 5);               -- Hello
SELECT substring('2025-06-04' from 1 for 4); -- 2025
```

::: warning 别用 char(n)
`char(n)` 用空格填充到固定长度，几乎无适用场景。`varchar(n)` 和 `text` 是更好的选择。参考：[PostgreSQL 官方文档 - Character Types](https://www.postgresql.org/docs/16/datatype-character.html)
:::

## 日期时间

```sql
CREATE TABLE events (
    id          integer GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    event_date  date NOT NULL,                          -- 仅日期
    start_time  time NOT NULL,                          -- 仅时间
    created_at  timestamp with time zone DEFAULT now(), -- ✅ 推荐：带时区
    duration    interval                                -- 时间间隔
);

-- timestamptz vs timestamp 的关键区别
SELECT '2025-06-04 10:00:00'::timestamp;        -- 不带时区，存储原值
SELECT '2025-06-04 10:00:00+08'::timestamptz;   -- 转为 UTC 存储，显示时转换
```

```sql
-- interval 运算
SELECT now() + interval '1 day';
SELECT now() - interval '2 hours 30 minutes';
SELECT age(timestamp '2024-01-01', timestamp '2023-06-15');
-- 6 mons 17 days

-- 日期截断
SELECT date_trunc('month', now());     -- 当月1日 00:00:00
SELECT date_trunc('hour', now());      -- 当前小时 00:00
```

::: important 始终使用 timestamptz
`timestamp with time zone` (简写 `timestamptz`) 是唯一正确的选择：
- 存储时自动转为 UTC
- 显示时根据客户端时区转换
- 避免夏令时和时区混乱
- 参考文档：[Date/Time Types](https://www.postgresql.org/docs/16/datatype-datetime.html)
:::

## UUID

```sql
-- 需要先启用扩展
CREATE EXTENSION IF NOT EXISTS "uuid-ossp";

CREATE TABLE api_keys (
    id    uuid DEFAULT uuid_generate_v4() PRIMARY KEY,  -- 随机 UUID
    name  varchar(100) NOT NULL,
    key   uuid DEFAULT uuid_generate_v4() UNIQUE
);

-- 或者 PG 13+ 的 gen_random_uuid()
CREATE TABLE sessions (
    id  uuid DEFAULT gen_random_uuid() PRIMARY KEY,  -- 无需扩展
    user_id integer NOT NULL REFERENCES users(id)
);
```

## JSONB

```sql
CREATE TABLE products (
    id      integer GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    name    text NOT NULL,
    attrs   jsonb NOT NULL DEFAULT '{}'  -- ✅ 用 jsonb，不用 json
);

-- 插入 JSONB 数据
INSERT INTO products (name, attrs) VALUES
    ('MacBook Pro', '{"cpu": "M3 Pro", "ram": 18, "colors": ["银色", "深空黑"], "specs": {"weight": 1.55, "screen": 14}}'),
    ('iPhone 15', '{"cpu": "A17 Pro", "ram": 8, "colors": ["黑色", "蓝色"], "specs": {"weight": 0.171, "screen": 6.1}}');

-- 查询 JSONB 字段
SELECT name, attrs->>'cpu' AS cpu          -- ->> 返回 text
FROM products
WHERE attrs->>'cpu' = 'M3 Pro';

-- 嵌套查询
SELECT name, attrs#>>'{specs,screen}' AS screen  -- #>> 路径取 text
FROM products;

-- 包含查询（最常用，可以用 GIN 索引加速）
SELECT name FROM products WHERE attrs @> '{"cpu": "M3 Pro"}';
```

::: tip JSONB 详解
JSONB 的完整操作符、索引策略和性能优化，请参考 [JSONB 全攻略](../02_进阶篇/03_JSONB全攻略.md)。
:::

## 数组类型

```sql
-- 数组类型：PG 原生支持，无需额外表
CREATE TABLE blog_posts (
    id       integer GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    title    text NOT NULL,
    tags     text[] NOT NULL DEFAULT '{}',          -- 文本数组
    ratings  integer[] DEFAULT '{}',                  -- 整数数组
    metadata jsonb DEFAULT '{}'
);

-- 插入数组
INSERT INTO blog_posts (title, tags, ratings) VALUES
    ('PostgreSQL 入门', ARRAY['数据库', 'PG', '入门'], ARRAY[5, 4, 5]),
    ('MySQL 对比', ARRAY['数据库', 'MySQL', '对比'], ARRAY[4, 3]);

-- 查询：包含某个元素
SELECT title FROM blog_posts WHERE 'PG' = ANY(tags);

-- 查询：包含所有指定元素
SELECT title FROM blog_posts WHERE tags @> ARRAY['数据库', 'PG'];

-- 查询：有交集
SELECT title FROM blog_posts WHERE tags && ARRAY['MySQL', 'PG'];

-- 数组函数
SELECT title, array_length(tags, 1) AS tag_count,
       array_to_string(tags, ', ') AS tag_string
FROM blog_posts;
```

```sql
-- GIN 索引加速数组查询
CREATE INDEX idx_blog_posts_tags ON blog_posts USING GIN (tags);

-- 验证索引使用
EXPLAIN SELECT title FROM blog_posts WHERE tags @> ARRAY['PG'];
-- Index Scan using idx_blog_posts_tags on blog_posts
```

::: tip 数组 vs 关联表
- **数组**：适合标签、少量固定属性，查询简单，无需 JOIN
- **关联表**：适合需要独立管理、统计、多对多关系的实体
- 经验法则：如果元素不需要单独查询/更新，用数组
:::

## 范围类型

```sql
-- 内置范围类型
-- int4range / int8range: 整数范围
-- numrange: 数值范围
-- daterange: 日期范围
-- tstzrange: 时间戳范围（最常用）

CREATE TABLE room_bookings (
    id        integer GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    room_id   integer NOT NULL,
    guest_id  integer NOT NULL,
    stay      tstzrange NOT NULL,  -- 入住时间段
    EXCLUDE USING gist (room_id WITH =, stay WITH &&)  -- 排除约束：同房间不重叠
);

-- 插入预订
INSERT INTO room_bookings (room_id, guest_id, stay) VALUES
    (101, 1, '[2025-06-01 14:00, 2025-06-03 12:00)'),   -- [ ] 包含, ( ) 排除
    (101, 2, '[2025-06-03 14:00, 2025-06-05 12:00)');   -- ✅ 不重叠

-- 重叠预订会被阻止
INSERT INTO room_bookings (room_id, guest_id, stay) VALUES
    (101, 3, '[2025-06-02 14:00, 2025-06-04 12:00)');
-- ERROR: conflicting key value violates exclusion constraint

-- 查询：某时间段的预订
SELECT * FROM room_bookings
WHERE stay && '[2025-06-02, 2025-06-04)'::tstzrange;
```

## 表设计实践

### Schema 组织

```sql
-- 使用 schema 组织表，而非全部放在 public
CREATE SCHEMA auth;     -- 认证相关
CREATE SCHEMA app;      -- 业务数据
CREATE SCHEMA audit;    -- 审计日志

-- Supabase 的 auth.users 表就是这种模式
CREATE TABLE auth.users (
    id              uuid DEFAULT gen_random_uuid() PRIMARY KEY,
    email           varchar(255) UNIQUE NOT NULL,
    encrypted_pass  text NOT NULL,
    email_confirmed timestamptz,
    created_at      timestamptz DEFAULT now()
);

CREATE TABLE app.profiles (
    id       uuid PRIMARY KEY REFERENCES auth.users(id) ON DELETE CASCADE,
    username varchar(50) UNIQUE NOT NULL,
    full_name text,
    avatar_url text,
    updated_at timestamptz DEFAULT now()
);

-- 搜索路径设置
SET search_path TO app, auth, public;
```

### 表继承

```sql
-- PG 独有的表继承
CREATE TABLE vehicles (
    id          integer GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    brand       text NOT NULL,
    model       text NOT NULL,
    year        integer NOT NULL
);

CREATE TABLE cars (
    fuel_type   text DEFAULT '汽油',
    doors       integer DEFAULT 4
) INHERITS (vehicles);

CREATE TABLE motorcycles (
    engine_cc   integer,
    has_sidecar boolean DEFAULT false
) INHERITS (vehicles);

-- 查询所有车辆（包含子表）
SELECT * FROM vehicles;           -- 包含 cars 和 motorcycles

-- 仅查询基表
SELECT * FROM ONLY vehicles;       -- 不包含子表
```

::: warning 表继承的限制
- 外键约束不会继承到子表
- 唯一约束不会继承
- 实际项目中，**分区表**更常用，继承用得越来越少
- 参考文档：[Table Inheritance](https://www.postgresql.org/docs/16/ddl-inherit.html)
:::

### 声明式分区表

```sql
-- 按范围分区（最常用：按时间）
CREATE TABLE orders (
    id          integer GENERATED ALWAYS AS IDENTITY,
    user_id     integer NOT NULL,
    amount      numeric(10, 2) NOT NULL,
    status      text DEFAULT 'pending',
    created_at  timestamptz DEFAULT now(),
    PRIMARY KEY (id, created_at)  -- 分区键必须包含在主键中
) PARTITION BY RANGE (created_at);

-- 创建分区
CREATE TABLE orders_2025_q1 PARTITION OF orders
    FOR VALUES FROM ('2025-01-01') TO ('2025-04-01');

CREATE TABLE orders_2025_q2 PARTITION OF orders
    FOR VALUES FROM ('2025-04-01') TO ('2025-07-01');

CREATE TABLE orders_2025_q3 PARTITION OF orders
    FOR VALUES FROM ('2025-07-01') TO ('2025-10-01');

CREATE TABLE orders_2025_q4 PARTITION OF orders
    FOR VALUES FROM ('2025-10-01') TO ('2026-01-01');

-- 默认分区（兜底）
CREATE TABLE orders_default PARTITION OF orders DEFAULT;

-- 插入数据自动路由到对应分区
INSERT INTO orders (user_id, amount, created_at) VALUES
    (1, 299.00, '2025-02-15'),  -- → orders_2025_q1
    (2, 599.00, '2025-05-20');  -- → orders_2025_q2
```

```sql
-- 按列表分区（如按地区）
CREATE TABLE users (
    id      integer GENERATED ALWAYS AS IDENTITY,
    name    text NOT NULL,
    region  text NOT NULL,
    PRIMARY KEY (id, region)
) PARTITION BY LIST (region);

CREATE TABLE users_cn PARTITION OF users FOR VALUES IN ('CN', 'HK', 'TW');
CREATE TABLE users_us PARTITION OF users FOR VALUES IN ('US', 'CA');
CREATE TABLE users_eu PARTITION OF users FOR VALUES IN ('UK', 'DE', 'FR');
```

```sql
-- 按哈希分区（均匀分布）
CREATE TABLE logs (
    id      bigint GENERATED ALWAYS AS IDENTITY,
    content text,
    PRIMARY KEY (id)
) PARTITION BY HASH (id);

CREATE TABLE logs_p0 PARTITION OF logs FOR VALUES WITH (MODULUS 4, REMAINDER 0);
CREATE TABLE logs_p1 PARTITION OF logs FOR VALUES WITH (MODULUS 4, REMAINDER 1);
CREATE TABLE logs_p2 PARTITION OF logs FOR VALUES WITH (MODULUS 4, REMAINDER 2);
CREATE TABLE logs_p3 PARTITION OF logs FOR VALUES WITH (MODULUS 4, REMAINDER 3);
```

### 列约束与 CHECK 约束

```sql
CREATE TABLE employees (
    id          integer GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    name        text NOT NULL CHECK (length(name) >= 2),
    email       text NOT NULL UNIQUE CHECK (email ~ '^[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Za-z]{2,}$'),
    age         integer CHECK (age >= 18 AND age <= 120),
    salary      numeric(10, 2) NOT NULL CHECK (salary > 0),
    hire_date   date NOT NULL DEFAULT CURRENT_DATE,
    status      text NOT NULL DEFAULT 'active'
                CHECK (status IN ('active', 'inactive', 'on_leave')),
    manager_id  integer REFERENCES employees(id) ON DELETE SET NULL,

    -- 表级约束
    CONSTRAINT valid_hire_date CHECK (hire_date <= CURRENT_DATE),
    CONSTRAINT salary_not_zero CHECK (salary > 0)
);

-- 命名约束便于后续管理
ALTER TABLE employees
    DROP CONSTRAINT employees_email_check;
```

## 实战：Supabase 的 auth.users 表设计

```sql
-- Supabase 实际使用的 auth.users 表结构（简化版）
CREATE TABLE auth.users (
    instance_id       uuid,
    id                uuid DEFAULT gen_random_uuid() PRIMARY KEY,
    aud               varchar(255),
    role              varchar(255),
    email             varchar(255) UNIQUE,
    encrypted_password varchar(255),
    email_confirmed_at timestamptz,
    phone             varchar(15) UNIQUE,
    phone_confirmed_at timestamptz,
    confirmation_token varchar(255),
    recovery_token     varchar(255),
    email_change       varchar(255),
    email_change_token varchar(255),
    created_at         timestamptz DEFAULT now(),
    updated_at         timestamptz DEFAULT now(),
    raw_app_meta_data  jsonb DEFAULT '{}',    -- ✅ JSONB 存储应用元数据
    raw_user_meta_data jsonb DEFAULT '{}',     -- ✅ JSONB 存储用户元数据
    is_super_admin     boolean DEFAULT false,
    banned_until       timestamptz,
    CONSTRAINT users_email_check CHECK (email ~ '^[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Za-z]{2,}$')
);

-- 注意 Supabase 的设计决策：
-- 1. 用 uuid 作为主键（分布式友好）
-- 2. 用 jsonb 存储可变元数据（避免频繁 ALTER TABLE）
-- 3. 用 timestamptz 存储时间（时区正确）
-- 4. CHECK 约束验证邮箱格式（数据库层校验）
```

## 面试技巧

::: tip 面试高频问题
1. **SERIAL 和 IDENTITY 有什么区别？**
   - SERIAL 是旧语法，本质是 integer + sequence + default，仍可手动插入值
   - IDENTITY 是 SQL 标准，ALWAYS 模式禁止手动插入，更安全
   - 新项目应始终使用 `GENERATED ALWAYS AS IDENTITY`

2. **什么时候用数组，什么时候用关联表？**
   - 数组：标签、少量属性、不需要单独管理
   - 关联表：需要独立 CRUD、统计、多对多关系

3. **json 和 jsonb 怎么选？**
   - 始终选 jsonb：二进制存储、支持索引、查询更快
   - json 仅在需要保留原始格式/顺序时使用（极少场景）

4. **分区表的分区键选择原则？**
   - 选择查询最常用的过滤条件作为分区键
   - 时间范围最常见（按月/季度分区）
   - 分区键必须包含在主键中
   - 数据量 < 100 万不需要分区
:::
