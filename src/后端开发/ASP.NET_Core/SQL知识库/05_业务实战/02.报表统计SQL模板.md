# 报表统计 SQL 模板

报表 SQL 基本就那几个套路，遇到类似的需求直接改改就能用。

## 同比环比

```sql
;WITH Monthly AS (
    SELECT YEAR(OrderDate) AS Y, MONTH(OrderDate) AS M,
        SUM(Amount) AS Total
    FROM Orders WHERE Status = 1
    GROUP BY YEAR(OrderDate), MONTH(OrderDate)
)
SELECT cur.Y, cur.M, cur.Total,
    -- 环比（和上月比）
    prev.Total AS LastMonth,
    CASE WHEN prev.Total > 0
        THEN (cur.Total - prev.Total) * 100.0 / prev.Total END AS MoM,
    -- 同比（和去年同月比）
    yoy.Total AS LastYear,
    CASE WHEN yoy.Total > 0
        THEN (cur.Total - yoy.Total) * 100.0 / yoy.Total END AS YoY
FROM Monthly cur
LEFT JOIN Monthly prev ON cur.Y = prev.Y AND cur.M = prev.M + 1
LEFT JOIN Monthly yoy ON cur.Y = yoy.Y + 1 AND cur.M = yoy.M
ORDER BY cur.Y, cur.M;
```

用 LAG 更简洁（但同比不太方便）：

```sql
SELECT Y, M, Total,
    LAG(Total) OVER (ORDER BY Y, M) AS LastMonth,
    (Total - LAG(Total) OVER (ORDER BY Y, M)) * 100.0
        / NULLIF(LAG(Total) OVER (ORDER BY Y, M), 0) AS MoM
FROM Monthly;
```

## 累计汇总

```sql
-- 每月销售额 + 年度累计
SELECT Y, M, Total,
    SUM(Total) OVER (PARTITION BY Y ORDER BY M) AS YTD
FROM Monthly;
```

## 占比分析

```sql
-- 每个产品类别占总销售额的百分比
SELECT Category,
    SUM(Amount) AS CategoryTotal,
    SUM(Amount) * 100.0 / SUM(SUM(Amount)) OVER () AS Pct
FROM Orders o
JOIN Products p ON o.ProductId = p.ProductId
GROUP BY Category
ORDER BY Pct DESC;
```

## 交叉表（月度 × 部门）

```sql
SELECT DeptName,
    SUM(CASE WHEN M = 1 THEN Total END) AS [1月],
    SUM(CASE WHEN M = 2 THEN Total END) AS [2月],
    SUM(CASE WHEN M = 3 THEN Total END) AS [3月],
    -- ... 到 12 月
    SUM(Total) AS [合计]
FROM (
    SELECT d.DeptName, MONTH(o.OrderDate) AS M, SUM(o.Amount) AS Total
    FROM Orders o JOIN Departments d ON o.DeptId = d.DeptId
    WHERE YEAR(o.OrderDate) = 2024
    GROUP BY d.DeptName, MONTH(o.OrderDate)
) t
GROUP BY DeptName;
```

## 排名分析

```sql
-- 销售额 Top 10 客户 + 排名 + 占比
SELECT CustomerName, Total,
    RANK() OVER (ORDER BY Total DESC) AS Rank,
    Total * 100.0 / SUM(Total) OVER () AS Pct,
    SUM(Total) OVER (ORDER BY Total DESC) * 100.0 / SUM(Total) OVER () AS CumPct
FROM (
    SELECT c.CustomerName, SUM(o.Amount) AS Total
    FROM Orders o JOIN Customers c ON o.CustomerId = c.CustomerId
    GROUP BY c.CustomerName
) t;
-- CumPct = 累计占比，可以做 ABC 分析：Top 20% 的客户贡献了多少销售额
```

## 无数据补零

报表里某些月份没有数据，直接 GROUP BY 会缺行。用日历表补：

```sql
;WITH Months AS (
    SELECT 1 AS M UNION ALL SELECT M + 1 FROM Months WHERE M < 12
)
SELECT m.M, ISNULL(t.Total, 0) AS Total
FROM Months m
LEFT JOIN (
    SELECT MONTH(OrderDate) AS M, SUM(Amount) AS Total
    FROM Orders WHERE YEAR(OrderDate) = 2024
    GROUP BY MONTH(OrderDate)
) t ON m.M = t.M;
```

## 漏斗分析

```sql
-- 电商漏斗：浏览 → 加购 → 下单 → 支付
SELECT '浏览' AS Step, COUNT(DISTINCT UserId) AS Users FROM PageViews WHERE Page = 'product'
UNION ALL
SELECT '加购', COUNT(DISTINCT UserId) FROM CartItems
UNION ALL
SELECT '下单', COUNT(DISTINCT UserId) FROM Orders
UNION ALL
SELECT '支付', COUNT(DISTINCT UserId) FROM Orders WHERE IsPaid = 1;
```
