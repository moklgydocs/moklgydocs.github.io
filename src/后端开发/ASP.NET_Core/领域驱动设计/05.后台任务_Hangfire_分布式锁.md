## Hangfire 实现

```csharp
// ============================================
// Hangfire/HangfireJobExecutor.cs
// 核心执行器：统一异常处理 + 日志 + 计时
// ============================================

using Microsoft.Extensions.DependencyInjection;
using Microsoft.Extensions.Logging;
using Platform.BackgroundJobs.Abstractions;
using System.Diagnostics;

namespace Platform.BackgroundJobs.Hangfire;

/// <summary>
/// Hangfire 任务执行器
/// 
/// Hangfire 调度 → 调用此类 → 从 DI 获取 Job → 执行
/// 
/// 为什么不让 Hangfire 直接调 IBackgroundJob.ExecuteAsync:
///   1. 统一异常处理和日志
///   2. 统一超时控制
///   3. 统一性能监控
///   4. 方便加 Tracing
/// </summary>
public class HangfireJobExecutor
{
    private readonly IServiceScopeFactory _scopeFactory;
    private readonly ILogger<HangfireJobExecutor> _logger;

    public HangfireJobExecutor(
        IServiceScopeFactory scopeFactory,
        ILogger<HangfireJobExecutor> logger)
    {
        _scopeFactory = scopeFactory;
        _logger = logger;
    }

    /// <summary>
    /// 执行任务（Hangfire 入口）
    /// </summary>
    public async Task ExecuteAsync<TJob, TArgs>(
        TArgs args,
        CancellationToken cancellationToken = default)
        where TJob : class, IBackgroundJob<TArgs>
        where TArgs : class
    {
        var jobName = typeof(TJob).Name;
        var argsName = typeof(TArgs).Name;

        _logger.LogInformation(
            "[BackgroundJob] Start: {JobName} ({ArgsName})",
            jobName, argsName);

        var sw = Stopwatch.StartNew();

        try
        {
            // 每个 Job 执行在独立的 DI Scope
            using var scope = _scopeFactory.CreateScope();
            var job = scope.ServiceProvider.GetRequiredService<TJob>();

            await job.ExecuteAsync(args, cancellationToken);

            sw.Stop();
            _logger.LogInformation(
                "[BackgroundJob] Completed: {JobName} in {Elapsed}ms",
                jobName, sw.ElapsedMilliseconds);
        }
        catch (OperationCanceledException)
        {
            _logger.LogWarning(
                "[BackgroundJob] Cancelled: {JobName} after {Elapsed}ms",
                jobName, sw.ElapsedMilliseconds);
            throw; // Hangfire 会标记为 Deleted
        }
        catch (Exception ex)
        {
            sw.Stop();
            _logger.LogError(ex,
                "[BackgroundJob] Failed: {JobName} after {Elapsed}ms - {Error}",
                jobName, sw.ElapsedMilliseconds, ex.Message);
            throw; // Hangfire 会自动重试
        }
    }
}
```

```csharp
// ============================================
// Hangfire/HangfireJobManager.cs
// ============================================

using Hangfire;
using Platform.BackgroundJobs.Abstractions;

namespace Platform.BackgroundJobs.Hangfire;

public class HangfireJobManager : IBackgroundJobManager
{
    private readonly IBackgroundJobClient _client;

    public HangfireJobManager(IBackgroundJobClient client)
    {
        _client = client;
    }

    public Task<string> EnqueueAsync<TArgs>(
        TArgs args,
        CancellationToken cancellationToken = default)
        where TArgs : class
    {
        // 找到 TArgs 对应的 IBackgroundJob<TArgs>
        // Hangfire 序列化 args，入队
        var jobId = _client.Enqueue<HangfireJobExecutor>(
            executor => executor.ExecuteAsync<IBackgroundJob<TArgs>, TArgs>(
                args, CancellationToken.None));

        return Task.FromResult(jobId);
    }

    public Task<string> EnqueueAsync<TArgs>(
        TArgs args,
        TimeSpan delay,
        CancellationToken cancellationToken = default)
        where TArgs : class
    {
        var jobId = _client.Schedule<HangfireJobExecutor>(
            executor => executor.ExecuteAsync<IBackgroundJob<TArgs>, TArgs>(
                args, CancellationToken.None),
            delay);

        return Task.FromResult(jobId);
    }

    public Task<string> ScheduleAsync<TArgs>(
        TArgs args,
        DateTime executeAt,
        CancellationToken cancellationToken = default)
        where TArgs : class
    {
        var offset = new DateTimeOffset(executeAt, TimeSpan.Zero);

        var jobId = _client.Schedule<HangfireJobExecutor>(
            executor => executor.ExecuteAsync<IBackgroundJob<TArgs>, TArgs>(
                args, CancellationToken.None),
            offset);

        return Task.FromResult(jobId);
    }

    public Task<bool> DeleteAsync(
        string jobId,
        CancellationToken cancellationToken = default)
    {
        return Task.FromResult(_client.Delete(jobId));
    }
}
```

```csharp
using Hangfire;
using Platform.BackgroundJobs.Abstractions;

namespace Platform.BackgroundJobs.Hangfire;

public class HangfireRecurringManager : IRecurringJobManager
{
    public void AddOrUpdate<TJob>(
        string jobId,
        string cronExpression,
        TimeZoneInfo? timeZone = null)
        where TJob : class
    {
        RecurringJob.AddOrUpdate<TJob>(
            jobId,
            job => ((dynamic)job).ExecuteAsync(CancellationToken.None),
            cronExpression,
            new RecurringJobOptions
            {
                TimeZone = timeZone ?? TimeZoneInfo.Utc,
            });
    }

    public void AddOrUpdate<TJob, TArgs>(
        string jobId,
        TArgs args,
        string cronExpression,
        TimeZoneInfo? timeZone = null)
        where TJob : class, IBackgroundJob<TArgs>
        where TArgs : class
    {
        RecurringJob.AddOrUpdate<HangfireJobExecutor>(
            jobId,
            executor => executor.ExecuteAsync<TJob, TArgs>(
                args, CancellationToken.None),
            cronExpression,
            new RecurringJobOptions
            {
                TimeZone = timeZone ?? TimeZoneInfo.Utc,
            });
    }

    public void Remove(string jobId)
    {
        RecurringJob.RemoveIfExists(jobId);
    }

    public void Trigger(string jobId)
    {
        RecurringJob.TriggerJob(jobId);
    }
}
```

```csharp
using Hangfire.Dashboard;

namespace Platform.BackgroundJobs.Hangfire;

/// <summary>
/// Hangfire Dashboard 鉴权过滤器
/// 
/// 开发环境: 允许所有人访问
/// 生产环境: 检查是否已登录 + 是否管理员
/// </summary>
public class HangfireAuthFilter : IDashboardAuthorizationFilter
{
    private readonly bool _isDevelopment;
    private readonly string[] _allowedRoles;

    public HangfireAuthFilter(bool isDevelopment, string[]? allowedRoles = null)
    {
        _isDevelopment = isDevelopment;
        _allowedRoles = allowedRoles ?? ["admin"];
    }

    public bool Authorize(DashboardContext context)
    {
        if (_isDevelopment)
            return true;

        var httpContext = context.GetHttpContext();
        var user = httpContext.User;

        if (!user.Identity?.IsAuthenticated ?? true)
            return false;

        return _allowedRoles.Any(role => user.IsInRole(role));
    }
}
```

```csharp
// ============================================
// Extensions/ServiceCollectionExtensions.cs
// ============================================

using Hangfire;
using Hangfire.PostgreSql;
using Microsoft.AspNetCore.Builder;
using Microsoft.Extensions.DependencyInjection;
using Microsoft.Extensions.Hosting;
using Platform.BackgroundJobs.Abstractions;
using Platform.BackgroundJobs.Hangfire;

namespace Platform.BackgroundJobs.Extensions;

public static class ServiceCollectionExtensions
{
    /// <summary>
    /// 注册后台任务模块
    /// 
    /// 用法:
    ///   builder.Services.AddPlatformBackgroundJobs(options =>
    ///   {
    ///       options.StorageType = JobStorageType.PostgreSql;
    ///       options.ConnectionString = conn;
    ///   },
    ///   typeof(SendEmailJob).Assembly);
    /// </summary>
    public static IServiceCollection AddPlatformBackgroundJobs(
        this IServiceCollection services,
        Action<BackgroundJobOptions> configure,
        params System.Reflection.Assembly[] assemblies)
    {
        var options = new BackgroundJobOptions();
        configure(options);

        // Hangfire 配置
        services.AddHangfire(config =>
        {
            switch (options.StorageType)
            {
                case JobStorageType.PostgreSql:
                    config.UsePostgreSqlStorage(c =>
                        c.UseNpgsqlConnection(options.ConnectionString!));
                    break;
                case JobStorageType.InMemory:
                    config.UseInMemoryStorage();
                    break;
                default:
                    throw new NotSupportedException(
                        $"Storage type {options.StorageType} not yet supported");
            }

            config.UseSimpleAssemblyNameTypeSerializer()
                  .UseRecommendedSerializerSettings();
        });

        services.AddHangfireServer(serverOptions =>
        {
            serverOptions.WorkerCount = options.WorkerCount;
        });

        // 注册管理器
        services.AddSingleton<IBackgroundJobManager, HangfireJobManager>();
        services.AddSingleton<IRecurringJobManager, HangfireRecurringManager>();
        services.AddSingleton<HangfireJobExecutor>();

        // 自动发现并注册所有 IBackgroundJob<T>
        foreach (var assembly in assemblies)
        {
            var jobTypes = assembly.GetTypes()
                .Where(t => !t.IsAbstract && !t.IsInterface)
                .Where(t => t.GetInterfaces()
                    .Any(i => i.IsGenericType &&
                         i.GetGenericTypeDefinition() == typeof(IBackgroundJob<>)));

            foreach (var jobType in jobTypes)
            {
                services.AddScoped(jobType);
            }
        }

        // 配置选项
        services.Configure<BackgroundJobOptions>(opt =>
        {
            opt.StorageType = options.StorageType;
            opt.ConnectionString = options.ConnectionString;
            opt.DashboardPath = options.DashboardPath;
            opt.EnableDashboard = options.EnableDashboard;
            opt.DashboardAllowedRoles = options.DashboardAllowedRoles;
            opt.WorkerCount = options.WorkerCount;
            opt.DefaultRetryCount = options.DefaultRetryCount;
        });

        // 全局重试策略
        GlobalJobFilters.Filters.Add(
            new AutomaticRetryAttribute
            {
                Attempts = options.DefaultRetryCount
            });

        return services;
    }

    /// <summary>
    /// 配置 Hangfire Dashboard
    /// </summary>
    public static IApplicationBuilder UsePlatformBackgroundJobs(
        this IApplicationBuilder app)
    {
        var env = app.ApplicationServices
            .GetRequiredService<IHostEnvironment>();
        var options = app.ApplicationServices
            .GetRequiredService<Microsoft.Extensions.Options.IOptions<BackgroundJobOptions>>()
            .Value;

        if (options.EnableDashboard)
        {
            app.UseHangfireDashboard(options.DashboardPath,
                new DashboardOptions
                {
                    Authorization =
                    [
                        new HangfireAuthFilter(
                            env.IsDevelopment(),
                            options.DashboardAllowedRoles)
                    ],
                    DashboardTitle = "Background Jobs",
                });
        }

        return app;
    }
}
```

```csharp
namespace Platform.EventBus.Abstractions;

/// <summary>
/// 集成事件接口（跨服务通信）
/// 
/// 和领域事件的区别:
///   领域事件(IDomainEvent): 进程内，同一个聚合根内部通信
///     由 DomainEventInterceptor 在 SaveChanges 后分发
///     例: OrderCreatedDomainEvent → 同服务内更新统计

///     例: OrderCreatedIntegrationEvent → 通知仓库系统备货
///     
/// 通常的流转:
///   聚合根 → 领域事件 → 领域事件Handler → 发布集成事件 → 消息总线 → 其他服务
/// </summary>
public interface IIntegrationEvent
{
    /// <summary>
    /// 事件唯一标识（幂等用）
    /// </summary>
    Guid EventId { get; }

    /// <summary>
    /// 事件创建时间
    /// </summary>
    DateTime CreatedAt { get; }

    /// <summary>
    /// 事件类型名（路由用）
    /// </summary>
    string EventTypeName => GetType().Name;
}
```

```csharp
// ============================================
// Local/InMemoryEventBus.cs
// 开发环境用，单机内进程内通信
// ============================================

using System.Collections.Concurrent;
using Microsoft.Extensions.DependencyInjection;
using Microsoft.Extensions.Logging;
using Platform.EventBus.Abstractions;

namespace Platform.EventBus.Local;

/// <summary>
/// 内存事件总线
/// 
/// 适用:
///   开发环境
///   单元测试
///   单机部署（不需要跨服务通信）
///   
/// 不适用:
///   多实例部署（消息只在当前进程内）
/// </summary>
public class InMemoryEventBus : IEventBus
{
    private readonly IServiceScopeFactory _scopeFactory;
    private readonly ILogger<InMemoryEventBus> _logger;

    // 订阅关系: EventType → List<HandlerType>
    private readonly ConcurrentDictionary<Type, List<Type>> _handlers = new();

    public InMemoryEventBus(
        IServiceScopeFactory scopeFactory,
        ILogger<InMemoryEventBus> logger)
    {
        _scopeFactory = scopeFactory;
        _logger = logger;
    }

    public async Task PublishAsync<TEvent>(
        TEvent @event,
        CancellationToken cancellationToken = default)
        where TEvent : IIntegrationEvent
    {
        var eventType = typeof(TEvent);
        var eventName = eventType.Name;

        _logger.LogInformation(
            "[EventBus] Publishing: {EventName} ({EventId})",
            eventName, @event.EventId);

        if (!_handlers.TryGetValue(eventType, out var handlerTypes))
        {
            _logger.LogDebug(
                "[EventBus] No handlers for: {EventName}", eventName);
            return;
        }

        foreach (var handlerType in handlerTypes)
        {
            try
            {
                using var scope = _scopeFactory.CreateScope();
                var handler = scope.ServiceProvider.GetRequiredService(handlerType);

                var method = handlerType.GetMethod("HandleAsync");
                if (method != null)
                {
                    var task = (Task)method.Invoke(
                        handler, [@event, cancellationToken])!;
                    await task;
                }

                _logger.LogDebug(
                    "[EventBus] Handled: {EventName} by {HandlerName}",
                    eventName, handlerType.Name);
            }
            catch (Exception ex)
            {
                _logger.LogError(ex,
                    "[EventBus] Handler failed: {EventName} by {HandlerName}",
                    eventName, handlerType.Name);
                // 内存模式不重试，直接记录错误
            }
        }
    }

    public async Task PublishAsync<TEvent>(
        TEvent @event,
        TimeSpan delay,
        CancellationToken cancellationToken = default)
        where TEvent : IIntegrationEvent
    {
        await Task.Delay(delay, cancellationToken);
        await PublishAsync(@event, cancellationToken);
    }

    public void Subscribe<TEvent, THandler>()
        where TEvent : IIntegrationEvent
        where THandler : IIntegrationEventHandler<TEvent>
    {
        var handlers = _handlers.GetOrAdd(typeof(TEvent), _ => []);

        if (!handlers.Contains(typeof(THandler)))
        {
            handlers.Add(typeof(THandler));

            _logger.LogInformation(
                "[EventBus] Subscribed: {EventName} → {HandlerName}",
                typeof(TEvent).Name, typeof(THandler).Name);
        }
    }

    public void Unsubscribe<TEvent, THandler>()
        where TEvent : IIntegrationEvent
        where THandler : IIntegrationEventHandler<TEvent>
    {
        if (_handlers.TryGetValue(typeof(TEvent), out var handlers))
        {
            handlers.Remove(typeof(THandler));
        }
    }
}
```

```csharp
using Microsoft.Extensions.Logging;
using RabbitMQ.Client;

namespace Platform.EventBus.RabbitMQ;

/// <summary>
/// RabbitMQ 持久连接
/// 
/// 功能:
///   自动重连
///   连接复用
///   线程安全
/// </summary>
public class RabbitMqConnection : IDisposable
{
    private readonly ConnectionFactory _factory;
    private readonly ILogger<RabbitMqConnection> _logger;
    private readonly SemaphoreSlim _semaphore = new(1, 1);
    private IConnection? _connection;
    private bool _disposed;

    public RabbitMqConnection(
        RabbitMqOptions options,
        ILogger<RabbitMqConnection> logger)
    {
        _factory = new ConnectionFactory
        {
            HostName = options.HostName,
            Port = options.Port,
            UserName = options.UserName,
            Password = options.Password,
            VirtualHost = options.VirtualHost,
            AutomaticRecoveryEnabled = true,
            NetworkRecoveryInterval = TimeSpan.FromSeconds(10),
            DispatchConsumersAsync = true, // 异步消费
        };
        _logger = logger;
    }

    public bool IsConnected =>
        _connection is { IsOpen: true } && !_disposed;

    public IModel CreateChannel()
    {
        if (!IsConnected)
            TryConnect();

        if (!IsConnected)
            throw new InvalidOperationException(
                "No RabbitMQ connection available");

        return _connection!.CreateModel();
    }

    private void TryConnect()
    {
        _semaphore.Wait();
        try
        {
            if (IsConnected) return;

            _logger.LogInformation("[RabbitMQ] Connecting...");
            _connection = _factory.CreateConnection();
            _logger.LogInformation("[RabbitMQ] Connected");

            _connection.ConnectionShutdown += (_, args) =>
            {
                _logger.LogWarning(
                    "[RabbitMQ] Connection shutdown: {Reason}",
                    args.ReplyText);
            };
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "[RabbitMQ] Connection failed");
            throw;
        }
        finally
        {
            _semaphore.Release();
        }
    }

    public void Dispose()
    {
        if (_disposed) return;
        _disposed = true;
        _connection?.Dispose();
    }
}
```

```csharp
using System.Collections.Concurrent;
using System.Text;
using System.Text.Json;
using Microsoft.Extensions.Logging;
using Platform.EventBus.Abstractions;

namespace Platform.EventBus.RabbitMQ;

public class RabbitMqEventBus : IEventBus, IDisposable
{

    private readonly ILogger<RabbitMqEventBus> _logger;
    private readonly ConcurrentDictionary<Type, List<Type>> _handlers = new();
    private IModel? _publishChannel;

    public RabbitMqEventBus(
        RabbitMqConnection connection,
        RabbitMqOptions options,
        ILogger<RabbitMqEventBus> logger)
    {
        _connection = connection;
        _options = options;
        _logger = logger;
    }

    public Task PublishAsync<TEvent>(
        TEvent @event,
        CancellationToken cancellationToken = default)
        where TEvent : IIntegrationEvent
    {
        EnsurePublishChannel();

        var eventName = typeof(TEvent).Name;
        var json = JsonSerializer.Serialize(@event);
        var body = Encoding.UTF8.GetBytes(json);

        // 消息属性
        var properties = _publishChannel!.CreateBasicProperties();
        properties.DeliveryMode = 2; // 持久化
        properties.MessageId = @event.EventId.ToString();
        properties.Timestamp = new AmqpTimestamp(
            new DateTimeOffset(@event.CreatedAt).ToUnixTimeSeconds());
        properties.Type = eventName;
        properties.Headers = new Dictionary<string, object>
        {
            ["event-type"] = eventName,
            ["retry-count"] = 0,
        };

        // 发布到 Exchange，routingKey 为事件名
        _publishChannel.BasicPublish(
            exchange: _options.ExchangeName,
            routingKey: eventName,
            basicProperties: properties,
            body: body);

        _logger.LogInformation(
            "[EventBus:RabbitMQ] Published: {EventName} ({EventId})",
            eventName, @event.EventId);

        return Task.CompletedTask;
    }

    public async Task PublishAsync<TEvent>(
        TEvent @event,
        TimeSpan delay,
        CancellationToken cancellationToken = default)
        where TEvent : IIntegrationEvent
    {
        // RabbitMQ 延迟消息通过 TTL + 死信实现
        // 简化处理：用 Task.Delay 模拟
        // 生产建议：使用 rabbitmq_delayed_message_exchange 插件
        await Task.Delay(delay, cancellationToken);
        await PublishAsync(@event, cancellationToken);
    }

    public void Subscribe<TEvent, THandler>()
        where TEvent : IIntegrationEvent
        where THandler : IIntegrationEventHandler<TEvent>
    {
        var eventType = typeof(TEvent);
        var handlers = _handlers.GetOrAdd(eventType, _ => []);

        if (!handlers.Contains(typeof(THandler)))
        {
            handlers.Add(typeof(THandler));
            _logger.LogInformation(
                "[EventBus:RabbitMQ] Subscribed: {Event} → {Handler}",
                eventType.Name, typeof(THandler).Name);
        }
    }

    public void Unsubscribe<TEvent, THandler>()
        where TEvent : IIntegrationEvent
        where THandler : IIntegrationEventHandler<TEvent>
    {
        if (_handlers.TryGetValue(typeof(TEvent), out var handlers))
            handlers.Remove(typeof(THandler));
    }

    internal ConcurrentDictionary<Type, List<Type>> GetHandlers() => _handlers;

    private void EnsurePublishChannel()
    {
        if (_publishChannel is { IsOpen: true })
            return;

        _publishChannel = _connection.CreateChannel();

        // 声明 Exchange
        _publishChannel.ExchangeDeclare(
            exchange: _options.ExchangeName,
            type: ExchangeType.Topic,
            durable: true,
            autoDelete: false);
    }

    public void Dispose()
    {
        _publishChannel?.Dispose();
    }
}
```

```csharp
// ============================================
// RabbitMQ/RabbitMqConsumerService.cs
// 消费者后台服务
// ============================================

using System.Collections.Concurrent;
using System.Text;
using System.Text.Json;
using Microsoft.Extensions.DependencyInjection;
using Microsoft.Extensions.Hosting;
using Microsoft.Extensions.Logging;
using Platform.EventBus.Abstractions;
using RabbitMQ.Client;
using RabbitMQ.Client.Events;

namespace Platform.EventBus.RabbitMQ;

/// <summary>
/// RabbitMQ 消费者后台服务
/// 
/// 启动时:
///   1. 为每个订阅的事件创建 Queue
///   2. 绑定 Queue 到 Exchange
///   3. 启动消费
///   
/// 消息处理:
///   反序列化 → 找 Handler → 执行 → ACK
///   失败 → 重试（Header 记录次数）→ 超过次数 → 死信
/// </summary>
public class RabbitMqConsumerService : BackgroundService
{
    private readonly RabbitMqConnection _connection;
    private readonly RabbitMqEventBus _eventBus;
    private readonly RabbitMqOptions _options;
    private readonly IServiceScopeFactory _scopeFactory;
    private readonly ILogger<RabbitMqConsumerService> _logger;
    private IModel? _consumerChannel;

    public RabbitMqConsumerService(
        RabbitMqConnection connection,
        RabbitMqEventBus eventBus,
        RabbitMqOptions options,
        IServiceScopeFactory scopeFactory,
        ILogger<RabbitMqConsumerService> logger)
    {
        _connection = connection;
        _eventBus = eventBus;
        _options = options;
        _scopeFactory = scopeFactory;
        _logger = logger;
    }

    protected override Task ExecuteAsync(CancellationToken stoppingToken)
    {
        _consumerChannel = _connection.CreateChannel();

        // 声明死信交换机
        _consumerChannel.ExchangeDeclare(
            _options.DeadLetterExchange,
            ExchangeType.Topic, durable: true);

        // 声明主 Exchange
        _consumerChannel.ExchangeDeclare(
            _options.ExchangeName,
            ExchangeType.Topic, durable: true);

        _consumerChannel.BasicQos(0, (ushort)_options.PrefetchCount, false);

        // 为每种事件创建队列并绑定
        var handlers = _eventBus.GetHandlers();
        foreach (var (eventType, _) in handlers)
        {
            var eventName = eventType.Name;
            var queueName = $"{_options.QueuePrefix}{eventName}";

            // 声明队列（带死信配置）
            _consumerChannel.QueueDeclare(
                queue: queueName,
                durable: true,
                exclusive: false,
                autoDelete: false,
                arguments: new Dictionary<string, object>
                {
                    ["x-dead-letter-exchange"] = _options.DeadLetterExchange,
                    ["x-dead-letter-routing-key"] = $"dlx.{eventName}",
                });

            // 绑定到 Exchange
            _consumerChannel.QueueBind(
                queue: queueName,
                exchange: _options.ExchangeName,
                routingKey: eventName);

            // 启动消费
            var consumer = new AsyncEventingBasicConsumer(_consumerChannel);
            consumer.Received += async (_, ea) =>
            {
                await ProcessMessageAsync(eventType, ea, stoppingToken);
            };

            _consumerChannel.BasicConsume(
                queue: queueName,
                autoAck: false, // 手动确认
                consumer: consumer);

            _logger.LogInformation(
                "[EventBus:RabbitMQ] Consuming queue: {Queue}",
                queueName);
        }

        return Task.CompletedTask;
    }

    private async Task ProcessMessageAsync(
        Type eventType,
        BasicDeliverEventArgs ea,
        CancellationToken ct)
    {
        var eventName = eventType.Name;
        var json = Encoding.UTF8.GetString(ea.Body.Span);

        try
        {
            var @event = JsonSerializer.Deserialize(json, eventType);
            if (@event == null)
            {
                _logger.LogWarning(
                    "[EventBus:RabbitMQ] Failed to deserialize: {EventName}",
                    eventName);
                _consumerChannel?.BasicNack(ea.DeliveryTag, false, false);
                return;
            }

            var handlers = _eventBus.GetHandlers();
            if (!handlers.TryGetValue(eventType, out var handlerTypes))
            {
                _consumerChannel?.BasicAck(ea.DeliveryTag, false);
                return;
            }

            foreach (var handlerType in handlerTypes)
            {
                using var scope = _scopeFactory.CreateScope();
                var handler = scope.ServiceProvider.GetRequiredService(handlerType);

                var method = handlerType.GetMethod("HandleAsync");
                if (method != null)
                {
                    var task = (Task)method.Invoke(handler, [@event, ct])!;
                    await task;
                }
            }

            _consumerChannel?.BasicAck(ea.DeliveryTag, false);

            _logger.LogDebug(
                "[EventBus:RabbitMQ] Processed: {EventName}",
                eventName);
        }
        catch (Exception ex)
        {
            _logger.LogError(ex,
                "[EventBus:RabbitMQ] Failed to process: {EventName}",
                eventName);

            // 检查重试次数
            var retryCount = GetRetryCount(ea.BasicProperties);
            if (retryCount < _options.RetryCount)
            {
                // 重新入队（会带上更新后的重试次数）
                _consumerChannel?.BasicNack(ea.DeliveryTag, false, true);
            }
            else
            {
                // 超过重试次数 → 进死信队列
                _consumerChannel?.BasicNack(ea.DeliveryTag, false, false);
                _logger.LogError(
                    "[EventBus:RabbitMQ] Message sent to DLX: {EventName}",
                    eventName);
            }
        }
    }

    private static int GetRetryCount(IBasicProperties properties)
    {
        if (properties.Headers != null &&
            properties.Headers.TryGetValue("retry-count", out var value))
        {
            return Convert.ToInt32(value);
        }
        return 0;
    }

    public override void Dispose()
    {
        _consumerChannel?.Dispose();
        base.Dispose();
    }
}
```

```csharp
///   2. 反序列化为事件
///   3. 发布到 EventBus
///   4. 标记为 Succeeded
///   5. 失败则标记错误，下次重试
/// </summary>
public class OutboxProcessorService : BackgroundService
{
    private readonly IServiceScopeFactory _scopeFactory;
    private readonly ILogger<OutboxProcessorService> _logger;
    private readonly TimeSpan _interval = TimeSpan.FromSeconds(5);
    private readonly int _batchSize = 20;

    public OutboxProcessorService(
        IServiceScopeFactory scopeFactory,
        ILogger<OutboxProcessorService> logger)
    {
        _scopeFactory = scopeFactory;
        _logger = logger;
    }

    protected override async Task ExecuteAsync(CancellationToken stoppingToken)
    {
        _logger.LogInformation("[Outbox] Processor started");

        while (!stoppingToken.IsCancellationRequested)
        {
            try
            {
                await ProcessBatchAsync(stoppingToken);
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "[Outbox] Processor error");
            }

            await Task.Delay(_interval, stoppingToken);
        }
    }

    private async Task ProcessBatchAsync(CancellationToken ct)
    {
        using var scope = _scopeFactory.CreateScope();
        var store = scope.ServiceProvider.GetRequiredService<IOutboxStore>();
        var eventBus = scope.ServiceProvider.GetRequiredService<IEventBus>();

        var messages = await store.GetPendingAsync(_batchSize, ct);
        if (messages.Count == 0) return;

        foreach (var message in messages)
        {
            try
            {
                // 反序列化
                var eventType = Type.GetType(message.EventTypeName);
                if (eventType == null)
                {
                    await store.MarkAsFailedAsync(message.Id,
                        $"Type not found: {message.EventTypeName}", ct);
                    continue;
                }

                var @event = JsonSerializer.Deserialize(
                    message.Payload, eventType) as IIntegrationEvent;

                if (@event == null)
                {
                    await store.MarkAsFailedAsync(message.Id,
                        "Deserialization returned null", ct);
                    continue;
                }

                // 发布
                await eventBus.PublishAsync(@event, ct);
                await store.MarkAsProcessedAsync(message.Id, ct);

                _logger.LogDebug(
                    "[Outbox] Processed: {EventType} ({Id})",
                    message.EventTypeName, message.Id);
            }
            catch (Exception ex)
            {
                await store.MarkAsFailedAsync(
                    message.Id, ex.Message, ct);

                _logger.LogWarning(ex,
                    "[Outbox] Failed: {Id}", message.Id);
            }
        }
    }
}
```

```csharp
// ============================================
// Redis/RedisDistributedLockProvider.cs
// ============================================

using Microsoft.Extensions.Logging;
using Platform.DistributedLocking.Abstractions;
using StackExchange.Redis;

namespace Platform.DistributedLocking.Redis;

/// <summary>
/// Redis 分布式锁
/// 
/// 实现原理:
///   获取锁: SET key lockId NX PX expiry
///     NX: 只在 key 不存在时设置（原子操作）
///     PX: 过期时间（毫秒）防死锁
///     lockId: 随机值，保证只释放自己的锁
///     
///   释放锁: Lua 脚本（原子操作）
///     if redis.call("get", key) == lockId then
///       return redis.call("del", key)
///     end
///     
///   续约: Lua 脚本
///     if redis.call("get", key) == lockId then
///       return redis.call("pexpire", key, expiry)
///     end
///     
/// 为什么不用 RedLock:
///   单 Redis 实例够用
///   RedLock 用于多 Redis 实例（更高可用）
///   绝大部分场景单实例足够
/// </summary>
public class RedisDistributedLockProvider : IDistributedLockProvider
{
    private readonly IConnectionMultiplexer _redis;
    private readonly RedisLockOptions _options;
    private readonly ILogger<RedisDistributedLockProvider> _logger;

    // 释放锁的 Lua 脚本（原子操作：只释放自己持有的锁）
    private const string ReleaseLuaScript = @"
        if redis.call('get', KEYS[1]) == ARGV[1] then
            return redis.call('del', KEYS[1])
        else
            return 0
        end";

    // 续约锁的 Lua 脚本
    private const string ExtendLuaScript = @"
        if redis.call('get', KEYS[1]) == ARGV[1] then
            return redis.call('pexpire', KEYS[1], ARGV[2])
        else
            return 0
        end";

    public RedisDistributedLockProvider(
        IConnectionMultiplexer redis,
        RedisLockOptions options,
        ILogger<RedisDistributedLockProvider> logger)
    {
        _redis = redis;
        _options = options;
        _logger = logger;
    }

    public async Task<IDistributedLockHandle?> TryAcquireAsync(
        string key,
        TimeSpan expiry,
        CancellationToken cancellationToken = default)
    {
        var fullKey = $"{_options.KeyPrefix}{key}";
        var lockId = Guid.NewGuid().ToString("N");

        var db = _redis.GetDatabase();

        // SET key lockId NX PX expiry
        var acquired = await db.StringSetAsync(
            fullKey, lockId, expiry, When.NotExists);

        if (!acquired)
        {
            _logger.LogDebug(
                "[DistributedLock] Failed to acquire: {Key}", key);
            return null;
        }

        _logger.LogDebug(
            "[DistributedLock] Acquired: {Key} (LockId: {LockId}, Expiry: {Expiry})",
            key, lockId[..8], expiry);

        return new RedisDistributedLockHandle(
            db, fullKey, lockId, _logger);
    }

    public async Task<IDistributedLockHandle> AcquireAsync(
        string key,
        TimeSpan expiry,
        TimeSpan? waitTimeout = null,
        TimeSpan? retryInterval = null,
        CancellationToken cancellationToken = default)
    {
        var timeout = waitTimeout ?? _options.DefaultWaitTimeout;
        var retry = retryInterval ?? _options.DefaultRetryInterval;
        var deadline = DateTime.UtcNow + timeout;

        while (DateTime.UtcNow < deadline)
        {
            cancellationToken.ThrowIfCancellationRequested();

            var handle = await TryAcquireAsync(key, expiry, cancellationToken);
            if (handle != null)
                return handle;

            await Task.Delay(retry, cancellationToken);
        }

        throw new TimeoutException(
            $"Failed to acquire distributed lock '{key}' " +
            $"within {timeout.TotalSeconds}s");
    }
}

/// <summary>
/// Redis 锁句柄
/// </summary>
internal class RedisDistributedLockHandle : IDistributedLockHandle
{
    private readonly IDatabase _db;
    private readonly ILogger _logger;
    private bool _isReleased;

    public string Key { get; }
    public string LockId { get; }
    public bool IsReleased => _isReleased;

    private const string ReleaseLuaScript = @"
        if redis.call('get', KEYS[1]) == ARGV[1] then
            return redis.call('del', KEYS[1])
        else
            return 0
        end";

    private const string ExtendLuaScript = @"
        if redis.call('get', KEYS[1]) == ARGV[1] then
            return redis.call('pexpire', KEYS[1], ARGV[2])
        else
            return 0
        end";

    internal RedisDistributedLockHandle(
        IDatabase db,
        string key,
        string lockId,
        ILogger logger)
    {
        _db = db;
        Key = key;
        LockId = lockId;
        _logger = logger;
    }

    public async Task<bool> ExtendAsync(TimeSpan extension)
    {
        if (_isReleased) return false;

        var result = await _db.ScriptEvaluateAsync(
            ExtendLuaScript,
            new RedisKey[] { Key },
            new RedisValue[] { LockId, (long)extension.TotalMilliseconds });

        var success = (int)result == 1;

        _logger.LogDebug(
            "[DistributedLock] Extend {Key}: {Result}",
            Key, success ? "OK" : "Failed");

        return success;
    }

    public async Task ReleaseAsync()
    {
        if (_isReleased) return;

        var result = await _db.ScriptEvaluateAsync(
            ReleaseLuaScript,
            new RedisKey[] { Key },
            new RedisValue[] { LockId });

        _isReleased = true;

        _logger.LogDebug(
            "[DistributedLock] Released: {Key} (Result: {Result})",
            Key, (int)result);
    }

    public async ValueTask DisposeAsync()
    {
        if (!_isReleased)
        {
            await ReleaseAsync();
        }
    }
}
```

## 内存实现（开发用）

```csharp
// ============================================
// Local/InMemoryDistributedLockProvider.cs
// ============================================

using System.Collections.Concurrent;
using Microsoft.Extensions.Logging;
using Platform.DistributedLocking.Abstractions;

namespace Platform.DistributedLocking.Local;

/// <summary>
/// 内存分布式锁（开发 / 测试用）
/// 
/// 单机有效，多实例无效
/// 不需要 Redis 依赖
/// </summary>
public class InMemoryDistributedLockProvider : IDistributedLockProvider
{
    private readonly ConcurrentDictionary<string, LockEntry> _locks = new();
    private readonly ILogger<InMemoryDistributedLockProvider> _logger;

    public InMemoryDistributedLockProvider(
        ILogger<InMemoryDistributedLockProvider> logger)
    {
        _logger = logger;
    }

    public Task<IDistributedLockHandle?> TryAcquireAsync(
        string key,
        TimeSpan expiry,
        CancellationToken cancellationToken = default)
    {
        CleanExpired();

        var lockId = Guid.NewGuid().ToString("N");
        var entry = new LockEntry(lockId, DateTime.UtcNow + expiry);

        if (_locks.TryAdd(key, entry))
        {
            _logger.LogDebug(
                "[DistributedLock:Memory] Acquired: {Key}", key);

            IDistributedLockHandle handle =
                new InMemoryLockHandle(key, lockId, _locks, _logger);
            return Task.FromResult<IDistributedLockHandle?>(handle);
        }

        // 检查是否过期
        if (_locks.TryGetValue(key, out var existing) &&
            existing.ExpiresAt < DateTime.UtcNow)
        {
            // 过期了，替换
            if (_locks.TryUpdate(key, entry, existing))
            {
                IDistributedLockHandle handle =
                    new InMemoryLockHandle(key, lockId, _locks, _logger);
                return Task.FromResult<IDistributedLockHandle?>(handle);
            }
        }

        return Task.FromResult<IDistributedLockHandle?>(null);
    }

    public async Task<IDistributedLockHandle> AcquireAsync(
        string key,
        TimeSpan expiry,
        TimeSpan? waitTimeout = null,
        TimeSpan? retryInterval = null,
        CancellationToken cancellationToken = default)
    {
        var timeout = waitTimeout ?? TimeSpan.FromSeconds(10);
        var retry = retryInterval ?? TimeSpan.FromMilliseconds(100);
        var deadline = DateTime.UtcNow + timeout;

        while (DateTime.UtcNow < deadline)
        {
            cancellationToken.ThrowIfCancellationRequested();

            var handle = await TryAcquireAsync(key, expiry, cancellationToken);
            if (handle != null) return handle;

            await Task.Delay(retry, cancellationToken);
        }

        throw new TimeoutException(
            $"Failed to acquire lock '{key}' within {timeout.TotalSeconds}s");
    }

    private void CleanExpired()
    {
        var now = DateTime.UtcNow;
        var expiredKeys = _locks
            .Where(kvp => kvp.Value.ExpiresAt < now)
            .Select(kvp => kvp.Key)
            .ToList();

        foreach (var key in expiredKeys)
        {
            _locks.TryRemove(key, out _);
        }
    }

    private record LockEntry(string LockId, DateTime ExpiresAt);

    private class InMemoryLockHandle : IDistributedLockHandle
    {
        private readonly ConcurrentDictionary<string, LockEntry> _locks;
        private readonly ILogger _logger;
        private bool _isReleased;

        public string Key { get; }
        public string LockId { get; }
        public bool IsReleased => _isReleased;

        internal InMemoryLockHandle(
            string key,
            string lockId,
            ConcurrentDictionary<string, LockEntry> locks,
            ILogger logger)
        {
            Key = key;
            LockId = lockId;
            _locks = locks;
            _logger = logger;
        }

        public Task<bool> ExtendAsync(TimeSpan extension)
        {
            if (_isReleased) return Task.FromResult(false);

            if (_locks.TryGetValue(Key, out var entry) &&
                entry.LockId == LockId)
            {
                var newEntry = new LockEntry(
                    LockId, DateTime.UtcNow + extension);
                _locks.TryUpdate(Key, newEntry, entry);
                return Task.FromResult(true);
            }

            return Task.FromResult(false);
        }

        public Task ReleaseAsync()
        {
            if (_isReleased) return Task.CompletedTask;

            if (_locks.TryGetValue(Key, out var entry) &&
                entry.LockId == LockId)
            {
                _locks.TryRemove(Key, out _);
            }

            _isReleased = true;
            _logger.LogDebug(
                "[DistributedLock:Memory] Released: {Key}", Key);
            return Task.CompletedTask;
        }

        public async ValueTask DisposeAsync()
        {
            if (!_isReleased) await ReleaseAsync();
        }
    }
}
```

```csharp
public class CreatePurchaseCommandHandler
    : ICommandHandler<CreatePurchaseCommand, Guid>
{
    private readonly IRepository<PurchaseOrder, Guid> _repo;
    private readonly IBackgroundJobManager _jobManager;
    private readonly IEventBus _eventBus;

    public CreatePurchaseCommandHandler(
        IRepository<PurchaseOrder, Guid> repo,
        IBackgroundJobManager jobManager,
        IEventBus eventBus)
    {
        _repo = repo;
        _jobManager = jobManager;
        _eventBus = eventBus;
    }

    public async Task<Guid> HandleAsync(
        CreatePurchaseCommand command, CancellationToken ct)
    {
        // 1. 创建聚合根
        var order = new PurchaseOrder(
            Guid.NewGuid(),
            command.SupplierName,
            command.Remark);

        foreach (var line in command.Lines)
            order.AddLine(Guid.NewGuid(), line.ProductName,
                line.Quantity, line.UnitPrice);

        await _repo.InsertAsync(order, ct);

        // 2. 后台发送邮件通知（不阻塞主流程）
        await _jobManager.EnqueueAsync(new SendPurchaseEmailArgs
        {
            OrderId = order.Id,
            SupplierName = command.SupplierName,
            TotalAmount = order.TotalAmount,
        });

        // 3. 发布集成事件（通知其他服务）
        await _eventBus.PublishAsync(new PurchaseCreatedEvent(
            order.Id, order.OrderNo, order.TotalAmount), ct);

        return order.Id;

        // UnitOfWorkBehavior 自动 SaveChanges ✅
        // AuditingInterceptor 自动填充审计字段 ✅
    }
}
```

```csharp
// 场景2: 库存扣减（分布式锁 + 事务）

public class DeductStockCommandHandler
    : ICommandHandler<DeductStockCommand>
{
    private readonly IRepository<StockRecord, Guid> _repo;
    private readonly IDistributedLockProvider _lockProvider;

    public DeductStockCommandHandler(
        IRepository<StockRecord, Guid> repo,
        IDistributedLockProvider lockProvider)
    {
        _repo = repo;
        _lockProvider = lockProvider;
    }

    public async Task<Unit> HandleAsync(
        DeductStockCommand command, CancellationToken ct)
    {
        // 分布式锁：防止同一商品并发扣减导致超卖
        var lockKey = $"stock:deduct:{command.ProductId}";

        await using var lockHandle = await _lockProvider.AcquireAsync(
            lockKey,
            expiry: TimeSpan.FromSeconds(10),
            waitTimeout: TimeSpan.FromSeconds(5));

        // 获取锁成功，执行业务
        var stock = await _repo.FirstOrDefaultAsync(
            s => s.ProductId == command.ProductId, ct);

        if (stock == null)
            throw new BusinessException("ERP:Stock:NotFound", "库存记录不存在");

        if (stock.Quantity < command.Quantity)
            throw new BusinessException("ERP:Stock:Insufficient", "库存不足");

        stock.Deduct(command.Quantity, command.Reason);

        await _repo.UpdateAsync(stock, ct);
        return Unit.Value;

        // lockHandle Dispose → 自动释放锁 ✅
        // UnitOfWorkBehavior → 自动 SaveChanges ✅
    }
}
```

```csharp
        _db = db;
        _logger = logger;
    }

    public async Task ExecuteAsync(
        CleanExpiredTokensArgs args, CancellationToken ct)
    {
        var cutoff = DateTime.UtcNow.AddDays(-args.RetentionDays);

        var count = await _db.Set<OpenIddictEntityFrameworkCoreToken>()
            .Where(t => t.ExpirationDate < cutoff)
            .ExecuteDeleteAsync(ct);

        _logger.LogInformation(
            "Cleaned {Count} expired tokens (older than {Days} days)",
            count, args.RetentionDays);
    }
}

        => _recurring = recurring;

            "clean-expired-tokens",

            "0 2 * * *");
```
