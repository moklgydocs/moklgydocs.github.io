## CurrentUser

```csharp
// ============================================
// CurrentUser/HttpContextCurrentUser.cs
// ============================================

using System.Security.Claims;
using Microsoft.AspNetCore.Http;
using MokFramework.EntityFrameworkCore;

namespace MokFramework.AspNetCore.CurrentUser;

/// <summary>
/// 从 HttpContext 获取当前用户信息
/// 
/// 支持的 Claim 类型:
///   sub / ClaimTypes.NameIdentifier → UserId
///   tenant_id → TenantId
///   name / ClaimTypes.Name → UserName
///   email / ClaimTypes.Email → Email
///   role / ClaimTypes.Role → Roles
/// </summary>
public class HttpContextCurrentUser : ICurrentUser
{
    private readonly IHttpContextAccessor _accessor;

    public HttpContextCurrentUser(IHttpContextAccessor accessor)
    {
        _accessor = accessor;
    }

    private ClaimsPrincipal? Principal => _accessor.HttpContext?.User;

    public Guid? Id
    {
        get
        {
            var sub = Principal?.FindFirstValue("sub")
                   ?? Principal?.FindFirstValue(ClaimTypes.NameIdentifier);

            return Guid.TryParse(sub, out var id) ? id : null;
        }
    }

    public Guid? TenantId
    {
        get
        {
            var value = Principal?.FindFirstValue("tenant_id");
            return Guid.TryParse(value, out var id) ? id : null;
        }
    }

    public bool IsAuthenticated =>
        Principal?.Identity?.IsAuthenticated ?? false;

    /// <summary>
    /// 当前用户名
    /// </summary>
    public string? UserName =>
        Principal?.FindFirstValue("name")
        ?? Principal?.FindFirstValue(ClaimTypes.Name);

    /// <summary>
    /// 当前用户邮箱
    /// </summary>
    public string? Email =>
        Principal?.FindFirstValue("email")
        ?? Principal?.FindFirstValue(ClaimTypes.Email);

    /// <summary>
    /// 当前用户角色
    /// </summary>
    public string[] Roles =>
        Principal?.FindAll("role")
            .Concat(Principal.FindAll(ClaimTypes.Role))
            .Select(c => c.Value)
            .Distinct()
            .ToArray() ?? [];
}
```

```csharp
// ============================================
// ApiResult/ApiResult.cs
// ============================================

using System.Text.Json.Serialization;

namespace MokFramework.AspNetCore.ApiResult;

/// <summary>
/// 统一 API 响应模型
/// 
/// 所有 API 的响应格式统一为:
/// {
///   "success": true,
///   "code": 200,
///   "message": "操作成功",
///   "data": { ... },
///   "timestamp": 1704067200000,
///   "traceId": "abc-123"
/// }
/// 
/// 好处:
///   前端只需判断 success 即可
///   错误码统一
///   链路追踪
/// </summary>
public class ApiResult
{
    /// <summary>
    /// 是否成功
    /// </summary>
    public bool Success { get; set; }

    /// <summary>
    /// 业务状态码（非 HTTP 状态码）
    /// 200 = 成功
    /// 400 = 参数错误
    /// 401 = 未认证
    /// 403 = 无权限
    /// 404 = 未找到
    /// 500 = 服务器错误
    /// 自定义: 10001 = 业务错误...
    /// </summary>
    public int Code { get; set; }

    /// <summary>
    /// 消息
    /// </summary>
    public string? Message { get; set; }

    /// <summary>
    /// 错误详情（仅开发环境返回）
    /// </summary>
    [JsonIgnore(Condition = JsonIgnoreCondition.WhenWritingNull)]
    public string? Details { get; set; }

    /// <summary>
    /// 错误码（业务错误编码）
    /// 例: "SSO:User:DuplicateEmail"
    /// </summary>
    [JsonIgnore(Condition = JsonIgnoreCondition.WhenWritingNull)]
    public string? ErrorCode { get; set; }

    /// <summary>
    /// 时间戳（毫秒）
    /// </summary>
    public long Timestamp { get; set; } = DateTimeOffset.UtcNow.ToUnixTimeMilliseconds();

    /// <summary>
    /// 链路追踪 ID
    /// </summary>
    [JsonIgnore(Condition = JsonIgnoreCondition.WhenWritingNull)]
    public string? TraceId { get; set; }

    // ============ 静态工厂方法 ============

    public static ApiResult Ok(string? message = "操作成功")
    {
        return new ApiResult
        {
            Success = true,
            Code = 200,
            Message = message,
        };
    }

    public static ApiResult<T> Ok<T>(T data, string? message = "操作成功")
    {
        return new ApiResult<T>
        {
            Success = true,
            Code = 200,
            Message = message,
            Data = data,
        };
    }

    public static ApiResult Fail(
        string? message = "操作失败",
        int code = 400,
        string? errorCode = null)
    {
        return new ApiResult
        {
            Success = false,
            Code = code,
            Message = message,
            ErrorCode = errorCode,
        };
    }

    public static ApiResult<T> Fail<T>(
        string? message = "操作失败",
        int code = 400,
        string? errorCode = null)
    {
        return new ApiResult<T>
        {
            Success = false,
            Code = code,
            Message = message,
            ErrorCode = errorCode,
        };
    }
}

/// <summary>
/// 带数据的统一响应
/// </summary>
public class ApiResult<T> : ApiResult
{
    /// <summary>
    /// 响应数据
    /// </summary>
    [JsonIgnore(Condition = JsonIgnoreCondition.WhenWritingDefault)]
    public T? Data { get; set; }
}
```

```csharp
/// <summary>
/// API 结果包装选项
/// </summary>
public class ApiResultWrapOptions
{
    /// <summary>
    /// 是否自动包装成功响应

    [
        "/swagger",
        "/health",
        "/hangfire",
        "/_framework",
        "/favicon.ico",

    /// 不需要包装的返回类型

    /// </summary>
    public List<Type> ExcludeTypes { get; set; } =
    [
        typeof(Microsoft.AspNetCore.Mvc.FileResult),
        typeof(Microsoft.AspNetCore.Mvc.FileStreamResult),
        typeof(Microsoft.AspNetCore.Mvc.FileContentResult),
    ];

    /// <summary>
    /// 开发环境是否返回异常详情
    /// </summary>
    public bool ShowDetailsInDevelopment { get; set; } = true;
}
```

```csharp
// ============================================
// ApiResult/ApiResultFilter.cs
// 自动包装成功响应
// ============================================

using Microsoft.AspNetCore.Mvc;
using Microsoft.AspNetCore.Mvc.Filters;
using Microsoft.Extensions.Options;

namespace MokFramework.AspNetCore.ApiResult;

/// <summary>
/// API 结果自动包装过滤器
/// 
/// 作用:
///   Controller 返回 new { Name = "张三" }
///   自动包装为 { success: true, code: 200, data: { name: "张三" } }
///   
/// 不包装的情况:
///   已经是 ApiResult 类型
///   文件下载
///   排除的路径（/swagger 等）
///   标记了 [SkipApiResultWrap] 特性
/// </summary>
public class ApiResultFilter : IAsyncResultFilter
{
    private readonly ApiResultWrapOptions _options;

    public ApiResultFilter(IOptions<ApiResultWrapOptions> options)
    {
        _options = options.Value;
    }

    public async Task OnResultExecutionAsync(
        ResultExecutingContext context,
        ResultExecutionDelegate next)
    {
        if (!_options.AutoWrapSuccessResult)
        {
            await next();
            return;
        }

        // 检查是否排除
        if (ShouldSkip(context))
        {
            await next();
            return;
        }

        // 包装 ObjectResult
        if (context.Result is ObjectResult objectResult)
        {
            var value = objectResult.Value;

            // 已经是 ApiResult，不重复包装
            if (value is ApiResult)
            {
                SetTraceId(context, (ApiResult)value);
                await next();
                return;
            }

            // 包装
            var wrapped = ApiResult.Ok(value);
            SetTraceId(context, wrapped);
            objectResult.Value = wrapped;
            objectResult.StatusCode = 200;
        }
        // 包装 EmptyResult（void 返回的 Action）
        else if (context.Result is EmptyResult)
        {
            var wrapped = ApiResult.Ok();
            SetTraceId(context, wrapped);
            context.Result = new ObjectResult(wrapped) { StatusCode = 200 };
        }
        // 包装 StatusCodeResult（如 NoContent）
        else if (context.Result is StatusCodeResult statusCode
                 && statusCode.StatusCode >= 200
                 && statusCode.StatusCode < 300)
        {
            var wrapped = ApiResult.Ok();
            SetTraceId(context, wrapped);
            context.Result = new ObjectResult(wrapped) { StatusCode = 200 };
        }

        await next();
    }

    private bool ShouldSkip(ResultExecutingContext context)
    {
        // 检查 [SkipApiResultWrap] 特性
        var hasSkipAttribute = context.ActionDescriptor.EndpointMetadata
            .Any(m => m is SkipApiResultWrapAttribute);
        if (hasSkipAttribute) return true;

        // 检查排除路径
        var path = context.HttpContext.Request.Path.Value ?? "";
        if (_options.ExcludePaths.Any(p =>
            path.StartsWith(p, StringComparison.OrdinalIgnoreCase)))
            return true;

        // 检查排除类型
        if (context.Result is ObjectResult obj &&
            obj.Value != null &&
            _options.ExcludeTypes.Any(t => t.IsInstanceOfType(obj.Value)))
            return true;

        // 检查文件结果
        if (context.Result is FileResult) return true;

        return false;
    }

    private static void SetTraceId(ResultExecutingContext context, ApiResult result)
    {
        result.TraceId = context.HttpContext.TraceIdentifier;
    }
}

/// <summary>
/// 跳过 ApiResult 包装
/// 
/// 用法:
///   [HttpGet("export")]
///   [SkipApiResultWrap]
///   public FileResult Export() => File(bytes, "application/xlsx");
/// </summary>
[AttributeUsage(AttributeTargets.Class | AttributeTargets.Method)]
public class SkipApiResultWrapAttribute : Attribute
{
}
```

## 全局异常处理

```csharp
// ============================================
// Exceptions/GlobalExceptionFilter.cs
// ============================================

using Microsoft.AspNetCore.Hosting;
using Microsoft.AspNetCore.Mvc;
using Microsoft.AspNetCore.Mvc.Filters;
using Microsoft.Extensions.Hosting;
using Microsoft.Extensions.Logging;
using Microsoft.Extensions.Options;
using MokFramework.Exceptions;

namespace MokFramework.AspNetCore.Exceptions;

/// <summary>
/// 全局异常过滤器
/// 
/// 所有未处理的异常在这里统一转为 ApiResult
/// 
/// 异常分类处理:
///   BusinessException → 400（业务错误，直接返回给用户）
///   EntityNotFoundException → 404
///   UnauthorizedAccessException → 401
///   DbUpdateConcurrencyException → 409（并发冲突）
///   其他异常 → 500（系统错误，不暴露详情）
/// </summary>
public class GlobalExceptionFilter : IAsyncExceptionFilter
{
    private readonly ILogger<GlobalExceptionFilter> _logger;
    private readonly IWebHostEnvironment _env;
    private readonly ApiResultWrapOptions _options;

    public GlobalExceptionFilter(
        ILogger<GlobalExceptionFilter> logger,
        IWebHostEnvironment env,
        IOptions<ApiResultWrapOptions> options)
    {
        _logger = logger;
        _env = env;
        _options = options.Value;
    }

    public Task OnExceptionAsync(ExceptionContext context)
    {
        if (context.ExceptionHandled) return Task.CompletedTask;

        var exception = context.Exception;
        var result = ConvertToApiResult(exception);

        // 设置 TraceId
        result.TraceId = context.HttpContext.TraceIdentifier;

        // 开发环境返回详情
        if (_env.IsDevelopment() && _options.ShowDetailsInDevelopment)
        {
            result.Details = exception.ToString();
        }

        // 日志
        LogException(exception, result);

        // 设置响应
        var httpStatusCode = GetHttpStatusCode(exception);
        context.Result = new ObjectResult(result)
        {
            StatusCode = httpStatusCode
        };

        context.ExceptionHandled = true;
        return Task.CompletedTask;
    }

    private ApiResult ConvertToApiResult(Exception exception)
    {
        return exception switch
        {
            // 业务异常 → 返回用户可见的错误信息
            BusinessException biz => new ApiResult
            {
                Success = false,
                Code = biz.HttpStatusCode,
                Message = biz.Message ?? "业务处理失败",
                ErrorCode = biz.Code,
                Details = biz.Details,
            },

            // 实体未找到
            EntityNotFoundException notFound => new ApiResult
            {
                Success = false,
                Code = 404,
                Message = notFound.Message,
                ErrorCode = notFound.Code,
            },

            // 未认证
            UnauthorizedAccessException => new ApiResult
            {
                Success = false,
                Code = 401,
                Message = "未登录或登录已过期",
            },

            // 并发冲突
            Microsoft.EntityFrameworkCore.DbUpdateConcurrencyException => new ApiResult
            {
                Success = false,
                Code = 409,
                Message = "数据已被其他人修改，请刷新后重试",
                ErrorCode = "Platform:ConcurrencyConflict",
            },

            // 操作取消（客户端断开连接）
            OperationCanceledException => new ApiResult
            {
                Success = false,
                Code = 499,
                Message = "请求已取消",
            },

            // 参数异常
            ArgumentException arg => new ApiResult
            {
                Success = false,
                Code = 400,
                Message = _env.IsDevelopment() ? arg.Message : "参数错误",
            },

            // 其他所有异常 → 500
            _ => new ApiResult
            {
                Success = false,
                Code = 500,
                Message = "服务器内部错误",
                ErrorCode = "Platform:InternalError",
            },
        };
    }

    private static int GetHttpStatusCode(Exception exception)
    {
        return exception switch
        {
            IHasHttpStatusCode hasCode => hasCode.HttpStatusCode,
            UnauthorizedAccessException => 401,
            Microsoft.EntityFrameworkCore.DbUpdateConcurrencyException => 409,
            OperationCanceledException => 499,
            _ => 500,
        };
    }

    private void LogException(Exception exception, ApiResult result)
    {
        // 业务异常: Warning 级别（预期的错误）
        // 系统异常: Error 级别（非预期的错误）
        if (exception is BusinessException)
        {
            _logger.LogWarning(exception,
                "Business exception: {ErrorCode} - {Message}",
                result.ErrorCode, result.Message);
        }
        else if (exception is OperationCanceledException)
        {
            _logger.LogInformation(
                "Request cancelled: {TraceId}", result.TraceId);
        }
        else
        {
            _logger.LogError(exception,
                "Unhandled exception: {Message} | TraceId: {TraceId}",
                exception.Message, result.TraceId);
        }
    }
}
```

```csharp
// Exceptions/ModelValidationFilter.cs
// 模型验证统一处理
// ============================================

using Microsoft.AspNetCore.Mvc;
using Microsoft.AspNetCore.Mvc.Filters;

namespace MokFramework.AspNetCore.Exceptions;

/// <summary>
/// 模型验证过滤器
/// 
/// 自动处理 [Required] [StringLength] 等验证特性的失败
/// 统一返回格式:
/// {
///   "success": false,
///   "code": 400,
///   "message": "参数验证失败",
///   "data": {
///     "UserName": ["用户名不能为空"],
///     "Email": ["邮箱格式不正确"]
///   }
/// }
/// </summary>
public class ModelValidationFilter : IAsyncActionFilter
{
    public async Task OnActionExecutionAsync(
        ActionExecutingContext context,
        ActionExecutionDelegate next)
    {
        if (!context.ModelState.IsValid)
        {
            var errors = context.ModelState
                .Where(e => e.Value?.Errors.Count > 0)
                .ToDictionary(
                    kvp => kvp.Key,
                    kvp => kvp.Value!.Errors
                        .Select(e => e.ErrorMessage)
                        .ToArray());

            var result = new ApiResult.ApiResult<Dictionary<string, string[]>>
            {
                Success = false,
                Code = 400,
                Message = "参数验证失败",
                Data = errors,
                TraceId = context.HttpContext.TraceIdentifier,
            };

            context.Result = new ObjectResult(result)
            {
                StatusCode = 400
            };
            return;
        }

        await next();
    }
}
```

```csharp
// ============================================
// Swagger/SwaggerExtensions.cs
// ============================================

using Microsoft.AspNetCore.Builder;
using Microsoft.Extensions.DependencyInjection;
using Microsoft.OpenApi.Models;
using Swashbuckle.AspNetCore.SwaggerGen;

namespace MokFramework.AspNetCore.Swagger;

/// <summary>
/// Swagger 一键配置
/// 
/// 用法:
///   builder.Services.AddPlatformSwagger("SSO 鉴权中心", "v1");
///   app.UsePlatformSwagger();
/// </summary>
public static class SwaggerExtensions
{
    public static IServiceCollection AddPlatformSwagger(
        this IServiceCollection services,
        string title,
        string version = "v1",
        string? description = null,
        Action<SwaggerGenOptions>? configure = null)
    {
        services.AddSwaggerGen(options =>
        {
            options.SwaggerDoc(version, new OpenApiInfo
            {
                Title = title,
                Version = version,
                Description = description ?? $"{title} API 文档",
            });

            // Bearer Token 认证
            options.AddSecurityDefinition("Bearer", new OpenApiSecurityScheme
            {
                Name = "Authorization",
                Type = SecuritySchemeType.Http,
                Scheme = "Bearer",
                BearerFormat = "JWT",
                In = ParameterLocation.Header,
                Description = "输入 JWT Token\n\n示例: eyJhbGciOiJSUzI1NiIs...",
            });

            options.AddSecurityRequirement(new OpenApiSecurityRequirement
            {
                {
                    new OpenApiSecurityScheme
                    {
                        Reference = new OpenApiReference
                        {
                            Type = ReferenceType.SecurityScheme,
                            Id = "Bearer"
                        }
                    },
                    Array.Empty<string>()
                }
            });

            // 枚举显示描述
            options.SchemaFilter<EnumSchemaFilter>();

            // 加载 XML 注释
            LoadXmlComments(options);

            // 自定义配置
            configure?.Invoke(options);
        });

        return services;
    }

    public static IApplicationBuilder UsePlatformSwagger(
        this IApplicationBuilder app,
        string version = "v1")
    {
        app.UseSwagger();
        app.UseSwaggerUI(options =>
        {
            options.SwaggerEndpoint($"/swagger/{version}/swagger.json", version);
            options.RoutePrefix = "swagger";
            options.DocExpansion(
                Swashbuckle.AspNetCore.SwaggerUI.DocExpansion.None);
            options.DefaultModelsExpandDepth(-1); // 折叠 Models
            options.DisplayRequestDuration();
        });

        return app;
    }

    private static void LoadXmlComments(SwaggerGenOptions options)
    {
        // 自动加载所有项目的 XML 注释文件
        var xmlFiles = Directory.GetFiles(
            AppContext.BaseDirectory, "*.xml",
            SearchOption.TopDirectoryOnly);

        foreach (var xmlFile in xmlFiles)
        {
            try
            {
                options.IncludeXmlComments(xmlFile, includeControllerXmlComments: true);
            }
            catch
            {
                // 忽略无法加载的 XML 文件
            }
        }
    }
}
```

```csharp
// ============================================
// Swagger/EnumSchemaFilter.cs
// ============================================

using Microsoft.OpenApi.Any;
using Microsoft.OpenApi.Models;
using Swashbuckle.AspNetCore.SwaggerGen;
using System.ComponentModel;
using System.Reflection;

namespace MokFramework.AspNetCore.Swagger;

/// <summary>
/// 枚举 Schema 过滤器
/// 
/// 让 Swagger 文档中的枚举显示中文描述
/// 
/// 枚举定义:
///   public enum OrderStatus
///   {
///       [Description("待审批")] Pending = 0,
///       [Description("已审批")] Approved = 1,
///   }
///   
/// Swagger 显示:
///   0 = Pending (待审批)
///   1 = Approved (已审批)
/// </summary>
public class EnumSchemaFilter : ISchemaFilter
{
    public void Apply(OpenApiSchema schema, SchemaFilterContext context)
    {
        if (!context.Type.IsEnum) return;

        var descriptions = new List<string>();

        foreach (var value in Enum.GetValues(context.Type))
        {
            var name = Enum.GetName(context.Type, value)!;
            var member = context.Type.GetMember(name).FirstOrDefault();
            var desc = member?.GetCustomAttribute<DescriptionAttribute>()?.Description;

            descriptions.Add(desc != null
                ? $"{(int)value} = {name} ({desc})"
                : $"{(int)value} = {name}");
        }

        schema.Description = string.Join("\n", descriptions);
    }
}
```

```csharp
using Microsoft.AspNetCore.Http;

namespace MokFramework.AspNetCore.Logging;

/// <summary>
/// 链路追踪 ID 中间件
/// 
/// 如果请求头带了 X-Correlation-Id，使用它
/// 否则生成一个新的
/// 写入响应头，方便排查问题
/// </summary>
public class CorrelationIdMiddleware
{
    private const string HeaderName = "X-Correlation-Id";
    private readonly RequestDelegate _next;

    public CorrelationIdMiddleware(RequestDelegate next)
    {
        _next = next;
    }

    public async Task InvokeAsync(HttpContext context)
    {
        // 从请求头获取或生成新的
        var correlationId = context.Request.Headers[HeaderName].FirstOrDefault()
            ?? Guid.NewGuid().ToString("N")[..16]; // 短 ID，够用

        // 设置 TraceIdentifier（全局可用）
        context.TraceIdentifier = correlationId;

        // 写入响应头
        context.Response.OnStarting(() =>
        {
            context.Response.Headers[HeaderName] = correlationId;
            return Task.CompletedTask;
        });

        await _next(context);
    }
}
```

```csharp
// ============================================
// Logging/RequestLoggingMiddleware.cs
// ============================================

using System.Diagnostics;
using Microsoft.AspNetCore.Http;
using Microsoft.Extensions.Logging;

namespace MokFramework.AspNetCore.Logging;

/// <summary>
/// 请求日志中间件
/// 
/// 记录每个请求的:
///   方法 / 路径 / 状态码 / 耗时 / 用户 / IP
///   
/// 日志格式:
///   HTTP GET /api/users → 200 (35ms) | User: admin | IP: 192.168.1.1
/// </summary>
public class RequestLoggingMiddleware
{
    private readonly RequestDelegate _next;
    private readonly ILogger<RequestLoggingMiddleware> _logger;
    private readonly RequestLoggingOptions _options;

    public RequestLoggingMiddleware(
        RequestDelegate next,
        ILogger<RequestLoggingMiddleware> logger,
        RequestLoggingOptions? options = null)
    {
        _next = next;
        _logger = logger;
        _options = options ?? new RequestLoggingOptions();
    }

    public async Task InvokeAsync(HttpContext context)
    {
        // 跳过不需要记录的路径
        var path = context.Request.Path.Value ?? "";
        if (_options.ExcludePaths.Any(p =>
            path.StartsWith(p, StringComparison.OrdinalIgnoreCase)))
        {
            await _next(context);
            return;
        }

        var sw = Stopwatch.StartNew();
        var method = context.Request.Method;
        var traceId = context.TraceIdentifier;

        try
        {
            await _next(context);
            sw.Stop();

            var statusCode = context.Response.StatusCode;
            var userId = context.User?.FindFirst("sub")?.Value ?? "anonymous";
            var ip = GetClientIp(context);

            var level = statusCode >= 500 ? LogLevel.Error
                      : statusCode >= 400 ? LogLevel.Warning
                      : LogLevel.Information;

            _logger.Log(level,
                "HTTP {Method} {Path} → {StatusCode} ({Elapsed}ms) | " +
                "User: {UserId} | IP: {IP} | TraceId: {TraceId}",
                method, path, statusCode, sw.ElapsedMilliseconds,
                userId, ip, traceId);
        }
        catch (Exception ex)
        {
            sw.Stop();
            _logger.LogError(ex,
                "HTTP {Method} {Path} → FAILED ({Elapsed}ms) | " +
                "TraceId: {TraceId}",
                method, path, sw.ElapsedMilliseconds, traceId);
            throw;
        }
    }

    private static string GetClientIp(HttpContext context)
    {
        // 优先从反向代理头获取
        var forwarded = context.Request.Headers["X-Forwarded-For"]
            .FirstOrDefault();
        if (!string.IsNullOrEmpty(forwarded))
            return forwarded.Split(',')[0].Trim();

        var realIp = context.Request.Headers["X-Real-IP"].FirstOrDefault();
        if (!string.IsNullOrEmpty(realIp))
            return realIp;

        return context.Connection.RemoteIpAddress?.ToString() ?? "unknown";
    }
}

public class RequestLoggingOptions
{
    public List<string> ExcludePaths { get; set; } =
    [
        "/swagger",
        "/health",
        "/favicon.ico",
        "/_framework",
    ];
}
```

```csharp
using Microsoft.AspNetCore.Http;

namespace MokFramework.AspNetCore.Security;

/// <summary>
/// 安全响应头中间件
/// 
/// 添加标准安全头，防御常见攻击
/// </summary>
public class SecurityHeadersMiddleware
{
    private readonly RequestDelegate _next;

    public SecurityHeadersMiddleware(RequestDelegate next)
    {
        _next = next;
    }

    public async Task InvokeAsync(HttpContext context)
    {
        // 防止 MIME 类型嗅探
        context.Response.Headers["X-Content-Type-Options"] = "nosniff";

        // 防止点击劫持
        context.Response.Headers["X-Frame-Options"] = "DENY";

        // XSS 保护
        context.Response.Headers["X-XSS-Protection"] = "1; mode=block";

        // 引用策略
        context.Response.Headers["Referrer-Policy"] = "strict-origin-when-cross-origin";

        // 权限策略
        context.Response.Headers["Permissions-Policy"] =
            "camera=(), microphone=(), geolocation=()";

        // 移除 Server 头（不暴露服务器信息）
        context.Response.Headers.Remove("Server");
        context.Response.Headers.Remove("X-Powered-By");

        await _next(context);
    }
}
```

```csharp
// ============================================
// Pagination/PagedResult.cs
// ============================================

namespace MokFramework.AspNetCore.Pagination;

/// <summary>
/// 分页结果
/// 
/// 响应示例:
/// {
///   "success": true,
///   "data": {
///     "items": [...],
///     "totalCount": 100,
///     "page": 1,
///     "pageSize": 20,
///     "totalPages": 5,
///     "hasPrevious": false,
///     "hasNext": true
///   }
/// }
/// </summary>
public class PagedResult<T>
{
    /// <summary>
    /// 数据列表
    /// </summary>
    public List<T> Items { get; set; } = [];

    /// <summary>
    /// 总记录数
    /// </summary>
    public long TotalCount { get; set; }

    /// <summary>
    /// 当前页
    /// </summary>
    public int Page { get; set; }

    /// <summary>
    /// 每页数量
    /// </summary>
    public int PageSize { get; set; }

    /// <summary>
    /// 总页数
    /// </summary>
    public int TotalPages => PageSize > 0
        ? (int)Math.Ceiling((double)TotalCount / PageSize)
        : 0;

    /// <summary>
    /// 是否有上一页
    /// </summary>
    public bool HasPrevious => Page > 1;

    /// <summary>
    /// 是否有下一页
    /// </summary>
    public bool HasNext => Page < TotalPages;

    public PagedResult() { }

    public PagedResult(List<T> items, long totalCount, int page, int pageSize)
    {
        Items = items;
        TotalCount = totalCount;
        Page = page;
        PageSize = pageSize;
    }

    /// <summary>
    /// 空结果
    /// </summary>
    public static PagedResult<T> Empty(int page = 1, int pageSize = 20)
    {
        return new PagedResult<T>([], 0, page, pageSize);
    }

    /// <summary>
    /// 映射到另一个类型
    /// var dtoResult = entityResult.MapTo(entity => new UserDto(entity));
    /// </summary>
    public PagedResult<TTarget> MapTo<TTarget>(Func<T, TTarget> mapper)
    {
        return new PagedResult<TTarget>
        {
            Items = Items.Select(mapper).ToList(),
            TotalCount = TotalCount,
            Page = Page,
            PageSize = PageSize,
        };
    }
}
```

```csharp
using System.Security.Claims;

namespace MokFramework.AspNetCore.Extensions;

/// <summary>
/// ClaimsPrincipal 扩展方法
/// 简化从 User 获取信息的写法
/// </summary>
public static class ClaimsPrincipalExtensions
{
    public static Guid? GetUserId(this ClaimsPrincipal principal)
    {
        var sub = principal.FindFirstValue("sub")
               ?? principal.FindFirstValue(ClaimTypes.NameIdentifier);
        return Guid.TryParse(sub, out var id) ? id : null;
    }

    public static Guid GetRequiredUserId(this ClaimsPrincipal principal)
    {
        return principal.GetUserId()
            ?? throw new UnauthorizedAccessException("无法获取用户ID");
    }

    public static string? GetUserName(this ClaimsPrincipal principal)
    {
        return principal.FindFirstValue("name")
            ?? principal.FindFirstValue(ClaimTypes.Name);
    }

    public static string? GetEmail(this ClaimsPrincipal principal)
    {
        return principal.FindFirstValue("email")
            ?? principal.FindFirstValue(ClaimTypes.Email);
    }

    public static string[] GetRoles(this ClaimsPrincipal principal)
    {
        return principal.FindAll("role")
            .Concat(principal.FindAll(ClaimTypes.Role))
            .Select(c => c.Value)
            .Distinct()
            .ToArray();
    }

    public static bool HasRole(this ClaimsPrincipal principal, string role)
    {
        return principal.GetRoles().Contains(role, StringComparer.OrdinalIgnoreCase);
    }

    public static Guid? GetTenantId(this ClaimsPrincipal principal)
    {
        var value = principal.FindFirstValue("tenant_id");
        return Guid.TryParse(value, out var id) ? id : null;
    }

    public static string? GetClientId(this ClaimsPrincipal principal)
    {
        return principal.FindFirstValue("client_id");
    }

    /// <summary>
    /// 获取自定义 Claim
    /// </summary>
    public static string? FindClaimValue(
        this ClaimsPrincipal principal, string claimType)
    {
        return principal.FindFirstValue(claimType);
    }
```

## 路由约定

```csharp
// ============================================
// Conventions/KebabCaseRouteConvention.cs
// ============================================

using System.Text.RegularExpressions;
using Microsoft.AspNetCore.Mvc.ApplicationModels;

namespace MokFramework.AspNetCore.Conventions;

/// <summary>
/// kebab-case 路由约定
/// 
/// 自动将 Controller 和 Action 名转为 kebab-case
/// 
/// UserRoleController.GetPagedList → /api/user-role/get-paged-list
/// PurchaseOrderController.Create → /api/purchase-order/create
/// </summary>
public partial class KebabCaseRouteConvention : IControllerModelConvention
{
    public void Apply(ControllerModel controller)
    {
        // Controller 名转换
        foreach (var selector in controller.Selectors)
        {
            if (selector.AttributeRouteModel != null)
            {
                selector.AttributeRouteModel.Template =
                    ToKebabCase(selector.AttributeRouteModel.Template);
            }
        }

        // Action 名转换
        foreach (var action in controller.Actions)
        {
            foreach (var selector in action.Selectors)
            {
                if (selector.AttributeRouteModel != null)
                {
                    selector.AttributeRouteModel.Template =
                        ToKebabCase(selector.AttributeRouteModel.Template);
                }
            }
        }
    }

    private static string? ToKebabCase(string? input)
    {
        if (string.IsNullOrEmpty(input)) return input;

        // 保留路由参数 {id} 不转换
        return KebabCaseRegex().Replace(input, match =>
        {
            var value = match.Value;
            if (value.StartsWith('{')) return value; // 路由参数不转换
            return PascalToKebab(value);
        });
    }

    private static string PascalToKebab(string input)
    {
        if (string.IsNullOrEmpty(input)) return input;

        return SplitRegex()
            .Replace(input, "-$1")
            .Trim('-')
            .ToLowerInvariant();
    }

    [GeneratedRegex(@"({[^}]+}|[^{/]+)")]
    private static partial Regex KebabCaseRegex();

    [GeneratedRegex(@"(?<!^)([A-Z][a-z]|(?<=[a-z])[A-Z])")]
    private static partial Regex SplitRegex();
}
```

```csharp
using System.Text.Json;
using System.Text.Json.Serialization;
using Microsoft.AspNetCore.Http;
using Microsoft.AspNetCore.Mvc;
using Microsoft.Extensions.DependencyInjection;
using MokFramework.AspNetCore.ApiResult;
using MokFramework.AspNetCore.Conventions;
using MokFramework.AspNetCore.CurrentUser;
using MokFramework.AspNetCore.Exceptions;
using MokFramework.EntityFrameworkCore;

namespace MokFramework.AspNetCore.Extensions;

/// <summary>
/// 平台 ASP.NET Core 层 DI 注册
/// 
/// 一行代码搞定:
///   builder.Services.AddPlatformAspNetCore();
/// </summary>
public static class ServiceCollectionExtensions
{
    public static IServiceCollection AddPlatformAspNetCore(
        this IServiceCollection services,
        Action<PlatformAspNetCoreOptions>? configure = null)
    {
        var options = new PlatformAspNetCoreOptions();
        configure?.Invoke(options);

        // HttpContextAccessor
        services.AddHttpContextAccessor();

        // ICurrentUser
        services.AddScoped<ICurrentUser, HttpContextCurrentUser>();

        // ApiResult 包装选项
        services.Configure<ApiResultWrapOptions>(opt =>
        {
            opt.AutoWrapSuccessResult = options.AutoWrapResult;
            opt.ShowDetailsInDevelopment = options.ShowExceptionDetails;

            if (options.ExcludePaths?.Count > 0)
                opt.ExcludePaths.AddRange(options.ExcludePaths);
        });

        // Controllers + Filters + JSON 配置
        services.AddControllers(mvcOptions =>
        {
            // 全局过滤器
            mvcOptions.Filters.Add<GlobalExceptionFilter>();
            mvcOptions.Filters.Add<ModelValidationFilter>();

            if (options.AutoWrapResult)
                mvcOptions.Filters.Add<ApiResultFilter>();

            // 路由约定
            if (options.UseKebabCaseRouting)
                mvcOptions.Conventions.Add(new KebabCaseRouteConvention());
        })
        .AddJsonOptions(jsonOptions =>
        {
            // JSON 序列化配置
            jsonOptions.JsonSerializerOptions.PropertyNamingPolicy =
                JsonNamingPolicy.CamelCase;
            jsonOptions.JsonSerializerOptions.DefaultIgnoreCondition =
                JsonIgnoreCondition.WhenWritingNull;
            jsonOptions.JsonSerializerOptions.Converters.Add(
                new JsonStringEnumConverter(JsonNamingPolicy.CamelCase));
            jsonOptions.JsonSerializerOptions.PropertyNameCaseInsensitive = true;

            // DateTime 格式
            jsonOptions.JsonSerializerOptions.Converters.Add(
                new DateTimeJsonConverter());
        });

        // 禁用默认的模型验证（由 ModelValidationFilter 处理）
        services.Configure<ApiBehaviorOptions>(opt =>
        {
            opt.SuppressModelStateInvalidFilter = true;
        });

        // CORS
        if (options.EnableCors)
        {
            services.AddCors(corsOptions =>
            {
                corsOptions.AddDefaultPolicy(policy =>
                {
                    if (options.AllowedOrigins?.Length > 0)
                    {
                        policy.WithOrigins(options.AllowedOrigins);
                    }
                    else
                    {
                        policy.AllowAnyOrigin();
                    }

                    policy.AllowAnyHeader()
                          .AllowAnyMethod();

                    if (options.AllowedOrigins?.Length > 0)
                    {
                        policy.AllowCredentials();
                    }
                });
            });
        }

        return services;
    }
}

/// <summary>
/// DateTime JSON 转换器
/// 统一为 ISO 8601 格式
/// </summary>
public class DateTimeJsonConverter : JsonConverter<DateTime>
{
    private const string Format = "yyyy-MM-ddTHH:mm:ss.fffZ";

    public override DateTime Read(
        ref Utf8JsonReader reader, Type typeToConvert, JsonSerializerOptions options)
    {
        return DateTime.Parse(reader.GetString()!).ToUniversalTime();
    }

    public override void Write(
        Utf8JsonWriter writer, DateTime value, JsonSerializerOptions options)
    {
        writer.WriteStringValue(value.ToUniversalTime().ToString(Format));
    }
}
```

```csharp
using Microsoft.AspNetCore.Builder;

/// <summary>
/// 中间件管道一键配置
/// 
/// 用法:
///   app.UsePlatformPipeline();
/// </summary>
public static class ApplicationBuilderExtensions
{
    public static IApplicationBuilder UsePlatformPipeline(
        this WebApplication app,
        Action<PlatformPipelineOptions>? configure = null)
    {
        var options = new PlatformPipelineOptions();
        configure?.Invoke(options);

        // 1. 链路追踪
        app.UseMiddleware<CorrelationIdMiddleware>();

        // 2. 安全头
        if (options.UseSecurityHeaders)
            app.UseMiddleware<SecurityHeadersMiddleware>();

        // 3. 请求日志
        if (options.UseRequestLogging)
            app.UseMiddleware<RequestLoggingMiddleware>();

        // 4. CORS（必须在认证之前）
        if (options.UseCors)
            app.UseCors();

        // 5. Swagger（开发环境或配置为始终开启）
        if (app.Environment.IsDevelopment() || options.AlwaysEnableSwagger)
            app.UsePlatformSwagger();

        // 6. 路由
        app.UseRouting();

        // 7. 认证
        app.UseAuthentication();

        // 8. 授权
        app.UseAuthorization();

        // 9. 端点
        app.MapControllers();

        // 10. 健康检查
        if (options.UseHealthChecks)
            app.MapHealthChecks("/health");

        return app;
    }
}

public class PlatformPipelineOptions
{
    public bool UseSecurityHeaders { get; set; } = true;
    public bool UseRequestLogging { get; set; } = true;
    public bool UseCors { get; set; } = true;
    public bool AlwaysEnableSwagger { get; set; } = false;
```

```csharp
// ============================================
// Controller 的写法（极简）
// ============================================

[ApiController]
[Route("api/[controller]")]
public class UsersController : ControllerBase
{
    private readonly IRepository<SsoUser, Guid> _userRepo;
    private readonly IUnitOfWork _uow;

    public UsersController(
        IRepository<SsoUser, Guid> userRepo,
        IUnitOfWork uow)
    {
        _userRepo = userRepo;
        _uow = uow;
    }

    /// <summary>
    /// 分页查询用户
    /// </summary>
    [HttpGet]
    public async Task<PagedResult<UserDto>> GetList(
        [FromQuery] PagedRequest request)
    {
        var query = _userRepo.AsQueryable();

        if (!string.IsNullOrEmpty(request.Keyword))
        {
            query = query.Where(u =>
                u.UserName.Contains(request.Keyword) ||
                u.Email.Contains(request.Keyword));
        }

        var totalCount = await query.LongCountAsync();
        var items = await query
            .OrderByDescending(u => u.CreationTime)
            .Skip(request.SkipCount)
            .Take(request.PageSize)
            .Select(u => new UserDto
            {
                Id = u.Id,
                UserName = u.UserName,
                Email = u.Email,
                CreationTime = u.CreationTime,
            })
            .ToListAsync();

        return new PagedResult<UserDto>(
            items, totalCount, request.Page, request.PageSize);
    }

    // 返回值自动包装为:
    // {
    //   "success": true,
    //   "code": 200,
    //   "message": "操作成功",
    //   "data": {
    //     "items": [...],
    //     "totalCount": 100,
    //     "page": 1,
    //     "pageSize": 20,
    //     "totalPages": 5,
    //     "hasNext": true
    //   },
    //   "traceId": "abc123"
    // }

    /// <summary>
    /// 获取用户详情
    /// </summary>
    [HttpGet("{id:guid}")]
    public async Task<UserDto> Get(Guid id)
    {
        var user = await _userRepo.GetAsync(id);
        // 如果不存在 → 自动抛 EntityNotFoundException → 自动返回 404

        return new UserDto
        {
            Id = user.Id,
            UserName = user.UserName,
            Email = user.Email,
        };
    }

    /// <summary>
    /// 创建用户
    /// </summary>
    [HttpPost]
    public async Task<UserDto> Create([FromBody] CreateUserRequest request)
    {
        // 模型验证失败 → ModelValidationFilter 自动返回 400

        // 业务校验
        var exists = await _userRepo.AnyAsync(
            u => u.Email == request.Email);
        if (exists)
            throw new BusinessException("SSO:User:DuplicateEmail", "邮箱已存在");
        // → GlobalExceptionFilter 自动返回 400 + 错误码

        var user = new SsoUser(Guid.NewGuid(), request.UserName, request.Email);
        await _userRepo.InsertAsync(user);
        await _uow.SaveChangesAsync();
        // → AuditingInterceptor 自动填充 CreationTime + CreatorId
        // → DomainEventInterceptor 自动分发领域事件

        return new UserDto { Id = user.Id, UserName = user.UserName };
    }

    /// <summary>
    /// 删除用户
    /// </summary>
    [HttpDelete("{id:guid}")]
    public async Task Delete(Guid id)
    {
        await _userRepo.DeleteAsync(id);
        await _uow.SaveChangesAsync();
        // → SoftDeleteInterceptor 自动转为软删除
        // → 设置 IsDeleted=true, DeletionTime, DeleterId
        // → 不会真正删除数据

        // 返回 void → ApiResultFilter 自动包装为:
        // { "success": true, "code": 200, "message": "操作成功" }
    }
}
```
