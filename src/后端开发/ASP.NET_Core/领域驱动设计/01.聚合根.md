# 聚合根 MokFramework.Domain 实现

---

## 设计目标

```
对标 ABP 的 Volo.Abp.Domain.Shared + Volo.Abp.Domain

但是:
  ❌ 不依赖 ABP 框架
  ❌ 不引入 ABP 的模块系统、依赖注入体系
  ✅ 只提取精髓: 实体、聚合根、审计、软删除、领域事件、值对象
  ✅ 生产级: 并发控制、序列化友好、EF Core 深度集成
  ✅ 所有项目共享这一个 NuGet 包
```

```csharp
// ============================================
// Data/ExtraPropertyDictionary.cs
// ============================================

using System.Text.Json;
using System.Text.Json.Serialization;

namespace MokFramework.Data;

/// <summary>
/// 扩展属性字典
/// 继承 Dictionary 以便 JSON 序列化
/// 提供类型安全的存取方法
/// </summary>
[JsonConverter(typeof(ExtraPropertyDictionaryJsonConverter))]
public class ExtraPropertyDictionary : Dictionary<string, object?>
{
    public ExtraPropertyDictionary() : base(StringComparer.Ordinal) { }

    public ExtraPropertyDictionary(IDictionary<string, object?> dictionary)
        : base(dictionary, StringComparer.Ordinal) { }

    /// <summary>
    /// 类型安全获取
    /// var score = extraProps.GetProperty<int>("Score");
    /// </summary>
    public T? GetProperty<T>(string name, T? defaultValue = default)
    {
        if (!TryGetValue(name, out var value) || value == null)
            return defaultValue;

        if (value is T typed)
            return typed;

        // JSON 反序列化时可能是 JsonElement
        if (value is JsonElement element)
        {
            return JsonSerializer.Deserialize<T>(element.GetRawText());
        }

        try
        {
            return (T)Convert.ChangeType(value, typeof(T));
        }
        catch
        {
            return defaultValue;
        }
    }

    /// <summary>
    /// 设置属性
    /// </summary>
    public void SetProperty<T>(string name, T? value)
    {
        this[name] = value;
    }

    /// <summary>
    /// 移除属性
    /// </summary>
    public bool RemoveProperty(string name)
    {
        return Remove(name);
    }

    /// <summary>
    /// 是否有属性
    /// </summary>
    public bool HasProperty(string name)
    {
        return ContainsKey(name);
    }
}

/// <summary>
/// ExtraPropertyDictionary 的 JSON 转换器
/// 确保序列化/反序列化正确
/// </summary>
public class ExtraPropertyDictionaryJsonConverter
    : JsonConverter<ExtraPropertyDictionary>
{
    public override ExtraPropertyDictionary? Read(
        ref Utf8JsonReader reader,
        Type typeToConvert,
        JsonSerializerOptions options)
    {
        var dict = JsonSerializer.Deserialize<Dictionary<string, object?>>(
            ref reader, options);
        return dict == null ? new() : new ExtraPropertyDictionary(dict);
    }

    public override void Write(
        Utf8JsonWriter writer,
        ExtraPropertyDictionary value,
        JsonSerializerOptions options)
    {
        JsonSerializer.Serialize(writer,
            (Dictionary<string, object?>)value, options);
    }
}
```

```csharp
// ============================================
// Entities/Entity.cs
// ============================================

namespace MokFramework.Entities;

/// <summary>
/// 实体基类（无主键）
/// 用于组合主键场景
/// 
/// ABP 的 Entity 复刻
/// </summary>
public abstract class Entity : IEntity
{
    public abstract object?[] GetKeys();

    public override string ToString()
    {
        return $"[ENTITY: {GetType().Name}] Keys = {string.Join(", ", GetKeys())}";
    }

    /// <summary>
    /// 实体相等性: 基于主键比较
    /// 两个实体主键相同则相等（即使其他属性不同）
    /// 这是 DDD 的核心概念
    /// </summary>
    public override bool Equals(object? obj)
    {
        if (obj is not Entity other)
            return false;

        if (ReferenceEquals(this, other))
            return true;

        if (GetType() != other.GetType())
            return false;

        var keys = GetKeys();
        var otherKeys = other.GetKeys();

        if (keys.Length != otherKeys.Length)
            return false;

        for (var i = 0; i < keys.Length; i++)
        {
            if (!Equals(keys[i], otherKeys[i]))
                return false;
        }

        return true;
    }

    public override int GetHashCode()
    {
        var keys = GetKeys();
        if (keys.Length == 0)
            return base.GetHashCode();

        var hash = new HashCode();
        foreach (var key in keys)
        {
            hash.Add(key);
        }
        return hash.ToHashCode();
    }

    public static bool operator ==(Entity? left, Entity? right)
    {
        return Equals(left, right);
    }

    public static bool operator !=(Entity? left, Entity? right)
    {
        return !Equals(left, right);
    }
}
```

```csharp
using MokFramework.Concurrency;
using MokFramework.Data;
using MokFramework.Events;

namespace MokFramework.Entities;

/// <summary>
/// 聚合根接口
/// 
/// DDD 核心概念:
///   聚合根是聚合的入口
///   外部只能通过聚合根访问聚合内部
///   聚合根负责维护不变量（业务规则）
///   聚合根是持久化的边界
///   聚合根是事务的边界
///   
/// ABP 的聚合根额外带:
///   并发戳（乐观并发控制）
///   扩展属性（动态属性）
///   领域事件（解耦副作用）
/// </summary>
public interface IAggregateRoot : IEntity, IHasConcurrencyStamp, IHasExtraProperties
{
    /// <summary>
    /// 获取领域事件（只读）
    /// </summary>
    IReadOnlyCollection<IDomainEvent> DomainEvents { get; }

    /// <summary>
    /// 清除领域事件（发布后调用）
    /// </summary>
    void ClearDomainEvents();
}

public interface IAggregateRoot<TKey> : IAggregateRoot, IEntity<TKey>
    where TKey : notnull
{
}
```

```csharp
// ============================================
// Entities/AggregateRoot.cs
// ============================================

using MokFramework.Concurrency;
using MokFramework.Data;
using MokFramework.Events;

namespace MokFramework.Entities;

/// <summary>
/// 聚合根基类（无主键）
/// </summary>
public abstract class AggregateRoot : Entity, IAggregateRoot
{
    // ============ 并发控制 ============
    
    /// <inheritdoc />
    public virtual string ConcurrencyStamp { get; set; } = Guid.NewGuid().ToString("N");

    // ============ 扩展属性 ============
    
    /// <inheritdoc />
    public virtual ExtraPropertyDictionary ExtraProperties { get; protected set; } = new();

    // ============ 领域事件 ============
    
    private readonly List<IDomainEvent> _domainEvents = [];

    /// <inheritdoc />
    public IReadOnlyCollection<IDomainEvent> DomainEvents => _domainEvents.AsReadOnly();

    /// <summary>
    /// 添加领域事件
    /// 
    /// 用法（在实体方法内）:
    ///   public void Approve(Guid approverId)
    ///   {
    ///       Status = OrderStatus.Approved;
    ///       AddDomainEvent(new OrderApprovedEvent(Id, approverId));
    ///   }
    /// </summary>
    protected void AddDomainEvent(IDomainEvent domainEvent)
    {
        _domainEvents.Add(domainEvent);
    }

    /// <summary>
    /// 移除领域事件
    /// </summary>
    protected void RemoveDomainEvent(IDomainEvent domainEvent)
    {
        _domainEvents.Remove(domainEvent);
    }

    /// <inheritdoc />
    public void ClearDomainEvents()
    {
        _domainEvents.Clear();
    }
}
```

```csharp
// ============================================
// ValueObjects/ValueObject.cs

namespace MokFramework.ValueObjects;

/// <summary>
/// 值对象基类
/// 
/// DDD 核心概念:
///   值对象没有唯一标识
///   两个值对象的所有属性相同则相等
///   值对象是不可变的
///   
/// 用法:
///   public class Address : ValueObject
///   {
///       public string Province { get; private set; }
///       public string City { get; private set; }
///       public string Street { get; private set; }
///       public string ZipCode { get; private set; }
///       
///       public Address(string province, string city, string street, string zipCode)
///       {
///           Province = province;
///           City = city;
///           Street = street;
///           ZipCode = zipCode;
///       }
///       
///       protected override IEnumerable&lt;object?&gt; GetAtomicValues()
///       {
///           yield return Province;
///           yield return City;
///           yield return Street;
///           yield return ZipCode;
///       }
///   }
///   
///   var addr1 = new Address("浙江", "杭州", "西湖路", "310000");
///   var addr2 = new Address("浙江", "杭州", "西湖路", "310000");
///   addr1 == addr2 // true（值相同）
/// </summary>
public abstract class ValueObject : IEquatable<ValueObject>
{
    /// <summary>
    /// 获取值对象的所有属性值（用于比较）
    /// 子类必须实现
    /// </summary>
    protected abstract IEnumerable<object?> GetAtomicValues();

    public bool Equals(ValueObject? other)
    {
        if (other is null || GetType() != other.GetType())
            return false;

        return GetAtomicValues().SequenceEqual(other.GetAtomicValues());
    }

    public override bool Equals(object? obj)
    {
        return obj is ValueObject other && Equals(other);
    }

    public override int GetHashCode()
    {
        return GetAtomicValues()
            .Aggregate(new HashCode(), (hash, value) =>
            {
                hash.Add(value);
                return hash;
            })
            .ToHashCode();
    }

    public static bool operator ==(ValueObject? left, ValueObject? right)
    {
        return Equals(left, right);
    }

    public static bool operator !=(ValueObject? left, ValueObject? right)
    {
        return !Equals(left, right);
    }

    /// <summary>
    /// 值对象是不可变的，"修改"通过创建新实例实现
    /// </summary>
    public virtual ValueObject Clone()
    {
        return (ValueObject)MemberwiseClone();
    }
}
```

```csharp
// ============================================
// Specifications/Specification.cs
// ============================================

using System.Linq.Expressions;

namespace MokFramework.Specifications;

/// <summary>
/// 规约基类
/// 
/// 支持组合: And (&amp;) / Or (|) / Not (!)
/// 
/// 用法:
///   public class ActiveOrderSpec : Specification&lt;Order&gt;
///   {
///       public override Expression&lt;Func&lt;Order, bool&gt;&gt; ToExpression()
///       {
///           return o => o.Status == OrderStatus.Active && !o.IsDeleted;
///       }
///   }
///   
///   public class HighValueOrderSpec : Specification&lt;Order&gt;
///   {
///       private readonly decimal _minAmount;
///       public HighValueOrderSpec(decimal minAmount) => _minAmount = minAmount;
///       
///       public override Expression&lt;Func&lt;Order, bool&gt;&gt; ToExpression()
///       {
///           return o => o.TotalAmount >= _minAmount;
///       }
///   }
///   
///   // 组合使用:
///   var spec = new ActiveOrderSpec() & new HighValueOrderSpec(1000);
///   var orders = await repo.GetListAsync(spec.ToExpression());
/// </summary>
public abstract class Specification<T> : ISpecification<T>
{
    public virtual bool IsSatisfiedBy(T entity)
    {
        return ToExpression().Compile()(entity);
    }

    public abstract Expression<Func<T, bool>> ToExpression();

    /// <summary>
    /// 隐式转换为 Expression
    /// 方便直接传给 Where
    /// </summary>
    public static implicit operator Expression<Func<T, bool>>(Specification<T> spec)
    {
        return spec.ToExpression();
    }

    /// <summary>
    /// And 组合
    /// </summary>
    public static Specification<T> operator &(Specification<T> left, Specification<T> right)
    {
        return new AndSpecification<T>(left, right);
    }

    /// <summary>
    /// Or 组合
    /// </summary>
    public static Specification<T> operator |(Specification<T> left, Specification<T> right)
    {
        return new OrSpecification<T>(left, right);
    }

    /// <summary>
    /// Not
    /// </summary>
    public static Specification<T> operator !(Specification<T> spec)
    {
        return new NotSpecification<T>(spec);
    }
}

// ============================================
// 组合规约
// ============================================

internal class AndSpecification<T> : Specification<T>
{
    private readonly Specification<T> _left;
    private readonly Specification<T> _right;

    public AndSpecification(Specification<T> left, Specification<T> right)
    {
        _left = left;
        _right = right;
    }

    public override Expression<Func<T, bool>> ToExpression()
    {
        var leftExpr = _left.ToExpression();
        var rightExpr = _right.ToExpression();

        var parameter = Expression.Parameter(typeof(T));

        var body = Expression.AndAlso(
            Expression.Invoke(leftExpr, parameter),
            Expression.Invoke(rightExpr, parameter));

        return Expression.Lambda<Func<T, bool>>(body, parameter);
    }
}

internal class OrSpecification<T> : Specification<T>
{
    private readonly Specification<T> _left;
    private readonly Specification<T> _right;

    public OrSpecification(Specification<T> left, Specification<T> right)
    {
        _left = left;
        _right = right;
    }

    public override Expression<Func<T, bool>> ToExpression()
    {
        var leftExpr = _left.ToExpression();
        var rightExpr = _right.ToExpression();

        var parameter = Expression.Parameter(typeof(T));

        var body = Expression.OrElse(
            Expression.Invoke(leftExpr, parameter),
            Expression.Invoke(rightExpr, parameter));

        return Expression.Lambda<Func<T, bool>>(body, parameter);
    }
}

internal class NotSpecification<T> : Specification<T>
{
    private readonly Specification<T> _spec;

    public NotSpecification(Specification<T> spec)
    {
        _spec = spec;
    }

    public override Expression<Func<T, bool>> ToExpression()
    {
        var expr = _spec.ToExpression();
        var parameter = Expression.Parameter(typeof(T));

        var body = Expression.Not(Expression.Invoke(expr, parameter));

        return Expression.Lambda<Func<T, bool>>(body, parameter);
    }
}
```

```csharp
// ============================================
// Guids/SequentialGuidGenerator.cs
// ============================================

using System.Security.Cryptography;

namespace MokFramework.Guids;

/// <summary>
/// 有序 GUID 生成器
/// 
/// 为什么不用 Guid.NewGuid():
///   普通 GUID 完全随机 → 数据库索引分裂 → 插入性能差
///   有序 GUID 前几位有序 → 索引连续 → 插入性能提升 5-10 倍
///   
/// ABP 的 SequentialGuidGenerator 复刻
/// 支持不同数据库的排序策略
/// </summary>
public class SequentialGuidGenerator : IGuidGenerator
{
    private readonly SequentialGuidType _guidType;

    /// <summary>
    /// 默认使用 AtEnd（适合 SQL Server / PostgreSQL）
    /// </summary>
    public SequentialGuidGenerator(
        SequentialGuidType guidType = SequentialGuidType.AtEnd)
    {
        _guidType = guidType;
    }

    public Guid Create()
    {
        return Create(_guidType);
    }

    /// <summary>
    /// 生成有序 GUID
    /// 参考: https://github.com/jhtodd/SequentialGuid
    /// </summary>
    public static Guid Create(SequentialGuidType guidType)
    {
        // 随机部分（10字节）
        var randomBytes = new byte[10];
        RandomNumberGenerator.Fill(randomBytes);

        // 时间部分（6字节，毫秒精度）
        var timestamp = DateTime.UtcNow.Ticks / 10000L;
        var timestampBytes = BitConverter.GetBytes(timestamp);

        // 只取低 6 字节
        if (BitConverter.IsLittleEndian)
            Array.Reverse(timestampBytes);

        var tsBytes = new byte[6];
        Buffer.BlockCopy(timestampBytes, 2, tsBytes, 0, 6);

        var guidBytes = new byte[16];

        switch (guidType)
        {
            case SequentialGuidType.AtBegin:
                // 时间戳放前面（适合 MySQL）
                Buffer.BlockCopy(tsBytes, 0, guidBytes, 0, 6);
                Buffer.BlockCopy(randomBytes, 0, guidBytes, 6, 10);
                break;

            case SequentialGuidType.AtEnd:
                // 时间戳放后面（适合 SQL Server / PostgreSQL）
                Buffer.BlockCopy(randomBytes, 0, guidBytes, 0, 10);
                Buffer.BlockCopy(tsBytes, 0, guidBytes, 10, 6);
                break;

            case SequentialGuidType.AsBinary:
                // 二进制排序（适合 Oracle）
                Buffer.BlockCopy(tsBytes, 0, guidBytes, 0, 6);
                Buffer.BlockCopy(randomBytes, 0, guidBytes, 6, 10);
                // 调整字节序
                Array.Reverse(guidBytes, 0, 4);
                Array.Reverse(guidBytes, 4, 2);
                break;
        }

        return new Guid(guidBytes);
    }
}

/// <summary>
/// 有序 GUID 类型
/// 不同数据库对 GUID 的排序方式不同
/// </summary>
public enum SequentialGuidType
{
    /// <summary>
    /// 时间戳在前（MySQL / SQLite）
    /// </summary>
    AtBegin,

    /// <summary>
    /// 时间戳在后（SQL Server / PostgreSQL）
    /// 默认推荐
    /// </summary>
    AtEnd,

    /// <summary>
    /// 二进制排序（Oracle）
    /// </summary>
    AsBinary,
}
```

```csharp
public class Money : ValueObject
{
    public decimal Amount { get; private set; }
    public string Currency { get; private set; }

    public Money(decimal amount, string currency = "CNY")
    {
        if (amount < 0)
            throw new BusinessException("PM:Money:NegativeAmount");
        Amount = amount;
        Currency = currency;
    }

    public Money Add(Money other)
    {
        if (Currency != other.Currency)
            throw new BusinessException("PM:Money:CurrencyMismatch");
        return new Money(Amount + other.Amount, Currency);
    }

    protected override IEnumerable<object?> GetAtomicValues()
    {
        yield return Amount;
        yield return Currency;
    }
}
```
