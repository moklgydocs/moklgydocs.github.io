## 第零步：MediatR vs 自研——决策

```
┌──────────────────┬──────────────────────┬─────────────────────┐
│ 维度             │ 直接用 MediatR        │ 自研                │
├──────────────────┼──────────────────────┼─────────────────────┤
│ 开发成本         │ 0（拿来就用）          │ 3-5天               │
│ 管道行为         │ 内置 IPipelineBehavior │ 自己实现             │
│ 社区生态         │ GitHub 10k+ stars     │ 无                  │
│ 性能             │ 有反射开销（微小）      │ 可以极致优化         │
│ 学习成本         │ 团队基本都会           │ 需要学你的API        │
│ 定制自由度       │ 中（够用）             │ 完全自由             │
│ 维护成本         │ 第三方维护             │ 自己维护             │
│ 和你的UoW集成    │ 写个Behavior即可       │ 写个Behavior即可     │
│ 和领域事件集成   │ 天然支持               │ 天然支持             │
│ 版本升级风险     │ 有（但MediatR很稳定）  │ 无                  │
├──────────────────┴──────────────────────┴─────────────────────┤
│                                                               │
│  结论：用 MediatR 做底层传输，上面包一层你自己的抽象             │
│                                                               │
│  理由：                                                        │
│  1. 你团队3人，业务开发为主，不应该花时间造消息管道             │
│  2. MediatR 的 Pipeline Behavior 机制成熟，直接用              │
│  3. 包一层自己的 ICommand/IQuery 接口，业务代码不直接依赖MediatR │
│  4. 未来要换掉 MediatR，只改 Platform.Cqrs，业务代码零改动     │
│                                                               │
│  这就是 ABP 的做法：底层用MediatR，上层包抽象                   │
└───────────────────────────────────────────────────────────────┘
```

```csharp
using MediatR;

namespace Platform.Cqrs.Commands;

/// <summary>
/// 命令接口（无返回值）
/// 
/// 语义：改变系统状态的操作
/// 例：CreateOrder, DeleteUser, ApproveRequest
/// 
/// 为什么继承 MediatR.IRequest：
///   底层用 MediatR 的管道机制
///   但业务代码只看到 ICommand，不直接依赖 MediatR
///   未来换掉 MediatR，业务代码不改
/// </summary>
public interface ICommand : ICommand<Unit>
{
}

/// <summary>
/// 命令接口（有返回值）
/// 
/// 例：CreateOrder → 返回 Guid（新订单ID）
///     ImportUsers → 返回 int（导入数量）
/// </summary>
public interface ICommand<out TResult> : IRequest<TResult>
{
}
```

```csharp
// ============================================
// Commands/ICommandHandler.cs
// ============================================

using MediatR;

namespace Platform.Cqrs.Commands;

/// <summary>
/// 命令处理器（无返回值）
/// 
/// 用法：
///   public class DeleteUserHandler : ICommandHandler&lt;DeleteUserCommand&gt;
///   {
///       public async Task HandleAsync(DeleteUserCommand command, CancellationToken ct)
///       {
///           // 删除逻辑
///       }
///   }
/// </summary>
public interface ICommandHandler<in TCommand> : ICommandHandler<TCommand, Unit>
    where TCommand : ICommand
{
}

/// <summary>
/// 命令处理器（有返回值）
/// 
/// 用法：
///   public class CreateOrderHandler : ICommandHandler&lt;CreateOrderCommand, Guid&gt;
///   {
///       public async Task&lt;Guid&gt; HandleAsync(CreateOrderCommand command, CancellationToken ct)
///       {
///           var order = new Order(...);
///           await _repo.InsertAsync(order);
///           return order.Id;
///       }
///   }
///   
/// 注意：不需要手动调 SaveChangesAsync
///       UnitOfWorkBehavior 会在 Handler 执行后自动提交
/// </summary>
public interface ICommandHandler<in TCommand, TResult> : IRequestHandler<TCommand, TResult>
    where TCommand : ICommand<TResult>
{
    /// <summary>
    /// 处理命令
    /// </summary>
    new Task<TResult> HandleAsync(TCommand command, CancellationToken cancellationToken = default);

    // 桥接 MediatR 的 Handle 方法
    async Task<TResult> IRequestHandler<TCommand, TResult>.Handle(
        TCommand request, CancellationToken cancellationToken)
    {
        return await HandleAsync(request, cancellationToken);
    }
}
```

```csharp
// ============================================

namespace Platform.Cqrs.Commands;
```

```csharp
namespace Platform.Cqrs.Mediator;

        Queries.IQuery<TResult> query,
```

```csharp
        await _mediator.Send(command, cancellationToken);

        IQuery<TResult> query,
        CancellationToken cancellationToken = default)
```

### 日志管道

```csharp
// ============================================
// Behaviors/LoggingBehavior.cs
// ============================================

using System.Text.Json;
using MediatR;
using Microsoft.Extensions.Logging;
using Platform.Cqrs.Commands;

namespace Platform.Cqrs.Behaviors;

/// <summary>
/// 日志管道行为
/// 
/// 记录每个 Command/Query 的执行：
///   开始：请求类型 + 参数
///   成功：耗时
///   失败：异常信息
///   
/// Command 和 Query 都走此管道
/// </summary>
public class LoggingBehavior<TRequest, TResponse>
    : IPipelineBehavior<TRequest, TResponse>
    where TRequest : notnull
{
    private readonly ILogger<LoggingBehavior<TRequest, TResponse>> _logger;

    public LoggingBehavior(
        ILogger<LoggingBehavior<TRequest, TResponse>> logger)
    {
        _logger = logger;
    }

    public async Task<TResponse> Handle(
        TRequest request,
        RequestHandlerDelegate<TResponse> next,
        CancellationToken cancellationToken)
    {
        var requestName = typeof(TRequest).Name;
        var isCommand = request is ICommand<TResponse>;
        var kind = isCommand ? "Command" : "Query";

        _logger.LogInformation(
            "[CQRS] {Kind} Start: {RequestName} {@Request}",
            kind, requestName, SanitizeForLog(request));

        try
        {
            var response = await next();

            _logger.LogInformation(
                "[CQRS] {Kind} Success: {RequestName}",
                kind, requestName);

            return response;
        }
        catch (Exception ex)
        {
            _logger.LogError(ex,
                "[CQRS] {Kind} Failed: {RequestName} - {Error}",
                kind, requestName, ex.Message);
            throw;
        }
    }

    /// <summary>
    /// 脱敏处理（隐藏密码等敏感字段）
    /// </summary>
    private static object SanitizeForLog(TRequest request)
    {
        try
        {
            var json = JsonSerializer.Serialize(request);
            // 简单脱敏：替换 password/secret/token 等字段
            json = System.Text.RegularExpressions.Regex.Replace(
                json,
                @"""(password|secret|token|accessToken|refreshToken)"":\s*""[^""]*""",
                @"""$1"":""***""",
                System.Text.RegularExpressions.RegexOptions.IgnoreCase);
            return json;
        }
        catch
        {
            return request?.GetType().Name ?? "Unknown";
        }
    }
}
```

### 性能监控管道

```csharp
// ============================================
// Behaviors/PerformanceBehavior.cs
// ============================================

using System.Diagnostics;
using MediatR;
using Microsoft.Extensions.Logging;

namespace Platform.Cqrs.Behaviors;

/// <summary>
/// 性能监控管道
/// 
/// 功能：
///   记录每个请求的耗时
///   超过阈值（默认500ms）发出警告
///   可配合 APM 工具使用
/// </summary>
public class PerformanceBehavior<TRequest, TResponse>
    : IPipelineBehavior<TRequest, TResponse>
    where TRequest : notnull
{
    private readonly ILogger<PerformanceBehavior<TRequest, TResponse>> _logger;
    private readonly CqrsOptions _options;

    public PerformanceBehavior(
        ILogger<PerformanceBehavior<TRequest, TResponse>> logger,
        Microsoft.Extensions.Options.IOptions<CqrsOptions> options)
    {
        _logger = logger;
        _options = options.Value;
    }

    public async Task<TResponse> Handle(
        TRequest request,
        RequestHandlerDelegate<TResponse> next,
        CancellationToken cancellationToken)
    {
        var sw = Stopwatch.StartNew();

        var response = await next();

        sw.Stop();
        var elapsed = sw.ElapsedMilliseconds;

        if (elapsed > _options.SlowRequestThresholdMs)
        {
            _logger.LogWarning(
                "[CQRS] SLOW Request: {RequestName} took {Elapsed}ms " +
                "(Threshold: {Threshold}ms)",
                typeof(TRequest).Name,
                elapsed,
                _options.SlowRequestThresholdMs);
        }
        else
        {
            _logger.LogDebug(
                "[CQRS] {RequestName} completed in {Elapsed}ms",
                typeof(TRequest).Name, elapsed);
        }

        return response;
    }
}
```

### 验证管道

```csharp
// ============================================
// Behaviors/ValidationBehavior.cs
// ============================================

using FluentValidation;
using MediatR;
using Platform.Cqrs.Validation;

namespace Platform.Cqrs.Behaviors;

/// <summary>
/// FluentValidation 验证管道
/// 
/// 在 Handler 执行前自动验证请求参数
/// 
/// 工作原理：
///   1. 从 DI 容器获取 IValidator&lt;TRequest&gt;
///   2. 如果有验证器，执行验证
///   3. 验证失败 → 抛 CqrsValidationException（不到 Handler）
///   4. 验证通过 → 继续执行 Handler
///   
/// 用法：
///   // 定义命令
///   public record CreateOrderCommand(string ProductName, int Qty) : ICommand&lt;Guid&gt;;
///   
///   // 定义验证器（和命令放在一起）
///   public class CreateOrderCommandValidator : AbstractValidator&lt;CreateOrderCommand&gt;
///   {
///       public CreateOrderCommandValidator()
///       {
///           RuleFor(x => x.ProductName).NotEmpty().MaximumLength(200);
///           RuleFor(x => x.Qty).GreaterThan(0);
///       }
///   }
///   
///   // 不需要其他配置，自动生效 ✅
/// </summary>
public class ValidationBehavior<TRequest, TResponse>
    : IPipelineBehavior<TRequest, TResponse>
    where TRequest : notnull
{
    private readonly IEnumerable<IValidator<TRequest>> _validators;

    public ValidationBehavior(IEnumerable<IValidator<TRequest>> validators)
    {
        _validators = validators;
    }

    public async Task<TResponse> Handle(
        TRequest request,
        RequestHandlerDelegate<TResponse> next,
        CancellationToken cancellationToken)
    {
        if (!_validators.Any())
            return await next();

        var context = new ValidationContext<TRequest>(request);

        // 并行执行所有验证器
        var validationResults = await Task.WhenAll(
            _validators.Select(v => v.ValidateAsync(context, cancellationToken)));

        var failures = validationResults
            .SelectMany(r => r.Errors)
            .Where(f => f != null)
            .ToList();

        if (failures.Count > 0)
        {
            throw new CqrsValidationException(failures);
        }

        return await next();
    }
}
```

### UnitOfWork 管道（核心）

```csharp
// ============================================
// Behaviors/UnitOfWorkBehavior.cs
// ============================================

using MediatR;
using Microsoft.Extensions.Logging;
using Platform.Cqrs.Commands;
using Platform.Cqrs.Markers;
using Platform.Cqrs.Queries;
using Platform.EntityFrameworkCore.UnitOfWork;

namespace Platform.Cqrs.Behaviors;

/// <summary>
/// UnitOfWork 管道行为（最核心的管道）
/// 
/// 规则：
///   ICommand → 自动 SaveChanges（除非标记 INoUnitOfWork）
///   ICommand + ITransactional → 包事务（Begin + Commit/Rollback）
///   IQuery → 什么都不做（查询不需要提交）
///   
/// 为什么不在 Handler 里手动 SaveChanges：
///   1. 减少重复代码（每个Handler都要写一遍）
///   2. 保证一致性（不会忘记提交）
///   3. 事务边界统一管理
///   4. 和 EF Core 的领域事件拦截器配合：
///      SaveChanges → 触发 DomainEventInterceptor → 分发事件
///      
/// 和现有 EfCoreUnitOfWork 的关系：
///   直接用 IUnitOfWork（已经注册好了）
///   SaveChanges → EF Core 的 SaveChangesInterceptor 链：
///     AuditingInterceptor → 填审计字段
///     SoftDeleteInterceptor → 软删除
///     ConcurrencyStampInterceptor → 刷新并发戳
///     DomainEventInterceptor → 分发领域事件（SavedChanges 后）
///   一切自动 ✅
/// </summary>
public class UnitOfWorkBehavior<TRequest, TResponse>
    : IPipelineBehavior<TRequest, TResponse>
    where TRequest : notnull
{
    private readonly IUnitOfWork _unitOfWork;
    private readonly ILogger<UnitOfWorkBehavior<TRequest, TResponse>> _logger;

    public UnitOfWorkBehavior(
        IUnitOfWork unitOfWork,
        ILogger<UnitOfWorkBehavior<TRequest, TResponse>> logger)
    {
        _unitOfWork = unitOfWork;
        _logger = logger;
    }

    public async Task<TResponse> Handle(
        TRequest request,
        RequestHandlerDelegate<TResponse> next,
        CancellationToken cancellationToken)
    {
        // Query → 直接执行，不走 UoW
        if (request is not ICommand<TResponse>)
        {
            return await next();
        }

        // 标记了 INoUnitOfWork → 直接执行
        if (request is INoUnitOfWork)
        {
            _logger.LogDebug(
                "[CQRS] Skipping UoW for {RequestName} (INoUnitOfWork)",
                typeof(TRequest).Name);
            return await next();
        }

        // 标记了 ITransactional → 包事务
        if (request is ITransactional)
        {
            return await ExecuteWithTransactionAsync(next, cancellationToken);
        }

        // 普通 Command → 只 SaveChanges
        return await ExecuteWithSaveChangesAsync(next, cancellationToken);
    }

    /// <summary>
    /// 普通命令：Handler 执行后 SaveChanges
    /// </summary>
    private async Task<TResponse> ExecuteWithSaveChangesAsync(
        RequestHandlerDelegate<TResponse> next,
        CancellationToken cancellationToken)
    {
        var response = await next();

        var changes = await _unitOfWork.SaveChangesAsync(cancellationToken);

        _logger.LogDebug(
            "[CQRS] SaveChanges for {RequestName}: {Changes} entities affected",
            typeof(TRequest).Name, changes);

        return response;
    }

    /// <summary>
    /// 事务命令：包在 Transaction 里
    /// </summary>
    private async Task<TResponse> ExecuteWithTransactionAsync(
        RequestHandlerDelegate<TResponse> next,
        CancellationToken cancellationToken)
    {
        _logger.LogDebug(
            "[CQRS] Begin transaction for {RequestName}",
            typeof(TRequest).Name);

        await _unitOfWork.BeginTransactionAsync(cancellationToken);

        try
        {
            var response = await next();
            await _unitOfWork.CommitTransactionAsync(cancellationToken);

            _logger.LogDebug(
                "[CQRS] Transaction committed for {RequestName}",
                typeof(TRequest).Name);

            return response;
        }
        catch
        {
            await _unitOfWork.RollbackTransactionAsync(cancellationToken);

            _logger.LogWarning(
                "[CQRS] Transaction rolled back for {RequestName}",
                typeof(TRequest).Name);

            throw;
        }
    }
}
```

```csharp
// ============================================
// Behaviors/ExceptionBehavior.cs
// ============================================

using MediatR;
using Microsoft.EntityFrameworkCore;
using Microsoft.Extensions.Logging;
using Platform.Exceptions;

namespace Platform.Cqrs.Behaviors;

/// <summary>
/// 异常处理管道
/// 
/// 将底层异常转为业务友好的异常
/// 
/// 转换规则：
///   DbUpdateConcurrencyException → BusinessException（并发冲突）
///   DbUpdateException + UniqueConstraint → BusinessException（重复数据）
/// </summary>
public class ExceptionBehavior<TRequest, TResponse>
    : IPipelineBehavior<TRequest, TResponse>
    where TRequest : notnull
{
    private readonly ILogger<ExceptionBehavior<TRequest, TResponse>> _logger;

    public ExceptionBehavior(
        ILogger<ExceptionBehavior<TRequest, TResponse>> logger)
    {
        _logger = logger;
    }

    public async Task<TResponse> Handle(
        TRequest request,
        RequestHandlerDelegate<TResponse> next,
        CancellationToken cancellationToken)
    {
        try
        {
            return await next();
        }
        catch (BusinessException)
        {
            // 业务异常直接抛出，不需要转换
            throw;
        }
        catch (DbUpdateConcurrencyException ex)
        {
            _logger.LogWarning(ex,
                "Concurrency conflict in {RequestName}",
                typeof(TRequest).Name);

            throw new BusinessException(
                "Platform:ConcurrencyConflict",
                "数据已被其他人修改，请刷新后重试",
                httpStatusCode: 409);
        }
        catch (DbUpdateException ex)
            when (ex.InnerException?.Message
                .Contains("unique", StringComparison.OrdinalIgnoreCase) == true
               || ex.InnerException?.Message
                .Contains("duplicate", StringComparison.OrdinalIgnoreCase) == true)
        {
            _logger.LogWarning(ex,
                "Duplicate key in {RequestName}",
                typeof(TRequest).Name);

            throw new BusinessException(
                "Platform:DuplicateKey",
                "数据已存在，请勿重复提交",
                httpStatusCode: 409);
        }
    }
}
```

```csharp
// ============================================
// Extensions/ServiceCollectionExtensions.cs
// ============================================

using System.Reflection;
using FluentValidation;
using MediatR;
using Microsoft.Extensions.DependencyInjection;
using Platform.Cqrs.Behaviors;
using Platform.Cqrs.Mediator;

namespace Platform.Cqrs.Extensions;

/// <summary>
/// CQRS 模块一键注册
/// 
/// 用法：
///   builder.Services.AddPlatformCqrs(options =>
///   {
///       options.SlowRequestThresholdMs = 500;
///   },
///   typeof(Program).Assembly,           // Handler 所在程序集
///   typeof(CreateOrderValidator).Assembly // Validator 所在程序集（可选）
///   );
/// </summary>
public static class ServiceCollectionExtensions
{
    public static IServiceCollection AddPlatformCqrs(
        this IServiceCollection services,
        Action<CqrsOptions>? configure = null,
        params Assembly[] assemblies)
    {
        var options = new CqrsOptions();
        configure?.Invoke(options);
        services.Configure<CqrsOptions>(opt =>
        {
            opt.SlowRequestThresholdMs = options.SlowRequestThresholdMs;
            opt.EnableLogging = options.EnableLogging;
            opt.EnablePerformanceMonitoring = options.EnablePerformanceMonitoring;
            opt.EnableValidation = options.EnableValidation;
            opt.EnableUnitOfWork = options.EnableUnitOfWork;
            opt.EnableAuthorization = options.EnableAuthorization;
            opt.EnableExceptionHandling = options.EnableExceptionHandling;
        });

        // 如果没指定程序集，用调用方的程序集
        if (assemblies.Length == 0)
        {
            assemblies = [Assembly.GetCallingAssembly()];
        }

        // 注册 MediatR + 所有 Handler
        services.AddMediatR(config =>
        {
            config.RegisterServicesFromAssemblies(assemblies);

            // 按顺序注册管道行为（顺序很重要）
            // 最外层 → 最内层

            if (options.EnableLogging)
                config.AddBehavior(typeof(IPipelineBehavior<,>),
                    typeof(LoggingBehavior<,>));

            if (options.EnablePerformanceMonitoring)
                config.AddBehavior(typeof(IPipelineBehavior<,>),
                    typeof(PerformanceBehavior<,>));

            if (options.EnableExceptionHandling)
                config.AddBehavior(typeof(IPipelineBehavior<,>),
                    typeof(ExceptionBehavior<,>));

            if (options.EnableAuthorization)
                config.AddBehavior(typeof(IPipelineBehavior<,>),
                    typeof(AuthorizationBehavior<,>));

            if (options.EnableValidation)
                config.AddBehavior(typeof(IPipelineBehavior<,>),
                    typeof(ValidationBehavior<,>));

            if (options.EnableUnitOfWork)
                config.AddBehavior(typeof(IPipelineBehavior<,>),
                    typeof(UnitOfWorkBehavior<,>));
        });

        // 注册 FluentValidation 的所有 Validator
        if (options.EnableValidation)
        {
            services.AddValidatorsFromAssemblies(assemblies);
        }

        // 注册 IPlatformMediator
        services.AddScoped<IPlatformMediator, MediatRMediator>();

        return services;
    }
}
```

```csharp
// ============================================
// 5. 查询处理器
// ERP.Application/Purchase/Queries/GetPurchaseListQueryHandler.cs
// ============================================

using Microsoft.EntityFrameworkCore;
using Platform.AspNetCore.Pagination;
using Platform.Cqrs.Queries;
using Platform.Repositories;

namespace ERP.Application.Purchase.Queries;

public class GetPurchaseListQueryHandler
    : IQueryHandler<GetPurchaseListQuery, PagedResult<PurchaseListDto>>
{
    private readonly IRepository<PurchaseOrder, Guid> _repo;

    public GetPurchaseListQueryHandler(
        IRepository<PurchaseOrder, Guid> repo)
    {
        _repo = repo;
    }

    public async Task<PagedResult<PurchaseListDto>> HandleAsync(
        GetPurchaseListQuery query,
        CancellationToken cancellationToken = default)
    {
        var q = _repo.AsQueryable();

        // 筛选
        if (!string.IsNullOrWhiteSpace(query.Keyword))
        {
            q = q.Where(o =>
                o.OrderNo.Contains(query.Keyword) ||
                o.SupplierName.Contains(query.Keyword));
        }

        if (query.Status.HasValue)
            q = q.Where(o => o.Status == query.Status.Value);

        if (query.StartDate.HasValue)
            q = q.Where(o => o.CreationTime >= query.StartDate.Value);

        if (query.EndDate.HasValue)
            q = q.Where(o => o.CreationTime <= query.EndDate.Value);

        // 总数
        var totalCount = await q.LongCountAsync(cancellationToken);

        // 分页
        var items = await q
            .OrderByDescending(o => o.CreationTime)
            .Skip(query.SkipCount)
            .Take(query.PageSize)
            .Select(o => new PurchaseListDto
            {
                Id = o.Id,
                OrderNo = o.OrderNo,
                SupplierName = o.SupplierName,
                TotalAmount = o.TotalAmount,
                Status = o.Status,
                CreationTime = o.CreationTime,
            })
            .ToListAsync(cancellationToken);

        return new PagedResult<PurchaseListDto>(
            items, totalCount, query.Page, query.PageSize);

        // Query 不会触发 SaveChanges ✅
    }
}
```
