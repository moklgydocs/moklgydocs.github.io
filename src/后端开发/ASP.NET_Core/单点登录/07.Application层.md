```csharp
// Sso.Application/Contracts/Clients/ClientDto.cs
namespace Sso.Application.Contracts.Clients;

public class ClientDto
{
    public string ClientId { get; set; } = default!;
    public string? DisplayName { get; set; }
    public List<string> RedirectUris { get; set; } = [];
    public List<string> PostLogoutRedirectUris { get; set; } = [];
    public List<string> Permissions { get; set; } = [];

    public DateTime? CreatedAt { get; set; }

{
    public string ClientId { get; set; } = default!;
    public string? ClientSecret { get; set; }
    public string? DisplayName { get; set; }
    public List<string> RedirectUris { get; set; } = [];
    public List<string> PostLogoutRedirectUris { get; set; } = [];

    /// <summary>
    /// 客户端类型: public(SPA/移动端) / confidential(后端服务)
    /// </summary>

    /// <summary>
    /// 授权流程: authorization_code, client_credentials, password, refresh_token
    /// </summary>

    /// <summary>
    /// 允许的 Scope
    /// </summary>

{
    public string? DisplayName { get; set; }
    public List<string>? RedirectUris { get; set; }
    public List<string>? PostLogoutRedirectUris { get; set; }
    public List<string>? Scopes { get; set; }
    public List<string>? GrantTypes { get; set; }
}
```

## 6.2 客户端管理服务

```csharp
// Sso.Application/Services/ClientAppService.cs
using OpenIddict.Abstractions;
using Sso.Application.Contracts.Clients;
using static OpenIddict.Abstractions.OpenIddictConstants;

namespace Sso.Application.Services;

public class ClientAppService
{
    private readonly IOpenIddictApplicationManager _applicationManager;

    public ClientAppService(IOpenIddictApplicationManager applicationManager)
    {
        _applicationManager = applicationManager;
    }

    public async Task<ClientDto> CreateAsync(CreateClientRequest request, CancellationToken ct = default)
    {
        // 检查是否已存在
        var existing = await _applicationManager.FindByClientIdAsync(request.ClientId, ct);
        if (existing != null)
            throw new InvalidOperationException($"客户端 '{request.ClientId}' 已存在");

        var descriptor = new OpenIddictApplicationDescriptor
        {
            ClientId = request.ClientId,
            DisplayName = request.DisplayName,
            ClientType = request.Type switch
            {
                "public" => ClientTypes.Public,
                _ => ClientTypes.Confidential
            }
        };

        // Secret（仅 confidential 客户端）
        if (request.Type != "public" && !string.IsNullOrEmpty(request.ClientSecret))
        {
            descriptor.ClientSecret = request.ClientSecret;
        }

        // Redirect URIs
        foreach (var uri in request.RedirectUris)
            descriptor.RedirectUris.Add(new Uri(uri));

        foreach (var uri in request.PostLogoutRedirectUris)
            descriptor.PostLogoutRedirectUris.Add(new Uri(uri));

        // Permissions: 授权类型
        foreach (var grantType in request.GrantTypes)
        {
            descriptor.Permissions.Add(grantType switch
            {
                "authorization_code" => Permissions.GrantTypes.AuthorizationCode,
                "client_credentials" => Permissions.GrantTypes.ClientCredentials,
                "password" => Permissions.GrantTypes.Password,
                "refresh_token" => Permissions.GrantTypes.RefreshToken,
                _ => throw new ArgumentException($"不支持的授权类型: {grantType}")
            });
        }

        // Permissions: 端点
        if (request.GrantTypes.Contains("authorization_code"))
        {
            descriptor.Permissions.Add(Permissions.Endpoints.Authorization);
            descriptor.Permissions.Add(Permissions.Endpoints.Logout);
            descriptor.Permissions.Add(Permissions.ResponseTypes.Code);
        }
        descriptor.Permissions.Add(Permissions.Endpoints.Token);

        // Permissions: Scopes
        foreach (var scope in request.Scopes)
        {
            descriptor.Permissions.Add(scope switch
            {
                "openid" => Permissions.Scopes.Profile, // openid 自动包含
                _ => Permissions.Prefixes.Scope + scope
            });
            // 确保 openid scope 也加上
        }
        descriptor.Permissions.Add(Permissions.Scopes.Email);
        descriptor.Permissions.Add(Permissions.Scopes.Profile);
        descriptor.Permissions.Add(Permissions.Scopes.Roles);

        var app = await _applicationManager.CreateAsync(descriptor, ct);

        return await MapToDtoAsync(app, ct);
    }

    public async Task<ClientDto?> GetByClientIdAsync(string clientId, CancellationToken ct = default)
    {
        var app = await _applicationManager.FindByClientIdAsync(clientId, ct);
        return app == null ? null : await MapToDtoAsync(app, ct);
    }

    public async Task<List<ClientDto>> GetAllAsync(CancellationToken ct = default)
    {
        var results = new List<ClientDto>();

        await foreach (var app in _applicationManager.ListAsync(ct))
        {
            results.Add(await MapToDtoAsync(app, ct));
        }

        return results;
    }

    public async Task UpdateAsync(string clientId, UpdateClientRequest request, CancellationToken ct = default)
    {
        var app = await _applicationManager.FindByClientIdAsync(clientId, ct)
            ?? throw new KeyNotFoundException($"客户端 '{clientId}' 不存在");

        var descriptor = new OpenIddictApplicationDescriptor();
        await _applicationManager.PopulateAsync(descriptor, app, ct);

        if (request.DisplayName != null)
            descriptor.DisplayName = request.DisplayName;

        if (request.RedirectUris != null)
        {
            descriptor.RedirectUris.Clear();
            foreach (var uri in request.RedirectUris)
                descriptor.RedirectUris.Add(new Uri(uri));
        }

        if (request.PostLogoutRedirectUris != null)
        {
            descriptor.PostLogoutRedirectUris.Clear();
            foreach (var uri in request.PostLogoutRedirectUris)
                descriptor.PostLogoutRedirectUris.Add(new Uri(uri));
        }

        await _applicationManager.UpdateAsync(app, descriptor, ct);
    }

    public async Task DeleteAsync(string clientId, CancellationToken ct = default)
    {
        var app = await _applicationManager.FindByClientIdAsync(clientId, ct)
            ?? throw new KeyNotFoundException($"客户端 '{clientId}' 不存在");

        await _applicationManager.DeleteAsync(app, ct);
    }

    private async Task<ClientDto> MapToDtoAsync(object app, CancellationToken ct)
    {
        var redirectUris = new List<string>();
        var postLogoutUris = new List<string>();

        await foreach (var uri in _applicationManager.GetRedirectUrisAsync(app, ct).ToAsyncEnumerable())
            redirectUris.Add(uri);

        await foreach (var uri in _applicationManager.GetPostLogoutRedirectUrisAsync(app, ct).ToAsyncEnumerable())
            postLogoutUris.Add(uri);

        return new ClientDto
        {
            ClientId = (await _applicationManager.GetClientIdAsync(app, ct))!,
            DisplayName = await _applicationManager.GetDisplayNameAsync(app, ct),
            RedirectUris = redirectUris,
            PostLogoutRedirectUris = postLogoutUris,
            Type = await _applicationManager.GetClientTypeAsync(app, ct)
        };
    }
}

// 扩展方法让 GetRedirectUrisAsync 等返回 IAsyncEnumerable
file static class ManagerExtensions
{
    public static async IAsyncEnumerable<string> ToAsyncEnumerable(
        this ValueTask<System.Collections.Immutable.ImmutableArray<string>> task)
    {
        var items = await task;
        foreach (var item in items)
            yield return item;
    }
}
```

```csharp
// Sso.Application/Services/ScopeAppService.cs
using OpenIddict.Abstractions;
using Sso.Application.Contracts.Scopes;

namespace Sso.Application.Services;

public class ScopeAppService
{
    private readonly IOpenIddictScopeManager _scopeManager;

    public ScopeAppService(IOpenIddictScopeManager scopeManager)
    {
        _scopeManager = scopeManager;
    }

    public async Task<ScopeDto> CreateAsync(CreateScopeRequest request, CancellationToken ct = default)
    {
        var existing = await _scopeManager.FindByNameAsync(request.Name, ct);
        if (existing != null)
            throw new InvalidOperationException($"Scope '{request.Name}' 已存在");

        var descriptor = new OpenIddictScopeDescriptor
        {
            Name = request.Name,
            DisplayName = request.DisplayName,
            Description = request.Description
        };

        foreach (var resource in request.Resources)
            descriptor.Resources.Add(resource);

        await _scopeManager.CreateAsync(descriptor, ct);

        return new ScopeDto
        {
            Name = request.Name,
            DisplayName = request.DisplayName,
            Description = request.Description,
            Resources = request.Resources
        };
    }

    public async Task<List<ScopeDto>> GetAllAsync(CancellationToken ct = default)
    {
        var results = new List<ScopeDto>();

        await foreach (var scope in _scopeManager.ListAsync(ct))
        {
            results.Add(new ScopeDto
            {
                Name = (await _scopeManager.GetNameAsync(scope, ct))!,
                DisplayName = await _scopeManager.GetDisplayNameAsync(scope, ct),
                Description = await _scopeManager.GetDescriptionAsync(scope, ct)
            });
        }

        return results;
    }

    public async Task DeleteAsync(string name, CancellationToken ct = default)
    {
        var scope = await _scopeManager.FindByNameAsync(name, ct)
            ?? throw new KeyNotFoundException($"Scope '{name}' 不存在");
        await _scopeManager.DeleteAsync(scope, ct);
    }
}
```

```csharp
// Sso.Application/Services/UserAppService.cs
using Microsoft.AspNetCore.Identity;
using Microsoft.EntityFrameworkCore;
using Sso.Application.Contracts.Users;
using Sso.Domain.Entities;
using Sso.Shared.Results;

namespace Sso.Application.Services;

public class UserAppService
{
    private readonly UserManager<SsoUser> _userManager;
    private readonly RoleManager<SsoRole> _roleManager;

    public UserAppService(
        UserManager<SsoUser> userManager,
        RoleManager<SsoRole> roleManager)
    {
        _userManager = userManager;
        _roleManager = roleManager;
    }

    public async Task<UserDto> CreateAsync(CreateUserRequest request)
    {
        var user = new SsoUser
        {
            UserName = request.UserName,
            Email = request.Email,
            DisplayName = request.DisplayName ?? request.UserName,
            IsActive = true,
            CreatedAt = DateTime.UtcNow
        };

        var result = await _userManager.CreateAsync(user, request.Password);
        if (!result.Succeeded)
        {
            var errors = string.Join("; ", result.Errors.Select(e => e.Description));
            throw new InvalidOperationException($"创建用户失败: {errors}");
        }

        // 分配角色
        if (request.Roles?.Any() == true)
        {
            foreach (var roleName in request.Roles)
            {
                if (!await _roleManager.RoleExistsAsync(roleName))
                    await _roleManager.CreateAsync(new SsoRole { Name = roleName });
            }
            await _userManager.AddToRolesAsync(user, request.Roles);
        }

        return await MapToDtoAsync(user);
    }

    public async Task<PagedResult<UserDto>> GetPagedAsync(PagedQuery query)
    {
        var queryable = _userManager.Users.AsNoTracking();

        if (!string.IsNullOrWhiteSpace(query.Keyword))
        {
            queryable = queryable.Where(u =>
                u.UserName!.Contains(query.Keyword) ||
                u.Email!.Contains(query.Keyword) ||
                (u.DisplayName != null && u.DisplayName.Contains(query.Keyword)));
        }

        var totalCount = await queryable.CountAsync();
        var users = await queryable
            .OrderByDescending(u => u.CreatedAt)
            .Skip((query.PageIndex - 1) * query.PageSize)
            .Take(query.PageSize)
            .ToListAsync();

        var dtos = new List<UserDto>();
        foreach (var user in users)
        {
            dtos.Add(await MapToDtoAsync(user));
        }

        return new PagedResult<UserDto>
        {
            Items = dtos,
            TotalCount = totalCount,
            PageIndex = query.PageIndex,
            PageSize = query.PageSize
        };
    }

    public async Task<UserDto?> GetByIdAsync(Guid id)
    {
        var user = await _userManager.FindByIdAsync(id.ToString());
        return user == null ? null : await MapToDtoAsync(user);
    }

    public async Task UpdateAsync(Guid id, UpdateUserRequest request)
    {
        var user = await _userManager.FindByIdAsync(id.ToString())
            ?? throw new KeyNotFoundException("用户不存在");

        if (request.DisplayName != null) user.DisplayName = request.DisplayName;
        if (request.Email != null) user.Email = request.Email;
        if (request.Avatar != null) user.Avatar = request.Avatar;
        if (request.IsActive.HasValue) user.IsActive = request.IsActive.Value;

        await _userManager.UpdateAsync(user);
    }

    public async Task ResetPasswordAsync(Guid id, string newPassword)
    {
        var user = await _userManager.FindByIdAsync(id.ToString())
            ?? throw new KeyNotFoundException("用户不存在");

        var token = await _userManager.GeneratePasswordResetTokenAsync(user);
        var result = await _userManager.ResetPasswordAsync(user, token, newPassword);

        if (!result.Succeeded)
        {
            var errors = string.Join("; ", result.Errors.Select(e => e.Description));
            throw new InvalidOperationException($"重置密码失败: {errors}");
        }
    }

    private async Task<UserDto> MapToDtoAsync(SsoUser user)
    {
        var roles = await _userManager.GetRolesAsync(user);
        return new UserDto
        {
            Id = user.Id,
            UserName = user.UserName!,
            Email = user.Email,
            DisplayName = user.DisplayName,
            Avatar = user.Avatar,
            IsActive = user.IsActive,
            CreatedAt = user.CreatedAt,
            LastLoginAt = user.LastLoginAt,
            Roles = roles.ToList()
        };
    }
}
```
