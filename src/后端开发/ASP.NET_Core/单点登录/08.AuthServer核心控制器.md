# AuthServer — 核心控制器

## 7.1 授权控制器 (OAuth2/OIDC 核心端点)

**这是整个SSO的核心，处理 OAuth2 各种 flow：**

```csharp
// Sso.AuthServer/Controllers/AuthorizationController.cs
using System.Security.Claims;
using Microsoft.AspNetCore;
using Microsoft.AspNetCore.Authentication;
using Microsoft.AspNetCore.Identity;
using Microsoft.AspNetCore.Mvc;
using OpenIddict.Abstractions;
using OpenIddict.Server.AspNetCore;
using Sso.Domain.Entities;
using static OpenIddict.Abstractions.OpenIddictConstants;

namespace Sso.AuthServer.Controllers;

[ApiController]
public class AuthorizationController : ControllerBase
{
    private readonly IOpenIddictApplicationManager _applicationManager;
    private readonly IOpenIddictScopeManager _scopeManager;
    private readonly SignInManager<SsoUser> _signInManager;
    private readonly UserManager<SsoUser> _userManager;

    public AuthorizationController(
        IOpenIddictApplicationManager applicationManager,
        IOpenIddictScopeManager scopeManager,
        SignInManager<SsoUser> signInManager,
        UserManager<SsoUser> userManager)
    {
        _applicationManager = applicationManager;
        _scopeManager = scopeManager;
        _signInManager = signInManager;
        _userManager = userManager;
    }

    /// <summary>
    /// 授权端点 - 授权码流程入口
    /// GET/POST /connect/authorize
    /// </summary>
    [HttpGet("connect/authorize")]
    [HttpPost("connect/authorize")]
    public async Task<IActionResult> Authorize()
    {
        var request = HttpContext.GetOpenIddictServerRequest()
            ?? throw new InvalidOperationException("无法获取 OpenID Connect 请求");

        // 尝试从 cookie 获取已认证的用户
        var result = await HttpContext.AuthenticateAsync(IdentityConstants.ApplicationScheme);

        if (!result.Succeeded || request.HasPrompt(Prompts.Login))
        {
            // 用户未登录 → 返回一个特殊响应，告诉前端需要登录
            // 前端是 React SPA，所以我们返回 401 + 登录地址信息
            if (IsApiRequest())
            {
                return Unauthorized(new
                {
                    error = "login_required",
                    login_url = "/account/login",
                    return_url = Request.Path + Request.QueryString
                });
            }

            // 也可以做传统重定向（用于非 SPA 场景）
            var prompt = string.Join(" ", request.GetPrompts().Remove(Prompts.Login));
            var parameters = Request.HasFormContentType
                ? Request.Form.Where(p => p.Key != Parameters.Prompt).ToList()
                : Request.Query.Where(p => p.Key != Parameters.Prompt).ToList();

            if (!string.IsNullOrEmpty(prompt))
                parameters.Add(KeyValuePair.Create(Parameters.Prompt, new Microsoft.Extensions.Primitives.StringValues(prompt)));

            return Challenge(
                authenticationSchemes: IdentityConstants.ApplicationScheme,
                properties: new AuthenticationProperties
                {
                    RedirectUri = Request.PathBase + Request.Path + QueryString.Create(parameters)
                });
        }

        var user = await _userManager.GetUserAsync(result.Principal)
            ?? throw new InvalidOperationException("无法获取用户信息");

        // 检查用户是否被禁用
        if (!user.IsActive)
        {
            return Forbid(
                authenticationSchemes: OpenIddictServerAspNetCoreDefaults.AuthenticationScheme,
                properties: new AuthenticationProperties(new Dictionary<string, string?>
                {
                    [OpenIddictServerAspNetCoreConstants.Properties.Error] = Errors.AccessDenied,
                    [OpenIddictServerAspNetCoreConstants.Properties.ErrorDescription] = "用户已被禁用"
                }));
        }

        // 构建 ClaimsPrincipal
        var principal = await CreatePrincipalAsync(user, request);

        // 直接授权（跳过 consent 页面，也可以加入 consent 逻辑）
        return SignIn(principal, OpenIddictServerAspNetCoreDefaults.AuthenticationScheme);
    }

    /// <summary>
    /// Token 端点 - 发放/刷新令牌
    /// POST /connect/token
    /// </summary>
    [HttpPost("connect/token")]
    public async Task<IActionResult> Exchange()
    {
        var request = HttpContext.GetOpenIddictServerRequest()
            ?? throw new InvalidOperationException("无法获取 OpenID Connect 请求");

        // ====== 授权码模式 ======
        if (request.IsAuthorizationCodeGrantType() || request.IsRefreshTokenGrantType())
        {
            var result = await HttpContext.AuthenticateAsync(
                OpenIddictServerAspNetCoreDefaults.AuthenticationScheme);

            var userId = result.Principal?.GetClaim(Claims.Subject);
            if (string.IsNullOrEmpty(userId))
            {
                return Forbid(
                    authenticationSchemes: OpenIddictServerAspNetCoreDefaults.AuthenticationScheme,
                    properties: new AuthenticationProperties(new Dictionary<string, string?>
                    {
                        [OpenIddictServerAspNetCoreConstants.Properties.Error] = Errors.InvalidGrant,
                        [OpenIddictServerAspNetCoreConstants.Properties.ErrorDescription] = "令牌已失效"
                    }));
            }

            var user = await _userManager.FindByIdAsync(userId);
            if (user == null || !user.IsActive)
            {
                return Forbid(
                    authenticationSchemes: OpenIddictServerAspNetCoreDefaults.AuthenticationScheme,
                    properties: new AuthenticationProperties(new Dictionary<string, string?>
                    {
                        [OpenIddictServerAspNetCoreConstants.Properties.Error] = Errors.InvalidGrant,
                        [OpenIddictServerAspNetCoreConstants.Properties.ErrorDescription] = "用户不存在或已被禁用"
                    }));
            }

            var principal = await CreatePrincipalAsync(user, request);

            // 更新最后登录时间
            user.LastLoginAt = DateTime.UtcNow;
            await _userManager.UpdateAsync(user);

            return SignIn(principal, OpenIddictServerAspNetCoreDefaults.AuthenticationScheme);
        }

        // ====== 密码模式（可选，用于移动端等信任客户端）======
        if (request.IsPasswordGrantType())
        {
            var user = await _userManager.FindByNameAsync(request.Username!)
                       ?? await _userManager.FindByEmailAsync(request.Username!);

            if (user == null)
            {
                return Forbid(
                    authenticationSchemes: OpenIddictServerAspNetCoreDefaults.AuthenticationScheme,
                    properties: new AuthenticationProperties(new Dictionary<string, string?>
                    {
                        [OpenIddictServerAspNetCoreConstants.Properties.Error] = Errors.InvalidGrant,
                        [OpenIddictServerAspNetCoreConstants.Properties.ErrorDescription] = "用户名或密码错误"
                    }));
            }

            var result = await _signInManager.CheckPasswordSignInAsync(user, request.Password!, lockoutOnFailure: true);
            if (!result.Succeeded)
            {
                var desc = result.IsLockedOut ? "账户已被锁定，请稍后再试" : "用户名或密码错误";
                return Forbid(
                    authenticationSchemes: OpenIddictServerAspNetCoreDefaults.AuthenticationScheme,
                    properties: new AuthenticationProperties(new Dictionary<string, string?>
                    {
                        [OpenIddictServerAspNetCoreConstants.Properties.Error] = Errors.InvalidGrant,
                        [OpenIddictServerAspNetCoreConstants.Properties.ErrorDescription] = desc
                    }));
            }

            if (!user.IsActive)
            {
                return Forbid(
                    authenticationSchemes: OpenIddictServerAspNetCoreDefaults.AuthenticationScheme,
                    properties: new AuthenticationProperties(new Dictionary<string, string?>
                    {
                        [OpenIddictServerAspNetCoreConstants.Properties.Error] = Errors.AccessDenied,
                        [OpenIddictServerAspNetCoreConstants.Properties.ErrorDescription] = "用户已被禁用"
                    }));
            }

            var principal = await CreatePrincipalAsync(user, request);

            user.LastLoginAt = DateTime.UtcNow;
            await _userManager.UpdateAsync(user);

            return SignIn(principal, OpenIddictServerAspNetCoreDefaults.AuthenticationScheme);
        }

        // ====== 客户端凭证模式（服务间通信）======
        if (request.IsClientCredentialsGrantType())
        {
            var application = await _applicationManager.FindByClientIdAsync(request.ClientId!)
                ?? throw new InvalidOperationException("找不到客户端应用");

            var identity = new ClaimsIdentity(
                authenticationType: OpenIddictServerAspNetCoreDefaults.AuthenticationScheme,
                nameType: Claims.Name,
                roleType: Claims.Role);

            identity.SetClaim(Claims.Subject, await _applicationManager.GetClientIdAsync(application));
            identity.SetClaim(Claims.Name, await _applicationManager.GetDisplayNameAsync(application));

            identity.SetScopes(request.GetScopes());
            identity.SetResources(await GetResourcesAsync(request.GetScopes()));

            identity.SetDestinations(GetDestinations);

            return SignIn(new ClaimsPrincipal(identity),
                OpenIddictServerAspNetCoreDefaults.AuthenticationScheme);
        }

        throw new InvalidOperationException("不支持的授权类型");
    }

    /// <summary>
    /// 用户信息端点
    /// GET/POST /connect/userinfo
    /// </summary>
    [HttpGet("connect/userinfo")]
    [HttpPost("connect/userinfo")]
    public async Task<IActionResult> Userinfo()
    {
        var claimsPrincipal = (await HttpContext.AuthenticateAsync(
            OpenIddictServerAspNetCoreDefaults.AuthenticationScheme)).Principal!;

        var userId = claimsPrincipal.GetClaim(Claims.Subject)!;
        var user = await _userManager.FindByIdAsync(userId);

        if (user == null)
            return Challenge(
                authenticationSchemes: OpenIddictServerAspNetCoreDefaults.AuthenticationScheme,
                properties: new AuthenticationProperties(new Dictionary<string, string?>
                {
                    [OpenIddictServerAspNetCoreConstants.Properties.Error] = Errors.InvalidToken,
                    [OpenIddictServerAspNetCoreConstants.Properties.ErrorDescription] = "用户不存在"
                }));

        var claims = new Dictionary<string, object>(StringComparer.Ordinal)
        {
            [Claims.Subject] = user.Id.ToString()
        };

        if (claimsPrincipal.HasScope(Scopes.Email))
        {
            claims[Claims.Email] = user.Email ?? "";
            claims[Claims.EmailVerified] = user.EmailConfirmed;
        }

        if (claimsPrincipal.HasScope(Scopes.Profile))
        {
            claims[Claims.Name] = user.DisplayName ?? user.UserName ?? "";
            claims[Claims.PreferredUsername] = user.UserName ?? "";
            if (user.Avatar != null)
                claims[Claims.Picture] = user.Avatar;
        }

        if (claimsPrincipal.HasScope(Scopes.Roles))
        {
            var roles = await _userManager.GetRolesAsync(user);
            claims[Claims.Role] = roles.ToArray();
        }

        return Ok(claims);
    }

    /// <summary>
    /// 登出端点
    /// GET/POST /connect/logout
    /// </summary>
    [HttpGet("connect/logout")]
    [HttpPost("connect/logout")]
    public async Task<IActionResult> Logout()
    {
        await _signInManager.SignOutAsync();

        return SignOut(
            authenticationSchemes: OpenIddictServerAspNetCoreDefaults.AuthenticationScheme,
            properties: new AuthenticationProperties { RedirectUri = "/" });
    }

    // =============== 私有方法 ===============

    private async Task<ClaimsPrincipal> CreatePrincipalAsync(
        SsoUser user, OpenIddictRequest request)
    {
        var principal = await _signInManager.CreateUserPrincipalAsync(user);
        var identity = (ClaimsIdentity)principal.Identity!;

        // 添加自定义 Claims
        identity.SetClaim(Claims.Subject, user.Id.ToString())
                .SetClaim(Claims.Email, user.Email)
                .SetClaim(Claims.Name, user.DisplayName ?? user.UserName)
                .SetClaim(Claims.PreferredUsername, user.UserName);

        if (user.Avatar != null)
            identity.SetClaim(Claims.Picture, user.Avatar);

        // 角色
        var roles = await _userManager.GetRolesAsync(user);
        foreach (var role in roles)
        {
            identity.AddClaim(Claims.Role, role);
        }

        // 设置请求的 Scopes
        principal.SetScopes(request.GetScopes());

        // 设置 Resources（哪些 API 可以使用这个 Token）
        principal.SetResources(await GetResourcesAsync(request.GetScopes()));

        // 设置 Claim Destinations（哪些 Claim 放入 access_token / id_token）
        foreach (var claim in principal.Claims)
        {
            claim.SetDestinations(GetDestinations(claim));
        }

        return principal;
    }

    private async Task<ImmutableArray<string>> GetResourcesAsync(
        ImmutableArray<string> scopes)
    {
        var resources = new List<string>();
        await foreach (var resource in _scopeManager.ListResourcesAsync(scopes))
        {
            resources.Add(resource);
        }
        return [.. resources];
    }

    private static IEnumerable<string> GetDestinations(Claim claim)
    {
        switch (claim.Type)
        {
            case Claims.Name or Claims.PreferredUsername:
                yield return Destinations.AccessToken;
                if (claim.Subject?.HasScope(Scopes.Profile) == true)
                    yield return Destinations.IdentityToken;
                yield break;

            case Claims.Email:
                yield return Destinations.AccessToken;
                if (claim.Subject?.HasScope(Scopes.Email) == true)
                    yield return Destinations.IdentityToken;
                yield break;

            case Claims.Role:
                yield return Destinations.AccessToken;
                if (claim.Subject?.HasScope(Scopes.Roles) == true)
                    yield return Destinations.IdentityToken;
                yield break;

            case Claims.Picture:
                yield return Destinations.AccessToken;
                if (claim.Subject?.HasScope(Scopes.Profile) == true)
                    yield return Destinations.IdentityToken;
                yield break;

            // 关键：Subject 和 audience 不要放入 id_token（已自动处理）
            case Claims.Subject:
                yield return Destinations.AccessToken;
                yield return Destinations.IdentityToken;
                yield break;

            // 默认只放 access_token
            default:
                yield return Destinations.AccessToken;
                yield break;
        }
    }

    private bool IsApiRequest()
    {
        return Request.Headers.Accept.Any(h =>
            h != null && h.Contains("application/json")) ||
            Request.Headers.ContainsKey("X-Requested-With");
    }
}
```

## 7.2 账户控制器（登录/注册 API，供 React 前端调用）

```csharp
// Sso.AuthServer/Controllers/AccountController.cs
using Microsoft.AspNetCore.Authorization;
using Microsoft.AspNetCore.Identity;
using Microsoft.AspNetCore.Mvc;
using Sso.Domain.Entities;
using Sso.Shared.Results;

namespace Sso.AuthServer.Controllers;

[ApiController]
[Route("api/account")]
public class AccountController : ControllerBase
{
    private readonly SignInManager<SsoUser> _signInManager;
    private readonly UserManager<SsoUser> _userManager;

    public AccountController(
        SignInManager<SsoUser> signInManager,
        UserManager<SsoUser> userManager)
    {
        _signInManager = signInManager;
        _userManager = userManager;
    }

    /// <summary>
    /// 用户登录（Cookie 认证，用于授权码流程）
    /// </summary>
    [HttpPost("login")]
    [AllowAnonymous]
    public async Task<IActionResult> Login([FromBody] LoginRequest request)
    {
        if (string.IsNullOrEmpty(request.Username) || string.IsNullOrEmpty(request.Password))
            return BadRequest(ApiResult.Fail("用户名和密码不能为空"));

        var user = await _userManager.FindByNameAsync(request.Username)
                   ?? await _userManager.FindByEmailAsync(request.Username);

        if (user == null)
            return BadRequest(ApiResult.Fail("用户名或密码错误"));

        if (!user.IsActive)
            return BadRequest(ApiResult.Fail("用户已被禁用"));

        var result = await _signInManager.PasswordSignInAsync(
            user, request.Password,
            isPersistent: request.RememberMe,
            lockoutOnFailure: true);

        if (result.IsLockedOut)
            return BadRequest(ApiResult.Fail("账户已被锁定，请15分钟后再试"));

        if (result.RequiresTwoFactor)
            return Ok(ApiResult<object>.Ok(new { requiresTwoFactor = true }, "需要双因素认证"));

        if (!result.Succeeded)
            return BadRequest(ApiResult.Fail("用户名或密码错误"));

        // 更新最后登录时间
        user.LastLoginAt = DateTime.UtcNow;
        await _userManager.UpdateAsync(user);

        return Ok(ApiResult<object>.Ok(new
        {
            user.Id,
            user.UserName,
            user.DisplayName,
            user.Email,
            returnUrl = request.ReturnUrl
        }, "登录成功"));
    }

    /// <summary>
    /// 登出
    /// </summary>
    [HttpPost("logout")]
    public async Task<IActionResult> Logout()
    {
        await _signInManager.SignOutAsync();
        return Ok(ApiResult.Ok("已登出"));
    }

    /// <summary>
    /// 获取当前登录用户信息
    /// </summary>
    [HttpGet("current")]
    [Authorize]
    public async Task<IActionResult> GetCurrentUser()
    {
        var user = await _userManager.GetUserAsync(User);
        if (user == null)
            return Unauthorized(ApiResult.Fail("未登录", 401));

        var roles = await _userManager.GetRolesAsync(user);

        return Ok(ApiResult<object>.Ok(new
        {
            user.Id,
            user.UserName,
            user.DisplayName,
            user.Email,
            user.Avatar,
            Roles = roles
        }));
    }

    /// <summary>
    /// 修改密码
    /// </summary>
    [HttpPost("change-password")]
    [Authorize]
    public async Task<IActionResult> ChangePassword([FromBody] ChangePasswordRequest request)
    {
        var user = await _userManager.GetUserAsync(User);
        if (user == null)
            return Unauthorized(ApiResult.Fail("未登录", 401));

        var result = await _userManager.ChangePasswordAsync(
            user, request.CurrentPassword, request.NewPassword);

        if (!result.Succeeded)
        {
            var errors = string.Join("; ", result.Errors.Select(e => e.Description));
            return BadRequest(ApiResult.Fail(errors));
        }

        return Ok(ApiResult.Ok("密码修改成功"));
    }
}

// DTO
public record LoginRequest(
    string Username,
    string Password,
    bool RememberMe = false,
    string? ReturnUrl = null);

public record ChangePasswordRequest(
    string CurrentPassword,
    string NewPassword);
```

## 7.3 客户端管理 API

```csharp
// Sso.AuthServer/Controllers/ClientManageController.cs
using Microsoft.AspNetCore.Authorization;
using Microsoft.AspNetCore.Mvc;
using Sso.Application.Contracts.Clients;
using Sso.Application.Services;
using Sso.Shared.Results;

namespace Sso.AuthServer.Controllers;

[ApiController]
[Route("api/manage/clients")]
[Authorize(Roles = "admin")] // 需要管理员角色
public class ClientManageController : ControllerBase
{
    private readonly ClientAppService _clientAppService;

    public ClientManageController(ClientAppService clientAppService)
    {
        _clientAppService = clientAppService;
    }

    [HttpGet]
    public async Task<IActionResult> GetAll(CancellationToken ct)
    {
        var clients = await _clientAppService.GetAllAsync(ct);
        return Ok(ApiResult<List<ClientDto>>.Ok(clients));
    }

    [HttpGet("{clientId}")]
    public async Task<IActionResult> GetByClientId(string clientId, CancellationToken ct)
    {
        var client = await _clientAppService.GetByClientIdAsync(clientId, ct);
        if (client == null)
            return NotFound(ApiResult.Fail("客户端不存在", 404));

        return Ok(ApiResult<ClientDto>.Ok(client));
    }

    [HttpPost]
    public async Task<IActionResult> Create(
        [FromBody] CreateClientRequest request, CancellationToken ct)
    {
        try
        {
            var client = await _clientAppService.CreateAsync(request, ct);
            return Ok(ApiResult<ClientDto>.Ok(client, "客户端创建成功"));
        }
        catch (InvalidOperationException ex)
        {
            return BadRequest(ApiResult.Fail(ex.Message));
        }
    }

    [HttpPut("{clientId}")]
    public async Task<IActionResult> Update(
        string clientId,
        [FromBody] UpdateClientRequest request,
        CancellationToken ct)
    {
        try
        {
            await _clientAppService.UpdateAsync(clientId, request, ct);
            return Ok(ApiResult.Ok("更新成功"));
        }
        catch (KeyNotFoundException ex)
        {
            return NotFound(ApiResult.Fail(ex.Message, 404));
        }
    }

    [HttpDelete("{clientId}")]
    public async Task<IActionResult> Delete(string clientId, CancellationToken ct)
    {
        try
        {
            await _clientAppService.DeleteAsync(clientId, ct);
            return Ok(ApiResult.Ok("删除成功"));
        }
        catch (KeyNotFoundException ex)
        {
            return NotFound(ApiResult.Fail(ex.Message, 404));
        }
    }
}
```

```csharp
// Sso.AuthServer/Controllers/UserManageController.cs
using Microsoft.AspNetCore.Authorization;
using Microsoft.AspNetCore.Mvc;
using Sso.Application.Contracts.Users;
using Sso.Application.Services;
using Sso.Shared.Results;

namespace Sso.AuthServer.Controllers;

[ApiController]
[Route("api/manage/users")]
[Authorize(Roles = "admin")]
public class UserManageController : ControllerBase
{
    private readonly UserAppService _userAppService;

    public UserManageController(UserAppService userAppService)
    {
        _userAppService = userAppService;
    }

    [HttpGet]
    public async Task<IActionResult> GetPaged([FromQuery] PagedQuery query)
    {
        var result = await _userAppService.GetPagedAsync(query);
        return Ok(ApiResult<PagedResult<UserDto>>.Ok(result));
    }

    [HttpGet("{id:guid}")]
    public async Task<IActionResult> GetById(Guid id)
    {
        var user = await _userAppService.GetByIdAsync(id);
        if (user == null)
            return NotFound(ApiResult.Fail("用户不存在", 404));
        return Ok(ApiResult<UserDto>.Ok(user));
    }

    [HttpPost]
    public async Task<IActionResult> Create([FromBody] CreateUserRequest request)
    {
        try
        {
            var user = await _userAppService.CreateAsync(request);
            return Ok(ApiResult<UserDto>.Ok(user, "用户创建成功"));
        }
        catch (InvalidOperationException ex)
        {
            return BadRequest(ApiResult.Fail(ex.Message));
        }
    }

    [HttpPut("{id:guid}")]
    public async Task<IActionResult> Update(Guid id, [FromBody] UpdateUserRequest request)
    {
        try
        {
            await _userAppService.UpdateAsync(id, request);
            return Ok(ApiResult.Ok("更新成功"));
        }
        catch (KeyNotFoundException ex)
        {
            return NotFound(ApiResult.Fail(ex.Message, 404));
        }
    }

    [HttpPost("{id:guid}/reset-password")]
    public async Task<IActionResult> ResetPassword(Guid id, [FromBody] ResetPasswordRequest request)
    {
        try
        {
            await _userAppService.ResetPasswordAsync(id, request.NewPassword);
            return Ok(ApiResult.Ok("密码重置成功"));
        }
        catch (Exception ex)
        {
            return BadRequest(ApiResult.Fail(ex.Message));
        }
    }
}

public record ResetPasswordRequest(string NewPassword);
```
