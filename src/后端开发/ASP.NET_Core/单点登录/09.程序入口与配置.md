# Program.cs 主入口与配置

## 第八步：Program.cs 主入口

```csharp
// Sso.AuthServer/Program.cs
using Sso.Application;
using Sso.Infrastructure;
using Sso.Infrastructure.OpenIddict;
using Sso.Shared.Options;

var builder = WebApplication.CreateBuilder(args);

// ====== 1. 配置（已有的从数据库加载配置的逻辑在此之前已执行） ======
var configuration = builder.Configuration;

// 自动注册 [ConfigSection] 标注的选项类
builder.Services.AddAutoOptions(configuration,
    typeof(SsoOpenIddictOptions).Assembly);  // Sso.Shared

// ====== 2. 基础设施层 ======
builder.Services.AddSsoInfrastructure(configuration);

// ====== 3. OpenIddict ======
builder.Services.AddSsoOpenIddict(configuration, builder.Environment);

// ====== 4. 应用服务层 ======
builder.Services.AddSsoApplication();

// ====== 5. ASP.NET Core ======
builder.Services.AddControllers();
builder.Services.AddEndpointsApiExplorer();
builder.Services.AddSwaggerGen();

// CORS（React 前端）
builder.Services.AddCors(options =>
{
    options.AddPolicy("SsoFrontend", policy =>
    {
        policy.WithOrigins(
                configuration.GetSection("Cors:Origins").Get<string[]>() ?? ["http://localhost:3000"])
              .AllowAnyHeader()
              .AllowAnyMethod()
              .AllowCredentials();
    });
});

// Cookie 配置（用于授权码流程中的用户认证）
builder.Services.ConfigureApplicationCookie(options =>
{
    options.Cookie.Name = "sso.auth";
    options.Cookie.HttpOnly = true;
    options.Cookie.SameSite = SameSiteMode.Lax;
    options.Cookie.SecurePolicy = CookieSecurePolicy.SameAsRequest;
    options.ExpireTimeSpan = TimeSpan.FromHours(2);
    options.SlidingExpiration = true;

    // API 模式：不自动重定向，返回 401
    options.Events.OnRedirectToLogin = context =>
    {
        context.Response.StatusCode = 401;
        return Task.CompletedTask;
    };
    options.Events.OnRedirectToAccessDenied = context =>
    {
        context.Response.StatusCode = 403;
        return Task.CompletedTask;
    };
});

var app = builder.Build();

// ====== 中间件管道 ======
if (app.Environment.IsDevelopment())
{
    app.UseSwagger();
    app.UseSwaggerUI();
}

app.UseHttpsRedirection();
app.UseCors("SsoFrontend");

app.UseAuthentication();
app.UseAuthorization();

app.MapControllers();

// ====== 初始化种子数据 ======
await SeedDataAsync(app);

app.Run();

// ====== 种子数据 ======
static async Task SeedDataAsync(WebApplication app)
{
    using var scope = app.Services.CreateScope();
    var services = scope.ServiceProvider;

    // 确保数据库已创建/迁移
    var context = services.GetRequiredService<Sso.Infrastructure.Data.SsoDbContext>();
    await context.Database.EnsureCreatedAsync();

    // 创建管理员用户
    var userManager = services.GetRequiredService<Microsoft.AspNetCore.Identity.UserManager<Sso.Domain.Entities.SsoUser>>();
    var roleManager = services.GetRequiredService<Microsoft.AspNetCore.Identity.RoleManager<Sso.Domain.Entities.SsoRole>>();

    if (!await roleManager.RoleExistsAsync("admin"))
    {
        await roleManager.CreateAsync(new Sso.Domain.Entities.SsoRole
        {
            Name = "admin",
            Description = "系统管理员"
        });
    }

    var adminUser = await userManager.FindByNameAsync("admin");
    if (adminUser == null)
    {
        adminUser = new Sso.Domain.Entities.SsoUser
        {
            UserName = "admin",
            Email = "admin@sso.local",
            DisplayName = "系统管理员",
            IsActive = true,
            EmailConfirmed = true
        };

        await userManager.CreateAsync(adminUser, "Admin@123456");
        await userManager.AddToRoleAsync(adminUser, "admin");
    }

    // 创建默认 Scope
    var scopeManager = services.GetRequiredService<OpenIddict.Abstractions.IOpenIddictScopeManager>();
    if (await scopeManager.FindByNameAsync("api") == null)
    {
        await scopeManager.CreateAsync(new OpenIddict.Abstractions.OpenIddictScopeDescriptor
        {
            Name = "api",
            DisplayName = "API 访问",
            Resources = { "resource-server" }
        });
    }
}
```
