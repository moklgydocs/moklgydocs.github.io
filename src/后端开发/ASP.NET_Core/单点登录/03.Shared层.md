# Shared 层 — 统一返回结果 & 选项基础

## 2.1 统一 API 返回

```csharp
// Sso.Shared/Results/ApiResult.cs
namespace Sso.Shared.Results;

public class ApiResult<T>
{
    public bool Success { get; init; }
    public int Code { get; init; }
    public string? Message { get; init; }
    public T? Data { get; init; }
    public IDictionary<string, string[]>? Errors { get; init; }

    public static ApiResult<T> Ok(T data, string? message = null) => new()
    {
        Success = true,
        Code = 200,
        Message = message ?? "操作成功",
        Data = data
    };

    public static ApiResult<T> Fail(string message, int code = 400) => new()
    {
        Success = false,
        Code = code,
        Message = message
    };

    public static ApiResult<T> ValidationFail(IDictionary<string, string[]> errors) => new()
    {
        Success = false,
        Code = 422,
        Message = "参数校验失败",
        Errors = errors
    };
}

public class ApiResult : ApiResult<object>
{
    public static ApiResult Ok(string? message = null) => new()
    {
        Success = true,
        Code = 200,
        Message = message ?? "操作成功"
    };

    public new static ApiResult Fail(string message, int code = 400) => new()
    {
        Success = false,
        Code = code,
        Message = message
    };
}
```

## 2.2 分页

```csharp
// Sso.Shared/Results/PagedResult.cs
namespace Sso.Shared.Results;

public class PagedResult<T>
{
    public IReadOnlyList<T> Items { get; init; } = [];
    public int TotalCount { get; init; }
    public int PageIndex { get; init; }
    public int PageSize { get; init; }
    public int TotalPages => (int)Math.Ceiling(TotalCount / (double)PageSize);
    public bool HasPrevious => PageIndex > 1;
    public bool HasNext => PageIndex < TotalPages;
}

public class PagedQuery
{
    private int _pageIndex = 1;
    private int _pageSize = 20;

    public int PageIndex
    {
        get => _pageIndex;
        set => _pageIndex = value < 1 ? 1 : value;
    }

    public int PageSize
    {
        get => _pageSize;
        set => _pageSize = value switch
        {
            < 1 => 20,
            > 100 => 100,
            _ => value
        };
    }

    public string? Keyword { get; set; }
}
```

## 2.3 已有的选项绑定特性 (假设有类似这样的)

```csharp
// Sso.Shared/Options/ConfigSectionAttribute.cs
namespace Sso.Shared.Options;

/// <summary>
/// 标记选项类对应的配置节名称，自动绑定
/// </summary>
[AttributeUsage(AttributeTargets.Class, Inherited = false)]
public sealed class ConfigSectionAttribute : Attribute
{
    public string SectionName { get; }

    public ConfigSectionAttribute(string sectionName)
    {
        SectionName = sectionName;
    }
}
```

```csharp
// Sso.Shared/Options/OptionsRegistrationExtensions.cs
using System.Reflection;
using Microsoft.Extensions.Configuration;
using Microsoft.Extensions.DependencyInjection;

namespace Sso.Shared.Options;

public static class OptionsRegistrationExtensions
{
    /// <summary>
    /// 扫描程序集，自动注册带 [ConfigSection] 特性的选项类
    /// </summary>
    public static IServiceCollection AddAutoOptions(
        this IServiceCollection services,
        IConfiguration configuration,
        params Assembly[] assemblies)
    {
        var optionTypes = assemblies
            .SelectMany(a => a.GetExportedTypes())
            .Where(t => t.GetCustomAttribute<ConfigSectionAttribute>() != null && t.IsClass && !t.IsAbstract);

        foreach (var type in optionTypes)
        {
            var attr = type.GetCustomAttribute<ConfigSectionAttribute>()!;
            var section = configuration.GetSection(attr.SectionName);

            // 调用 services.Configure<T>(section)
            var method = typeof(OptionsConfigurationServiceCollectionExtensions)
                .GetMethods()
                .First(m => m.Name == "Configure"
                            && m.GetParameters().Length == 2
                            && m.GetParameters()[1].ParameterType == typeof(IConfiguration));

            var generic = method.MakeGenericMethod(type);
            generic.Invoke(null, [services, section]);
        }

        return services;
    }
}
```
