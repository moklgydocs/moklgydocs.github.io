## 4.2 DbContext

```csharp
// Sso.Infrastructure/Data/SsoDbContext.cs
using Microsoft.AspNetCore.Identity.EntityFrameworkCore;
using Microsoft.EntityFrameworkCore;
using Sso.Domain.Entities;

namespace Sso.Infrastructure.Data;

public class SsoDbContext : IdentityDbContext<SsoUser, SsoRole, Guid>
{
    public SsoDbContext(DbContextOptions<SsoDbContext> options) : base(options) { }

    protected override void OnModelCreating(ModelBuilder builder)
    {
        base.OnModelCreating(builder);

        // 自定义表名前缀，避免与业务表冲突
        builder.Entity<SsoUser>(b =>
        {
            b.ToTable("SsoUsers");
            b.Property(u => u.DisplayName).HasMaxLength(100);
            b.Property(u => u.Avatar).HasMaxLength(500);
            b.HasIndex(u => u.TenantId);
        });

        builder.Entity<SsoRole>(b =>
        {
            b.ToTable("SsoRoles");
            b.Property(r => r.Description).HasMaxLength(256);
        });

        // 统一前缀
        builder.Entity<Microsoft.AspNetCore.Identity.IdentityUserRole<Guid>>().ToTable("SsoUserRoles");
        builder.Entity<Microsoft.AspNetCore.Identity.IdentityUserClaim<Guid>>().ToTable("SsoUserClaims");
        builder.Entity<Microsoft.AspNetCore.Identity.IdentityUserLogin<Guid>>().ToTable("SsoUserLogins");
        builder.Entity<Microsoft.AspNetCore.Identity.IdentityUserToken<Guid>>().ToTable("SsoUserTokens");
        builder.Entity<Microsoft.AspNetCore.Identity.IdentityRoleClaim<Guid>>().ToTable("SsoRoleClaims");
    }
}
```

## 4.3 基础设施注册

```csharp
// Sso.Infrastructure/InfrastructureServiceExtensions.cs
using Microsoft.EntityFrameworkCore;
using Microsoft.Extensions.Configuration;
using Microsoft.Extensions.DependencyInjection;
using Sso.Domain.Entities;
using Sso.Infrastructure.Data;

namespace Sso.Infrastructure;

public static class InfrastructureServiceExtensions
{
    public static IServiceCollection AddSsoInfrastructure(
        this IServiceCollection services,
        IConfiguration configuration)
    {
        var connectionString = configuration.GetConnectionString("Default")
            ?? throw new InvalidOperationException("连接字符串 'Default' 未配置");

        // 1. EF Core
        services.AddDbContext<SsoDbContext>(options =>
        {
            options.UseSqlServer(connectionString, sql =>
            {
                sql.MigrationsAssembly(typeof(SsoDbContext).Assembly.FullName);
                sql.EnableRetryOnFailure(3);
            });

            // 注册 OpenIddict 实体
            options.UseOpenIddict();
        });

        // 2. ASP.NET Core Identity
        services.AddIdentity<SsoUser, SsoRole>(options =>
        {
            // 密码策略
            options.Password.RequiredLength = 8;
            options.Password.RequireDigit = true;
            options.Password.RequireLowercase = true;
            options.Password.RequireUppercase = false;
            options.Password.RequireNonAlphanumeric = false;

            // 锁定策略
            options.Lockout.MaxFailedAccessAttempts = 5;
            options.Lockout.DefaultLockoutTimeSpan = TimeSpan.FromMinutes(15);
            options.Lockout.AllowedForNewUsers = true;

            // 用户
            options.User.RequireUniqueEmail = true;

            // Claims identity
            options.ClaimsIdentity.UserNameClaimType = OpenIddict.Abstractions.OpenIddictConstants.Claims.Name;
            options.ClaimsIdentity.UserIdClaimType = OpenIddict.Abstractions.OpenIddictConstants.Claims.Subject;
            options.ClaimsIdentity.RoleClaimType = OpenIddict.Abstractions.OpenIddictConstants.Claims.Role;
            options.ClaimsIdentity.EmailClaimType = OpenIddict.Abstractions.OpenIddictConstants.Claims.Email;
        })
        .AddEntityFrameworkStores<SsoDbContext>()
        .AddDefaultTokenProviders();

        return services;
    }
}
```
