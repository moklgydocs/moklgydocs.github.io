# 客户端没有被授权使用 `password` 授权类型问题处理

## 问题定位

让我们先查看数据库中 `react-app` 客户端实际存储了哪些权限：

```bash
# 查看客户端信息
curl -X GET https://localhost:7008/api/manage/clients/react-app \
  -b cookies.txt
```

**核心问题在 `ClientAppService.CreateAsync` 中的权限注册逻辑有漏洞：**

```
创建客户端时你传的 GrantTypes = ["authorization_code", "refresh_token"]
但测试用的是 grant_type=password  ← 这个根本没注册！

即使你传了 password，Scope 权限的注册逻辑也有 bug
```

## 完整修复

### 1. 修复 ClientAppService 的权限构建逻辑

```csharp
// Sso.Application/Services/ClientAppService.cs
using OpenIddict.Abstractions;
using Sso.Application.Contracts.Clients;
using Sso.Domain.Entities;
using Sso.Shared.Results;
using static OpenIddict.Abstractions.OpenIddictConstants;

namespace Sso.Application.Services;

public class ClientAppService
{
    private readonly IOpenIddictApplicationManager _applicationManager;
    private readonly Infrastructure.Data.SsoDbContext _dbContext;

    public ClientAppService(
        IOpenIddictApplicationManager applicationManager,
        Infrastructure.Data.SsoDbContext dbContext)
    {
        _applicationManager = applicationManager;
        _dbContext = dbContext;
    }

    public async Task<ClientDto> CreateAsync(CreateClientRequest request, CancellationToken ct = default)
    {
        var existing = await _applicationManager.FindByClientIdAsync(request.ClientId, ct);
        if (existing != null)
            throw new InvalidOperationException($"客户端 '{request.ClientId}' 已存在");

        var descriptor = new OpenIddictApplicationDescriptor
        {
            ClientId = request.ClientId,
            DisplayName = request.DisplayName,
            ClientType = request.Type switch
            {
                "public" => ClientTypes.Public,
                _ => ClientTypes.Confidential
            }
        };

        // Secret
        if (request.Type != "public" && !string.IsNullOrEmpty(request.ClientSecret))
            descriptor.ClientSecret = request.ClientSecret;

        // Redirect URIs
        foreach (var uri in request.RedirectUris)
            descriptor.RedirectUris.Add(new Uri(uri));

        foreach (var uri in request.PostLogoutRedirectUris)
            descriptor.PostLogoutRedirectUris.Add(new Uri(uri));

        // ==========================================
        // ▼▼▼ 修复：完整的权限构建 ▼▼▼
        // ==========================================
        BuildPermissions(descriptor, request.GrantTypes, request.Scopes);

        var app = await _applicationManager.CreateAsync(descriptor, ct);

        // 扩展字段
        if (app is SsoApplication ssoApp)
        {
            ssoApp.Description = request.Description;
            ssoApp.Group = request.Group;
            ssoApp.LogoUri = request.LogoUri;
            ssoApp.AccessTokenLifetimeMinutes = request.AccessTokenLifetimeMinutes;
            ssoApp.IsEnabled = true;
            ssoApp.CreatedAt = DateTime.UtcNow;
            await _applicationManager.UpdateAsync(app, ct);
        }

        return await MapToDtoAsync(app, ct);
    }

    /// <summary>
    /// 核心：根据授权类型和 Scope 构建完整的权限列表
    /// OpenIddict 要求三类权限必须同时注册：
    /// 1. GrantType 权限  → 允许使用哪种授权类型
    /// 2. Endpoint 权限   → 允许访问哪些端点
    /// 3. Scope 权限      → 允许请求哪些 Scope
    /// 4. ResponseType 权限 → 允许哪种响应类型（授权码流程需要）
    /// </summary>
    private static void BuildPermissions(
        OpenIddictApplicationDescriptor descriptor,
        List<string> grantTypes,
        List<string> scopes)
    {
        // ====== 1. GrantType 权限 ======
        foreach (var grantType in grantTypes)
        {
            var permission = grantType switch
            {
                "authorization_code" => Permissions.GrantTypes.AuthorizationCode,
                "client_credentials" => Permissions.GrantTypes.ClientCredentials,
                "password"           => Permissions.GrantTypes.Password,
                "refresh_token"      => Permissions.GrantTypes.RefreshToken,
                "device_code"        => Permissions.GrantTypes.DeviceCode,
                _ => throw new ArgumentException($"不支持的授权类型: {grantType}")
            };
            descriptor.Permissions.Add(permission);
        }

        // ====== 2. Endpoint 权限（根据 GrantType 自动推断） ======
        // Token 端点 → 所有授权类型都需要
        descriptor.Permissions.Add(Permissions.Endpoints.Token);

        if (grantTypes.Contains("authorization_code"))
        {
            descriptor.Permissions.Add(Permissions.Endpoints.Authorization);
            descriptor.Permissions.Add(Permissions.Endpoints.Logout);

            // ====== 4. ResponseType 权限（授权码必须） ======
            descriptor.Permissions.Add(Permissions.ResponseTypes.Code);
        }

        if (grantTypes.Contains("device_code"))
        {
            descriptor.Permissions.Add(Permissions.Endpoints.Device);
        }

        // Introspection & Revocation（可选，一般都开）
        descriptor.Permissions.Add(Permissions.Endpoints.Introspection);
        descriptor.Permissions.Add(Permissions.Endpoints.Revocation);

        // ====== 3. Scope 权限 ======
        foreach (var scope in scopes)
        {
            // OpenIddict 的 Scope 权限格式：scp:xxx
            descriptor.Permissions.Add(Permissions.Prefixes.Scope + scope);
        }

        // 确保基础 Scope 都在
        EnsureScopePermission(descriptor, "openid");
        EnsureScopePermission(descriptor, "profile");
        EnsureScopePermission(descriptor, "email");
        EnsureScopePermission(descriptor, "roles");

        // 如果有 refresh_token，确保 offline_access scope
        if (grantTypes.Contains("refresh_token"))
        {
            EnsureScopePermission(descriptor, "offline_access");
        }
    }

    private static void EnsureScopePermission(
        OpenIddictApplicationDescriptor descriptor, string scope)
    {
        var permission = Permissions.Prefixes.Scope + scope;
        if (!descriptor.Permissions.Contains(permission))
        {
            descriptor.Permissions.Add(permission);
        }
    }

    // ... 其余方法保持不变
}
```

### 2. 现在有两种修复路径

#### 路径 A：删除旧客户端重新创建（推荐，干净）

```bash
# 1. 登录
curl -X POST https://localhost:7008/api/account/login \
  -H "Content-Type: application/json" \
  -d '{"username":"admin","password":"Admin@123456"}' \
  -c cookies.txt

# 2. 删除旧的
curl -X DELETE https://localhost:7008/api/manage/clients/react-app \
  -b cookies.txt

# 3. 重新创建（注意 grantTypes 包含 password）
curl -X POST https://localhost:7008/api/manage/clients \
  -H "Content-Type: application/json" \
  -b cookies.txt \
  -d '{
    "clientId": "react-app",
    "displayName": "React 前端应用",
    "type": "public",
    "redirectUris": ["http://localhost:3000/callback"],
    "postLogoutRedirectUris": ["http://localhost:3000"],
    "grantTypes": ["authorization_code", "password", "refresh_token"],
    "scopes": ["openid", "profile", "email", "roles", "api", "offline_access"]
  }'
```

#### 路径 B：通过种子数据自动修复

如果你更喜欢在代码中管理客户端，修改 `Program.cs` 的种子数据：

```csharp
// Program.cs 中 SeedDataAsync 方法
static async Task SeedDataAsync(WebApplication app)
{
    using var scope = app.Services.CreateScope();
    var services = scope.ServiceProvider;

    var context = services.GetRequiredService<Sso.Infrastructure.Data.SsoDbContext>();
    await context.Database.EnsureCreatedAsync();

    // ====== 种子用户 ======
    var userManager = services.GetRequiredService<UserManager<SsoUser>>();
    var roleManager = services.GetRequiredService<RoleManager<SsoRole>>();

    if (!await roleManager.RoleExistsAsync("admin"))
    {
        await roleManager.CreateAsync(new SsoRole
        {
            Name = "admin",
            Description = "系统管理员"
        });
    }

    var adminUser = await userManager.FindByNameAsync("admin");
    if (adminUser == null)
    {
        adminUser = new SsoUser
        {
            UserName = "admin",
            Email = "admin@sso.local",
            DisplayName = "系统管理员",
            IsActive = true,
            EmailConfirmed = true
        };
        await userManager.CreateAsync(adminUser, "Admin@123456");
        await userManager.AddToRoleAsync(adminUser, "admin");
    }

    // ====== 种子 Scope ======
    var scopeManager = services.GetRequiredService<IOpenIddictScopeManager>();
    if (await scopeManager.FindByNameAsync("api") == null)
    {
        await scopeManager.CreateAsync(new OpenIddictScopeDescriptor
        {
            Name = "api",
            DisplayName = "API 访问",
            Resources = { "resource-server" }
        });
    }

    // ====== 种子客户端 ======
    var appManager = services.GetRequiredService<IOpenIddictApplicationManager>();
    await SeedClientAsync(appManager);
}

static async Task SeedClientAsync(IOpenIddictApplicationManager manager)
{
    // ====== React SPA 客户端 ======
    var reactApp = await manager.FindByClientIdAsync("react-app");

    // 如果已存在先删除（开发阶段方便调试，生产环境去掉这行）
    if (reactApp != null)
    {
        await manager.DeleteAsync(reactApp);
    }

    await manager.CreateAsync(new OpenIddictApplicationDescriptor
    {
        ClientId = "react-app",
        DisplayName = "React 前端应用",
        ClientType = ClientTypes.Public,   // SPA 用 public

        RedirectUris =
        {
            new Uri("http://localhost:3000/callback"),
            new Uri("http://localhost:5173/callback")  // Vite 默认端口
        },
        PostLogoutRedirectUris =
        {
            new Uri("http://localhost:3000"),
            new Uri("http://localhost:5173")
        },

        Permissions =
        {
            // ====== GrantType ======
            Permissions.GrantTypes.AuthorizationCode,
            Permissions.GrantTypes.Password,
            Permissions.GrantTypes.RefreshToken,

            // ====== Endpoint ======
            Permissions.Endpoints.Authorization,
            Permissions.Endpoints.Token,
            Permissions.Endpoints.Logout,
            Permissions.Endpoints.Revocation,

            // ====== ResponseType ======
            Permissions.ResponseTypes.Code,

            // ====== Scope ======
            Permissions.Prefixes.Scope + "openid",
            Permissions.Prefixes.Scope + "profile",
            Permissions.Prefixes.Scope + "email",
            Permissions.Prefixes.Scope + "roles",
            Permissions.Prefixes.Scope + "api",
            Permissions.Prefixes.Scope + "offline_access",
        },

        // PKCE 要求（public 客户端安全增强）
        Requirements =
        {
            Requirements.Features.ProofKeyForCodeExchange
        }
    });

    // ====== 后端服务客户端 ======
    var backendApp = await manager.FindByClientIdAsync("backend-service");
    if (backendApp != null)
    {
        await manager.DeleteAsync(backendApp);
    }

    await manager.CreateAsync(new OpenIddictApplicationDescriptor
    {
        ClientId = "backend-service",
        ClientSecret = "backend-secret-123",
        DisplayName = "后端微服务",
        ClientType = ClientTypes.Confidential,

        Permissions =
        {
            Permissions.GrantTypes.ClientCredentials,
            Permissions.Endpoints.Token,
            Permissions.Endpoints.Introspection,
            Permissions.Endpoints.Revocation,
            Permissions.Prefixes.Scope + "api",
        }
    });
}
```

## 权限模型速查表

```
OpenIddict 权限检查逻辑（每次请求都会验证）:

POST /connect/token
  grant_type=password
  client_id=react-app
  scope=openid profile email api

OpenIddict 会检查 react-app 客户端是否同时具有:
  ✅ gt:password              → 允许密码模式
  ✅ ept:token                → 允许访问 token 端点
  ✅ scp:openid               → 允许请求 openid scope
  ✅ scp:profile              → 允许请求 profile scope
  ✅ scp:email                → 允许请求 email scope
  ✅ scp:api                  → 允许请求 api scope

缺任何一个 → 返回 unauthorized_client ❌

权限前缀速查:
  gt:   = GrantType
  ept:  = Endpoint  
  scp:  = Scope
  rst:  = ResponseType
```
