# OpenIddict 核心配置

## 5.1 OpenIddict 选项类（利用的特性绑定）

```csharp
// Sso.Shared/Options/OpenIddictServerOptions.cs
namespace Sso.Shared.Options;

[ConfigSection("Sso:OpenIddict")]
public class SsoOpenIddictOptions
{
    /// <summary>
    /// 签名证书路径（生产环境必须用证书）
    /// </summary>
    public string? SigningCertificatePath { get; set; }
    public string? SigningCertificatePassword { get; set; }

    /// <summary>
    /// 加密证书路径
    /// </summary>
    public string? EncryptionCertificatePath { get; set; }
    public string? EncryptionCertificatePassword { get; set; }

    /// <summary>
    /// Access Token 过期时间（分钟）
    /// </summary>
    public int AccessTokenLifetimeMinutes { get; set; } = 60;

    /// <summary>
    /// Refresh Token 过期时间（天）
    /// </summary>
    public int RefreshTokenLifetimeDays { get; set; } = 14;

    /// <summary>
    /// 是否启用开发者证书（仅开发环境）
    /// </summary>
    public bool UseDevelopmentCertificates { get; set; } = false;

    /// <summary>
    /// Issuer
    /// </summary>
    public string? Issuer { get; set; }
}
```

## 5.2 OpenIddict 注册

```csharp
// Sso.Infrastructure/OpenIddict/OpenIddictServiceExtensions.cs
using System.Security.Cryptography.X509Certificates;
using Microsoft.Extensions.Configuration;
using Microsoft.Extensions.DependencyInjection;
using Microsoft.Extensions.Hosting;
using OpenIddict.Abstractions;
using Sso.Shared.Options;

namespace Sso.Infrastructure.OpenIddict;

public static class OpenIddictServiceExtensions
{
    public static IServiceCollection AddSsoOpenIddict(
        this IServiceCollection services,
        IConfiguration configuration,
        IHostEnvironment environment)
    {
        var oidcOptions = configuration
            .GetSection("Sso:OpenIddict")
            .Get<SsoOpenIddictOptions>() ?? new SsoOpenIddictOptions();

        services.AddOpenIddict()
            // 1. Core: 实体存储
            .AddCore(options =>
            {
                options.UseEntityFrameworkCore()
                       .UseDbContext<Data.SsoDbContext>();

                // 启用 Quartz 自动清理过期 Token（生产必备）
                options.UseQuartz();
            })

            // 2. Server: OAuth2 / OIDC 服务端
            .AddServer(options =>
            {
                // ====== 端点配置 ======
                options.SetAuthorizationEndpointUris("connect/authorize")
                       .SetTokenEndpointUris("connect/token")
                       .SetUserinfoEndpointUris("connect/userinfo")
                       .SetLogoutEndpointUris("connect/logout")
                       .SetIntrospectionEndpointUris("connect/introspect")
                       .SetRevocationEndpointUris("connect/revoke");

                // ====== 支持的授权流程 ======
                options.AllowAuthorizationCodeFlow()   // 标准授权码（Web应用）
                       .AllowRefreshTokenFlow()        // 刷新令牌
                       .AllowClientCredentialsFlow()   // 服务间通信
                       .AllowPasswordFlow();           // 密码模式（移动端/信任客户端，可选）

                // ====== 支持 PKCE（安全增强）======
                options.RequireProofKeyForCodeExchange();

                // ====== Scope 注册 ======
                options.RegisterScopes(
                    OpenIddictConstants.Scopes.Email,
                    OpenIddictConstants.Scopes.Profile,
                    OpenIddictConstants.Scopes.Roles,
                    OpenIddictConstants.Scopes.OfflineAccess,
                    "api"  // 自定义 API scope
                );

                // ====== Token 生命周期 ======
                options.SetAccessTokenLifetime(
                    TimeSpan.FromMinutes(oidcOptions.AccessTokenLifetimeMinutes));
                options.SetRefreshTokenLifetime(
                    TimeSpan.FromDays(oidcOptions.RefreshTokenLifetimeDays));

                // ====== 签名和加密 ======
                if (environment.IsDevelopment() || oidcOptions.UseDevelopmentCertificates)
                {
                    options.AddDevelopmentEncryptionCertificate()
                           .AddDevelopmentSigningCertificate();
                }
                else
                {
                    if (!string.IsNullOrEmpty(oidcOptions.SigningCertificatePath))
                    {
                        var signingCert = new X509Certificate2(
                            oidcOptions.SigningCertificatePath,
                            oidcOptions.SigningCertificatePassword);
                        options.AddSigningCertificate(signingCert);
                    }

                    if (!string.IsNullOrEmpty(oidcOptions.EncryptionCertificatePath))
                    {
                        var encryptionCert = new X509Certificate2(
                            oidcOptions.EncryptionCertificatePath,
                            oidcOptions.EncryptionCertificatePassword);
                        options.AddEncryptionCertificate(encryptionCert);
                    }
                }

                // ====== ASP.NET Core 集成 ======
                options.UseAspNetCore()
                       .EnableAuthorizationEndpointPassthrough()
                       .EnableTokenEndpointPassthrough()
                       .EnableUserinfoEndpointPassthrough()
                       .EnableLogoutEndpointPassthrough()
                       .EnableStatusCodePagesIntegration();
            })

            // 3. Validation: Token 验证（鉴权中心自身也验证 Token）
            .AddValidation(options =>
            {
                options.UseLocalServer();
                options.UseAspNetCore();
            });

        // Quartz 用于自动清理过期 Token
        services.AddQuartz(options =>
        {
            options.UseSimpleTypeLoader();
            options.UseInMemoryStore();
        });
        services.AddQuartzHostedService(options =>
        {
            options.WaitForJobsToComplete = true;
        });

        return services;
    }
}
```
