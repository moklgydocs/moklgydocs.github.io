# 第四章：存储过程与企业级规范

> **写存储过程不难，写"不会给自己挖坑"的存储过程才难。** 规范决定了你的代码能不能上生产。

---

## 一、命名规范

```
✅ 推荐命名：
  存储过程：usp_模块_动作        例：usp_Order_GetByStatus
  函数：    ufn_模块_动作        例：ufn_Order_CalcTotal
  视图：    vw_模块_名称         例：vw_Order_Summary
  索引：    IX_表_列             例：IX_Orders_OrderDate
  主键：    PK_表                例：PK_Orders
  外键：    FK_子表_父表         例：FK_OrderItems_Orders

❌ 不要用 sp_ 前缀
  SQL Server 看到 sp_ 会先去 master 库查找
  多一次额外查找 → 微小性能损失 + 语义不清
```

## 二、存储过程标准模板

```sql
-- ============================================
-- 存储过程标准模板
-- 作者：Moklgy
-- 创建日期：2025-04-13
-- 描述：按状态分页查询订单
-- 修改记录：
--   2025-04-13 Moklgy 初始创建
-- ============================================
CREATE OR ALTER PROCEDURE usp_Order_GetByStatus
    @Status NVARCHAR(50),
    @PageIndex INT = 1,
    @PageSize INT = 20,
    @TotalCount INT OUTPUT
AS
BEGIN
    SET NOCOUNT ON;  -- 🔥 必须加！减少网络传输
    SET XACT_ABORT ON;  -- 🔥 错误时自动回滚事务

    -- 参数校验
    IF @PageIndex < 1 SET @PageIndex = 1;
    IF @PageSize < 1 OR @PageSize > 100 SET @PageSize = 20;

    BEGIN TRY
        -- 查询总数
        SELECT @TotalCount = COUNT(*)
        FROM Orders
        WHERE Status = @Status;

        -- 分页查询
        SELECT 
            o.OrderId, o.OrderDate, o.TotalAmount,
            c.CustomerName
        FROM Orders o
        INNER JOIN Customers c ON o.CustomerId = c.CustomerId
        WHERE o.Status = @Status
        ORDER BY o.OrderDate DESC
        OFFSET (@PageIndex - 1) * @PageSize ROWS 
        FETCH NEXT @PageSize ROWS ONLY;

    END TRY
    BEGIN CATCH
        -- 如果在事务中，回滚
        IF @@TRANCOUNT > 0
            ROLLBACK TRANSACTION;

        -- 记录错误日志
        INSERT INTO ErrorLog(ErrorNumber, ErrorMessage, ErrorProcedure, ErrorLine, ErrorTime)
        VALUES (
            ERROR_NUMBER(),
            ERROR_MESSAGE(),
            ERROR_PROCEDURE(),
            ERROR_LINE(),
            GETDATE()
        );

        -- 重新抛出错误
        THROW;
    END CATCH
END;
GO
```

## 三、事务处理规范

```sql
-- ✅ 标准事务模板
CREATE OR ALTER PROCEDURE usp_Order_Create
    @CustomerId INT,
    @Items udt_OrderItem READONLY,  -- 表值参数
    @OrderId INT OUTPUT
AS
BEGIN
    SET NOCOUNT ON;
    SET XACT_ABORT ON;

    BEGIN TRY
        BEGIN TRANSACTION;

        -- 1. 创建订单头
        INSERT INTO Orders(CustomerId, OrderDate, Status, TotalAmount)
        VALUES (@CustomerId, GETDATE(), 'Pending', 0);
        
        SET @OrderId = SCOPE_IDENTITY();

        -- 2. 创建订单明细
        INSERT INTO OrderItems(OrderId, ProductId, Quantity, UnitPrice)
        SELECT @OrderId, ProductId, Quantity, UnitPrice
        FROM @Items;

        -- 3. 更新订单总额
        UPDATE Orders 
        SET TotalAmount = (
            SELECT SUM(Quantity * UnitPrice) 
            FROM OrderItems WHERE OrderId = @OrderId
        )
        WHERE OrderId = @OrderId;

        -- 4. 扣库存（原子操作）
        UPDATE p SET p.Stock = p.Stock - i.Quantity
        FROM Products p
        INNER JOIN @Items i ON p.ProductId = i.ProductId
        WHERE p.Stock >= i.Quantity;

        -- 检查是否所有商品都扣成功
        IF @@ROWCOUNT <> (SELECT COUNT(*) FROM @Items)
        BEGIN
            RAISERROR('库存不足', 16, 1);
            -- XACT_ABORT=ON 时，RAISERROR severity>=16 会自动触发CATCH
        END

        COMMIT TRANSACTION;
    END TRY
    BEGIN CATCH
        IF @@TRANCOUNT > 0
            ROLLBACK TRANSACTION;
        THROW;
    END CATCH
END;
GO
```

## 四、动态SQL

### 4.1 安全的动态SQL（防注入）

```sql
-- ❌ 危险！SQL注入
DECLARE @sql NVARCHAR(MAX);
SET @sql = 'SELECT * FROM Users WHERE Name = ''' + @UserInput + '''';
EXEC(@sql);
-- 如果 @UserInput = "'; DROP TABLE Users; --"  → 灾难！

-- ✅ 参数化动态SQL
DECLARE @sql NVARCHAR(MAX);
DECLARE @params NVARCHAR(MAX);

SET @sql = N'SELECT * FROM Users WHERE Name = @Name';
SET @params = N'@Name NVARCHAR(100)';

EXEC sp_executesql @sql, @params, @Name = @UserInput;
-- 参数化后，输入内容只会被当作值，不会被当作SQL
```

### 4.2 动态条件查询

```sql
-- 场景：多条件可选过滤（管理后台常见需求）
CREATE OR ALTER PROCEDURE usp_Order_Search
    @Status NVARCHAR(50) = NULL,
    @CustomerId INT = NULL,
    @StartDate DATE = NULL,
    @EndDate DATE = NULL,
    @MinAmount DECIMAL(18,2) = NULL
AS
BEGIN
    SET NOCOUNT ON;

    DECLARE @sql NVARCHAR(MAX);
    DECLARE @params NVARCHAR(MAX);

    SET @sql = N'
        SELECT o.OrderId, o.OrderDate, o.TotalAmount, o.Status,
               c.CustomerName
        FROM Orders o
        INNER JOIN Customers c ON o.CustomerId = c.CustomerId
        WHERE 1=1';  -- 1=1 技巧：方便拼接AND

    IF @Status IS NOT NULL
        SET @sql += N' AND o.Status = @Status';
    
    IF @CustomerId IS NOT NULL
        SET @sql += N' AND o.CustomerId = @CustomerId';
    
    IF @StartDate IS NOT NULL
        SET @sql += N' AND o.OrderDate >= @StartDate';
    
    IF @EndDate IS NOT NULL
        SET @sql += N' AND o.OrderDate < DATEADD(DAY, 1, @EndDate)';
    
    IF @MinAmount IS NOT NULL
        SET @sql += N' AND o.TotalAmount >= @MinAmount';

    SET @sql += N' ORDER BY o.OrderDate DESC';

    SET @params = N'
        @Status NVARCHAR(50),
        @CustomerId INT,
        @StartDate DATE,
        @EndDate DATE,
        @MinAmount DECIMAL(18,2)';

    EXEC sp_executesql @sql, @params, 
        @Status, @CustomerId, @StartDate, @EndDate, @MinAmount;
END;
GO
```

### 4.3 非动态SQL替代方案（推荐优先考虑）

```sql
-- 用 OR + IS NULL 技巧避免动态SQL
SELECT o.OrderId, o.OrderDate, o.TotalAmount, c.CustomerName
FROM Orders o
INNER JOIN Customers c ON o.CustomerId = c.CustomerId
WHERE (@Status IS NULL OR o.Status = @Status)
  AND (@CustomerId IS NULL OR o.CustomerId = @CustomerId)
  AND (@StartDate IS NULL OR o.OrderDate >= @StartDate)
  AND (@EndDate IS NULL OR o.OrderDate < DATEADD(DAY, 1, @EndDate))
  AND (@MinAmount IS NULL OR o.TotalAmount >= @MinAmount)
ORDER BY o.OrderDate DESC
OPTION (RECOMPILE);  -- 🔥 关键！每次重新编译，避免参数嗅探
-- 不加 RECOMPILE → 第一次NULL参数编译的计划可能不适合后续非NULL参数
```

## 五、批量操作

### 5.1 表值参数（TVP）

```sql
-- 1. 创建表类型
CREATE TYPE udt_OrderItem AS TABLE
(
    ProductId INT NOT NULL,
    Quantity INT NOT NULL,
    UnitPrice DECIMAL(18,2) NOT NULL
);
GO

-- 2. 存储过程接收TVP
CREATE PROCEDURE usp_OrderItem_BatchInsert
    @OrderId INT,
    @Items udt_OrderItem READONLY  -- 必须是READONLY
AS
BEGIN
    INSERT INTO OrderItems(OrderId, ProductId, Quantity, UnitPrice)
    SELECT @OrderId, ProductId, Quantity, UnitPrice
    FROM @Items;
END;
GO
```

```csharp
// C# 传递 TVP
var table = new DataTable();
table.Columns.Add("ProductId", typeof(int));
table.Columns.Add("Quantity", typeof(int));
table.Columns.Add("UnitPrice", typeof(decimal));

foreach (var item in orderItems)
{
    table.Rows.Add(item.ProductId, item.Quantity, item.UnitPrice);
}

using var cmd = new SqlCommand("usp_OrderItem_BatchInsert", connection);
cmd.CommandType = CommandType.StoredProcedure;
cmd.Parameters.AddWithValue("@OrderId", orderId);
cmd.Parameters.AddWithValue("@Items", table);
await cmd.ExecuteNonQueryAsync();
```

### 5.2 MERGE 语句（Upsert）

```sql
-- 存在则更新，不存在则插入
MERGE INTO Products AS target
USING @ProductUpdates AS source
ON target.ProductId = source.ProductId
WHEN MATCHED THEN
    UPDATE SET 
        target.Name = source.Name,
        target.Price = source.Price,
        target.UpdatedAt = GETDATE()
WHEN NOT MATCHED BY TARGET THEN
    INSERT (ProductId, Name, Price, CreatedAt)
    VALUES (source.ProductId, source.Name, source.Price, GETDATE())
WHEN NOT MATCHED BY SOURCE THEN
    DELETE;  -- 源中没有的记录删除（可选）
-- ⚠️ MERGE 有已知bug，建议加 HOLDLOCK 提示
-- 或在高并发下用 INSERT + UPDATE 替代
```

### 5.3 大批量数据处理（分批）

```sql
-- 分批删除（避免长事务+锁升级）
DECLARE @BatchSize INT = 5000;
DECLARE @Deleted INT = 1;

WHILE @Deleted > 0
BEGIN
    DELETE TOP (@BatchSize) FROM OrderLogs
    WHERE CreatedAt < DATEADD(YEAR, -2, GETDATE());
    
    SET @Deleted = @@ROWCOUNT;
    
    -- 每批之间短暂等待，让其他事务有机会执行
    WAITFOR DELAY '00:00:01';
END
```

## 六、错误日志表设计

```sql
CREATE TABLE ErrorLog
(
    ErrorLogId BIGINT IDENTITY(1,1) PRIMARY KEY,
    ErrorNumber INT,
    ErrorSeverity INT,
    ErrorState INT,
    ErrorProcedure NVARCHAR(200),
    ErrorLine INT,
    ErrorMessage NVARCHAR(4000),
    ErrorTime DATETIME2 DEFAULT GETDATE(),
    UserName NVARCHAR(128) DEFAULT SYSTEM_USER,
    HostName NVARCHAR(128) DEFAULT HOST_NAME()
);

-- 通用错误记录过程
CREATE OR ALTER PROCEDURE usp_Error_Log
AS
BEGIN
    INSERT INTO ErrorLog(ErrorNumber, ErrorSeverity, ErrorState, 
                         ErrorProcedure, ErrorLine, ErrorMessage)
    VALUES (
        ERROR_NUMBER(),
        ERROR_SEVERITY(),
        ERROR_STATE(),
        ERROR_PROCEDURE(),
        ERROR_LINE(),
        ERROR_MESSAGE()
    );
END;
GO
```
