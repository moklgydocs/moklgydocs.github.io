# 格式化与数据处理工具函数：中后台的瑞士军刀

> **TL;DR**
> 每个中后台项目的 `src/utils/` 目录都需要一套"瑞士军刀"——字符串格式化、数字处理、日期工具、对象/数组操作。这些函数看似简单，但在 200 个页面中被调用上千次，一个 Bug 就会波及全系统。本章提供一套**经过 TypeScript 严格类型化、零依赖（或轻量依赖）**的生产级工具函数集。

## 1. 核心架构：utils 目录组织

```
src/utils/
├── format.ts        # 字符串与格式化
├── number.ts        # 数字与货币
├── date.ts          # 日期与时间
├── object.ts        # 对象与数组操作
├── storage.ts       # 本地存储
├── dom.ts           # DOM 操作工具
├── validator.ts     # 校验工具
├── cn.ts            # 样式合并（shadcn/ui 标配）
└── index.ts         # 统一导出
```

## 2. 字符串格式化工具

```tsx
// src/utils/format.ts — 字符串处理

/**
 * 文本截断，超长时追加省略号
 * truncateText("GlobalMart跨国电商平台管理系统", 10) → "GlobalMart..."
 */
function truncateText(text: string, maxLength: number, suffix = "..."): string {
  if (text.length <= maxLength) return text;
  return text.slice(0, maxLength) + suffix;
}

/**
 * 手机号脱敏：13812345678 → 138****5678
 */
function maskPhone(phone: string): string {
  return phone.replace(/(\d{3})\d{4}(\d{4})/, "$1****$2");
}

/**
 * 邮箱脱敏：john.doe@gmail.com → j*****e@gmail.com
 */
function maskEmail(email: string): string {
  const [name, domain] = email.split("@");
  if (!domain || name.length <= 2) return email;
  return `${name[0]}${"*".repeat(Math.min(name.length - 2, 5))}${name[name.length - 1]}@${domain}`;
}

/**
 * 银行卡号格式化：6222023456789012 → 6222 0234 5678 9012
 */
function formatBankCard(cardNo: string): string {
  return cardNo.replace(/\s/g, "").replace(/(\d{4})(?=\d)/g, "$1 ");
}

/**
 * 文件大小格式化：1536 → "1.5 KB"
 */
function formatFileSize(bytes: number): string {
  if (bytes === 0) return "0 B";
  const units = ["B", "KB", "MB", "GB", "TB"];
  const i = Math.floor(Math.log(bytes) / Math.log(1024));
  const size = bytes / Math.pow(1024, i);
  return `${size.toFixed(size >= 10 ? 0 : 1)} ${units[i]}`;
}

/**
 * 驼峰转短横线：orderStatus → order-status
 */
function camelToKebab(str: string): string {
  return str.replace(/([A-Z])/g, "-$1").toLowerCase().replace(/^-/, "");
}

/**
 * 短横线转驼峰：order-status → orderStatus
 */
function kebabToCamel(str: string): string {
  return str.replace(/-([a-z])/g, (_, char) => char.toUpperCase());
}

/**
 * 首字母大写：hello world → Hello world
 */
function capitalize(str: string): string {
  return str.charAt(0).toUpperCase() + str.slice(1);
}

/**
 * 生成随机 ID（非加密用途，用于 DOM id 或临时标识）
 */
function generateId(prefix = "id"): string {
  return `${prefix}_${Math.random().toString(36).slice(2, 11)}`;
}

export {
  truncateText,
  maskPhone,
  maskEmail,
  formatBankCard,
  formatFileSize,
  camelToKebab,
  kebabToCamel,
  capitalize,
  generateId,
};
```

## 3. 数字与货币工具

```tsx
// src/utils/number.ts — 数字格式化与计算

/**
 * 货币格式化（支持多币种）
 * formatCurrency(12345.6, "CNY") → "¥12,345.60"
 * formatCurrency(12345.6, "USD") → "$12,345.60"
 */
function formatCurrency(
  amount: number,
  currency: string = "CNY",
  locale: string = "zh-CN"
): string {
  return new Intl.NumberFormat(locale, {
    style: "currency",
    currency,
    minimumFractionDigits: 2,
    maximumFractionDigits: 2,
  }).format(amount);
}

/**
 * 千分位格式化：1234567 → "1,234,567"
 */
function formatNumber(num: number, locale: string = "zh-CN"): string {
  return new Intl.NumberFormat(locale).format(num);
}

/**
 * 大数缩写：12345 → "1.2万"，123456789 → "1.2亿"
 */
function abbreviateNumber(num: number): string {
  if (Math.abs(num) >= 1_0000_0000) {
    return `${(num / 1_0000_0000).toFixed(1)}亿`;
  }
  if (Math.abs(num) >= 1_0000) {
    return `${(num / 1_0000).toFixed(1)}万`;
  }
  return String(num);
}

/**
 * 百分比计算（安全除法，分母为 0 时返回 0）
 * calcPercent(3, 10) → 30
 * calcPercent(3, 10, 1) → 30.0
 */
function calcPercent(
  numerator: number,
  denominator: number,
  decimals: number = 0
): number {
  if (denominator === 0) return 0;
  return Number(((numerator / denominator) * 100).toFixed(decimals));
}

/**
 * 百分比显示：calcPercent 的格式化版本
 * formatPercent(3, 10) → "30%"
 */
function formatPercent(
  numerator: number,
  denominator: number,
  decimals: number = 1
): string {
  return `${calcPercent(numerator, denominator, decimals)}%`;
}

/**
 * 安全的浮点数加法（解决 0.1 + 0.2 !== 0.3 问题）
 */
function safeAdd(a: number, b: number): number {
  const precision = Math.max(
    (a.toString().split(".")[1] || "").length,
    (b.toString().split(".")[1] || "").length
  );
  const factor = Math.pow(10, precision);
  return (Math.round(a * factor) + Math.round(b * factor)) / factor;
}

/**
 * 将数字限制在指定范围内
 * clamp(150, 0, 100) → 100
 */
function clamp(value: number, min: number, max: number): number {
  return Math.min(Math.max(value, min), max);
}

export {
  formatCurrency,
  formatNumber,
  abbreviateNumber,
  calcPercent,
  formatPercent,
  safeAdd,
  clamp,
};
```

## 4. 日期与时间工具

```tsx
// src/utils/date.ts — 日期处理（轻量版，不依赖 date-fns 也能用）

/**
 * 格式化日期
 * formatDate(new Date()) → "2026-04-19"
 * formatDate(new Date(), "YYYY/MM/DD HH:mm") → "2026/04/19 14:30"
 */
function formatDate(
  date: Date | string | number,
  format: string = "YYYY-MM-DD"
): string {
  const d = new Date(date);
  if (isNaN(d.getTime())) return "";

  const map: Record<string, string> = {
    YYYY: String(d.getFullYear()),
    MM: String(d.getMonth() + 1).padStart(2, "0"),
    DD: String(d.getDate()).padStart(2, "0"),
    HH: String(d.getHours()).padStart(2, "0"),
    mm: String(d.getMinutes()).padStart(2, "0"),
    ss: String(d.getSeconds()).padStart(2, "0"),
  };

  return format.replace(/YYYY|MM|DD|HH|mm|ss/g, (match) => map[match]);
}

/**
 * 相对时间：3 分钟前、2 小时前、昨天、3 天前
 */
function timeAgo(date: Date | string | number): string {
  const now = Date.now();
  const then = new Date(date).getTime();
  const diff = now - then;

  const seconds = Math.floor(diff / 1000);
  const minutes = Math.floor(seconds / 60);
  const hours = Math.floor(minutes / 60);
  const days = Math.floor(hours / 24);

  if (seconds < 60) return "刚刚";
  if (minutes < 60) return `${minutes} 分钟前`;
  if (hours < 24) return `${hours} 小时前`;
  if (days === 1) return "昨天";
  if (days < 30) return `${days} 天前`;
  if (days < 365) return `${Math.floor(days / 30)} 个月前`;
  return `${Math.floor(days / 365)} 年前`;
}

/**
 * 判断是否是今天
 */
function isToday(date: Date | string | number): boolean {
  const d = new Date(date);
  const today = new Date();
  return (
    d.getFullYear() === today.getFullYear() &&
    d.getMonth() === today.getMonth() &&
    d.getDate() === today.getDate()
  );
}

/**
 * 获取日期范围快捷方式（用于中后台的日期筛选）
 */
type DateRangePreset = "today" | "yesterday" | "last7days" | "last30days" | "thisMonth" | "lastMonth";

function getDateRange(preset: DateRangePreset): [Date, Date] {
  const now = new Date();
  const startOfToday = new Date(now.getFullYear(), now.getMonth(), now.getDate());

  switch (preset) {
    case "today":
      return [startOfToday, now];
    case "yesterday": {
      const yesterday = new Date(startOfToday);
      yesterday.setDate(yesterday.getDate() - 1);
      return [yesterday, startOfToday];
    }
    case "last7days": {
      const start = new Date(startOfToday);
      start.setDate(start.getDate() - 7);
      return [start, now];
    }
    case "last30days": {
      const start = new Date(startOfToday);
      start.setDate(start.getDate() - 30);
      return [start, now];
    }
    case "thisMonth":
      return [new Date(now.getFullYear(), now.getMonth(), 1), now];
    case "lastMonth": {
      const start = new Date(now.getFullYear(), now.getMonth() - 1, 1);
      const end = new Date(now.getFullYear(), now.getMonth(), 0, 23, 59, 59);
      return [start, end];
    }
  }
}

export {
  formatDate,
  timeAgo,
  isToday,
  getDateRange,
  type DateRangePreset,
};
```

## 5. 对象与数组操作工具

```tsx
// src/utils/object.ts — 对象与数组操作

/**
 * 深拷贝（使用 structuredClone，现代浏览器原生支持）
 */
function deepClone<T>(obj: T): T {
  return structuredClone(obj);
}

/**
 * 从对象中挑选指定字段（运行时版本的 Pick）
 * pick({ a: 1, b: 2, c: 3 }, ["a", "c"]) → { a: 1, c: 3 }
 */
function pick<T extends Record<string, unknown>, K extends keyof T>(
  obj: T,
  keys: K[]
): Pick<T, K> {
  const result = {} as Pick<T, K>;
  for (const key of keys) {
    if (key in obj) {
      result[key] = obj[key];
    }
  }
  return result;
}

/**
 * 从对象中排除指定字段（运行时版本的 Omit）
 * omit({ a: 1, b: 2, c: 3 }, ["b"]) → { a: 1, c: 3 }
 */
function omit<T extends Record<string, unknown>, K extends keyof T>(
  obj: T,
  keys: K[]
): Omit<T, K> {
  const result = { ...obj };
  for (const key of keys) {
    delete result[key];
  }
  return result as Omit<T, K>;
}

/**
 * 移除对象中值为 undefined/null/空字符串 的字段
 * 常用于构建 API 请求参数（避免把空筛选条件传给后端）
 */
function removeEmpty<T extends Record<string, unknown>>(
  obj: T
): Partial<T> {
  const result: Record<string, unknown> = {};
  for (const [key, value] of Object.entries(obj)) {
    if (value !== undefined && value !== null && value !== "") {
      result[key] = value;
    }
  }
  return result as Partial<T>;
}

/**
 * 数组去重（支持通过 key 函数去重）
 * uniqueBy([{ id: 1, name: "a" }, { id: 1, name: "b" }], item => item.id)
 * → [{ id: 1, name: "a" }]
 */
function uniqueBy<T>(arr: T[], keyFn: (item: T) => unknown): T[] {
  const seen = new Set();
  return arr.filter((item) => {
    const key = keyFn(item);
    if (seen.has(key)) return false;
    seen.add(key);
    return true;
  });
}

/**
 * 数组分组：将数组按某个字段分组为 Map
 * groupBy(orders, order => order.status)
 * → Map { "pending" => [...], "shipped" => [...] }
 */
function groupBy<T, K>(arr: T[], keyFn: (item: T) => K): Map<K, T[]> {
  const map = new Map<K, T[]>();
  for (const item of arr) {
    const key = keyFn(item);
    const group = map.get(key) ?? [];
    group.push(item);
    map.set(key, group);
  }
  return map;
}

/**
 * 将扁平数组构建为树形结构（中后台菜单/部门/分类的经典需求）
 */
interface TreeNode {
  id: string;
  parentId: string | null;
  children?: TreeNode[];
  [key: string]: unknown;
}

function buildTree<T extends TreeNode>(
  items: T[],
  rootParentId: string | null = null
): (T & { children: T[] })[] {
  const map = new Map<string, T & { children: T[] }>();
  const roots: (T & { children: T[] })[] = [];

  for (const item of items) {
    map.set(item.id, { ...item, children: [] });
  }

  for (const item of map.values()) {
    if (item.parentId === rootParentId) {
      roots.push(item);
    } else {
      const parent = map.get(item.parentId!);
      if (parent) {
        parent.children.push(item);
      }
    }
  }

  return roots;
}

/**
 * 安全的 JSON 解析（不抛异常）
 */
function safeJsonParse<T>(json: string, fallback: T): T {
  try {
    return JSON.parse(json) as T;
  } catch {
    return fallback;
  }
}

export {
  deepClone,
  pick,
  omit,
  removeEmpty,
  uniqueBy,
  groupBy,
  buildTree,
  safeJsonParse,
  type TreeNode,
};
```

## 6. 避坑指南

### 坑一：formatDate 传入字符串在 Safari 上失败

```tsx
// Bad: Safari 不支持 "2026-04-19 14:30:00" 格式
new Date("2026-04-19 14:30:00") // Safari: Invalid Date

// Good: 使用 ISO 8601 格式，或用 T 分隔
new Date("2026-04-19T14:30:00") // 所有浏览器支持
```

### 坑二：浮点数精度导致金额计算错误

```tsx
// Bad: 0.1 + 0.2 = 0.30000000000000004
const total = price * quantity; // 可能出现 xxx.990000001

// Good: 用 safeAdd，或所有金额用"分"（整数）计算，显示时除以 100
const totalCents = priceCents * quantity;
const displayTotal = formatCurrency(totalCents / 100, "CNY");
```
