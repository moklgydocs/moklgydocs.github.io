# DOM 与浏览器工具函数：中后台必备浏览器交互工具

> **TL;DR**
> 中后台系统经常需要和浏览器打交道——复制到剪贴板、下载文件、滚动定位、检测设备类型、本地存储管理。这些操作看似简单，但浏览器兼容性、安全限制、错误处理的坑一个比一个深。本章提供一套封装好的浏览器工具函数。

## 1. 剪贴板工具

```tsx
// src/utils/dom.ts — DOM 与浏览器操作

/**
 * 复制文本到剪贴板
 * 优先使用 Clipboard API，降级到 execCommand
 */
async function copyToClipboard(text: string): Promise<boolean> {
  try {
    if (navigator.clipboard) {
      await navigator.clipboard.writeText(text);
      return true;
    }
    // 降级方案：创建临时 textarea
    const textarea = document.createElement("textarea");
    textarea.value = text;
    textarea.style.position = "fixed";
    textarea.style.left = "-9999px";
    document.body.appendChild(textarea);
    textarea.select();
    const success = document.execCommand("copy");
    document.body.removeChild(textarea);
    return success;
  } catch {
    return false;
  }
}
```

## 2. 下载工具

```tsx
/**
 * 下载 Blob 文件（Excel、PDF 等后端返回的二进制流）
 */
function downloadBlob(blob: Blob, filename: string): void {
  const url = URL.createObjectURL(blob);
  const link = document.createElement("a");
  link.href = url;
  link.download = filename;
  link.style.display = "none";
  document.body.appendChild(link);
  link.click();
  document.body.removeChild(link);
  URL.revokeObjectURL(url);
}

/**
 * 下载 URL 资源（图片、文件等远程资源）
 */
function downloadUrl(url: string, filename: string): void {
  const link = document.createElement("a");
  link.href = url;
  link.download = filename;
  link.target = "_blank";
  link.rel = "noopener noreferrer";
  document.body.appendChild(link);
  link.click();
  document.body.removeChild(link);
}

/**
 * 将 JSON 数据导出为 CSV 文件
 * 中后台经典需求：点击"导出"按钮，将表格数据下载为 CSV
 */
function exportToCsv<T extends Record<string, unknown>>(
  data: T[],
  columns: { key: keyof T; title: string }[],
  filename: string = "export.csv"
): void {
  // BOM 头确保 Excel 正确识别 UTF-8 编码
  const BOM = "\uFEFF";

  const header = columns.map((col) => `"${col.title}"`).join(",");
  const rows = data.map((row) =>
    columns
      .map((col) => {
        const value = row[col.key];
        const str = value === null || value === undefined ? "" : String(value);
        return `"${str.replace(/"/g, '""')}"`;
      })
      .join(",")
  );

  const csv = BOM + [header, ...rows].join("\n");
  const blob = new Blob([csv], { type: "text/csv;charset=utf-8" });
  downloadBlob(blob, filename);
}
```

## 3. 滚动工具

```tsx
/**
 * 平滑滚动到页面顶部
 */
function scrollToTop(behavior: ScrollBehavior = "smooth"): void {
  window.scrollTo({ top: 0, behavior });
}

/**
 * 平滑滚动到指定元素
 */
function scrollToElement(
  element: HTMLElement | string,
  options?: { offset?: number; behavior?: ScrollBehavior }
): void {
  const el =
    typeof element === "string" ? document.querySelector(element) : element;
  if (!el) return;

  const { offset = 0, behavior = "smooth" } = options ?? {};
  const top = el.getBoundingClientRect().top + window.scrollY + offset;
  window.scrollTo({ top, behavior });
}

/**
 * 判断元素是否在可视区域内
 */
function isElementInViewport(el: HTMLElement): boolean {
  const rect = el.getBoundingClientRect();
  return (
    rect.top >= 0 &&
    rect.left >= 0 &&
    rect.bottom <= window.innerHeight &&
    rect.right <= window.innerWidth
  );
}
```

## 4. 本地存储工具

```tsx
// src/utils/storage.ts — 类型安全的本地存储

/**
 * 带过期时间的 localStorage 封装
 */
interface StorageItem<T> {
  value: T;
  expireAt: number | null;
}

const storage = {
  /**
   * 存储数据，可选过期时间（毫秒）
   * storage.set("user", userData, 7 * 24 * 60 * 60 * 1000) → 7天过期
   */
  set<T>(key: string, value: T, ttl?: number): void {
    const item: StorageItem<T> = {
      value,
      expireAt: ttl ? Date.now() + ttl : null,
    };
    localStorage.setItem(key, JSON.stringify(item));
  },

  /**
   * 获取数据，过期则自动清除并返回 null
   */
  get<T>(key: string): T | null {
    const raw = localStorage.getItem(key);
    if (!raw) return null;

    try {
      const item: StorageItem<T> = JSON.parse(raw);
      if (item.expireAt && Date.now() > item.expireAt) {
        localStorage.removeItem(key);
        return null;
      }
      return item.value;
    } catch {
      return null;
    }
  },

  remove(key: string): void {
    localStorage.removeItem(key);
  },

  clear(): void {
    localStorage.clear();
  },
};

/**
 * SessionStorage 简易封装（标签页级别的临时数据）
 */
const sessionStore = {
  set<T>(key: string, value: T): void {
    sessionStorage.setItem(key, JSON.stringify(value));
  },

  get<T>(key: string): T | null {
    const raw = sessionStorage.getItem(key);
    if (!raw) return null;
    try {
      return JSON.parse(raw) as T;
    } catch {
      return null;
    }
  },

  remove(key: string): void {
    sessionStorage.removeItem(key);
  },
};

export { storage, sessionStore };
```

## 5. 设备与环境检测

```tsx
// src/utils/device.ts — 设备与环境检测

/**
 * 检测当前设备类型
 */
function isMobile(): boolean {
  return /Android|webOS|iPhone|iPad|iPod|BlackBerry|IEMobile|Opera Mini/i.test(
    navigator.userAgent
  );
}

/**
 * 检测是否为触屏设备
 */
function isTouchDevice(): boolean {
  return "ontouchstart" in window || navigator.maxTouchPoints > 0;
}

/**
 * 获取操作系统类型
 */
function getOS(): "Windows" | "Mac" | "Linux" | "iOS" | "Android" | "Unknown" {
  const ua = navigator.userAgent;
  if (/Windows/i.test(ua)) return "Windows";
  if (/Mac/i.test(ua) && !/iPhone|iPad/.test(ua)) return "Mac";
  if (/Linux/i.test(ua) && !/Android/.test(ua)) return "Linux";
  if (/iPhone|iPad|iPod/i.test(ua)) return "iOS";
  if (/Android/i.test(ua)) return "Android";
  return "Unknown";
}

/**
 * 获取快捷键修饰符提示文本（Mac 用 ⌘，其他用 Ctrl）
 */
function getModifierKey(): "⌘" | "Ctrl" {
  return getOS() === "Mac" ? "⌘" : "Ctrl";
}

export { isMobile, isTouchDevice, getOS, getModifierKey };
```

## 7. cn() 样式合并函数详解

```tsx
// src/lib/utils.ts — shadcn/ui 项目的灵魂函数

import { type ClassValue, clsx } from "clsx";
import { twMerge } from "tailwind-merge";

/**
 * cn() = clsx() + twMerge()
 *
 * clsx 负责：条件拼接类名
 *   cn("base", isActive && "active", isPrimary ? "bg-blue" : "bg-gray")
 *   → "base active bg-blue"（跳过 false/undefined/null）
 *
 * twMerge 负责：智能合并冲突的 Tailwind 类名
 *   cn("px-4 py-2", "px-6")
 *   → "py-2 px-6"（px-6 覆盖 px-4，而不是两个都保留）
 *
 * 为什么需要 twMerge？
 * 因为 CSS 的优先级由选择器权重决定，不是由书写顺序决定。
 * "px-4 px-6" 两个类都存在时，哪个生效取决于 CSS 文件中哪个在后面。
 * twMerge 帮你移除被覆盖的那个，确保最后写的生效。
 */
function cn(...inputs: ClassValue[]): string {
  return twMerge(clsx(inputs));
}

export { cn };
```

```tsx
// 实战示例：组件内使用 cn

import { cn } from "@/lib/utils";

interface ButtonProps {
  variant?: "primary" | "secondary";
  size?: "sm" | "lg";
  className?: string;
  children: React.ReactNode;
}

function Button({ variant = "primary", size, className, children }: ButtonProps) {
  return (
    <button
      className={cn(
        // 基础样式
        "rounded-md font-medium transition-colors",
        // 变体样式
        variant === "primary" && "bg-primary text-primary-foreground hover:bg-primary/90",
        variant === "secondary" && "bg-secondary text-secondary-foreground hover:bg-secondary/80",
        // 尺寸样式
        size === "sm" && "px-3 py-1.5 text-sm",
        size === "lg" && "px-6 py-3 text-lg",
        !size && "px-4 py-2 text-sm",
        // 外部传入的类名（可以覆盖上面的任何样式）
        className
      )}
    >
      {children}
    </button>
  );
}

// 使用：className 中的 bg-red-500 会覆盖组件内部的 bg-primary
<Button className="bg-red-500">自定义背景色</Button>
```

## 9. 避坑指南

### 坑一：localStorage 存储容量有限

```tsx
// Bad: 把大量数据塞进 localStorage
storage.set("allOrders", hugeOrderList); // 可能超出 5MB 限制

// Good: localStorage 只存少量配置和标识，大数据用 IndexedDB 或后端
storage.set("userPreferences", { theme: "dark", pageSize: 20 });
```

### 坑二：CSV 导出中文乱码

```tsx
// Bad: 没有 BOM 头，Excel 打开是乱码
const blob = new Blob([csvContent], { type: "text/csv" });

// Good: 加 BOM 头 "\uFEFF"
const blob = new Blob(["\uFEFF" + csvContent], { type: "text/csv;charset=utf-8" });
```
