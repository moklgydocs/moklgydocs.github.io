# useReducer 深入

## 学习目标

- 理解 useReducer 的工作机制和适用场景
- 掌握 action + reducer 的状态管理模式
- 知道什么时候该用 useReducer 而不是 useState

## 为什么需要 useReducer？

当组件状态逻辑变复杂时，useState 会变得难以维护：

```jsx
// ❌ 多个 useState 之间有关联，逻辑散落各处
function ShoppingCart() {
  const [items, setItems] = useState([]);
  const [total, setTotal] = useState(0);
  const [discount, setDiscount] = useState(0);
  
  const addItem = (item) => {
    setItems([...items, item]);
    setTotal(total + item.price);
    // 如果满 200 打 8 折...这个逻辑放哪？
  };
  
  const removeItem = (id) => {
    const item = items.find(i => i.id === id);
    setItems(items.filter(i => i.id !== id));
    setTotal(total - item.price);
    // 每个操作都要同时更新多个状态
  };
}
```

## useReducer 基本用法

```jsx
import { useReducer } from 'react';

// 1. 定义初始状态
const initialState = {
  items: [],
  total: 0,
  discount: 0,
};

// 2. 定义 reducer（纯函数：接收旧状态和动作，返回新状态）
function cartReducer(state, action) {
  switch (action.type) {
    case 'ADD_ITEM': {
      const newItems = [...state.items, action.payload];
      const newTotal = newItems.reduce((sum, item) => sum + item.price, 0);
      return {
        ...state,
        items: newItems,
        total: newTotal,
        discount: newTotal >= 200 ? newTotal * 0.2 : 0,
      };
    }
    case 'REMOVE_ITEM': {
      const newItems = state.items.filter(i => i.id !== action.payload);
      const newTotal = newItems.reduce((sum, item) => sum + item.price, 0);
      return {
        ...state,
        items: newItems,
        total: newTotal,
        discount: newTotal >= 200 ? newTotal * 0.2 : 0,
      };
    }
    case 'CLEAR_CART':
      return initialState;
    default:
      throw new Error(`未知 action: ${action.type}`);
  }
}

// 3. 在组件中使用
function ShoppingCart() {
  const [state, dispatch] = useReducer(cartReducer, initialState);

  return (
    <div>
      <h2>购物车 ({state.items.length} 件)</h2>
      {state.items.map(item => (
        <div key={item.id}>
          {item.name} - ¥{item.price}
          <button onClick={() => dispatch({ type: 'REMOVE_ITEM', payload: item.id })}>
            删除
          </button>
        </div>
      ))}
      <p>总计: ¥{state.total}</p>
      {state.discount > 0 && <p>优惠: -¥{state.discount}</p>}
      <button onClick={() => dispatch({ type: 'CLEAR_CART' })}>清空</button>
    </div>
  );
}
```

## useState vs useReducer 选择指南

| 场景 | 推荐 | 原因 |
|------|------|------|
| 单个简单值 | useState | 简单直接 |
| 多个相关状态 | useReducer | 状态逻辑集中管理 |
| 复杂状态转换 | useReducer | reducer 函数易于测试 |
| 需要在子组件中修改状态 | useReducer | dispatch 引用稳定 |
| 新状态依赖旧状态的多个字段 | useReducer | 在一个函数中访问完整状态 |

> 💡 **经验法则**：当你发现一个操作需要同时更新 3 个以上的 state，就是切换到 useReducer 的信号。

## useReducer + Context：轻量级 Redux

```jsx
const CartContext = createContext(null);

function CartProvider({ children }) {
  const [state, dispatch] = useReducer(cartReducer, initialState);
  return (
    <CartContext.Provider value={{ state, dispatch }}>
      {children}
    </CartContext.Provider>
  );
}

function useCart() {
  const context = useContext(CartContext);
  if (!context) throw new Error('useCart 必须在 CartProvider 内使用');
  return context;
}

// 任意深层组件都能访问购物车
function AddToCartButton({ product }) {
  const { dispatch } = useCart();
  return (
    <button onClick={() => dispatch({ type: 'ADD_ITEM', payload: product })}>
      加入购物车
    </button>
  );
}
```
