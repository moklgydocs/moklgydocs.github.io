# TodoApp 完整实战

## 项目目标

用我们学到的所有知识，从零构建一个**功能完整**的 Todo 应用，涵盖：

- ✅ 增删改查
- ✅ 状态管理（useReducer）
- ✅ 数据持久化（localStorage）
- ✅ 筛选与搜索
- ✅ 组件拆分
- ✅ TypeScript 类型

## 第一步：定义类型与状态

```tsx
// types.ts
export interface Todo {
  id: string;
  text: string;
  completed: boolean;
  createdAt: number;
}

export type Filter = 'all' | 'active' | 'completed';

export interface TodoState {
  todos: Todo[];
  filter: Filter;
}

export type TodoAction =
  | { type: 'ADD_TODO'; payload: string }
  | { type: 'TOGGLE_TODO'; payload: string }
  | { type: 'DELETE_TODO'; payload: string }
  | { type: 'EDIT_TODO'; payload: { id: string; text: string } }
  | { type: 'SET_FILTER'; payload: Filter }
  | { type: 'CLEAR_COMPLETED' };
```

## 第二步：Reducer 函数

```tsx
// todoReducer.ts
function todoReducer(state: TodoState, action: TodoAction): TodoState {
  switch (action.type) {
    case 'ADD_TODO':
      return {
        ...state,
        todos: [
          {
            id: crypto.randomUUID(),
            text: action.payload,
            completed: false,
            createdAt: Date.now(),
          },
          ...state.todos,
        ],
      };
    case 'TOGGLE_TODO':
      return {
        ...state,
        todos: state.todos.map(todo =>
          todo.id === action.payload
            ? { ...todo, completed: !todo.completed }
            : todo
        ),
      };
    case 'DELETE_TODO':
      return {
        ...state,
        todos: state.todos.filter(todo => todo.id !== action.payload),
      };
    case 'EDIT_TODO':
      return {
        ...state,
        todos: state.todos.map(todo =>
          todo.id === action.payload.id
            ? { ...todo, text: action.payload.text }
            : todo
        ),
      };
    case 'SET_FILTER':
      return { ...state, filter: action.payload };
    case 'CLEAR_COMPLETED':
      return {
        ...state,
        todos: state.todos.filter(todo => !todo.completed),
      };
    default:
      return state;
  }
}
```

## 第三步：组件拆分

```
TodoApp
├── TodoInput        → 新增输入框
├── TodoFilter       → 筛选栏（全部/未完成/已完成）
├── TodoList         → 列表容器
│   └── TodoItem     → 单个待办项（展示/编辑/删除）
└── TodoFooter       → 底部统计 + 清除已完成
```

## 第四步：核心组件实现

```tsx
// TodoApp.tsx - 根组件
function TodoApp() {
  const [state, dispatch] = useReducer(todoReducer, initialState, () => {
    const saved = localStorage.getItem('todo-state');
    return saved ? JSON.parse(saved) : initialState;
  });

  // 持久化
  useEffect(() => {
    localStorage.setItem('todo-state', JSON.stringify(state));
  }, [state]);

  // 派生状态
  const filteredTodos = useMemo(() => {
    switch (state.filter) {
      case 'active': return state.todos.filter(t => !t.completed);
      case 'completed': return state.todos.filter(t => t.completed);
      default: return state.todos;
    }
  }, [state.todos, state.filter]);

  const activeCount = state.todos.filter(t => !t.completed).length;

  return (
    <div className="todo-app">
      <h1>📝 Todo App</h1>
      <TodoInput onAdd={(text) => dispatch({ type: 'ADD_TODO', payload: text })} />
      <TodoFilter
        current={state.filter}
        onChange={(filter) => dispatch({ type: 'SET_FILTER', payload: filter })} 
      />
      <TodoList
        todos={filteredTodos}
        onToggle={(id) => dispatch({ type: 'TOGGLE_TODO', payload: id })}
        onDelete={(id) => dispatch({ type: 'DELETE_TODO', payload: id })}
        onEdit={(id, text) => dispatch({ type: 'EDIT_TODO', payload: { id, text } })}
      />
      <TodoFooter
        activeCount={activeCount}
        hasCompleted={state.todos.some(t => t.completed)}
        onClearCompleted={() => dispatch({ type: 'CLEAR_COMPLETED' })}
      />
    </div>
  );
}
```

```tsx
// TodoItem.tsx - 支持编辑的待办项
function TodoItem({ todo, onToggle, onDelete, onEdit }) {
  const [isEditing, setIsEditing] = useState(false);
  const [editText, setEditText] = useState(todo.text);

  const handleSubmit = () => {
    if (editText.trim()) {
      onEdit(todo.id, editText.trim());
      setIsEditing(false);
    }
  };

  if (isEditing) {
    return (
      <li>
        <input
          value={editText}
          onChange={e => setEditText(e.target.value)}
          onKeyDown={e => e.key === 'Enter' && handleSubmit()}
          onBlur={handleSubmit}
          autoFocus
        />
      </li>
    );
  }

  return (
    <li style={{ textDecoration: todo.completed ? 'line-through' : 'none' }}>
      <input type="checkbox" checked={todo.completed} onChange={() => onToggle(todo.id)} />
      <span onDoubleClick={() => setIsEditing(true)}>{todo.text}</span>
      <button onClick={() => onDelete(todo.id)}>🗑️</button>
    </li>
  );
}
```
