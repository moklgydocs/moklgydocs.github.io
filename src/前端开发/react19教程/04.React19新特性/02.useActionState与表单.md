# useActionState 与 React 19 表单革命

## 学习目标

- 掌握 useActionState 的用法
- 理解 form action 的工作原理
- 用 useFormStatus 获取表单提交状态

## 传统表单处理的痛点

React 18 及之前，处理表单提交需要大量样板代码：

```jsx
// ❌ 传统方式：太多样板代码
function LoginForm() {
  const [username, setUsername] = useState('');
  const [password, setPassword] = useState('');
  const [isPending, setIsPending] = useState(false);
  const [error, setError] = useState(null);

  const handleSubmit = async (e) => {
    e.preventDefault();
    setIsPending(true);
    setError(null);
    try {
      await login(username, password);
    } catch (err) {
      setError(err.message);
    } finally {
      setIsPending(false);
    }
  };

  return (
    <form onSubmit={handleSubmit}>
      <input value={username} onChange={e => setUsername(e.target.value)} />
      <input value={password} onChange={e => setPassword(e.target.value)} type="password" />
      <button disabled={isPending}>{isPending ? '登录中...' : '登录'}</button>
      {error && <p className="error">{error}</p>}
    </form>
  );
}
```

## useActionState：优雅的表单处理

```jsx
import { useActionState } from 'react';

function LoginForm() {
  const [error, submitAction, isPending] = useActionState(
    async (previousState, formData) => {
      const username = formData.get('username');
      const password = formData.get('password');
      
      const result = await login(username, password);
      if (result.error) {
        return result.error; // 返回值成为新的 state
      }
      redirect('/dashboard');
      return null;
    },
    null // 初始 state
  );

  return (
    <form action={submitAction}>
      <input name="username" placeholder="用户名" required />
      <input name="password" type="password" placeholder="密码" required />
      <button disabled={isPending}>
        {isPending ? '登录中...' : '登录'}
      </button>
      {error && <p className="error">{error}</p>}
    </form>
  );
}
```

**注意变化**：
- `<form action={submitAction}>` 取代了 `<form onSubmit={handleSubmit}>`
- 不需要 `e.preventDefault()`
- 不需要手动管理 `isPending` 和 `error`
- 使用 `formData.get()` 获取值，不需要受控组件

## useFormStatus：子组件中获取表单状态

```jsx
import { useFormStatus } from 'react-dom';

function SubmitButton() {
  const { pending } = useFormStatus();
  
  return (
    <button type="submit" disabled={pending}>
      {pending ? '⏳ 提交中...' : '📤 提交'}
    </button>
  );
}

// 使用：SubmitButton 必须在 <form> 内部
function ContactForm() {
  const [state, formAction] = useActionState(handleSubmit, null);
  
  return (
    <form action={formAction}>
      <input name="email" type="email" required />
      <textarea name="message" required />
      <SubmitButton />  {/* 自动获取表单的 pending 状态 */}
    </form>
  );
}
```

> 💡 **实战场景**：`useFormStatus` 特别适合封装通用的提交按钮组件——它能自动感知所在表单的提交状态，不需要通过 props 传递。

## 表单验证实战

```jsx
function RegistrationForm() {
  const [state, formAction, isPending] = useActionState(
    async (prevState, formData) => {
      const email = formData.get('email');
      const password = formData.get('password');
      const confirm = formData.get('confirm');
      
      // 客户端验证
      const errors = {};
      if (!email.includes('@')) errors.email = '邮箱格式不正确';
      if (password.length < 8) errors.password = '密码至少 8 位';
      if (password !== confirm) errors.confirm = '两次密码不一致';
      
      if (Object.keys(errors).length > 0) {
        return { errors, success: false };
      }
      
      // 服务端验证
      const result = await registerUser({ email, password });
      if (result.error) {
        return { errors: { server: result.error }, success: false };
      }
      
      return { errors: null, success: true };
    },
    { errors: null, success: false }
  );

  if (state.success) {
    return <p>✅ 注册成功！请查看邮箱验证。</p>;
  }

  return (
    <form action={formAction}>
      <div>
        <input name="email" type="email" placeholder="邮箱" />
        {state.errors?.email && <span className="error">{state.errors.email}</span>}
      </div>
      <div>
        <input name="password" type="password" placeholder="密码" />
        {state.errors?.password && <span className="error">{state.errors.password}</span>}
      </div>
      <div>
        <input name="confirm" type="password" placeholder="确认密码" />
        {state.errors?.confirm && <span className="error">{state.errors.confirm}</span>}
      </div>
      {state.errors?.server && <p className="error">{state.errors.server}</p>}
      <button disabled={isPending}>{isPending ? '注册中...' : '注册'}</button>
    </form>
  );
}
```
